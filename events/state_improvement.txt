namespace = state_improvement

##################################################################
# [#223] State-investment event chain. Backs the invest_in_state_button_* province buttons
# (common/scripted_guis/invest_in_state_buttons.txt) and the found-city / civ-cap buttons. This is
# vanilla Imperator: Rome content that Terra Indomita wires live (events/state_improvement.txt);
# imp19c copied the buttons + the state_improvement.N calls but left this event file a 0-byte
# placeholder, causing boot "'state_improvement.N' does not have a valid namespace" errors.
# Ported from TI (.1-.7 + the .12-.15/.18 notification relays), dropping TI's .8/.9 (dead AI
# auto-found, always=no), .10/.11/.16/.17 (Sparta _SPA — no SPA tag in this TC), and .19-.26
# (TI's own admin/manpower/academic/civ buttons, which imp19c does not have).
#
# Pattern: the button applies state_modification_in_progress (182d) immediately, then fires the
# matching .1-.4 event after 182 days to apply the permanent state_improvement_* modifier and
# relay a notification (.12-.15) to the owning country. .5/.6/.7 are the delayed city-status
# results for the civ-cap / found-city / found-metropolis buttons.
##################################################################

state_improvement.1 = { # Fortify the Province (military)
	type = state_event
	hidden = yes

	trigger = { exists = scope:state_improvement_target }

	immediate = {
		add_state_modifier = {
			name = state_improvement_military
			duration = -1
			mode = add
		}
		owner = {
			trigger_event = { id = state_improvement.12 }
		}
		increment_state_investment_count = yes
	}
}

state_improvement.5 = { # Coordinate Urban Development (city-level civilization)
	type = province_event
	hidden = yes

	trigger = { exists = owner }

	immediate = {
		save_scope_as = local_civilization_target
		add_province_modifier = {
			name = local_civilization_pmod_stack
			duration = -1
			mode = add
		}
		owner = {
			trigger_event = { id = state_improvement.18 }
		}
	}
}

state_improvement.6 = { # Found City (settlement -> city)
	type = province_event
	hidden = yes

	trigger = {
		exists = owner
		has_city_status = no
	}

	immediate = {
		set_city_status = city
	}
}

state_improvement.7 = { # Found Metropolis (city -> metropolis)
	type = province_event
	hidden = yes

	trigger = { exists = owner }

	immediate = {
		set_city_status = city_metropolis
	}
}

##################################################################
# Notification relays — fire on the owning country, narrate what happened in the state.
# show_as_tooltip only; the modifier itself was already applied by .1-.5 above.
##################################################################

state_improvement.12 = { # Fortify the Province
	type = minor_country_event
	title = "state_improvement.12.t"
	desc = "state_improvement.12.desc"
	left_portrait = scope:state_improvement_target.governor_or_ruler
	goto_location = scope:state_improvement_target.capital_scope
	interface_lock = no

	trigger = { exists = scope:state_improvement_target }

	immediate = {
		show_as_tooltip = {
			scope:state_improvement_target = {
				add_state_modifier = {
					name = state_improvement_military
					duration = -1
					mode = add
				}
			}
		}
	}

	option = { name = "state_improvement.12.a" }
}
