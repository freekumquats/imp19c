namespace = qing_gp_dispatch

qing_gp_dispatch.1 = {
	type = country_event
	title = "qing_gp_dispatch.1.t"
	desc = {
		triggered_desc = { trigger = { var:qing_gp_dispatch_evt_outcome = flag:success }  desc = "qing_gp_dispatch.1.desc_success" }
		triggered_desc = { trigger = { var:qing_gp_dispatch_evt_outcome = flag:failure }  desc = "qing_gp_dispatch.1.desc_failure" }
	}
	picture = council_country

	left_portrait = scope:qing_gp_dispatch_evt_diplomat

	# re-check in the event's own trigger (the roll happened at dispatch time; this just gates
	# whether the REPORT is still worth showing over the days-delay travel time).
	trigger = {
		tag = CHI
		has_variable = qing_gp_dispatch_evt_outcome
		exists = scope:qing_gp_dispatch_evt_diplomat
		scope:qing_gp_dispatch_evt_diplomat = { is_alive = yes  has_variable = qing_zongli_dispatched_gp_power }
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_gp_dispatch.1 (Great Game diplomat's report, outcome already resolved on dispatch)" }
		# the mission concludes with this report -- release him, unlike the play-dispatch
		# marker (which stays until the play itself resolves via AI_remove_diplomatic_play).
		scope:qing_gp_dispatch_evt_diplomat = {
			remove_variable = qing_zongli_dispatched_marker
			remove_variable = qing_zongli_dispatched_gp_power
		}
	}

	option = {
		name = "qing_gp_dispatch.1.a"
		remove_variable = qing_gp_dispatch_evt_outcome
	}
}
