########################################################################
# qing_gp_dispatch_events.txt — [fix 2026-08-24, user-requested] the delayed follow-up
# report for a Great Game "Dispatch a Diplomat" mission (QING_gp_dispatch_diplomat,
# se_QING_DIPLO.txt). Mirrors the Zongli play-dispatch report event
# (qing_zongli_dispatch_events.txt / se_QING_ZONGLI_DISPATCH.txt) but resolves in ONE
# shot rather than rolling indefinitely: a GP soothing errand is a single defined trip,
# not an open-ended posting that lasts until a play resolves. Queued with scope:
# qing_gp_dispatch_evt_diplomat (the working diplomat, carrying qing_zongli_dispatched_gp_power
# — a flag = britain|france|russia recording which power's tension he is working) already
# saved by QING_gp_dispatch_diplomat.
########################################################################

namespace = qing_gp_dispatch

qing_gp_dispatch.1 = {
	type = country_event
	title = "qing_gp_dispatch.1.t"
	desc = "qing_gp_dispatch.1.desc"
	picture = council_country

	left_portrait = scope:qing_gp_dispatch_evt_diplomat

	# re-check in the event's own trigger (the roll's gate can go stale over the days delay) --
	# mirrors qing_zongli_dispatch.1's own re-check comment.
	trigger = {
		tag = CHI
		exists = scope:qing_gp_dispatch_evt_diplomat
		scope:qing_gp_dispatch_evt_diplomat = { is_alive = yes  has_variable = qing_zongli_dispatched_gp_power }
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_gp_dispatch.1 (Great Game diplomat's report)" }
	}

	option = {
		name = "qing_gp_dispatch.1.a"
		# a single acknowledgement option -- the diplomat's own charisma decides the outcome
		# below, matching qing_zongli_dispatch.1's "based on the diplomat's skills" framing.
		random_list = {
			50 = {
				custom_tooltip = "qing_gp_dispatch.1.a.good.tt"
				modifier = {
					factor = 2
					scope:qing_gp_dispatch_evt_diplomat = { charisma >= 8 }
				}
				modifier = {
					factor = 0.5
					scope:qing_gp_dispatch_evt_diplomat = { charisma <= 3 }
				}
				if = {
					limit = { scope:qing_gp_dispatch_evt_diplomat = { var:qing_zongli_dispatched_gp_power = flag:britain } }
					QING_DECLINE_nudge = { var = qing_gp_tension_britain  amount = -6 }
				}
				else_if = {
					limit = { scope:qing_gp_dispatch_evt_diplomat = { var:qing_zongli_dispatched_gp_power = flag:france } }
					QING_DECLINE_nudge = { var = qing_gp_tension_france  amount = -6 }
				}
				else_if = {
					limit = { scope:qing_gp_dispatch_evt_diplomat = { var:qing_zongli_dispatched_gp_power = flag:russia } }
					QING_DECLINE_nudge = { var = qing_gp_tension_russia  amount = -6 }
				}
				LOG_line = { sys = QING  msg = "GP dispatch report: diplomat's follow-up lands well, tension eased further" }
			}
			50 = {
				custom_tooltip = "qing_gp_dispatch.1.a.bad.tt"
				modifier = {
					factor = 2
					scope:qing_gp_dispatch_evt_diplomat = { charisma <= 3 }
				}
				modifier = {
					factor = 0.5
					scope:qing_gp_dispatch_evt_diplomat = { charisma >= 8 }
				}
				if = {
					limit = { scope:qing_gp_dispatch_evt_diplomat = { var:qing_zongli_dispatched_gp_power = flag:britain } }
					QING_DECLINE_nudge = { var = qing_gp_tension_britain  amount = 6 }
				}
				else_if = {
					limit = { scope:qing_gp_dispatch_evt_diplomat = { var:qing_zongli_dispatched_gp_power = flag:france } }
					QING_DECLINE_nudge = { var = qing_gp_tension_france  amount = 6 }
				}
				else_if = {
					limit = { scope:qing_gp_dispatch_evt_diplomat = { var:qing_zongli_dispatched_gp_power = flag:russia } }
					QING_DECLINE_nudge = { var = qing_gp_tension_russia  amount = 6 }
				}
				LOG_line = { sys = QING  msg = "GP dispatch report: diplomat stumbles, tension partly rebounds" }
			}
		}
		# the mission concludes with this single report -- release him, unlike the play-dispatch
		# marker (which stays until the play itself resolves via AI_remove_diplomatic_play).
		scope:qing_gp_dispatch_evt_diplomat = {
			remove_variable = qing_zongli_dispatched_marker
			remove_variable = qing_zongli_dispatched_gp_power
		}
	}
}
