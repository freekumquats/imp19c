########################################################################
# qing_canton_events.txt — THE CANTON SYSTEM (粵海關 / 一口通商), task #389.
#
# Fired from se_QING_CANTON.txt's quarterly customs pulse. The Cohong (公行) — the
# licensed Hong-merchant guild that alone was permitted to trade with the Westerners —
# buckled recurrently under the levies and "squeeze" the Hoppo and the officials exacted;
# bankruptcies were frequent (hence the 1781 Consoo Fund, a 3% levy to cover failed hongs).
# When the Hoppo's over-exaction (qing_hoppo_squeeze) grows punishing, the merchants'
# debts come due and the throne must choose: bail out the guild from the customs surplus,
# rein in the Hoppo, or let the failures fall on the traders.
#
#   ROOT / this = the Qing (CHI, player)
#
# Namespace = qing_canton
#   .1  the Cohong debt crisis (public bailout / rein in the Hoppo / let it fall)
#   .2  [#111b] venal-Hoppo exposure (a censor's memorial names his graft hoard EARLY)
#   .3  [#111b] brilliant-Hoppo commendation (a high-finesse, honest Superintendent)
#
# [#111b] The Hoppo's take is GRADED on his stats (#111a: finesse lifts the customs, squeeze
# shaves them; #69: his corruption is the source of the squeeze, and his siphon is real
# confiscable wealth). These events surface that: the throne rewards a brilliant clean man,
# catches a venal one before the crisis, or weathers the Cohong collapse a punishing squeeze brings.
#
# Sources: Van Dyke, The Canton Trade (2005) on the Cohong + Consoo fund; H.B. Morse on
#   the Hoppo's squeeze; Chang Te-Ch'ang (JAS 1972) on the Neiwufu revenue routing.
########################################################################

namespace = qing_canton

########################################################################
# .1  THE COHONG DEBT CRISIS  (公行商欠)
# The Hong merchants, bled by the Hoppo's exactions, cannot meet their debts. The
# system that funnels the emperor's Canton silver is at risk of seizing up.
########################################################################
qing_canton.1 = {
	type = country_event
	title = "qing_canton.1.t"
	desc = "qing_canton.1.desc"
	picture = trade_port

	left_portrait = root.current_ruler


	trigger = {
		tag = CHI
		has_variable = qing_hoppo_squeeze
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_canton.1 (the Cohong debt crisis) [ROOT.GetTag]" }
	}

	# --- Bail the guild out from the customs surplus (公所 Consoo relief) ---------------
	option = {
		name = "qing_canton.1.a"
		custom_tooltip = "qing_canton.1.a.tt"
		# the relief comes out of the emperor's Canton haul (the money is HIS), not the 戶部.
		# [#389 review-fix] the privy-purse hit is gated on the SAME >= 30 condition as the
		# customs drawdown, so the tooltip's "spends ~30萬兩" is honest — a broke haul pays neither.
		if = {
			limit = { has_variable = qing_canton_customs  var:qing_canton_customs >= 30 }
			change_variable = { name = qing_canton_customs  subtract = 30 }
			if = { limit = { exists = current_ruler }  current_ruler = { add_gold = -24 } }
		}
		# the trade steadies and the squeeze eases as the throne signals restraint.
		QING_DECLINE_nudge = { var = qing_hoppo_squeeze  amount = -15 }
		add_stability = 1
	}

	# --- Rein in the Hoppo — dispatch an Inspector-General (御史) into the over-exaction ------
	option = {
		name = "qing_canton.1.b"
		custom_tooltip = "qing_canton.1.b.tt"
		# [#68 2026-08-08] A SIGNIFICANTLY CORRUPT Hoppo (corruption > 10) is not merely cautioned —
		# the Inspector-General's memorial brings him before the CENSORATE and he is actually impeached:
		# disgraced, stripped of the post, and his siphoned hoard (#69) confiscated into the treasury
		# (#70, via QING_censorate_impeach_uphold). Then clear qing_hoppo_holder so the Canton pulse
		# backfills a fresh, honest appointee. This is CHI country scope (ROOT = CHI), so saving the
		# holder as scope:qing_censorate_target and calling the uphold effect matches its other call sites.
		if = {
			limit = {
				has_variable = qing_hoppo_holder
				var:qing_hoppo_holder = { is_alive = yes  corruption > 10 }
			}
			var:qing_hoppo_holder = { save_scope_as = qing_censorate_target }
			# [#68 review-fix] strip the Hoppo marker off the disgraced man too (mirror the rotate path's
			# clean-up, se_QING_CANTON.txt) — the uphold clears offices but not this bespoke marker, so
			# without this the impeached ex-Hoppo would carry a stale qing_hoppo_marker forever.
			scope:qing_censorate_target = { if = { limit = { has_variable = qing_hoppo_marker }  remove_variable = qing_hoppo_marker } }
			QING_censorate_impeach_uphold = yes
			remove_variable = qing_hoppo_holder
			# the exposure of a venal customs chief eases the empire-wide graft (the Hoppo's squeeze
			# itself falls automatically next pulse, as it is now derived from the fresh backfill's low
			# corruption, #69 — so no direct squeeze nudge here).
			if = { limit = { has_variable = qing_corruption_level }  QING_DECLINE_nudge = { var = qing_corruption_level  amount = -5 } }
			add_legitimacy = 3
		}
		else = {
			# a broadly honest Hoppo (corruption <= 10) cannot be impeached — only cautioned. The
			# investigation RESTRAINS his exactions: reduce his own corruption (the concrete source the
			# squeeze is derived from, #69), so next pulse the squeeze reads lower. No fortune to seize.
			if = {
				limit = { has_variable = qing_hoppo_holder  var:qing_hoppo_holder = { is_alive = yes } }
				var:qing_hoppo_holder = { add_corruption = -8 }
			}
			if = { limit = { has_variable = qing_corruption_level }  QING_DECLINE_nudge = { var = qing_corruption_level  amount = -5 } }
			add_legitimacy = 3
		}
	}

	# --- Let the failures fall on the merchants (the guild absorbs its own) -------------
	option = {
		name = "qing_canton.1.c"
		custom_tooltip = "qing_canton.1.c.tt"
		# the throne is unmoved; the bankruptcies ripple and the trade sours, but no silver
		# is spent. The squeeze festers and the merchants' resentment simmers.
		QING_DECLINE_nudge = { var = qing_hoppo_squeeze  amount = 5 }
		QING_DECLINE_nudge_signed = { var = qing_currency_stress_residual  amount = 4 }
	}
}

########################################################################
# .2  VENAL-HOPPO EXPOSURE  (參奏 — a censor's memorial names the hoard)
# [#111b] A Canton censor memorialises that the seated Hoppo has amassed a great private
# fortune from his squeeze on the Hong merchants (his real #69 siphon wealth), while the
# trade has NOT yet buckled (squeeze still below the Cohong-crisis band). Caught EARLY, the
# throne can impeach him, quietly transfer him for a cut, or look away. Distinct from .1:
# keyed on his CONCRETE hoard (wealth), not the squeeze meter; a pre-emptive choice.
########################################################################
qing_canton.2 = {
	type = country_event
	title = "qing_canton.2.t"
	desc = "qing_canton.2.desc"
	picture = trade_port

	left_portrait = root.current_ruler
	right_portrait = var:qing_hoppo_holder

	trigger = {
		tag = CHI
		has_variable = qing_hoppo_holder
		var:qing_hoppo_holder = { is_alive = yes }
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_canton.2 (venal-Hoppo exposure) [ROOT.GetTag]" }
	}

	# --- Impeach him through the Censorate now (查辦抄家) -------------------------------------
	option = {
		name = "qing_canton.2.a"
		custom_tooltip = "qing_canton.2.a.tt"
		# reuse the PROVEN .1.b impeach sequence: save the holder, strip his Hoppo marker (the
		# uphold effect clears offices but not this bespoke marker), uphold the impeachment (抄家 —
		# confiscates his hoard to the treasury, hard-disgraces him, vacates his posts), then clear
		# the holder so the pulse backfills a fresh, honest man.
		if = {
			limit = { has_variable = qing_hoppo_holder  var:qing_hoppo_holder = { is_alive = yes } }
			var:qing_hoppo_holder = { save_scope_as = qing_censorate_target }
			scope:qing_censorate_target = { if = { limit = { has_variable = qing_hoppo_marker }  remove_variable = qing_hoppo_marker } }
			QING_censorate_impeach_uphold = yes
			remove_variable = qing_hoppo_holder
			if = { limit = { has_variable = qing_corruption_level }  QING_DECLINE_nudge = { var = qing_corruption_level  amount = -5 } }
			add_legitimacy = 3
		}
	}

	# --- Quietly transfer him and take the customary cut (調任) -------------------------------
	option = {
		name = "qing_canton.2.b"
		custom_tooltip = "qing_canton.2.b.tt"
		# no scandal: the corrupt-but-useful bondservant is rotated out, and the throne pockets a
		# THIRD of his hoard as an informal "gift". Read his wealth off a SAVED SCOPE (var:X.wealth
		# reads 0 silently — the se_QING_CANAL.txt:118-120 trap); he keeps the other two-thirds
		# (no strip). Then rotate via the existing lever (fresh man, squeeze eases).
		if = {
			limit = { has_variable = qing_hoppo_holder  var:qing_hoppo_holder = { is_alive = yes  wealth > 0 } }
			var:qing_hoppo_holder = { save_scope_as = qing_hoppo_seated }
			add_treasury = { value = scope:qing_hoppo_seated.wealth  divide = 3 }
			# [#108 fix 2026-08-15] stage into a var -- a compound amount= block trips TREASURY_LOG_it's
			# macro-arg validator (see common/scripted_effects/se_STATE_INVESTMENT.txt for the full note).
			set_variable = { name = TREASURY_LOG_amt_tmp  value = { value = scope:qing_hoppo_seated.wealth  divide = 3 } }
			TREASURY_LOG_it = { amount = var:TREASURY_LOG_amt_tmp }
			remove_variable = TREASURY_LOG_amt_tmp
		}
		QING_canton_rotate_hoppo = yes
	}

	# --- Keep the memorial in the palace (留中不發) — do nothing --------------------------------
	option = {
		name = "qing_canton.2.c"
		custom_tooltip = "qing_canton.2.c.tt"
		# the throne looks away; the graft festers and the squeeze creeps up.
		QING_DECLINE_nudge = { var = qing_hoppo_squeeze  amount = 4 }
		if = { limit = { has_variable = qing_corruption_level }  QING_DECLINE_nudge = { var = qing_corruption_level  amount = 2 } }
	}
}

########################################################################
# .3  BRILLIANT-HOPPO COMMENDATION  (嘉獎 — a rich, clean customs house)
# [#111b] The seated Hoppo (high finesse, honest — low squeeze, no graft hoard) has run the
# 粵海關 richly and straight. The throne may commend him with a plaque and a bonus, or promote
# him to a metropolitan post (losing Canton's ablest administrator). Positive counterpart to
# the venal/collapse beats; rewards the finesse #111a made matter.
########################################################################
qing_canton.3 = {
	type = country_event
	title = "qing_canton.3.t"
	desc = "qing_canton.3.desc"
	picture = trade_port

	left_portrait = root.current_ruler
	right_portrait = var:qing_hoppo_holder

	trigger = {
		tag = CHI
		has_variable = qing_hoppo_holder
		var:qing_hoppo_holder = { is_alive = yes }
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_canton.3 (brilliant-Hoppo commendation) [ROOT.GetTag]" }
	}

	# --- Commend him with a plaque and a bonus (嘉獎) ----------------------------------------
	option = {
		name = "qing_canton.3.a"
		custom_tooltip = "qing_canton.3.a.tt"
		# reward the man (loyalty + a modest honest bonus — NOT a siphon), a small customs surplus
		# his clean administration delivered, and a stability lift.
		if = {
			limit = { has_variable = qing_hoppo_holder  var:qing_hoppo_holder = { is_alive = yes } }
			var:qing_hoppo_holder = {
				add_loyalty = loyalty_qing_commended
				add_gold = 8
			}
		}
		if = { limit = { has_variable = qing_canton_customs }  change_variable = { name = qing_canton_customs  add = 10 } }
		add_stability = 1
	}

	# --- Promote him to a metropolitan post (內遷) — lose him at Canton ------------------------
	option = {
		name = "qing_canton.3.b"
		custom_tooltip = "qing_canton.3.b.tt"
		# he is too able to leave milking trade — recall him to the capital. The realm's meritocracy
		# is served, at the cost of Canton's best administrator (the rotate seats a likely lesser man).
		add_legitimacy = 2
		QING_canton_rotate_hoppo = yes
	}
}
