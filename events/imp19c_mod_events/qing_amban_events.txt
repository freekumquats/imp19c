# qing_amban_events.txt — the AMBAN / RESIDENT lifecycle events (#113).
#
# The 駐紮大臣 (amban, resident commissioner) posted to Inner Asian dependencies:
# Tibet, Mongolia, Xinjiang. Coupled to the LIFAN YUAN (理藩院) office. Events:
#   qing_amban.1 — a clash between the resident and the Lifan Yuan holder (recall
#                  crisis; player chooses to recall or back the resident)
#   qing_amban.2 — an amban goes native (or is co-opted by the local elite; player
#                  recalls him or lets him stay, accepting the drift)
#   qing_amban.4 — death or replacement of a long-serving resident (turnover)
# [#58 2026-08-10] qing_amban.3 ("Able Governance") DELETED — its able-resident reward is now
#   applied INLINE in the three integration negotiation-success arms (qing_integ.10.d/.12.d
#   unrest/strife = lesser bump; qing_integ.40.c decree/revolt = full commend), scaled by the
#   crisis's significance, on the specific event the player already saw rather than a generic follow-up.
#
# CHI player-only; fired from QING_amban_evaluate or as follow-ons. Namespace qing_amban.
########################################################################

namespace = qing_amban

# ======================================================================
# qing_amban.1 — CLASH: the resident and the Lifan Yuan holder disagree on how
# to manage the dependency. The holder demands recall; the resident defends his
# conduct. Player chooses: recall (back the Lifan holder, dependency stabilizes)
# or back the resident (holder loses face, resident stays, friction persists).
# ======================================================================
qing_amban.1 = {
	type = country_event
	title = "qing_amban.1.t"
	desc = "qing_amban.1.desc"
	picture = chinese_throne_room
	left_portrait = scope:qing_amban_crisis_holder
	right_portrait = scope:qing_amban_crisis_resident
	trigger = {
		tag = CHI
		exists = scope:qing_amban_crisis_subject
		exists = scope:qing_amban_crisis_resident
		exists = scope:qing_amban_crisis_holder
		scope:qing_amban_crisis_resident = { is_alive = yes  employer = scope:qing_amban_crisis_subject }
	}
	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_amban.1 (resident-holder clash) [ROOT.GetTag]" }
	}
	option = {
		name = "qing_amban.1.a"
		custom_tooltip = "qing_amban.1.a.tt"
		# RECALL: back the Lifan holder. Resident is recalled, dependency stabilizes.
		QING_amban_recall = { subject = scope:qing_amban_crisis_subject }
		# [#58 review-fix 2026-08-08] refresh the Lifan Yuan panel lists now — this option is dispatched
		# with a delay, so it runs AFTER its pulse's post-evaluate recompute; without this the recalled
		# seat shows a stale "?" until the next quarter. Safe inline (event option, never compile-inlined).
		QING_ministry_recompute_perf_lifanyuan = yes
		scope:qing_amban_crisis_subject = {
			add_opinion = { modifier = qing_amban_cooperation_opinion  target = ROOT }
		}
		# the Lifan holder gains standing (he prevailed)
		scope:qing_amban_crisis_holder = {
			QING_char_promote_standing = { pop = 3  prom = 2 }
		}
		# the recalled resident loses face
		scope:qing_amban_crisis_resident = {
			QING_char_promote_standing = { pop = -5  prom = -3 }
			add_loyalty = loyalty_qing_estranged
		}
	}
	option = {
		name = "qing_amban.1.b"
		custom_tooltip = "qing_amban.1.b.tt"
		# BACK THE RESIDENT: overrule the Lifan holder. Resident stays, but the
		# friction persists and the dependency drifts further. Holder loses face.
		scope:qing_amban_crisis_subject = {
			add_opinion = { modifier = qing_amban_clash_opinion  target = ROOT }
		}
		scope:qing_amban_crisis_holder = {
			QING_char_promote_standing = { pop = -4  prom = -2 }
			add_loyalty = loyalty_qing_estranged
		}
		# the resident gains standing (backed by the throne)
		scope:qing_amban_crisis_resident = {
			QING_char_promote_standing = { pop = 4  prom = 3 }
		}
	}
}

# ======================================================================
# qing_amban.2 — GOING NATIVE: the resident has been in post so long that he has
# gone native, or is co-opted by the local elite. The Lifan Yuan reports he is
# no longer reliably representing Peking's interests. Player recalls or accepts
# the drift (a nominal resident who is effectively a local figure).
# ======================================================================
qing_amban.2 = {
	type = country_event
	title = "qing_amban.2.t"
	desc = "qing_amban.2.desc"
	picture = senate_debate
	left_portrait = scope:qing_amban_native
	trigger = {
		tag = CHI
		# fires for a subject with a posted amban who has been in post >= 8 years
		# and has low affinity with the throne (< 40)
		any_subject = {
			has_variable = qing_amban_here
			var:qing_amban_here = {
				is_alive = yes
				employer = prev
				# [deferred-fix #2] `time_in_office` invalid — use the auto-expiring tenure timer.
				NOT = { has_variable = qing_amban_tenure_pending }
				has_variable = qing_char_affinity
				var:qing_char_affinity < 40
			}
		}
	}
	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_amban.2 (going native) [ROOT.GetTag]" }
		# pick one such subject + resident
		random_subject = {
			limit = {
				has_variable = qing_amban_here
				var:qing_amban_here = {
					is_alive = yes
					employer = prev
					# [deferred-fix #2] `time_in_office` is not a valid character token (proven
					# nowhere). Long tenure is expressed by the qing_amban_tenure_pending timer
					# (set at post time, days=2920) having auto-expired: no var = served >= 8 yrs.
					NOT = { has_variable = qing_amban_tenure_pending }
					has_variable = qing_char_affinity
					var:qing_char_affinity < 40
				}
			}
			save_scope_as = qing_amban_native_subject
			var:qing_amban_here = { save_scope_as = qing_amban_native }
		}
	}
	option = {
		name = "qing_amban.2.a"
		custom_tooltip = "qing_amban.2.a.tt"
		# RECALL: replace him. Dependency stabilizes briefly, but turnover cost.
		QING_amban_recall = { subject = scope:qing_amban_native_subject }
		# [#58 review-fix 2026-08-08] refresh the Lifan Yuan panel lists now — this option is dispatched
		# with a delay, so it runs AFTER its pulse's post-evaluate recompute; without this the recalled
		# seat shows a stale "?" until the next quarter. Safe inline (event option, never compile-inlined).
		QING_ministry_recompute_perf_lifanyuan = yes
		scope:qing_amban_native_subject = {
			add_opinion = { modifier = qing_amban_cooperation_opinion  target = ROOT }
		}
		add_treasury = -180
		TREASURY_LOG_it = { amount = -180 }
	}
	option = {
		name = "qing_amban.2.b"
		custom_tooltip = "qing_amban.2.b.tt"
		# LET HIM STAY: accept the drift. Dependency slips further, but no turnover cost.
		scope:qing_amban_native_subject = {
			add_opinion = { modifier = qing_amban_ineffective_opinion  target = ROOT }
		}
		# the resident's affinity falls further (he is now a local figure)
		scope:qing_amban_native = {
			QING_char_affinity = yes
			change_variable = { name = qing_char_affinity  subtract = 10 }
		}
	}
}

# ======================================================================
# qing_amban.4 — TURNOVER: a long-serving resident dies or is replaced due to age
# or health. Player appoints a replacement or leaves the post vacant (dependency
# drifts until a new resident is posted).
# ======================================================================
qing_amban.4 = {
	type = country_event
	title = "qing_amban.4.t"
	desc = "qing_amban.4.desc"
	picture = chinese_throne_room
	left_portrait = scope:qing_amban_old
	trigger = {
		tag = CHI
		# fires for a subject with a posted amban who is old (>= 65) or unhealthy (< 0.4)
		any_subject = {
			has_variable = qing_amban_here
			var:qing_amban_here = {
				is_alive = yes
				employer = prev
				OR = {
					age >= 65
					health < 0.4
				}
			}
		}
	}
	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_amban.4 (resident turnover) [ROOT.GetTag]" }
		random_subject = {
			limit = {
				has_variable = qing_amban_here
				var:qing_amban_here = {
					is_alive = yes
					employer = prev
					OR = {
						age >= 65
						health < 0.4
					}
				}
			}
			save_scope_as = qing_amban_old_subject
			var:qing_amban_here = { save_scope_as = qing_amban_old }
		}
	}
	option = {
		name = "qing_amban.4.a"
		custom_tooltip = "qing_amban.4.a.tt"
		# REPLACE: recall the old resident, post a new one. Cost in gold (appointment).
		QING_amban_recall = { subject = scope:qing_amban_old_subject }
		QING_amban_post = { subject = scope:qing_amban_old_subject }
		# [#114 review-fix] QING_amban_post can now no-op (no free banner graduate to draw), leaving
		# the seat vacant — the create_character fallback that used to guarantee a resident was
		# removed. Only charge the appointment cost when a resident was actually posted, or the
		# player pays 200 gold for a seat that ends empty.
		if = {
			limit = {
				scope:qing_amban_old_subject = { has_variable = qing_amban_here }
			}
			add_treasury = -200
			TREASURY_LOG_it = { amount = -200 }
		}
		# [#58 review-fix 2026-08-08] refresh the Lifan Yuan panel lists AFTER the repost so the new
		# resident shows immediately (this delayed-dispatch option runs after its pulse's recompute).
		# Safe inline (event option, never compile-inlined, unlike the scripted_gui buttons).
		QING_ministry_recompute_perf_lifanyuan = yes
	}
	option = {
		name = "qing_amban.4.b"
		custom_tooltip = "qing_amban.4.b.tt"
		# LEAVE VACANT: recall without replacement. Dependency drifts.
		QING_amban_recall = { subject = scope:qing_amban_old_subject }
		# [#58 review-fix 2026-08-08] refresh the Lifan Yuan panel lists now — this option is dispatched
		# with a delay, so it runs AFTER its pulse's post-evaluate recompute; without this the recalled
		# seat shows a stale "?" until the next quarter. Safe inline (event option, never compile-inlined).
		QING_ministry_recompute_perf_lifanyuan = yes
		scope:qing_amban_old_subject = {
			add_opinion = { modifier = qing_amban_ineffective_opinion  target = ROOT }
		}
	}
}


# ======================================================================
# [#26 2026-08-07] qing_amban.6 — HIDDEN trampoline for the PICKER post (a PLAYER-CHOSEN amban).
# The picker row-click (qing_amban_appoint_selected) saves the chosen character as scope:qing_amban_picked
# and the subject as scope:qing_amban_manual_target, then fires this. It PRESETS scope:qing_amban_new to
# the picked man and calls QING_amban_wire directly (the wire does NOT draw - review-confirmed
# se_QING_AMBAN.txt:163). So the player's pick is posted. Trampoline (not inlined in the scripted_gui,
# the #443 compile-inline crash class). ROOT = CHI.
# [#114 2026-08-11] The sibling qing_amban.5 (a manual-post trampoline that called QING_amban_post to
# DRAW the ablest graduate) is DELETED — the #26/#62 picker migration rerouted all manual posting
# through this event, orphaning .5 with zero live trigger_event callers (confirmed by whole-repo grep).
# Removed rather than left commented, so no future edit can silently rewire a button back onto the
# un-picker'd auto-draw path.
# ======================================================================
qing_amban.6 = {
	type = country_event
	hidden = yes
	trigger = { tag = CHI }
	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_amban.6 (picker post trampoline) [ROOT.GetTag]" }
		# [#62 2026-08-08] REPLACE MODE: the picker was opened over an OCCUPIED seat, so recall the sitting
		# incumbent FIRST (clears qing_amban_here) before the post guard below — otherwise the `NOT has
		# qing_amban_here` limit would reject the replacement. Recall is iterator-free (safe to call here),
		# and this is a runtime event anyway (not a compiled button). The player already paid 25 PI + 2
		# tyranny at the row-click; recall here carries no extra charge (it is part of the paid replace).
		if = {
			limit = {
				has_variable = qing_amban_pick_was_replace   # guard the read (unset in a plain-post flow — avoids read-before-set log noise)
				var:qing_amban_pick_was_replace = 1
				exists = scope:qing_amban_manual_target
				scope:qing_amban_manual_target = { has_variable = qing_amban_here }
			}
			QING_amban_recall = { subject = scope:qing_amban_manual_target }
		}
		if = {
			limit = {
				exists = scope:qing_amban_manual_target
				exists = scope:qing_amban_picked
				scope:qing_amban_picked = { is_alive = yes  employer = ROOT  NOT = { QING_char_holds_court_position = yes } }
				NOT = { scope:qing_amban_manual_target = { has_variable = qing_amban_here } }
			}
			scope:qing_amban_picked = { save_scope_as = qing_amban_new }
			QING_amban_wire = { subject = scope:qing_amban_manual_target }
			LOG_line = { sys = QING  msg = "amban: PICKED resident posted by the player" }
		}
		else = {
			# [#26 review-fix] the post failed (picked man ineligible at fire / post filled in the race window).
			# The row handler already charged 25 PI + flagged the subject manual — REFUND both so the player
			# isn't out the influence with an empty, manual-flagged dependency.
			add_political_influence = 25
			# [#62] a failed REPLACE also over-charged 2 tyranny at the row-click — refund it (subtract).
			# (The incumbent was already recalled above, so a failed replace degrades to a recall; that is
			# acceptable, but the player should not eat the replace tyranny for a post that never landed.)
			if = { limit = { has_variable = qing_amban_pick_was_replace  var:qing_amban_pick_was_replace = 1 }  add_tyranny = -2 }
			if = {
				limit = { exists = scope:qing_amban_manual_target  scope:qing_amban_manual_target = { has_variable = qing_amban_manual }  NOT = { scope:qing_amban_manual_target = { has_variable = qing_amban_here } } }
				scope:qing_amban_manual_target = { remove_variable = qing_amban_manual }
			}
			LOG_fail = { sys = QING  fn = "qing_amban.6"  reason = "picked candidate no longer eligible or seat filled before the trampoline fired" }
		}
		# [#62] clear the replace-mode flag so it never leaks into a later post/replace cycle.
		if = { limit = { has_variable = qing_amban_pick_was_replace }  remove_variable = qing_amban_pick_was_replace }
		# [#58/#59/#63 2026-08-08] refresh the Lifan Yuan panel lists after the picker post (success path
		# shows the new resident + Replace; refund path clears qing_amban_manual). ROOT = CHI; idempotent.
		QING_ministry_recompute_perf_lifanyuan = yes
	}
}

# #443/#34 compile-inline AV class). So the buttons fire THIS runtime event (trigger_event is a
# runtime ref, never compile-inlined) right after their inline recall/replace, and the heavy
# recompute lives here where the parser never expands it into the button chain. Mirrors the
# Rites/Personnel/Household trampoline pattern. ROOT = CHI (trigger tag=CHI). Hidden, no window.
# ======================================================================
qing_amban.7 = {
	type = country_event
	hidden = yes
	trigger = { tag = CHI }
	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_amban.7 (Lifan Yuan panel refresh) [ROOT.GetTag]" }
		QING_ministry_recompute_perf_lifanyuan = yes
	}
}
