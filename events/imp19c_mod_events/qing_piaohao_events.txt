########################################################################
# qing_piaohao_events.txt — THE RISE OF THE SHANXI DRAFT BANKS (票號, task #232).
#
# The Taiyuan piaohao seed was REMOVED from the 1763 static building start (#229)
# because it is anachronistic: the Shanxi piaohao (draft/remittance banks) did not
# exist at a 1763 or 1815 start. This event founds them at their historical time.
#
# History (research-sourced, EN+CN): the first documented piaohao — Rishengchang
# 日昇昌 in Pingyao 平遙, converted from the Xiyuecheng 西裕成 dye company under
# chief manager Lei Lutai 雷履泰 — emerged in the early Daoguang reign. "1823" is
# the conventional date (Pingyao museum: 道光四年/1824); the hard stele-bracket is
# 1819–1838 (黃鑑暉《山西票號史》; Beijing guild-hall steles via Li Hua). We do NOT
# claim Rishengchang was categorically first — it was the first well-documented and
# most successful; scholars disagree on absolute origin (五說). The empire-wide
# BRANCH NETWORK is a 1830s–40s phenomenon (Mao Hongsui's spin-off banks; ~11 more
# firms over the next decades). Sources: Morck & Yang, "The Shanxi Banks" (NBER WP
# 15884, 2010); 黃鑑暉《山西票號史》(1992); 張正明《晉商興衰史》.
#
# .1 fires ~1823 (day-offset from the 1763 start + a current_date >= 1823 re-check
# at fire time, so it lands correctly from any start date and only once). Seeds the
# qing_draft_bank_building at the Shanxi home base (Taiyuan, standing in for the
# Pingyao/Taiyuan/Qi-county piaohao heartland) and the principal remittance hubs the
# banks branched to (Beijing the political capital, the Jiangnan commercial cores).
#
#   ROOT / this = the Qing (CHI, player or AI).
#
# add_building_level bypasses the building `allow` tech gate (build-menu only) — same
# mechanism as the start-of-game seeds; guarded (exists + owner CHI + not present).
# Idempotent via qing_piaohao_founded.
#
# Namespace = qing_piaohao. All paths logged sys = QING. NO BOM (se/event convention).
########################################################################

namespace = qing_piaohao

########################################################################
# .1  票號之興 — THE RISE OF THE DRAFT BANKS (~1823)
# The Shanxi merchants' remittance houses formalise into the piaohao: paper drafts
# move silver across the empire without shipping bullion, financing the long-distance
# trade and, in time, the state's own transfers.
########################################################################
qing_piaohao.1 = {
	type = country_event
	title = "qing_piaohao.1.t"
	desc = "qing_piaohao.1.desc"
	picture = trade_port

	left_portrait = root.current_ruler

	is_triggered_only = yes

	# Land in the historical window and only once. The day-offset schedule targets
	# ~1823; this re-check makes it robust to any start date and idempotent.
	trigger = {
		tag = CHI
		NOT = { has_variable = qing_piaohao_founded }
		current_date >= 1822.12.1
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_piaohao.1 (rise of the Shanxi draft banks) for [ROOT.GetTag]" }
		set_variable = qing_piaohao_founded
		# Home base: the Shanxi piaohao heartland (Taiyuan stands in for Pingyao/Taiyuan/Qi).
		QING_piaohao_found_at = { P = 8501  NAME = "Taiyuan 太原 (Shanxi piaohao heartland)" }
		# Principal remittance hubs the network branched to: the political capital and the
		# Jiangnan commercial cores (Suzhou, Jiangning) + the southern trade capital (Canton).
		QING_piaohao_found_at = { P = 8363  NAME = "Beijing 北京 branch" }
		QING_piaohao_found_at = { P = 2588  NAME = "Suzhou 蘇州 branch" }
		QING_piaohao_found_at = { P = 6659  NAME = "Jiangning 江寧 branch" }
		QING_piaohao_found_at = { P = 9298  NAME = "Guangzhou 廣州 branch" }
	}

	# --- The paper draft comes of age --------------------------------------------
	option = {
		name = "qing_piaohao.1.a"
	}
}

# Seed one qing_draft_bank_building at province $P$, guarded (exists + CHI-owned +
# not already present). add_building_level bypasses the allow tech gate. Static LOG
# (no multi-word $NAME$ in the log string — log-string-macro rule); $NAME$ is only a
# readable label on the call site. Scope: any (called from CHI-scope immediate).
QING_piaohao_found_at = {
	if = {
		limit = {
			exists = p:$P$
			p:$P$ = {
				owner = c:CHI
				has_city_status = yes   # re-guard the building's own potential (code-review: don't rely on add_building_level bypassing potential; honest SKIP if a target ever lacks city status)
				NOT = { has_building = qing_draft_bank_building }
			}
		}
		p:$P$ = {
			add_building_level = qing_draft_bank_building
			LOG_line = { sys = QING  msg = "piaohao: qing_draft_bank_building founded (p:$P$)" }
		}
	}
	else = {
		LOG_line = { sys = QING  msg = "piaohao: SKIP draft bank (p:$P$) — not CHI-owned or already present" }
	}
}
