########################################################################
# qing_subject_integration.txt — reaction events for the Qing subject
# integration process (the promote/demote/integrate/absorb feature).
#
# Fired from se_SUBJECT_QING.txt (SUBJ_QING_roll_reaction) either right after
# a deliberate integration push, or on the ambient pulse for any subject with
# an active, unsuspended integration. In every event:
#     ROOT / this  = the overlord (player, Qing / CHI)
#     scope:target = the subject governorship being integrated
#
# The chain is deliberately Qing-only and player-only; AI never sees it.
# Each event is anchored to real Qing frontier history and cites it in the
# localisation. Options let the player push through, delay, suspend, or
# accelerate the process, and are wired into existing mechanics: treasury /
# currency, industrialisation, trade, pop happiness/unrest, stability,
# character loyalty, and the subject-relationship modifiers.
#
# Namespace = qing_integ
#   .10  NEGATIVE — popular unrest (Three Feudatories / White Lotus echo)
#   .11  NEGATIVE — corruption scandal (Heshen archetype)
#   .12  NEGATIVE — ethnic strife (Miao / Panthay / Dungan risings)
#   .20  POSITIVE — bountiful frontier harvest (tuntian colonies)
#   .21  POSITIVE — investiture festival / imperial favour
########################################################################

namespace = qing_integ

########################################################################
# .10  POPULAR UNREST
# Anchor: Revolt of the Three Feudatories (三藩之亂, 1673-81) — Kangxi's move
# to revoke the autonomous southern fiefs triggered an eight-year revolt; and
# the White Lotus Rebellion (白蓮教起義, 1794-1804), where a Daoguang decree
# admitted "it was extortion by local officials that goaded the people into
# rebellion" (官吏苛索逼民為亂). Tightening the screws provokes a rising.
########################################################################
qing_integ.10 = {
	type = country_event
	title = "qing_integ.10.t"
	# [#task-27 2026-08-04] base text + the shared actor-status addendum naming the local amban /
	# garrison when present (resolved in the immediate). .10 now carries both a garrison option (.10.e)
	# and an amban option (.10.d), so it names them on the ground like its sibling events do.
	desc = {
		desc = "qing_integ.10.desc"
		first_valid = {
			triggered_desc = { trigger = { has_variable = integ_amban_present  has_variable = integ_garrison_size  var:integ_garrison_size >= 1 }  desc = "qing_integ.actors_both" }
			triggered_desc = { trigger = { has_variable = integ_amban_present }  desc = "qing_integ.actors_amban" }
			triggered_desc = { trigger = { has_variable = integ_garrison_size  var:integ_garrison_size >= 1 }  desc = "qing_integ.actors_garrison" }
			triggered_desc = { trigger = { always = yes }  desc = "qing_integ.actors_none" }
		}
	}
	picture = revolt

	left_portrait = root.current_ruler
	right_portrait = scope:target.current_ruler

	goto_location = scope:unrest_province


	# [playtest loc-fix] Self-abort if the subject vanished during the 3-10 day
	# delay (absorbed to landless / deleted) — otherwise scope:target orphans and
	# the desc renders ERROR:[scope:target.GetName]. Mirrors the qing_integ.30 guard.
	trigger = {
		exists = scope:target
		scope:target = { is_subject_of = ROOT }
	}

	immediate = {
		LOG_line = { sys = SUBJ  msg = "EVENT qing_integ.10 (unrest) overlord=[ROOT.GetTag] subject=[scope:target.GetTag]" }
		# Anchor the event on the subject's most populous province, if any.
		if = {
			limit = { scope:target = { any_owned_province = { count >= 1 } } }
			scope:target = {
				ordered_owned_province = {
					order_by = total_population
					max = 1
					save_scope_as = unrest_province
				}
			}
		}
		# [#task-8] resolve the resident amban + garrison so the options below can be gated by them.
		SUBJ_QING_resolve_integ_actors = yes
	}

	# --- Push through with a show of force: BRING IN troops to crush it ---
	# [#task-27 R1 2026-08-04] This is the throne BRINGING IN a column to crush the rising (a heavy
	# hand, -5 stability). The case where a LOCAL garrison is already on the spot has been promoted to
	# its own cheaper option (.10.e). Always available (the fallback when no local garrison can act).
	option = {
		name = "qing_integ.10.a"
		custom_tooltip = "qing_integ.10.a.tt"
		add_war_exhaustion = 2
		add_stability = -5
		LOG_line = { sys = SUBJ  msg = "unrest: outside troops brought in to crush the rising, subject=[scope:target.GetTag]" }
		# The subject network resents the heavy hand (Three Feudatories echo).
		add_country_modifier = {
			name = qing_integ_frontier_resentment
			duration = 1825
		}
		if = {
			limit = { exists = scope:unrest_province }
			scope:unrest_province = {
				add_province_modifier = {
					name = local_unrest_harsh
					duration = 730
				}
			}
		}
		# But a crushed revolt does not stop the process.
		SUBJ_QING_clear_integ_actors = yes
	}

	# --- [#task-27 R1 2026-08-04] The LOCAL GARRISON restores order ------
	# Shown only when an imperial garrison strong enough to matter (edge >= 1) already stands in the
	# dependency: the banner troops on the spot put the rising down at once, cheaper and with far less
	# lingering resentment than marching in an outside column (.10.a). A dominant garrison (edge 2) does
	# it almost bloodlessly (-1 stability); rough parity (edge 1) is rougher (-3). The "local garrison
	# does it, cheaper" counterpart to .10.a. The process keeps moving.
	option = {
		name = "qing_integ.10.e"
		custom_tooltip = "qing_integ.10.e.tt"
		trigger = {
			has_variable = integ_garrison_edge
			var:integ_garrison_edge >= 1
		}
		if = {
			limit = { has_variable = integ_garrison_edge  var:integ_garrison_edge = 2 }
			add_stability = -1
			LOG_line = { sys = SUBJ  msg = "unrest: a dominant local garrison restores order almost bloodlessly, subject=[scope:target.GetTag]" }
		}
		else = {
			add_stability = -3
			LOG_line = { sys = SUBJ  msg = "unrest: the local garrison restores order, subject=[scope:target.GetTag]" }
		}
		SUBJ_QING_war_garrison_outcome = { delta = 2 }   # a garrison keeping the peace credits the Board of War
		if = {
			limit = { exists = scope:unrest_province }
			scope:unrest_province = { add_province_modifier = { name = local_unrest_harsh  duration = 365 } }
		}
		SUBJ_QING_clear_integ_actors = yes
	}

	# --- [#task-8; #task-27 req-9 light skill-check 2026-08-04] The resident amban restores order ---
	# Shown only when a steady-or-better resident amban (band >= 1) sits in the dependency: he OFFERS to
	# calm the unrest without the mailed fist. Whether he SUCCEEDS is a SKILL CHECK on his diplomatic
	# skills (charisma + finesse, qing_amban_negotiate_chance_svalue):
	#   • SUCCESS → he talks the crowds down: a little stability, the province eased, his standing and
	#               the Lifan Yuan credited.
	#   • FAILURE → his intervention falls flat; a small stability ding, and he is DISCREDITED (a
	#               prominence hit) and the Lifan Yuan marked down — but per "execute only in .41", a
	#               failed CALMING does NOT kill him (routine unrest is not the mortal gamble of .41).
	# Keeps the process moving either way. Absent / wavering amban => hidden.
	option = {
		name = "qing_integ.10.d"
		custom_tooltip = "qing_integ.10.d.tt"
		trigger = {
			has_variable = integ_amban_present
			var:integ_amban_band >= 1
			exists = scope:integ_amban
		}
		scope:integ_amban = {
			random = {
				chance = qing_amban_negotiate_chance_svalue
				set_variable = { name = integ_amban_negotiated_ok  value = 1 }
			}
		}
		if = {
			limit = { scope:integ_amban = { has_variable = integ_amban_negotiated_ok } }
			add_stability = 1
			scope:integ_amban = { add_prominence = 5  remove_variable = integ_amban_negotiated_ok }
			if = {
				limit = { exists = scope:unrest_province }
				scope:unrest_province = { add_province_modifier = { name = positive_food_modifier  duration = 365 } }
			}
			SUBJ_QING_lifan_amban_outcome = { delta = 4 }
			LOG_line = { sys = SUBJ  msg = "unrest: resident amban restores order by persuasion (skill check passed), subject=[scope:target.GetTag]" }
		}
		else = {
			add_stability = -1
			scope:integ_amban = { add_prominence = -5 }
			SUBJ_QING_lifan_amban_outcome = { delta = -3 }
			LOG_line = { sys = SUBJ  msg = "unrest: the amban's appeal fails and he is discredited, subject=[scope:target.GetTag]" }
		}
		SUBJ_QING_clear_integ_actors = yes
	}

	# --- Suspend the integration and let tempers cool --------------------
	option = {
		name = "qing_integ.10.b"
		custom_tooltip = "qing_integ.10.b.tt"
		SUBJ_QING_suspend_integration = yes
		SUBJ_QING_reduce_integration = { steps = 1 }
		add_stability = 2
		SUBJ_QING_clear_integ_actors = yes
	}

	# --- Give alms and remit taxes to pacify the frontier (rou yuan 柔遠) -
	option = {
		name = "qing_integ.10.c"
		custom_tooltip = "qing_integ.10.c.tt"
		trigger = { treasury >= 100 }
		add_treasury = -100
		if = {
			limit = { exists = scope:unrest_province }
			scope:unrest_province = {
				add_province_modifier = {
					name = positive_food_modifier
					duration = 730
				}
			}
		}
		# Applied to the overlord: subject_loyalty tightens the bond, and integrate_speed
		# now feeds the ENGINE's subject-integration timer (post-#subj-integ-rework: once the
		# 改土歸流 capstone hands a governorship to the engine's native integration, integrate_speed
		# genuinely accelerates the gradual absorption — it was inert on the old flat-+1 meter).
		add_country_modifier = {
			name = qing_integ_imperial_favor
			duration = 1095
		}
		SUBJ_QING_clear_integ_actors = yes
	}
}

########################################################################
# .11  CORRUPTION SCANDAL
# Anchor: Heshen (和珅, d. 1799), "the most corrupt official in Chinese
# history" — his confiscated estate (~1.1bn taels) reputedly equalled twelve
# years of imperial revenue. His graft diverted funds meant for canals and
# dams, worsening floods and famine (Hummel 1943). A grandee has been skimming
# the integration funds.
########################################################################
qing_integ.11 = {
	type = country_event
	title = "qing_integ.11.t"
	# [#task-20 2026-08-04] base text + shared actor-status addendum (amban/garrison present).
	desc = {
		desc = "qing_integ.11.desc"
		first_valid = {
			triggered_desc = { trigger = { has_variable = integ_amban_present  has_variable = integ_garrison_size  var:integ_garrison_size >= 1 }  desc = "qing_integ.actors_both" }
			triggered_desc = { trigger = { has_variable = integ_amban_present }  desc = "qing_integ.actors_amban" }
			triggered_desc = { trigger = { has_variable = integ_garrison_size  var:integ_garrison_size >= 1 }  desc = "qing_integ.actors_garrison" }
			triggered_desc = { trigger = { always = yes }  desc = "qing_integ.actors_none" }
		}
	}
	picture = dagger_behind_back

	left_portrait = root.current_ruler
	right_portrait = scope:corrupt_official


	# [playtest loc-fix] Self-abort if the subject vanished during the delay — the
	# LOG line and (via .a) the desc reference scope:target. corrupt_official is
	# re-derived with a fallback below, but target cannot be. Mirrors qing_integ.30.
	trigger = {
		exists = scope:target
		scope:target = { is_subject_of = ROOT }
	}

	immediate = {
		LOG_line = { sys = SUBJ  msg = "EVENT qing_integ.11 (scandal) overlord=[ROOT.GetTag] subject=[scope:target.GetTag]" }
		SUBJ_QING_resolve_integ_actors = yes  # [#task-20] resolve amban/garrison for the desc addendum
		# Find a disloyal official at court to pin the scandal on; fall back
		# to any adult courtier if none are outright disloyal.
		if = {
			limit = {
				any_character = {
					employer = ROOT
					is_ruler = no
					is_adult = yes
					prisoner = no
					loyalty < 50
				}
			}
			random_character = {
				limit = {
					employer = ROOT
					is_ruler = no
					is_adult = yes
					prisoner = no
					loyalty < 50
				}
				weight = {
					modifier = { factor = 2  prominence >= 50 }
					modifier = { factor = 2  has_any_office = yes }
				}
				save_scope_as = corrupt_official
			}
		}
		else_if = {
			limit = {
				any_character = { employer = ROOT  is_ruler = no  is_adult = yes  prisoner = no }
			}
			random_character = {
				limit = { employer = ROOT  is_ruler = no  is_adult = yes  prisoner = no }
				save_scope_as = corrupt_official
			}
		}
		else = {
			# Last-resort fallback so scope:corrupt_official is ALWAYS set — the
			# desc text and right_portrait reference it, and an unset scope would
			# render a broken portrait and an unresolved name. The ruler stands in.
			current_ruler = { save_scope_as = corrupt_official }
			LOG_line = { sys = SUBJ  msg = "qing_integ.11 no courtier available; ruler stands in as corrupt_official" }
		}
	}

	# --- Impeach and confiscate (the Jiaqing solution) -------------------
	option = {
		name = "qing_integ.11.a"
		custom_tooltip = "qing_integ.11.a.tt"
		trigger = { exists = scope:corrupt_official }
		# Seize the grandee's fortune — a Heshen-scale windfall to the treasury.
		add_treasury = 150
		add_stability = 3
		add_tyranny = 5
		scope:corrupt_official = {
			add_corruption = 20
			add_loyalty = inspired_disloyalty_other_l
		}
		SUBJ_QING_clear_integ_actors = yes  # [#task-20 hygiene] clear the resolver scratch vars
	}

	# --- Quietly settle it, losing money and momentum --------------------
	option = {
		name = "qing_integ.11.b"
		custom_tooltip = "qing_integ.11.b.tt"
		add_treasury = -80
		SUBJ_QING_reduce_integration = { steps = 1 }
		add_country_modifier = {
			name = qing_integ_corruption_scandal
			duration = 1095
		}
		SUBJ_QING_clear_integ_actors = yes  # [#task-20 hygiene] clear the resolver scratch vars
	}

	# --- Institute yanglian yin (養廉銀), "nourish honesty" stipends ------
	option = {
		name = "qing_integ.11.c"
		custom_tooltip = "qing_integ.11.c.tt"
		trigger = { treasury >= 120 }
		add_treasury = -120
		current_ruler = { add_popularity = 5 }
		if = {
			limit = { exists = scope:corrupt_official }
			scope:corrupt_official = {
				add_character_modifier = {
					name = qing_integ_bribed_official
					duration = 1825
				}
			}
		}
		SUBJ_QING_clear_integ_actors = yes  # [#task-20 hygiene] clear the resolver scratch vars
	}
}

# .12  ETHNIC STRIFE
# Anchor: the risings that gaitu guiliu (改土归流) and Han in-migration set off —
# the Miao Rebellion (苗民起義, 1795-1806, Han settlers displacing natives),
# the Panthay Rebellion (雲南回變, 1856-73, Du Wenxiu's 杜文秀 Pingnan Guo), and
# the Dungan Revolt (陝甘回變, 1862-77, ~21 million dead). Settlers and natives
# come to blows as direct rule replaces the old chieftaincies.
########################################################################
qing_integ.12 = {
	type = country_event
	title = "qing_integ.12.t"
	# [#task-20 2026-08-04] base text + an actor-status addendum naming the local amban / garrison
	# when they exist (resolved in the immediate by SUBJ_QING_resolve_integ_actors). The addenda are
	# mutually exclusive on presence so exactly one trailing paragraph shows. triggered_desc is the
	# proven conditional-desc idiom (qing_harem.1 / qing_princes.1).
	desc = {
		desc = "qing_integ.12.desc"
		first_valid = {
			triggered_desc = { trigger = { has_variable = integ_amban_present  has_variable = integ_garrison_size  var:integ_garrison_size >= 1 }  desc = "qing_integ.12.desc_actors_both" }
			triggered_desc = { trigger = { has_variable = integ_amban_present }  desc = "qing_integ.12.desc_actors_amban" }
			triggered_desc = { trigger = { has_variable = integ_garrison_size  var:integ_garrison_size >= 1 }  desc = "qing_integ.12.desc_actors_garrison" }
			triggered_desc = { trigger = { always = yes }  desc = "qing_integ.12.desc_actors_none" }
		}
	}
	picture = looting

	left_portrait = root.current_ruler

	goto_location = scope:riot_province


	# [playtest loc-fix] Self-abort if the subject vanished during the delay —
	# scope:target drives the desc + riot_province anchor. Mirrors qing_integ.30.
	trigger = {
		exists = scope:target
		scope:target = { is_subject_of = ROOT }
	}

	immediate = {
		LOG_line = { sys = SUBJ  msg = "EVENT qing_integ.12 (ethnic strife) overlord=[ROOT.GetTag] subject=[scope:target.GetTag]" }
		if = {
			limit = { scope:target = { any_owned_province = { count >= 1 } } }
			scope:target = {
				ordered_owned_province = {
					order_by = total_population
					max = 1
					save_scope_as = riot_province
				}
			}
		}
		# [#task-8] resolve the resident amban + garrison so the options below can be gated by them.
		SUBJ_QING_resolve_integ_actors = yes
	}

	# --- BRING IN the Green Standard Army to restore order ---------------
	# [#task-27 R1 2026-08-04] The throne BRINGS IN an outside column to pacify the strife (a bloodier
	# imposition, +5 AE). The case where a LOCAL garrison is already on the spot is promoted to its own
	# cheaper option (.12.e). Always available (the fallback when no local garrison can act).
	option = {
		name = "qing_integ.12.a"
		custom_tooltip = "qing_integ.12.a.tt"
		add_stability = -3
		add_aggressive_expansion = 5
		LOG_line = { sys = SUBJ  msg = "strife: outside troops brought in to restore order, subject=[scope:target.GetTag]" }
		# Applied to the overlord: subject_loyalty tightens the bond, and integrate_speed
		# now feeds the ENGINE's subject-integration timer (post-#subj-integ-rework: once the
		# 改土歸流 capstone hands a governorship to the engine's native integration, integrate_speed
		# genuinely accelerates the gradual absorption — it was inert on the old flat-+1 meter).
		add_country_modifier = {
			name = qing_integ_ethnic_strife
			duration = 1095
		}
		if = {
			limit = { exists = scope:riot_province }
			scope:riot_province = {
				add_province_modifier = {
					name = local_unrest_harsh
					duration = 730
				}
			}
		}
		SUBJ_QING_clear_integ_actors = yes
	}

	# --- [#task-27 R1 2026-08-04] The LOCAL GARRISON separates the communities ----
	# Shown only when an imperial garrison strong enough to matter (edge >= 1) already stands in the
	# dependency: the banner troops on the spot interpose between settlers and natives at once, with far
	# less bloodshed (less AE) than marching in an outside column (.12.a). A dominant garrison (edge 2)
	# does it with a minimum of force (+2 AE); rough parity (edge 1) is rougher (+3). The "local garrison
	# does it, cheaper" counterpart to .12.a.
	option = {
		name = "qing_integ.12.e"
		custom_tooltip = "qing_integ.12.e.tt"
		trigger = {
			has_variable = integ_garrison_edge
			var:integ_garrison_edge >= 1
		}
		add_stability = -1
		if = {
			limit = { has_variable = integ_garrison_edge  var:integ_garrison_edge = 2 }
			add_aggressive_expansion = 2
			LOG_line = { sys = SUBJ  msg = "strife: a dominant local garrison separates the communities with minimal force, subject=[scope:target.GetTag]" }
		}
		else = {
			add_aggressive_expansion = 3
			LOG_line = { sys = SUBJ  msg = "strife: the local garrison separates the communities, subject=[scope:target.GetTag]" }
		}
		SUBJ_QING_war_garrison_outcome = { delta = 2 }
		add_country_modifier = {
			name = qing_integ_ethnic_strife
			duration = 730
		}
		if = {
			limit = { exists = scope:riot_province }
			scope:riot_province = { add_province_modifier = { name = local_unrest_harsh  duration = 365 } }
		}
		SUBJ_QING_clear_integ_actors = yes
	}

	# --- Segregate settlers and natives; slow the process to keep peace --
	# (After the Miao rising the Qing built a segregating wall and forts.)
	option = {
		name = "qing_integ.12.b"
		custom_tooltip = "qing_integ.12.b.tt"
		SUBJ_QING_reduce_integration = { steps = 1 }
		add_stability = 2
		# Applied to the overlord: subject_loyalty tightens the bond, and integrate_speed
		# now feeds the ENGINE's subject-integration timer (post-#subj-integ-rework: once the
		# 改土歸流 capstone hands a governorship to the engine's native integration, integrate_speed
		# genuinely accelerates the gradual absorption — it was inert on the old flat-+1 meter).
		add_country_modifier = {
			name = qing_integ_imperial_favor
			duration = 730
		}
		SUBJ_QING_clear_integ_actors = yes
	}

	# --- Fund resettlement and irrigation to defuse the land dispute -----
	option = {
		name = "qing_integ.12.c"
		custom_tooltip = "qing_integ.12.c.tt"
		trigger = { treasury >= 100 }
		add_treasury = -100
		if = {
			limit = { exists = scope:riot_province }
			scope:riot_province = {
				add_province_modifier = {
					name = positive_food_modifier
					duration = 1095
				}
			}
		}
		SUBJ_QING_clear_integ_actors = yes
	}

	# --- [#task-8; #task-27 req-9 light skill-check 2026-08-04] The resident amban mediates ----------
	# Shown only when a steady-or-better resident amban (band >= 1) sits in the dependency: he OFFERS to
	# broker a local settlement between settlers and natives. Whether he SUCCEEDS is a SKILL CHECK on his
	# diplomatic skills (charisma + finesse, qing_amban_negotiate_chance_svalue):
	#   • SUCCESS → the least bloody path: a little stability, the province eased, his standing and the
	#               Lifan Yuan credited.
	#   • FAILURE → the mediation collapses; a small stability ding, he is DISCREDITED (prominence hit)
	#               and the Lifan Yuan marked down — but per "execute only in .41", a failed MEDIATION
	#               here does NOT kill him.
	# Keeps the process on track either way. Absent / wavering amban => hidden.
	option = {
		name = "qing_integ.12.d"
		custom_tooltip = "qing_integ.12.d.tt"
		trigger = {
			has_variable = integ_amban_present
			var:integ_amban_band >= 1
			exists = scope:integ_amban
		}
		scope:integ_amban = {
			random = {
				chance = qing_amban_negotiate_chance_svalue
				set_variable = { name = integ_amban_negotiated_ok  value = 1 }
			}
		}
		if = {
			limit = { scope:integ_amban = { has_variable = integ_amban_negotiated_ok } }
			add_stability = 1
			scope:integ_amban = { add_prominence = 5  remove_variable = integ_amban_negotiated_ok }
			if = {
				limit = { exists = scope:riot_province }
				scope:riot_province = { add_province_modifier = { name = positive_food_modifier  duration = 365 } }
			}
			SUBJ_QING_lifan_amban_outcome = { delta = 4 }
			LOG_line = { sys = SUBJ  msg = "strife: resident amban brokers a local settlement (skill check passed), subject=[scope:target.GetTag]" }
		}
		else = {
			add_stability = -1
			scope:integ_amban = { add_prominence = -5 }
			SUBJ_QING_lifan_amban_outcome = { delta = -3 }
			LOG_line = { sys = SUBJ  msg = "strife: the amban's mediation collapses and he is discredited, subject=[scope:target.GetTag]" }
		}
		SUBJ_QING_clear_integ_actors = yes
	}
