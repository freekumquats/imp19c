# ======================================================================
# qing_revenue.9 — THE RECKONING
# Resolves the chain. qing_saltreform_margin = authority accumulated across .1/.7/.10/.8/.11/
# .12/.13 (7 stages, extended 2026-08-17 per direct user instruction -- the chain was too
# short) minus the Commissioner's OWN fitness (QING_salt_assess_fitness, se_QING_SALT.txt --
# his competence at defending the status quo, not a new meter). [correction, direct user
# instruction] success now CONCLUDES the reform directly by changing qing_salt_admin_law to
# its qing_salt_reformed option (common/laws/00_qing_statutes_laws.txt) -- change_law is the
# proven effect for a scripted law change (oracle-attested, TI/Invictus 00_india_effects.txt
# etc; no player decision/UI action involved). qing_salt_gabelle_reformed is ALSO set
# directly here (redundant-but-safe: change_law's own on_enact should set it too, but this
# doesn't rely on that assumption). The short-term treasury cost and long-term production/
# revenue payoff both land HERE, at the moment the law actually changes -- not at the
# earlier decision that merely opened the struggle. Failure closes the door for a decade.
# Scope: country (CHI).
# ======================================================================
qing_revenue.9 = {
	type = country_event
	title = "qing_revenue.9.t"
	desc = {
		triggered_desc = {
			trigger = { var:qing_saltreform_margin > 0 }
			desc = "qing_revenue.9.desc_success"
		}
		triggered_desc = { trigger = { always = yes }  desc = "qing_revenue.9.desc_failure" }
	}
	picture = chinese_throne_room
	left_portrait = scope:revenue_minister
	right_portrait = scope:salt_commissioner

	trigger = {
		tag = CHI
		has_variable = qing_saltreform_authority
		has_variable = qing_office_revenue_holder
		var:qing_office_revenue_holder = { is_alive = yes  employer = ROOT }
		has_variable = qing_salt_commissioner_holder
		var:qing_salt_commissioner_holder = { is_alive = yes  employer = ROOT }
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_revenue.9 (the reckoning)" }
		var:qing_office_revenue_holder = { save_scope_as = revenue_minister }
		var:qing_salt_commissioner_holder = { save_scope_as = salt_commissioner }
		QING_salt_assess_fitness = yes
		set_variable = { name = qing_saltreform_margin  value = var:qing_saltreform_authority }
		change_variable = { name = qing_saltreform_margin  subtract = var:qing_salt_fitness }
	}

	# OPTION A: success -- the minister's authority prevails, the law changes now
	option = {
		name = "qing_revenue.9.a"
		custom_tooltip = "qing_revenue.9.a.tt"
		trigger = { var:qing_saltreform_margin > 0 }
		set_variable = { name = qing_salt_reform_chain_complete  value = 1 }
		remove_variable = qing_saltreform_authority
		remove_variable = qing_saltreform_timeout
		change_law = qing_salt_reformed
		set_variable = { name = qing_salt_gabelle_reformed  value = 1 }
		# short-term cost: the disruption of formally reforming the monopoly
		add_treasury = -120
		TREASURY_LOG_it = { amount = -120 }
		# long-term payoff: the same permanent tax-efficiency modifier the old one-shot event
		# already granted (proven, unchanged) ...
		add_country_modifier = { name = qing_salt_gabelle_reformed_mod  duration = -1 }
		# ... plus a genuine, durable salt-production/revenue cleanup: qing_salt_squeeze is the
		# ONLY term in the standing quarterly 鹽課 income (QING_salt_income, se_QING_SALT.txt)
		# that represents how much of the monopoly's real yield actually reaches the treasury
		# vs. the commissioner's own graft. Reforming sets an immediate, durable floor instead
		# of waiting on the slow quarterly QING_salt_reconcile drift.
		if = {
			limit = { has_variable = qing_salt_squeeze  var:qing_salt_squeeze > 10 }
			set_variable = { name = qing_salt_squeeze  value = 10 }
		}
		scope:revenue_minister = { QING_char_promote_standing = { pop = 10  prom = 5 } }
		LOG_line = { sys = QING  msg = "revenue: the minister's authority prevails -- the salt gabelle is formally reformed" }
	}

	# OPTION B: failure -- entrenched interests hold, closed for a decade
	option = {
		name = "qing_revenue.9.b"
		custom_tooltip = "qing_revenue.9.b.tt"
		trigger = { var:qing_saltreform_margin <= 0 }
		set_variable = { name = qing_salt_gabelle_reform_failed  days = 3650 }
		remove_variable = qing_saltreform_authority
		remove_variable = qing_saltreform_timeout
		LOG_line = { sys = QING  msg = "revenue: the commissioner's entrenchment holds -- the reform effort collapses" }
	}
}

# ======================================================================
# qing_revenue.2 — STOCK THE EVER-NORMAL GRANARIES (常平倉)
# A good harvest lets the Revenue minister stock the granaries (price-stabilising
# grain reserves). Stock them (unrest control via concrete granary building) or
# divert the surplus to other needs (revenue, but no unrest buff).
# ======================================================================
qing_revenue.2 = {
	type = country_event
	title = "qing_revenue.2.t"
	desc = "qing_revenue.2.desc"
	picture = senate
	left_portrait = root.current_ruler
	right_portrait = scope:revenue_minister

	trigger = {
		tag = CHI
		has_variable = qing_office_revenue_holder
		var:qing_office_revenue_holder = { is_alive = yes  employer = ROOT }
		# can only stock if we have suitable provinces
		any_owned_province = {
			OR = {
				province_id = 7229
				province_id = 7230
				province_id = 7234
				province_id = 1
				province_id = 7235
			}
			has_city_status = yes
		}
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_revenue.2 (granary stocking)" }
		var:qing_office_revenue_holder = { save_scope_as = revenue_minister }
		set_variable = { name = qing_revenue_task  value = flag:fiscal }
		QING_revenue_assess_fitness = { task = fiscal }
	}

	# OPTION A: stock the granaries — unrest control
	option = {
		name = "qing_revenue.2.a"
		custom_tooltip = "qing_revenue.2.a.tt"
		QING_revenue_build_granary = yes   # [#18] BUILD a granary (coverage); the market-price grain BUY is now the panel button only
		add_country_modifier = { name = qing_granaries_stocked  duration = 1825 }
		LOG_line = { sys = QING  msg = "revenue: granaries stocked (building +1, unrest buff)" }
	}

	# OPTION B: divert the surplus to the treasury
	option = {
		name = "qing_revenue.2.b"
		custom_tooltip = "qing_revenue.2.b.tt"
		add_treasury = 50
		TREASURY_LOG_it = { amount = 50 }
		# but no unrest buff, and the minister's reputation suffers (short-term thinking)
		scope:revenue_minister = {
			QING_char_promote_standing = { pop = -5  prom = 0 }
		}
		LOG_line = { sys = QING  msg = "revenue: granary surplus diverted to treasury" }
	}

	# OPTION C (alternate scenario): famine tests the reserves
	# This option only appears if we already have granaries to deplete.
	option = {
		name = "qing_revenue.2.c"
		custom_tooltip = "qing_revenue.2.c.tt"
		trigger = {
			any_owned_province = {
				OR = {
					province_id = 7229
					province_id = 7230
					province_id = 7234
					province_id = 1
					province_id = 7235
				}
				num_of_qing_granary_building >= 1
			}
		}
		# deplete a granary (famine consumes the reserves)
		QING_revenue_deplete_granary = yes
		# but the reserves cushioned the blow (temp unrest reduction)
		add_country_modifier = { name = qing_granaries_cushioned_famine  duration = 730 }
		LOG_line = { sys = QING  msg = "revenue: famine tested granaries (building -1, but unrest cushioned)" }
	}

# ======================================================================
# qing_revenue.3 — THE SILVER CRISIS RESPONSE (銀荒/銀貴錢賤)
# The silver crisis (opium drain + copper-cash inflation) is THE fiscal emergency
# of the 19thC. Can the Revenue minister act? A capable, loyal minister can nudge
# currency stress down (policy response); a venal or incapable one botches it
# (stress worsens).
# ======================================================================
qing_revenue.3 = {
	type = country_event
	title = "qing_revenue.3.t"
	desc = "qing_revenue.3.desc"
	picture = senate_debate
	left_portrait = root.current_ruler
	right_portrait = scope:revenue_minister

	trigger = {
		tag = CHI
		has_variable = qing_office_revenue_holder
		var:qing_office_revenue_holder = { is_alive = yes  employer = ROOT }
		has_variable = qing_currency_stress
		var:qing_currency_stress >= 30
		# [#86 2026-08-08] a Silver Crisis (銀荒) requires the 戶部銀庫 to be GENUINELY DRAWN DOWN — not just
		# an elevated abstract stress meter. The event was firing at a record-high reserve because
		# qing_currency_stress can reach 30 on opium-drain flow alone while the vaults are still near their
		# peak, which is not a crisis by any reasonable definition (the reported bug). Also require the real
		# engine backing reserve to have fallen below 60% of its historical peak. Guarded on the var existing
		# (seeded for CHI at boot); RHS is the dedicated cmpsvalue, not a var-ref (RHS-comparison rule).
		has_variable = silver_reserve_size
		var:silver_reserve_size < qing_silver_reserve_crisis_floor_cmpsvalue
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_revenue.3 (silver crisis response)" }
		var:qing_office_revenue_holder = { save_scope_as = revenue_minister }
		set_variable = { name = qing_revenue_task  value = flag:crisis }
		QING_revenue_assess_fitness = { task = crisis }
	}

	# OPTION A: decisive policy response (minister capable, stress down)
	option = {
		name = "qing_revenue.3.a"
		custom_tooltip = "qing_revenue.3.a.tt"
		trigger = {
			has_variable = qing_revenue_fitness
			var:qing_revenue_fitness >= 60
		}
		# effective intervention: currency stress down
		QING_DECLINE_nudge_signed = { var = qing_currency_stress_residual  amount = -10 }
		add_country_modifier = { name = qing_silver_crisis_managed  duration = 1825 }
		# [#86 2026-08-08] ACTUALLY MOVE THE MONETARY SYSTEM. Previously this "decisive response" only
		# nudged the abstract stress meter + docked the treasury; it never touched the real silver reserve
		# the crisis is ABOUT. A decisive policy (tighter Canton controls, bullion purchase, a crackdown on
		# the opium trade that is draining the vaults) REPLENISHES the engine backing reserve directly:
		# +2000 千兩 (~2.4% of peak — meaningful against the drift's ±100-250/pulse, not game-breaking).
		# Guarded on the var existing so a no-reserve edge never errors. This is the concrete face of
		# "arrest the silver drain" the option name + tooltip promise.
		if = {
			limit = { has_variable = silver_reserve_size }
			change_variable = { name = silver_reserve_size  add = 2000 }
		}
		# it costs the exchequer to mount (bullion purchase + enforcement)
		add_treasury = -310
		TREASURY_LOG_it = { amount = -310 }
		scope:revenue_minister = {
			add_loyalty = loyalty_qing_congenial
			QING_char_promote_standing = { pop = -8  prom = 8 }
		}
		LOG_line = { sys = QING  msg = "revenue: silver crisis managed (+silver reserve, -currency stress, -treasury, minister's standing +/-)" }
	}

	# OPTION B: half-measures (minister mediocre, modest effect)
	option = {
		name = "qing_revenue.3.b"
		custom_tooltip = "qing_revenue.3.b.tt"
		trigger = {
			has_variable = qing_revenue_fitness
			var:qing_revenue_fitness >= 40
			var:qing_revenue_fitness < 60
		}
		# partial intervention: small currency stress reduction
		QING_DECLINE_nudge_signed = { var = qing_currency_stress_residual  amount = -3 }
		# [#86 2026-08-08] half-measures also touch the real reserve, but only modestly (+600 千兩) —
		# enough to slow the bleed, not reverse it. Guarded on the var existing.
		if = {
			limit = { has_variable = silver_reserve_size }
			change_variable = { name = silver_reserve_size  add = 600 }
		}
		add_treasury = -220
		TREASURY_LOG_it = { amount = -220 }
		LOG_line = { sys = QING  msg = "revenue: silver crisis half-measures (+silver reserve modest, -3 currency stress, -30 treasury)" }
	}

	# OPTION C: botched response (minister incapable or corrupt, stress worsens)
	option = {
		name = "qing_revenue.3.c"
		custom_tooltip = "qing_revenue.3.c.tt"
		trigger = {
			has_variable = qing_revenue_fitness
			var:qing_revenue_fitness < 40
		}
		# failed intervention: currency stress INCREASES
		QING_DECLINE_nudge_signed = { var = qing_currency_stress_residual  amount = 5 }
		add_country_modifier = { name = qing_silver_crisis_botched  duration = 1825 }
		# minister's reputation collapses
		scope:revenue_minister = {
			add_loyalty = loyalty_qing_estranged
			QING_char_promote_standing = { pop = -15  prom = -5 }
		}
		LOG_line = { sys = QING  msg = "revenue: silver crisis botched (+currency stress, minister discredited)" }
	}

	# OPTION D: do nothing (stress continues to rise)
	option = {
		name = "qing_revenue.3.d"
		custom_tooltip = "qing_revenue.3.d.tt"
		QING_DECLINE_nudge_signed = { var = qing_currency_stress_residual  amount = 3 }
		LOG_line = { sys = QING  msg = "revenue: silver crisis unaddressed (+currency stress)" }
	}
}

# ======================================================================

	}

# qing_revenue.5 — THE TREASURY AT ITS PEAK (盈庫之盛)  [#372]
# Fired ONCE from QING_revenue_reserve_drift when the 戶部銀庫 (Board of Revenue silver
# treasury) first reaches its historic all-time high (~8182萬兩, 乾隆42/1777). A moment of
# the High Qing zenith: the fullest coffers the dynasty ever knew. The throne may lock the
# reserve away as a war-chest (stability + legitimacy, the prudent Yongzheng model), or spend

	# [rework 2026-08-25] the historic all-time peak of the ENTIRE dynasty's treasury

	# scaled up from the original 2/5/5 while keeping the same proven effect types.

		# ministry pulse. Fund a real building spree spanning BOTH the Public Works and
		# Infrastructure categories (gui/macro_builder_view.gui's PublicWorksItems/
		# InfrastructureItems groupings), not granaries alone. Each call mirrors
		# QING_revenue_build_granary's own idiom (ordered_owned_province, NOT-has_building
		# guard, max=1) so repeated calls land on distinct provinces, never stacking levels
		# on one. Infrastructure picks (INF_canal/INF_depot) are tech-gated exactly like
		# their own `allow` block, so on a country that hasn't researched that tech yet the

		QING_revenue_build_granary = yes
		QING_revenue_build_granary = yes
		QING_revenue_grand_works_build_dike = yes
		QING_revenue_grand_works_build_dike = yes
		QING_revenue_grand_works_build_canal_depot = yes
		QING_revenue_grand_works_build_canal_depot = yes

		QING_revenue_grand_works_build_INF_canal = yes
		QING_revenue_grand_works_build_INF_depot = yes
	}
