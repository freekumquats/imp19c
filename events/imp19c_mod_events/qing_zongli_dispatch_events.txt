qing_zongli_dispatch.1 = {
	type = country_event
	title = "qing_zongli_dispatch.1.t"
	desc = "qing_zongli_dispatch.1.desc"
	picture = council_country

	left_portrait = scope:qing_dispatch_evt_diplomat

	# re-check in the event's own trigger (the roll's gate can go stale over the days delay).
	trigger = {
		tag = CHI
		exists = scope:qing_dispatch_evt_diplomat
		exists = scope:qing_dispatch_evt_play
		scope:qing_dispatch_evt_diplomat = { is_alive = yes  has_variable = qing_zongli_dispatched_marker }
		scope:qing_dispatch_evt_play = { has_variable = is_diplomatic_play }
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_zongli_dispatch.1 (dispatched diplomat's report)" }
	}

	option = {
		name = "qing_zongli_dispatch.1.a"
		# a single acknowledgement option -- the diplomat's own skill decides the outcome below,
		# not the player's choice (matches the user's "based on the diplomat's skills" framing).
		random_list = {
			50 = {
				# the base roll: an even chance, skewed by the charisma modifiers below.
				custom_tooltip = "qing_zongli_dispatch.1.a.good.tt"
				modifier = {
					factor = 2
					scope:qing_dispatch_evt_diplomat = { charisma >= 8 }
				}
				modifier = {
					factor = 0.5
					scope:qing_dispatch_evt_diplomat = { charisma <= 3 }
				}
				scope:qing_dispatch_evt_play = { DIPLOMACY_modify_play_success = { amt = 8 } }
				LOG_line = { sys = QING  msg = "zongli dispatch: diplomat's gambit pays off, success raised" }
			}
			50 = {
				custom_tooltip = "qing_zongli_dispatch.1.a.bad.tt"
				modifier = {
					factor = 2
					scope:qing_dispatch_evt_diplomat = { charisma <= 3 }
				}
				modifier = {
					factor = 0.5
					scope:qing_dispatch_evt_diplomat = { charisma >= 8 }
				}
				scope:qing_dispatch_evt_play = { DIPLOMACY_modify_play_success = { amt = -8 } }
				LOG_line = { sys = QING  msg = "zongli dispatch: diplomat stumbles, success lowered" }
			}
		}
	}
}
