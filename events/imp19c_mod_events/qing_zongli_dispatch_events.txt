	title = "qing_zongli_dispatch.1.t"
	desc = {
		triggered_desc = { trigger = { var:qing_zongli_dispatch_evt_outcome = flag:success }  desc = "qing_zongli_dispatch.1.desc_success" }
		triggered_desc = { trigger = { var:qing_zongli_dispatch_evt_outcome = flag:failure }  desc = "qing_zongli_dispatch.1.desc_failure" }
	}
	picture = council_country

	left_portrait = scope:qing_dispatch_evt_diplomat

		has_variable = qing_zongli_dispatch_evt_outcome

		name = "qing_zongli_dispatch.1.a"
		custom_tooltip = {
			text = qing_zongli_dispatch.1.a.tt_success
			var:qing_zongli_dispatch_evt_outcome = flag:success
		}
		custom_tooltip = {
			text = qing_zongli_dispatch.1.a.tt_failure
			var:qing_zongli_dispatch_evt_outcome = flag:failure
		}
		remove_variable = qing_zongli_dispatch_evt_outcome
	}
}
