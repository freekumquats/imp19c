########################################################################
# qing_caravan_events.txt — [#370] Silk Road / Central Asian Caravan Trade (絲路商道 / 回疆貿易).
#
# The player-facing surface of the caravan trade-node model (se_QING_CARAVAN.txt): the Kokand
# ultimatum that dramatises the escalating khanate demands (the road to the 1832 settlement) and
# the route-cut crisis when a khoja scare severs the caravan road. CHI-only, player-focused,
# logged sys = QING.
#
#   .1  THE KOKAND ULTIMATUM (浩罕之求) — offered by QING_caravan_pulse when the trade is worth
#       extorting (prosperity >= 55), the aqsaqal is NOT yet granted, and Kokand is already ill-
#       disposed (a festering grievance). GRANT (許阿奇木, the 1832 settlement) vs REFUSE & GARRISON
#       (拒而設防, hold sovereignty, arm the khoja scare) vs TEMPORISE (羈縻拖延, buy time, re-offers).
#   .2  THE ROUTE IS CUT (商道斷絕) — offered when a khoja scare is pending and the route was worth
#       something. MILITARY ESCORT (護商, spend treasury to force it open) vs NEGOTIATE VIA KOKAND
#       (許阿奇木, grant the concession to reopen) vs LET IT LAPSE (聽其斷絕, the trade collapses).
#
# These do NOT own the prosperity meter — they act on it through the se_QING_CARAVAN.txt levers,
# which drive the SAME qing_caravan_prosperity counter the pulse reads, so the panel, the pulse, and
# the events all interlock. They also couple to the #367 Xinjiang state (qing_xinjiang_control grip,
# qing_xj_khoja_pending scare, the KOK opinion) that the caravan model shares.
#
# The once-only offered flags (qing_caravan_ultimatum_seen / _routecut_seen) are set in each event's
# OWN immediate (only when the event actually fires), NOT in the pulse — the #366/#368 flag-leak fix.
#
# HISTORY (Millward, Beyond the Pass / Eurasian Crossroads; Newby, The Empire and the Khanate;
# Fletcher, Cambridge History of China vol. 10): after 1759 the Khanate of Kokand pressed the Qing
# for the aqsaqal (阿奇木伯克) consul at Kashgar, the right to levy its OWN customs on third-party
# Andijani traders, and extraterritoriality — leverage it exercised by sheltering the White-Mountain
# khoja pretenders (Jahangir 1826; Yusuf 1830; the Seven Khojas 1847). The 1832 settlement conceded
# most of these demands, and is often called the first "unequal treaty" the Qing granted, a decade
# before Nanking. The alternative — the costly military reopening of the road (護商 / 進剿) — is the
# one Qianlong and the 善後 administrators had used, covered by the 協餉 subsidy.
########################################################################

namespace = qing_caravan

########################################################################
# .1  THE KOKAND ULTIMATUM (浩罕之求)
# Offered by QING_caravan_pulse when the Altishahr trade is prosperous, the aqsaqal concession is not
# yet granted, and Kokand is already emboldened. The khanate presses its demands.
########################################################################
qing_caravan.1 = {
	type = country_event
	title = "qing_caravan.1.t"
	desc = "qing_caravan.1.desc"
	# [boot-fix 2026-08-06] was picture = trade_port — a Mediterranean galley + toga figure, jarringly wrong
	# for an 1820s Central Asian Kokand envoy. steppe_cavalry (Central Asian horse) is the fitting register
	# (proven picture, rus_expansion_events.txt:46).
	picture = steppe_cavalry

	left_portrait = root.current_ruler


	trigger = {
		tag = CHI
		# [boot-fix 2026-08-06] DATE GATE: the Kokand aqsaqal pressure / khoja invasions are an 1820s-1832
		# phenomenon, NOT the High-Qing zenith. It fired 17 Mar 1763 because the 1763 caravan prosperity
		# opens >= 55 (the S=0.5 oasis-trade scale is uncalibrated/placeholder) and the event's own trigger
		# only gated on prosperity + not-granted. Gate on current_date so it cannot fire at the zenith
		# regardless of the prosperity opening (era-gate precedent: se_QING_OPIUM.txt:98-101 date ramp).
		current_date >= 1820.1.1
		has_variable = qing_caravan_prosperity
		var:qing_caravan_prosperity >= 55
		NOT = { has_variable = qing_caravan_aqsaqal_granted }
		# [#13 2026-08-07 LINK] a beaten or caravan.3-settled Kokand no longer issues ultimatums (belt-and-braces
		# with the pulse offer guard; design §B.2 both-guards form).
		NOT = { QING_kok_conquered_trigger = yes }
		NOT = { QING_kok_yielded_flag = yes }
	}

	immediate = {
		# [#370] set the once-only offered flag HERE (only when the event actually fires), not in the
		# pulse — so a re-check failure during the 5-20 day delay never strands it set (#366/#368 fix).
		set_variable = { name = qing_caravan_ultimatum_seen  value = 1 }
		LOG_line = { sys = QING  msg = "EVENT qing_caravan.1 (Kokand ultimatum) fired for CHI" }
	}

	# --- GRANT THE CONCESSION (許阿奇木) — the 1832 settlement: recognise the consul, mend relations,
	# defuse the pretenders, but swallow the humiliation and the revenue haircut. -----
	option = {
		name = "qing_caravan.1.grant"
		custom_tooltip = "qing_caravan.1.grant.tt"
		QING_caravan_grant_aqsaqal = yes
		# [#370-R2] clear the once-only flag on resolution — the aqsaqal_granted gate suppresses
		# re-offer while the concession stands, but a later QING_caravan_revoke_aqsaqal re-provokes
		# Kokand; if this flag stayed set forever the ultimatum could never re-fire post-revoke.
		if = {
			limit = { has_variable = qing_caravan_ultimatum_seen }
			remove_variable = qing_caravan_ultimatum_seen
		}
	}

	# --- REFUSE & GARRISON (拒而設防) — hold sovereignty, brace the frontier; but Kokand arms the
	# khoja pretenders and the road frays. The costly proud answer. -----
	option = {
		name = "qing_caravan.1.refuse"
		custom_tooltip = "qing_caravan.1.refuse.tt"
		# the affronted khanate looses its pretenders — set the #367 khoja scare and dent the grip.
		# [meter-concretize #10B] −4 control nudge DROPPED — the frayed road is stamped CONTESTED on an oasis;
		# control DERIVES down. Gated on not-fully-integrated (review H1: no post-integration khoja/rot).
		if = {
			limit = { NOT = { has_variable = qing_xj_fully_integrated } }
			set_variable = { name = qing_xj_khoja_pending  value = 1 }
			QING_ili_apply_prov_band = { INTENT = contested }
			QING_xj_derive_control = yes
		}
		current_ruler = { add_popularity = 10 }  # [T3 BT-5/6] was add_prestige (inert on a character)
		# [#370-R2] DELIBERATELY leave qing_caravan_ultimatum_seen SET here. REFUSE does not drop
		# prosperity, so clearing it would re-fire the .1 if-branch next quarter and PREEMPT the .2
		# route-cut (else_if) that this refusal is meant to provoke via the khoja scare just set.
		# Renewed Kokand hostility after a refusal expresses as the route-cut crisis, not a repeat
		# ultimatum — so the flag must stay set to let .2 fire. (The GRANT path clears it because
		# aqsaqal_granted independently gates .1 while the concession stands.)
	}

	# --- TEMPORISE (羈縻拖延) — the old loose-rein evasion: neither grant nor refuse. Buys time at a
	# little prestige, and the demand returns. -----
	option = {
		name = "qing_caravan.1.temporise"
		custom_tooltip = "qing_caravan.1.temporise.tt"
		current_ruler = { add_popularity = -5 }  # [T3 BT-5/6] was add_prestige (inert on a character)
		# the demand festers — clear the seen flag so the pulse re-offers while the pressure stands.
		if = {
			limit = { has_variable = qing_caravan_ultimatum_seen }
			remove_variable = qing_caravan_ultimatum_seen
		}
	}
}

########################################################################
# .2  THE ROUTE IS CUT (商道斷絕)
# Offered by QING_caravan_pulse when a khoja scare is pending (#367) and the caravan trade was worth
# something — the pretenders' raids sever the Kashgar–Yarkand road and the trade collapses.
########################################################################
qing_caravan.2 = {
	type = country_event
	title = "qing_caravan.2.t"
	desc = "qing_caravan.2.desc"
	picture = revolt

	left_portrait = root.current_ruler


	trigger = {
		tag = CHI
		has_variable = qing_xj_khoja_pending
		has_variable = qing_caravan_prosperity
		var:qing_caravan_prosperity >= 40
		# [#13 2026-08-07 LINK] a beaten or caravan.3-settled Kokand cannot cut the road (belt-and-braces with the
		# pulse offer guard; design §B.2). The khoja scare itself is now suppressed upstream in this case, so this
		# is defensive — but keeps the event honest if fired by any other path.
		NOT = { QING_kok_conquered_trigger = yes }
		NOT = { QING_kok_yielded_flag = yes }
	}

	immediate = {
		# [#370] set the once-only offered flag HERE (only when the event fires).
		set_variable = { name = qing_caravan_routecut_seen  value = 1 }
		LOG_line = { sys = QING  msg = "EVENT qing_caravan.2 (route cut) fired for CHI" }
	}

	# --- MILITARY ESCORT (護商) — force the road open with a garrison column; costly, firms the grip. --
	option = {
		name = "qing_caravan.2.escort"
		custom_tooltip = "qing_caravan.2.escort.tt"
		trigger = { treasury >= 695  manpower >= 2 }
		QING_caravan_military_escort = yes
		# [#370-R] clear the once-only flag on RESOLUTION too, not only on lapse — the crisis is
		# over, and a FUTURE khoja scare (se_QING_XINJIANG re-raises qing_xj_khoja_pending) must be
		# able to re-fire .2. Leaving it set permanently suppressed the recurring crisis after the
		# first escort/negotiate, dragging prosperity -15 with no player choice (the #366/#368 class).
		if = {
			limit = { has_variable = qing_caravan_routecut_seen }
			remove_variable = qing_caravan_routecut_seen
		}
	}

	# --- NEGOTIATE VIA KOKAND (許阿奇木) — grant the concession to have Kokand call off its pretenders
	# and reopen the road. The commercial surrender. -----
	# [route-cut fix 2026-08-06] Shown ONLY when the aqsaqal is NOT already granted (mirrors the .1
	# ultimatum's own gate). Previously unconditional: if the concession already stood,
	# QING_caravan_grant_aqsaqal no-oped (guarded NOT aqsaqal_granted) so this was a DEAD option that
	# neither re-granted nor cleared the scare — the player was offered "grant a concession you already
	# hold" and it could not resolve the crisis. The already-granted case now gets .2.invoke below.
	option = {
		name = "qing_caravan.2.negotiate"
		trigger = { NOT = { has_variable = qing_caravan_aqsaqal_granted } }
		custom_tooltip = "qing_caravan.2.negotiate.tt"
		QING_caravan_grant_aqsaqal = yes
		# [#370-R2] clear the once-only flag on THIS resolution too (escort + lapse already do).
		# Granting the aqsaqal ends THIS crisis; a FUTURE khoja scare after a later revoke must be
		# able to re-fire .2. Leaving it set permanently suppressed the recurring crisis on the
		# grant-then-revoke path (the #366/#368 flag-leak class the escort option was fixed against).
		if = {
			limit = { has_variable = qing_caravan_routecut_seen }
			remove_variable = qing_caravan_routecut_seen
		}
	}

	# --- INVOKE THE STANDING CONCESSION (援例阿奇木) — shown ONLY when the aqsaqal is ALREADY in force.
	# [route-cut fix 2026-08-06] The player already granted the concession (via the .1 ultimatum or an
	# earlier .2); leaning on the 1832 settlement they already pay for, the throne has Kokand rein in its
	# pretenders and reopen the road — no fresh grant, no new legitimacy hit, just the scare cleared and a
	# modest revival. Resolves the crisis the concession-holding player would otherwise have no non-military,
	# non-collapse way out of. -----
	option = {
		name = "qing_caravan.2.invoke"
		trigger = { has_variable = qing_caravan_aqsaqal_granted }
		custom_tooltip = "qing_caravan.2.invoke.tt"
		QING_caravan_invoke_aqsaqal = yes
		if = {
			limit = { has_variable = qing_caravan_routecut_seen }
			remove_variable = qing_caravan_routecut_seen
		}
	}

	# --- LET IT LAPSE (聽其斷絕) — the trade collapses; prosperity crashes and the oases sour. ---
	option = {
		name = "qing_caravan.2.lapse"
		custom_tooltip = "qing_caravan.2.lapse.tt"
		QING_DECLINE_nudge = { var = qing_caravan_prosperity  amount = -20 }
		# [meter-concretize #10B] −3 control nudge DROPPED — the soured oases are stamped CONTESTED (the road's
		# insecurity is real on the map); control DERIVES down. Gated on not-fully-integrated (review H1/MED-4).
		if = { limit = { NOT = { has_variable = qing_xj_fully_integrated } }  QING_ili_apply_prov_band = { INTENT = contested }  QING_xj_derive_control = yes }
		# the crisis can recur — clear the seen flag so the pulse re-offers while the scare stands.
		if = {
			limit = { has_variable = qing_caravan_routecut_seen }
			remove_variable = qing_caravan_routecut_seen
		}
	}
}

########################################################################
# .3  THE KHANATE YIELDS (浩罕屈服)  — [#13 2026-08-07]
# Offered by QING_caravan_pulse ONCE Kokand is beaten by force (QING_kok_conquered_trigger: subjugated as a
# CHI subject, OR its heartland seat Kokand city p:110 held). The CONQUEST alternative to managing the
# aqsaqal/khoja cycle: the throne now dictates the terms of the caravan trade rather than conceding them.
# A one-time settlement — the once-only guard qing_caravan_kok_yielded is set in THIS immediate (mirrors
# .1/.2), and each option's END-STATE flag (qing_caravan_kok_terms_dictated / _kok_absorbed) permanently
# suppresses the ultimatum, the route-cut crisis, AND the khoja scare (QING_kok_yielded_flag, §NEW-MEDIUM-2).
########################################################################
qing_caravan.3 = {
	type = country_event
	title = "qing_caravan.3.t"
	desc = "qing_caravan.3.desc"
	picture = steppe_cavalry

	left_portrait = root.current_ruler


	trigger = {
		tag = CHI
		QING_kok_conquered_trigger = yes
		NOT = { has_variable = qing_caravan_kok_yielded }
	}

	immediate = {
		# [#13] set the once-only offered flag HERE (only when the event actually fires), not in the pulse —
		# so a re-check failure during the 5-15 day delay never strands it set (the #366/#368 flag-leak fix).
		set_variable = { name = qing_caravan_kok_yielded  value = 1 }
		LOG_line = { sys = QING  msg = "EVENT qing_caravan.3 (the khanate yields) fired for CHI" }
	}

	# --- DICTATE TERMS (天朝定制) — reopen the trade on Qing terms: no foreign aqsaqal, no customs haircut. ---
	option = {
		name = "qing_caravan.3.dictate"
		# [#13-R impl-review MED-1] affordability gate, mirroring qing_caravan.2.escort's own trigger — the
		# tooltip advertises the 100 treasury as a hard cost, so don't let a cash-poor throne go negative.
		trigger = { treasury >= 400 }
		custom_tooltip = "qing_caravan.3.dictate.tt"
		QING_caravan_dictate_terms = yes
		add_treasury = -400
		TREASURY_LOG_it = { amount = -400 }
		current_ruler = { add_popularity = 20 }
	}

	# --- ABSORB THE ROUTE (併商道) — fold the Kokand merchants into a Qing state monopoly of the oasis trade. ---
	option = {
		name = "qing_caravan.3.absorb"
		# [#13-R impl-review MED-1] affordability gate — the tooltip advertises 40 political influence as a
		# hard cost; gate the option so it can't be taken partly free (mirrors .2.escort's trigger precedent).
		trigger = { political_influence >= 40 }
		custom_tooltip = "qing_caravan.3.absorb.tt"
		QING_caravan_absorb_route = yes
		add_political_influence = -40
	}

	# --- DEFER THE SETTLEMENT (緩議) — the always-available fallback. -----
	# [#13-R impl-review MED-1] Both settlement options above now carry affordability triggers, so a throne that
	# can afford NEITHER (a costly war just ended) would otherwise face an unclosable event — an Imperator
	# country_event needs one available option (cf. qing_caravan.2.lapse, the ungated escape). This defers:
	# clear the once-only offered flag so QING_caravan_pulse re-offers .3 later once the coffers recover.
	# Kokand stays beaten, so QING_kok_conquered_trigger keeps the ultimatum/route-cut/khoja-scare suppressed
	# in the meantime — nothing strands.
	option = {
		name = "qing_caravan.3.defer"
		custom_tooltip = "qing_caravan.3.defer.tt"
		if = {
			limit = { has_variable = qing_caravan_kok_yielded }
			remove_variable = qing_caravan_kok_yielded
		}
	}
}

# .4  A DISPUTE AT KASHGAR (交涉阿奇木)  — [#112c / #12]
# The recurring day-to-day friction between the seated Caravan Superintendent (參贊大臣) and the seated
# Kokandi aqsaqal (阿奇木伯克): a border/customs dispute the two must resolve. EVERY option is a contest of
# their RELATIVE attributes (SPIKE 2's two-sided odds, extended to three lenses — rev-112bc H2):
#   • NEGOTIATE (folded rendering) — super(finesse+charisma) vs aqsaqal(finesse+zeal); the diplomatic path.
#   • COERCE — super.martial vs aqsaqal.martial; threaten the escort, he invokes Kokandi raiders.
#   • COLLUDE — a venal super + a bribable aqsaqal split the skim; weights corruption, the aqsaqal's zeal resists.
# Each is a skill-check roll (the proven hidden_effect{ scope = { random{ chance = <svalue> } } } idiom,
# qing_integ.10.d) whose odds are RENDERED as summed percentages via the stash-and-read _shown vars
# (qing_amban_negotiate_chance_shown precedent). #12: each beat READS both chars' stats into the odds AND
# WRITES an outcome back to char state (aqsaqal loyalty/opinion, super corruption/prominence) — genuine
# bidirectional coupling, not two portraits + odds. Offered by QING_caravan_pulse under the MANDATORY
# court-slot + qing_dept_cd_caravan throttle (rev-112bc H3). Both chars guarded seated+alive before the
# scopes are saved (rev-112bc M3). CHI-player-only.
########################################################################
qing_caravan.4 = {
	type = country_event
	title = "qing_caravan.4.t"
	desc = "qing_caravan.4.desc"
	picture = steppe_cavalry

	# the two contestants, side by side: the Qing superintendent (left) + the foreign Kokandi consul (right).
	# Bare `var:X` char-var portrait form (proven, qing_canton_events.txt:137 right_portrait = var:qing_hoppo_holder).
	left_portrait = var:qing_caravan_super_holder
	right_portrait = var:qing_caravan_aqsaqal_holder

	trigger = {
		tag = CHI
		# the concession must still stand (belt-and-braces with the pulse gate — matches the event intent).
		has_variable = qing_caravan_aqsaqal_granted
		# both offices seated + alive (M3) — the pulse already gated this, re-check belt-and-braces.
		has_variable = qing_caravan_aqsaqal_holder
		var:qing_caravan_aqsaqal_holder = { is_alive = yes }
		has_variable = qing_caravan_super_holder
		var:qing_caravan_super_holder = { is_alive = yes }
	}

	immediate = {
		# save BOTH contestants as the scopes the contest svalues read (proven CHI-scope save_scope_as, NOT
		# the var:X.finesse-reads-0 trap). Guarded above; save here so the svalues + the _shown stash resolve.
		var:qing_caravan_super_holder = { save_scope_as = qing_super }
		var:qing_caravan_aqsaqal_holder = { save_scope_as = qing_aqsaqal }
		# stash all three lenses' odds (+ complements) ON CHI so the option tooltips render REAL percentages
		# that sum to 100 via [ROOT.MakeScope.GetVariable('X').GetValue] (the amban-negotiate render idiom).
		set_variable = { name = qing_super_negotiate_shown   value = qing_super_contest_chance_svalue }
		set_variable = { name = qing_super_negotiate_fail_shown  value = qing_super_contest_fail_svalue }
		set_variable = { name = qing_super_coerce_shown      value = qing_super_coerce_chance_svalue }
		set_variable = { name = qing_super_coerce_fail_shown     value = qing_super_coerce_fail_svalue }
		set_variable = { name = qing_super_collude_shown     value = qing_super_collude_chance_svalue }
		set_variable = { name = qing_super_collude_fail_shown    value = qing_super_collude_fail_svalue }
		LOG_line = { sys = QING  msg = "EVENT qing_caravan.4 (aqsaqal friction contest) fired for CHI" }
	}

	# --- NEGOTIATE (折衝) — settle it by diplomacy. super(finesse+charisma) vs aqsaqal(finesse+zeal). ---
	# WIN: the dispute resolves in the throne's favour — prosperity up, the aqsaqal's opinion of CHI warms
	# (WRITE-BACK), the superintendent's standing rises. LOSE: the consul out-talks him — a little prosperity
	# lost, the superintendent discredited (prominence down, WRITE-BACK).
	option = {
		name = "qing_caravan.4.negotiate"
		custom_tooltip = "qing_caravan.4.negotiate.tt"
		hidden_effect = {
			# [#123] Initialise the flag to 0 BEFORE the random roll. The old code set qing_caravan_contest_ok
			# only INSIDE random{}, so on a failed roll (and in the option's effect-preview pass, where the
			# random set does not commit) the var stayed unset — and the downstream reads then errored
			# ~48x/sec with "Failed to fetch variable / unset scope 'var' / invalid left side comparison".
			# Defining it up front makes every read hit a valid 0/1, killing the flood; the branch now tests
			# the VALUE (= 1) rather than existence. Semantics unchanged (chance-gated win vs lose).
			set_variable = { name = qing_caravan_contest_ok  value = 0 }
			random = {
				chance = qing_super_contest_chance_svalue
				set_variable = { name = qing_caravan_contest_ok  value = 1 }
			}
			if = {
				limit = { var:qing_caravan_contest_ok = 1 }
				QING_DECLINE_nudge = { var = qing_caravan_prosperity  amount = 6 }
				# WRITE-BACK: the consul respects a fair settlement — Kokand warms; the super is commended.
				if = { limit = { exists = c:KOK }  c:KOK = { add_opinion = { modifier = qing_gp_relation_opinion  target = ROOT } } }
				if = { limit = { var:qing_caravan_super_holder = { is_alive = yes } }  var:qing_caravan_super_holder = { add_popularity = 8 } }
				LOG_line = { sys = QING  msg = "caravan: the superintendent out-negotiates the aqsaqal (skill check passed) for" }
				set_variable = { name = qing_caravan_outcome  value = flag:negotiate_win }   # [#91] outcome for the notification event
			}
			else = {
				QING_DECLINE_nudge = { var = qing_caravan_prosperity  amount = -4 }
				# WRITE-BACK: the consul gets the better of him — the superintendent loses face.
				if = { limit = { var:qing_caravan_super_holder = { is_alive = yes } }  var:qing_caravan_super_holder = { add_popularity = -6 } }
				LOG_line = { sys = QING  msg = "caravan: the aqsaqal out-talks the superintendent (skill check failed) for" }
				set_variable = { name = qing_caravan_outcome  value = flag:negotiate_lose }   # [#91]
			}
			remove_variable = qing_caravan_contest_ok
		}
			trigger_event = { id = qing_caravan.5  days = 2 }   # [#91] fire the outcome notification (silent hidden_effect never told the player win/lose)
	}

	# --- COERCE (威嚇) — threaten the escort/force. super.martial vs aqsaqal.martial (he invokes raiders). ---
	# WIN: the consul backs down — the grip firms, but Kokand resents the strong-arm (WRITE-BACK: emboldened).
	# LOSE: the aqsaqal calls the bluff and stirs the frontier — a khoja scare, the superintendent shamed.
	option = {
		name = "qing_caravan.4.coerce"
		custom_tooltip = "qing_caravan.4.coerce.tt"
		hidden_effect = {
			set_variable = { name = qing_caravan_contest_ok  value = 0 }   # [#123] define before the roll — see the negotiate option
			random = {
				chance = qing_super_coerce_chance_svalue
				set_variable = { name = qing_caravan_contest_ok  value = 1 }
			}
			if = {
				limit = { var:qing_caravan_contest_ok = 1 }
				QING_DECLINE_nudge = { var = qing_caravan_prosperity  amount = 3 }
				# WRITE-BACK: coercion works but rankles — Kokand's grievance hardens even in defeat.
				if = { limit = { exists = c:KOK }  c:KOK = { add_opinion = { modifier = qing_xj_kokand_emboldened  target = ROOT } } }
				if = { limit = { var:qing_caravan_super_holder = { is_alive = yes } }  var:qing_caravan_super_holder = { add_popularity = 5 } }
				LOG_line = { sys = QING  msg = "caravan: the superintendent faces down the aqsaqal (coercion succeeded) for" }
				set_variable = { name = qing_caravan_outcome  value = flag:coerce_win }   # [#91]
			}
			else = {
				# the bluff is called: a khoja scare (only if the oasis isn't already normalised — #367 gate).
				if = {
					limit = { NOT = { has_variable = qing_xj_fully_integrated } }
					set_variable = { name = qing_xj_khoja_pending  value = 1 }
				}
				if = { limit = { var:qing_caravan_super_holder = { is_alive = yes } }  var:qing_caravan_super_holder = { add_popularity = -8 } }
				LOG_line = { sys = QING  msg = "caravan: the aqsaqal calls the superintendent's bluff and stirs the frontier (coercion failed) for" }
				set_variable = { name = qing_caravan_outcome  value = flag:coerce_lose }   # [#91]
			}
			remove_variable = qing_caravan_contest_ok
		}
			trigger_event = { id = qing_caravan.5  days = 2 }   # [#91] fire the outcome notification (silent hidden_effect never told the player win/lose)
	}

	# --- COLLUDE (通同) — split the customs skim with the consul. weights corruption; the aqsaqal's zeal resists. ---
	# WIN: the two pocket the difference — both gain personal wealth (WRITE-BACK: super corruption up), but the
	# throne's take suffers a touch. LOSE: the pious consul refuses and denounces the offer — the superintendent
	# exposed (corruption up AND prominence down), Kokand affronted.
	option = {
		name = "qing_caravan.4.collude"
		custom_tooltip = "qing_caravan.4.collude.tt"
		hidden_effect = {
			set_variable = { name = qing_caravan_contest_ok  value = 0 }   # [#123] define before the roll — see the negotiate option
			random = {
				chance = qing_super_collude_chance_svalue
				set_variable = { name = qing_caravan_contest_ok  value = 1 }
			}
			if = {
				limit = { var:qing_caravan_contest_ok = 1 }
				# WRITE-BACK: the skim succeeds — both men pocket silver; the super grows more venal; prosperity
				# dips a touch (the throne's take is shaved). add_gold touches only CHAR wealth (not treasury/
				# silver reserve) — cannot perturb the currency sim (the #112a siphon precedent).
				if = {
					limit = { var:qing_caravan_super_holder = { is_alive = yes } }
					var:qing_caravan_super_holder = { add_gold = 4  if = { limit = { corruption < 90 }  add_corruption = 3 } }
				}
				QING_DECLINE_nudge = { var = qing_caravan_prosperity  amount = -2 }
				LOG_line = { sys = QING  msg = "caravan: the superintendent and the aqsaqal quietly split the skim (collusion succeeded) for" }
				set_variable = { name = qing_caravan_outcome  value = flag:collude_win }   # [#91]
			}
			else = {
				# the pious consul refuses + denounces: the super is exposed, Kokand affronted.
				if = {
					limit = { var:qing_caravan_super_holder = { is_alive = yes } }
					var:qing_caravan_super_holder = { add_popularity = -10  if = { limit = { corruption < 90 }  add_corruption = 2 } }
				}
				if = { limit = { exists = c:KOK }  c:KOK = { add_opinion = { modifier = qing_xj_kokand_emboldened  target = ROOT } } }
				LOG_line = { sys = QING  msg = "caravan: the aqsaqal refuses the bribe and denounces the superintendent (collusion failed) for" }
				set_variable = { name = qing_caravan_outcome  value = flag:collude_lose }   # [#91]
			}
			remove_variable = qing_caravan_contest_ok
		}
			trigger_event = { id = qing_caravan.5  days = 2 }   # [#91] fire the outcome notification (silent hidden_effect never told the player win/lose)
	}
}
