flavor_eve.6 = {
	type = country_event
	title = "flavor_eve.6.t"
	desc = "flavor_eve.6.desc"
	picture = farming
	goto_location = scope:randomized_country_province
	left_portrait = root.current_ruler
	
	trigger = {
		num_of_provinces > 1						
	}
	
	immediate = {
		random_owned_province = {
			limit = {
				is_capital = no
				# TODO: Replace this trigger with shortages, or low amenities
				# Shortages should cause loss of amenities
			}
			save_scope_as = randomized_country_province
		}
	}
	
	fire_only_once = yes
	
	option = {
		name = flavor_eve.6.a 
		ai_chance = { factor = 75 }
		scope:randomized_country_province = {
			add_province_modifier = {
				name = positive_food_modifier # TODO: Replace this with an Imperatrix style thing
				duration = 1825
			}
		}
		# [#101 2026-08-08] SIGN FIX: sending aid must COST the treasury, not fill it. Was
		# `add = root.INCOME_national_total_quarterly` (a POSITIVE credit — the reported "gives
		# money instead of costing it" bug). Relief now DEBITS the treasury: a quarter of the
		# nation's quarterly income (~one month's take) is a proportionate one-province relief
		# sum, negated via multiply = -1 (the proven cost idiom, cf. shortage.1.a).
		add_treasury = {
			value = root.INCOME_national_total_quarterly
			divide = 4
			multiply = -1
		}
	}
	# [#101 2026-08-08] OPEN THE EVER-NORMAL GRANARIES (常平倉) — the historical Qing answer to a
	# grain shortage: ship real stored grain to the stricken locale instead of buying relief on the
	# market. CHI-only, shown only when the national pool holds grain. Ships food into the province's
	# STATE (add_state_food), drains the pool by the same amount, stamps qing_grain_relief — no
	# treasury cost, but spends down a finite reserve. Mirrors the proven shortage.1.c granary ship
	# (se_QING_DECLINE.txt ship-leg idiom): save the State scope, size = 1/5 capacity bounded by pool.
	option = {
		name = flavor_eve.6.c
		trigger = {
			tag = CHI
			exists = scope:randomized_country_province
			has_variable = qing_granary_food
			var:qing_granary_food > 0
		}
		scope:randomized_country_province.state = {
			save_scope_as = qing_hardtimes_state
			set_variable = { name = qing_gran_ht_ship_tmp  value = has_state_food_capacity }
			change_variable = { name = qing_gran_ht_ship_tmp  divide = 5 }
			if = {
				limit = { var:qing_gran_ht_ship_tmp > { value = ROOT.var:qing_granary_food } }
				set_variable = { name = qing_gran_ht_ship_tmp  value = ROOT.var:qing_granary_food }
			}
			if = {
				limit = { var:qing_gran_ht_ship_tmp > 0 }
				add_state_food = { value = var:qing_gran_ht_ship_tmp }
				ROOT = { change_variable = { name = qing_granary_food  subtract = scope:qing_hardtimes_state.var:qing_gran_ht_ship_tmp } }
			}
			remove_variable = qing_gran_ht_ship_tmp
		}
		scope:randomized_country_province = {
			if = {
				limit = { NOT = { has_province_modifier = qing_grain_relief } }
				add_province_modifier = { name = qing_grain_relief  duration = 365 }
			}
		}
		add_political_influence = -10
	}
	option = {
		name = flavor_eve.6.b 
		ai_chance = { factor = 25 }
		scope:randomized_country_province = {
			add_province_modifier = {
				name = local_unrest_capital
				duration = 1825
			}
		}
		add_stability = -3
	}
}

flavor_eve.13 = {
	type = country_event
	title = "flavor_eve.13.t"
	desc = "flavor_eve.13.desc"
	picture = looting
	left_portrait = current_ruler
	
	trigger = {
        NOT = { has_variable = student_riots_cooldown }
        stability < 75
    	any_owned_province = {
    		has_building = EDU_university
    		province_unrest >= 1
    	}
    }

    immediate = {
        # [cooldown-fix 2026-08-04] was set_variable industrial_building_cooldown — but this event's
        # trigger gates on student_riots_cooldown (line ~519), so the throttle never engaged and Student
        # Riots could re-fire every pulse. Set the var the trigger actually checks.
        set_variable = {
            name = student_riots_cooldown
            days = 100
        }
        random_owned_province = {
            limit = {
                has_building = EDU_university
                province_unrest >= 1
            }
            save_scope_as = student_riots_province
            # [loc-fix 2026-08-04] the desc (flavor_eve.13.desc) refers to [building_province.GetName]
            # but this event only saved scope:student_riots_province -> ERROR:[building_province.GetName]
            # (same class as flavor_eve.15). Save the same province under building_province so the desc
            # resolves; the option effects keep using student_riots_province.
            save_scope_as = building_province
        }
        # [#110 2026-08-08] Detect a garrison stationed in the riot province so option .d (the local
        # garrison restores order) can be gated on it. Count our own standing armies on that province's
        # soil (proven every_army + unit_location idiom, se_SUBJECT_QING.txt:849). In the capital
        # (Beijing) the Imperial/Bayara Guard supplies exactly such a garrison; elsewhere a raised host
        # parked in the province qualifies. 0 = no troops on hand -> option .d hidden.
        set_variable = { name = student_riots_garrison  value = 0 }
        every_army = {
            limit = {
                is_moving = no
                exists = unit_location
                unit_location = { this = scope:student_riots_province }
            }
            ROOT = { change_variable = { name = student_riots_garrison  add = 1 } }
        }
    }
	
	option = {
		name = flavor_eve.13.a
		add_manpower = -10
		add_tyranny = 6
		add_stability = 2		
	}

	# [#110 2026-08-08] Order the LOCAL garrison to restore order — shown only when a garrison already
	# stands in the riot province (student_riots_garrison >= 1; the capital's Imperial/Bayara Guard is
	# exactly this). Troops already on the spot quell the riot at far less political cost than marching
	# in the field army: no manpower drain, a lighter tyranny hit, and the local_unrest_harsh modifier
	# (an unrest REDUCTION, -0.30) settles the province.
	option = {
		name = flavor_eve.13.d
		trigger = {
			has_variable = student_riots_garrison
			var:student_riots_garrison >= 1
		}
		scope:student_riots_province = {
			add_province_modifier = {
				name = qing_order_restored   # [#110] positively-named unrest-reduction (-0.30); NOT local_unrest_harsh ("Severe Unrest" label would contradict)
				duration = 730
			}
		}
		add_tyranny = 2
		add_stability = 3
	}

	option = {
		name = flavor_eve.13.b
		scope:student_riots_province = {
			add_province_modifier = {
				name = local_unrest_mild
				duration = 730
			}
		} 	
	}

	option = {
		name = flavor_eve.13.c
		scope:student_riots_province = {
			scope:student_riots_province = {
				remove_building_level = EDU_university
			}
		}
		add_tyranny = 3
		add_stability = 5 	
	}
}
