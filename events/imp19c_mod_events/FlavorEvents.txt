flavor_eve.6 = {
	type = country_event
	title = "flavor_eve.6.t"
	desc = "flavor_eve.6.desc"
	picture = farming
	goto_location = scope:randomized_country_province
	left_portrait = root.current_ruler
	
	# [#37 2026-08-09 BUGFIX] Fire ONLY where a province's STATE is genuinely short of stored food.
	# The old trigger was `num_of_provinces > 1` with an EMPTY picker limit (a leftover TODO), so
	# "Hard Times" fired on a well-fed province (reported: Xingiang, whose Henan state held a maxed
	# 4120). Gate on the engine's own State food pool, the same signal shortage.1 uses (has_state_food
	# below 1/5 of capacity = "running low", a warning ahead of shortage.1's has_state_food<1 famine).
	trigger = {
		num_of_provinces > 1
		any_owned_province = {
			is_capital = no
			state = {
				has_state_food_capacity > 0
				has_state_food < { value = has_state_food_capacity  divide = 5 }
			}
		}
	}
	
	immediate = {
		random_owned_province = {
			# [#37] pick a NON-capital province whose STATE is actually short of stored food (mirrors the trigger).
			limit = {
				is_capital = no
				state = {
					has_state_food_capacity > 0
					has_state_food < { value = has_state_food_capacity  divide = 5 }
				}
			}
			save_scope_as = randomized_country_province
		}
		# [#38/#39/#47 2026-08-09] PRECOMPUTE the granary-relief ship amount ONCE, here in immediate — so
		# option .c only READS it and never sets-then-reads a var in the AI predict / tooltip pass (the #47
		# flood: flavor_eve.6.c produced ~50k error.log lines because set_variable does not commit during
		# effect prediction, so the following var: reads failed en masse). CHI-only, and only when the
		# national ever-normal pool holds grain ABOVE the reserve floor (#39 — a relief may never empty it).
		if = {
			limit = {
				tag = CHI
				exists = scope:randomized_country_province
				has_variable = qing_granary_food
				var:qing_granary_food > 400
			}
			scope:randomized_country_province.state = {
				save_scope_as = qing_hardtimes_state
				# target = 1/5 of the stricken state's food capacity...
				set_variable = { name = qing_hardtimes_ship_preview  value = has_state_food_capacity }
				change_variable = { name = qing_hardtimes_ship_preview  divide = 5 }
				# ...capped at a modest 400-shih local dole (#39; supersedes #17's 250 per the user's "~400")...
				if = { limit = { var:qing_hardtimes_ship_preview > 400 }  set_variable = { name = qing_hardtimes_ship_preview  value = 400 } }
				# ...and bounded so the national reserve is NEVER drawn below its 400 floor (#39): ship at most (pool - 400).
				set_variable = { name = qing_ht_avail_tmp  value = ROOT.var:qing_granary_food }
				change_variable = { name = qing_ht_avail_tmp  subtract = 400 }
				if = {
					limit = { var:qing_hardtimes_ship_preview > { value = var:qing_ht_avail_tmp } }
					set_variable = { name = qing_hardtimes_ship_preview  value = var:qing_ht_avail_tmp }
				}
				if = { limit = { var:qing_hardtimes_ship_preview < 0 }  set_variable = { name = qing_hardtimes_ship_preview  value = 0 } }
				remove_variable = qing_ht_avail_tmp
			}
		}
	}
	
	# [#37 decision 2026-08-09] KEPT fire_only_once = yes (matches the sibling flavour events flavor_eve.1/.7/.8);
	# with the real state-shortage gate above the event no longer fires spuriously, so a one-shot beat is fine.
	# (A repeatable-with-cooldown form was considered and rejected here to avoid introducing an unthrottled
	# recurring economic event on this pass.)
	fire_only_once = yes
	
	option = {
		name = flavor_eve.6.a
		ai_chance = { factor = 75 }
		scope:randomized_country_province = {
			add_province_modifier = {
				name = positive_food_modifier # TODO: Replace this with an Imperatrix style thing
				duration = 1825
			}
		}
		# [#101 2026-08-08] SIGN FIX: sending aid must COST the treasury, not fill it. Was
		# `add = root.INCOME_national_total_quarterly` (a POSITIVE credit — the reported "gives
		# money instead of costing it" bug). Relief now DEBITS the treasury: a quarter of the
		# nation's quarterly income (~one month's take) is a proportionate one-province relief
		# sum, negated via multiply = -1 (the proven cost idiom, cf. shortage.1.a — shortage.1.a's
		# value is a sum of good prices x population, always >= 0, so it never needed a clamp;
		# income here CAN go negative in a deficit quarter, and unlike shortage.1.a this had none).
		# [review-fix 2026-08-25] adversarial review of the flavor_eve.16/.11 fixes flagged this
		# as the same bug class, unfixed: with no clamp, a deficit quarter's negative income,
		# divided and negated, becomes a POSITIVE gain -- this "relief" event would then hand the
		# player money instead of costing them any, exactly the #101 bug this option was
		# originally written to fix. max=0 caps the negated result at a ceiling of 0 (never a gain).
		add_treasury = {
			value = root.INCOME_national_total_quarterly
			divide = 4
			multiply = -1
			max = 0
		}
		# [#108 fix 2026-08-15] stage into a var -- a compound amount= block trips TREASURY_LOG_it's
		# macro-arg validator (see se_STATE_INVESTMENT.txt for the full note).
		set_variable = { name = TREASURY_LOG_amt_tmp  value = { value = root.INCOME_national_total_quarterly  divide = 4  multiply = -1  max = 0 } }
		TREASURY_LOG_it = { amount = var:TREASURY_LOG_amt_tmp }
		remove_variable = TREASURY_LOG_amt_tmp
	}
	# [#101 2026-08-08] OPEN THE EVER-NORMAL GRANARIES (常平倉) — the historical Qing answer to a grain
	# shortage: ship real stored grain to the stricken locale instead of buying relief on the market.
	# CHI-only, shown only when the national pool holds grain ABOVE the floor. The ship amount was
	# PRECOMPUTED in immediate (qing_hardtimes_ship_preview on the saved state scope) — this option only
	# READS it (add_state_food into the state, drain the pool by the same amount), so no set-then-read
	# flood (#47) and the option loc can display the figure up front (#38).
	option = {
		name = flavor_eve.6.c
		custom_tooltip = "flavor_eve.6.c.tt"
		trigger = {
			tag = CHI
			exists = scope:qing_hardtimes_state
			has_variable = qing_granary_food
			var:qing_granary_food > 400
		}
		# [review LOW-2] gate the ENTIRE relief payload (ship + drain + relief modifier + influence cost)
		# on a real ship amount, so a tiny-capacity state that rounds to 0 doesn't cost influence / stamp
		# relief for no grain. [review LOW-1] clean up the preview var after use (symmetry with qing_ht_avail_tmp).
		if = {
			limit = { scope:qing_hardtimes_state.var:qing_hardtimes_ship_preview > 0 }
			scope:qing_hardtimes_state = { add_state_food = { value = var:qing_hardtimes_ship_preview } }
			change_variable = { name = qing_granary_food  subtract = scope:qing_hardtimes_state.var:qing_hardtimes_ship_preview }
			scope:randomized_country_province = {
				if = {
					limit = { NOT = { has_province_modifier = qing_grain_relief } }
					add_province_modifier = { name = qing_grain_relief  duration = 365 }
				}
			}
			add_political_influence = -10
		}
		scope:qing_hardtimes_state = { if = { limit = { has_variable = qing_hardtimes_ship_preview }  remove_variable = qing_hardtimes_ship_preview } }
	}
	option = {
		name = flavor_eve.6.b
		ai_chance = { factor = 25 }
		scope:randomized_country_province = {
			add_province_modifier = {
				name = local_unrest_capital
				duration = 1825
			}
		}
		add_stability = -3
	}
}

flavor_eve.7 = {
	type = country_event
	title = "flavor_eve.7.t"
	desc = "flavor_eve.7.desc"
	picture = war_council
	left_portrait = root.current_ruler
	right_portrait = scope:friendly_neighbor.current_ruler
	
	# [#43 2026-08-09] was an EMPTY trigger (fired any country/era) + random_country that could pick
	# ROOT itself. Mirror the #21/#22 Foreign Friends (flavor_eve.18) fix: require a real reachable
	# non-allied neighbour to exist.
	trigger = {
		any_neighbour_country = {
			has_land = yes
			is_subject = no
			NOT = { alliance_with = root }
			in_diplomatic_range = root
		}
	}
	
	immediate = {
		random_country = {
			limit = {
				has_land = yes
				is_subject = no
				NOT = { alliance_with = root }
				in_diplomatic_range = root
				NOT = { this = root }   # [#43] never pick ourselves (a self-gift is nonsensical)
			}
			save_scope_as = friendly_neighbor
		}
	}
	
	fire_only_once = yes
	
	# OPTION A: accept the gift graciously. [#43] The desc promises warming relations, but the old
	# option only added treasury. Make the improved relations REAL (bilateral opinion), and — for the
	# Qing with a seated Zongli Yamen (總理衙門) director — route the exchange through a REAL envoy from
	# the diplomat corps (mirrors the #22 flavor_eve.18 Yamen tie-in): draw the ablest diplomat, credit
	# him, and apply the stronger envoy-brokered warmth. Non-Yamen (or non-CHI) falls back to the
	# generic ruler-to-ruler warmth.
	option = {
		name = flavor_eve.7.a
		ai_chance = { factor = 75 }
		add_treasury = 15
		TREASURY_LOG_it = { amount = 15 }
		if = {
			# YAMEN-HANDLED (Qing, director seated): a real diplomat brokers the exchange.
			limit = {
				tag = CHI
				exists = scope:friendly_neighbor
				has_variable = qing_office_zongli_holder
				var:qing_office_zongli_holder = { is_alive = yes  employer = ROOT }
				has_variable = qing_zongli_diplomat_count
				var:qing_zongli_diplomat_count > 0
			}
			# the ablest Zongli diplomat handles it (the corps' governing skill), and is rewarded.
			ordered_in_list = {
				variable = qing_zongli_diplomats
				order_by = combined_stats_council_svalue
				check_range_bounds = no
				max = 1
				save_scope_as = qing_ff_envoy
			}
			# [#43 review HIGH-1] GUARD the envoy scope with exists (as qing_legation.3 does around the
			# identical call): qing_zongli_diplomat_count is bumped by the refill-sweep mint on a SEPARATE
			# path from the qing_zongli_diplomats LIST and only resyncs at pulse step 0, so count>0 does NOT
			# guarantee the list is non-empty this instant — ordered_in_list could save nothing, leaving
			# qing_ff_envoy unset (error) or STALE from a prior qing_legation.3 fire (promotes a wrong/dead
			# char). The exists-guard makes the reward fire only when a real envoy was actually drawn.
			if = {
				limit = { exists = scope:qing_ff_envoy }
				scope:qing_ff_envoy = { QING_char_promote_standing = { pop = 2  prom = 1 } }
			}
			# the director's own standing rises a touch for the corps' success.
			var:qing_office_zongli_holder = { QING_char_promote_standing = { pop = 1  prom = 1 } }
			# stronger, envoy-brokered bilateral warmth (the Yamen premium over the routine exchange).
			scope:friendly_neighbor = { add_opinion = { modifier = qing_zongli_envoy_opinion  target = ROOT } }
			add_opinion = { modifier = qing_zongli_envoy_opinion  target = scope:friendly_neighbor }
		}
		else = {
			# generic (non-Yamen / non-CHI): the Court receives the gift directly — real bilateral warmth.
			if = {
				limit = { exists = scope:friendly_neighbor }
				scope:friendly_neighbor = { add_opinion = { modifier = foreign_friends_opinion  target = ROOT } }
				add_opinion = { modifier = foreign_friends_opinion  target = scope:friendly_neighbor }
			}
		}
	}
	# OPTION B: reciprocate with a counter-gift of our own (spend treasury) — a PERSONAL bond between
	# the sovereigns AND the same real bilateral warming (the desc's "improved relations" applies here too).
	option = {
		name = flavor_eve.7.b
		ai_chance = { factor = 25 }
		add_treasury = -15
		TREASURY_LOG_it = { amount = -15 }
		if = {
			limit = { exists = scope:friendly_neighbor }
			current_ruler = {
				add_friend = scope:friendly_neighbor.current_ruler
			}
			scope:friendly_neighbor = { add_opinion = { modifier = foreign_friends_opinion  target = ROOT } }
			add_opinion = { modifier = foreign_friends_opinion  target = scope:friendly_neighbor }
		}
	}
}

				var:qing_office_works_holder = { is_alive = yes  employer = ROOT }
			}
			var:qing_office_works_holder = { add_loyalty = loyalty_qing_estranged }
			QING_DECLINE_nudge = { var = qing_reform_pressure  amount = 2 }
		}

		name = flavor_eve.8.b
		ai_chance = { factor = 25 }

			add_popularity = 5
		}

		# >=9, matching QING_works_build_dike's own expert-tier cutoff, se_QING_WORKS.txt:194) gets

		# original flat -50 unchanged.

			limit = {
				tag = CHI
				has_variable = qing_office_works_holder

			}

				limit = { var:qing_office_works_holder = { finesse >= 9 } }
				add_treasury = -40
				TREASURY_LOG_it = { amount = -40 }
			}
			else = {
				add_treasury = -50
				TREASURY_LOG_it = { amount = -50 }
			}
			var:qing_office_works_holder = { add_loyalty = loyalty_qing_commended }
		}
		else = {
			add_treasury = -50
			TREASURY_LOG_it = { amount = -50 }
		}
	}
}
