########################################################################
# qing_wenzhi_painter_events.txt — task #15, the Ruyiguan atelier's ambient flavour.
#
# The Court Painter mechanic (qing_court_artist, se_QING_WENZHI.txt) had no narrative
# events of its own — painters sat silently once commissioned. This adds the missing
# ambient payoff for maintaining the atelier at all: a finished work presented to the
# throne, rolled at low frequency from QING_wenzhi_pulse while at least one court
# painter is in residence.
#
#   ROOT / this = the Qing (CHI, player)
#
# Namespace = qing_wenzhi_painter
#   .1  the atelier presents a finished work (ambient, recurring, ~once per few years)
#   .2  a court painter has died (reactive, one-shot per painter; Castiglione gets a
#       named variant, drawn from QING_wenzhi_painter_death_check via on_character_death)
########################################################################

namespace = qing_wenzhi_painter

########################################################################
# .1  A FINISHED WORK PRESENTED (進呈畫作)
# The Ruyiguan atelier completes a commission — a court still life, a hunting scene,
# an imperial portrait fusing brush and Western perspective — and presents it to the
# throne.
########################################################################
qing_wenzhi_painter.1 = {
	type = country_event
	title = "qing_wenzhi_painter.1.t"
	desc = {
		# [fix, direct user instruction 2026-08-17] qing_work_painter is selected fresh in immediate
		# from the button-maintained qing_court_artist_count (staleness note above, se_QING_WENZHI.txt
		# lines 149-152) -- across this event's own 5-20 day trigger_event delay that painter could
		# in principle have died, so the named variant is triggered, not assumed, mirroring .2's own
		# triggered_desc/generic-fallback convention below.
		triggered_desc = {
			trigger = { exists = scope:qing_work_painter }
			desc = "qing_wenzhi_painter.1.desc_named"
		}
		triggered_desc = { trigger = { always = yes }  desc = "qing_wenzhi_painter.1.desc_generic" }
	}
	picture = chinese_throne_room

	left_portrait = root.current_ruler
	right_portrait = scope:qing_work_painter

	trigger = {
		tag = CHI
		is_ai = no
		var:qing_court_artist_count > 0
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_wenzhi_painter.1 (atelier presents a finished work)" }
		# [fix, direct user instruction 2026-08-17] the event used to name only the atelier, never
		# the actual painter, and gave him no personal payoff for the work he presented. Pick the
		# resident atelier's ablest hand (same order_by=finesse selection idiom as QING_salt_
		# commissioner_appoint / QING_canton_appoint_hoppo) and save him for the portrait, the desc,
		# and his own popularity gain below.
		ordered_character = {
			limit = { has_variable = qing_court_artist  is_alive = yes  employer = ROOT }
			order_by = finesse
			max = 1
			save_scope_as = qing_work_painter
		}
	}

	option = {
		name = "qing_wenzhi_painter.1.a"
		custom_tooltip = "qing_wenzhi_painter.1.a.tt"
		current_ruler = { add_popularity = 4 }
		QING_wenzhi_advance = { amount = 3 }
		if = {
			limit = { exists = scope:qing_work_painter }
			scope:qing_work_painter = { add_popularity = 6 }
		}
	}
}

########################################################################
# .2  A COURT PAINTER HAS DIED
# Reactive, fired from QING_wenzhi_painter_death_check (on_character_death). Castiglione
# (郎世寧), seeded by name at game start, gets a unique text; any other court artist gets
# a generic one.
########################################################################
qing_wenzhi_painter.2 = {
	type = country_event
	title = "qing_wenzhi_painter.2.t"
	desc = {
		triggered_desc = {
			trigger = {
				has_variable = qing_castiglione_char
				var:qing_castiglione_char = scope:qing_dead_painter
			}
			desc = "qing_wenzhi_painter.2.desc_castiglione"
		}
		triggered_desc = { trigger = { always = yes }  desc = "qing_wenzhi_painter.2.desc_generic" }
	}
	picture = chinese_throne_room

	left_portrait = root.current_ruler

	trigger = {
		tag = CHI
		is_ai = no
		exists = scope:qing_dead_painter
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_wenzhi_painter.2 (a court painter has died)" }
	}

	option = {
		name = "qing_wenzhi_painter.2.a"
		custom_tooltip = "qing_wenzhi_painter.2.a.tt"
		QING_wenzhi_advance = { amount = -4 }
	}
}
