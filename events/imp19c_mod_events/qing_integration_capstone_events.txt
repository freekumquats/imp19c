########################################################################
# qing_integration_capstone_events.txt — completion and resistance
# events for the Qing subject integration process.
#
# These extend the qing_integ namespace (see qing_subject_integration.txt)
# with two new mechanics:
#   1. COMPLETION CAPSTONE (.30): when a subject reaches the integration
#      threshold (progress >= 5), this fires BEFORE absorption to let the
#      player choose how to formally enact gaitu guiliu (改土歸流).
#   2. INCORPORATION DECREE + RESISTANCE (.40, .41): the player can force
#      the pace of integration (at cost/risk) when well-advanced, and proud
#      dependencies may resist.
#
# ROOT / this  = the overlord (player, Qing / CHI)
# scope:target = the subject governorship being integrated
#
# Namespace = qing_integ (continued from .21)
#   .30  CAPSTONE — integration completion / absorption ceremony
#   .40  INCORPORATION DECREE — force the pace of integration
#   .41  PROTECTORATE RESISTANCE — dependency resists coercion
########################################################################

namespace = qing_integ

########################################################################
# .30  INTEGRATION COMPLETION CAPSTONE
# Anchor: gaitu guiliu (改土歸流, "substitute chieftains with officials")
# — the Qing policy that converted native chieftaincies (土司 tusi) into
# directly-administered circuits under regular magistrates (流官 liuguan).
# The early waves (Kangxi-Yongzheng: Guizhou, Yunnan, Guangxi, Sichuan)
# were gradualist; the late push (Zhao Erfeng's 1904-11 Kham incorporation)
# was coercive and provoked revolt. This event fires when a subject has
# reached the integration threshold (progress >= 5) and offers the player
# a choice of absorption styles: ceremonial / bureaucratic / coercive.
# All paths absorb the subject, but with different consequences.
########################################################################
qing_integ.30 = {
	type = country_event
	title = "qing_integ.30.t"
	desc = "qing_integ.30.desc"
	picture = chinese_throne_room

	left_portrait = root.current_ruler
	right_portrait = scope:target.current_ruler


	trigger = {
		tag = CHI
		exists = scope:target
		scope:target = { is_subject_of = ROOT }
		scope:target = { is_subject_type = autonomous_governorship }
		scope:target = {
			has_variable = SUBJ_integration_progress
			var:SUBJ_integration_progress >= 5
		}
	}

	immediate = {
		LOG_line = { sys = SUBJ  msg = "EVENT qing_integ.30 (capstone) overlord=[ROOT.GetTag] subject=[scope:target.GetTag] progress=[scope:target.MakeScope.GetVariable('SUBJ_integration_progress').GetValue|0]/5" }
		# [#task-8] resolve the resident amban + garrison so the coercive path can key off a
		# garrison being on the spot (a coercive absorb with no banner troops present is dearer).
		SUBJ_QING_resolve_integ_actors = yes
	}

	# --- Grand ceremony: hold a full investiture festival ----------------
	# A triumphal procession, a solemn ritual, and a feast. Prestige and
	# legitimacy gain, but the cost in gold and time is high.
	option = {
		name = "qing_integ.30.a"
		custom_tooltip = "qing_integ.30.a.tt"
		trigger = { treasury >= 120 }
		add_treasury = -120
		add_legitimacy = 40
		add_legitimacy = 10
		add_stability = 5
		SUBJ_QING_clear_integ_actors = yes
		# Bureau capacity is taxed by the administrative overhead of the
		# newly incorporated provinces, but the ceremonial legitimacy offsets it.
		QING_DECLINE_nudge = { var = qing_bureau_capacity  amount = -5 }
		# [#subj-integ-rework 2026-07-22] AUTHORIZE the engine handoff instead of the old instant
		# LAND_transfer_provinces absorb: reparent any sub-subjects now (while the intermediary still
		# exists) and convert the subject to integrating_governorship so the engine's native Integrate
		# action lights up; the player finishes and the engine absorbs GRADUALLY. Costs/legitimacy/
		# modifiers below are unchanged — this option still IS the 改土歸流 decision, just no longer
		# same-tick teleportation of the provinces.
		SUBJ_QING_authorize_integration = yes
		# Apply a "newly incorporated circuit" modifier to CHI
		add_country_modifier = {
			name = qing_integ_new_circuit_solemn
			duration = 1825
		}
	}

	# --- Bureaucratic integration: dispatch officials and surveyors -------
	# The pragmatic path: a census, a tax roll, and a circuit magistrate.
	# Less glory, less gold, faster. Strains the bureaucracy.
	option = {
		name = "qing_integ.30.b"
		custom_tooltip = "qing_integ.30.b.tt"
		trigger = { treasury >= 60 }
		add_treasury = -60
		add_legitimacy = 5
		# The administrative load is heavier on this path.
		QING_DECLINE_nudge = { var = qing_bureau_capacity  amount = -10 }
		# [#subj-integ-rework 2026-07-22] authorize the engine handoff (see option .a).
		SUBJ_QING_authorize_integration = yes
		SUBJ_QING_clear_integ_actors = yes
		add_country_modifier = {
			name = qing_integ_new_circuit_bureaucratic
			duration = 1460
		}
	}

	# --- Coercive absorption: send in the Green Standard Army -------------
	# The Zhao Erfeng model: garrison first, administrate second. Cheap and
	# fast, but provokes unrest and tyranny. The frontier remembers force.
	option = {
		name = "qing_integ.30.c"
		custom_tooltip = "qing_integ.30.c.tt"
		add_tyranny = 8
		add_aggressive_expansion = 8
		# [#task-8] a garrison already on the spot (edge >= 1) means the "garrison first" model is
		# cheaper on stability and provokes a milder aftershock; with no banner troops present the
		# coercion is a raw imposition and bites harder.
		if = {
			limit = { has_variable = integ_garrison_edge  var:integ_garrison_edge >= 1 }
			add_stability = -3
			LOG_line = { sys = SUBJ  msg = "capstone coercive: garrison on the spot eases the imposition, subject=[scope:target.GetTag]" }
		}
		else = {
			add_stability = -5
			LOG_line = { sys = SUBJ  msg = "capstone coercive: no garrison present — a raw imposition, subject=[scope:target.GetTag]" }
		}
		QING_DECLINE_nudge = { var = qing_bureau_capacity  amount = -8 }
		# [#subj-integ-rework 2026-07-22] authorize the engine handoff (see option .a).
		SUBJ_QING_authorize_integration = yes
		SUBJ_QING_clear_integ_actors = yes
		add_country_modifier = {
			name = qing_integ_new_circuit_coercive
			duration = 1825
		}
		# A coercive push can seed a delayed resistance event (the chieftain's
		# kinsmen remember, and may rise later). Set a flag for the pulse.
		set_variable = { name = qing_integ_coercive_absorption_flag  days = 730  value = 1 }
	}
}

########################################################################
# .40  INCORPORATION DECREE
# The throne's active decision to force the pace of integration before it
# is fully complete: a push decree that spends legitimacy and bureau
# capacity to accelerate absorption, at the risk of provoking resistance
# (.41). Historical anchor: Zhao Erfeng's川邊改土歸流 (Chuanbian gaitu guiliu,
# 1904-11), where the Sichuan governor-general forcibly incorporated Kham
# and eastern Tibet, triggering the 1905 Batang Uprising and laying the
# groundwork for the post-1911 Tibetan independence movement.
#
# This event is triggered by the flavour roll when integration is well-
# advanced (progress >= 3 but < 5) and the player has the resources to
# force it. It offers a choice: spend now to finish fast, or wait.
########################################################################
qing_integ.40 = {
	type = country_event
	title = "qing_integ.40.t"
	# [#task-20 2026-08-04] base text + actor-status addendum naming the local amban / garrison when
	# present (resolved in the immediate by SUBJ_QING_resolve_integ_actors). Mutually exclusive on presence.
	desc = {
		desc = "qing_integ.40.desc"
		first_valid = {
			triggered_desc = { trigger = { has_variable = integ_amban_present  has_variable = integ_garrison_size  var:integ_garrison_size >= 1 }  desc = "qing_integ.40.desc_actors_both" }
			triggered_desc = { trigger = { has_variable = integ_amban_present }  desc = "qing_integ.40.desc_actors_amban" }
			triggered_desc = { trigger = { has_variable = integ_garrison_size  var:integ_garrison_size >= 1 }  desc = "qing_integ.40.desc_actors_garrison" }
			triggered_desc = { trigger = { always = yes }  desc = "qing_integ.40.desc_actors_none" }
		}
	}
	picture = senate_debate

	left_portrait = root.current_ruler
	right_portrait = scope:target.current_ruler


	trigger = {
		tag = CHI
		exists = scope:target
		scope:target = { is_subject_of = ROOT }
		scope:target = { is_subject_type = autonomous_governorship }
		scope:target = {
			has_variable = SUBJ_integration_progress
			var:SUBJ_integration_progress >= 3
			var:SUBJ_integration_progress < 5
		}
	}

	immediate = {
		LOG_line = { sys = SUBJ  msg = "EVENT qing_integ.40 (decree) overlord=[ROOT.GetTag] subject=[scope:target.GetTag] progress=[scope:target.MakeScope.GetVariable('SUBJ_integration_progress').GetValue|0]/5" }
		# [#task-20] resolve the resident amban + garrison so the decree's options + resistance roll
		# below can key off them (a garrison on the spot / a loyal amban makes forcing the pace safer).
		SUBJ_QING_resolve_integ_actors = yes
	}

	# --- Issue the incorporation decree: force the pace ------------------
	# Spend legitimacy, bureau capacity, and gold to skip the remaining
	# steps and jump straight to the threshold (progress = 5). Risky:
	# a proud or culturally alien dependency may resist (.41).
	option = {
		name = "qing_integ.40.a"
		custom_tooltip = "qing_integ.40.a.tt"
		trigger = {
			treasury >= 100
			legitimacy >= 20
		}
		add_treasury = -100
		add_legitimacy = -20
		QING_DECLINE_nudge = { var = qing_bureau_capacity  amount = -15 }
		# Jump to the threshold
		scope:target = {
			set_variable = { name = SUBJ_integration_progress  value = 5 }
		}
		# Roll for resistance: proud dependencies, culturally alien ones, or
		# those with high total_population are likelier to resist — but a garrison
		# on the spot and a loyal resident amban make forcing the pace SAFER.
		random_list = {
			40 = {
				# Resistance event fires
				modifier = {
					factor = 2
					scope:target = { NOT = { primary_culture = root.culture } }
				}
				modifier = {
					factor = 1.5
					scope:target = { any_owned_province = { total_population >= 30 } }
				}
				# [#task-20] a dominant imperial garrison overawes the dependency — a decree backed by
				# banner troops on the ground is far likelier to be swallowed without a rising (e.g. the
				# TWO garrisons in Yili). Rough parity still helps a little.
				modifier = {
					factor = 0.4
					has_variable = integ_garrison_edge
					var:integ_garrison_edge = 2
				}
				modifier = {
					factor = 0.7
					has_variable = integ_garrison_edge
					var:integ_garrison_edge = 1
				}
				# [#task-20] a staunch resident amban prepares the ground and softens the blow.
				modifier = {
					factor = 0.6
					has_variable = integ_amban_present
					var:integ_amban_band = 2
				}
				LOG_line = { sys = SUBJ  msg = "decree RESISTANCE subject=[scope:target.GetTag]" }
				trigger_event = { id = qing_integ.41  days = { 10 30 } }
			}
			60 = {
				# No resistance: the decree lands without incident
				LOG_line = { sys = SUBJ  msg = "decree ACCEPTED subject=[scope:target.GetTag]" }
				# Since we're at threshold, the absorption capstone (.30) should
				# fire next (via the normal integrate button flow or the pulse).
			}
		}
		SUBJ_QING_clear_integ_actors = yes
	}

	# --- [#task-20] Let the resident amban prepare the ground first -------
	# Shown only when a STAUNCH resident amban (band 2) sits in the dependency: rather than force
	# the decree cold, he spends months smoothing the local elite — the pace is forced AND resistance
	# is far less likely (skips the .41 roll). Cheaper on legitimacy than the raw decree; the amban's
	# standing rises for the service. Absent / lesser amban => option hidden.
	option = {
		name = "qing_integ.40.c"
		custom_tooltip = "qing_integ.40.c.tt"
		trigger = {
			treasury >= 80
			has_variable = integ_amban_present
			var:integ_amban_band = 2
			exists = scope:integ_amban
		}
		add_treasury = -80
		add_legitimacy = -10
		QING_DECLINE_nudge = { var = qing_bureau_capacity  amount = -10 }
		scope:target = {
			set_variable = { name = SUBJ_integration_progress  value = 5 }
		}
		scope:integ_amban = { add_prominence = 10  add_popularity = 5 }
		# a decree brokered by a trusted resident lands without a rising.
		LOG_line = { sys = SUBJ  msg = "decree SMOOTHED by resident amban (no resistance roll) subject=[scope:target.GetTag]" }
		SUBJ_QING_clear_integ_actors = yes
	}

	# --- Wait: let the process mature naturally ---------------------------
	# The patient path: no cost, no risk, but no acceleration.
	option = {
		name = "qing_integ.40.b"
		custom_tooltip = "qing_integ.40.b.tt"
		add_stability = 2
		LOG_line = { sys = SUBJ  msg = "decree DECLINED (patient path) subject=[scope:target.GetTag]" }
		SUBJ_QING_clear_integ_actors = yes
	}
}

########################################################################
# .41  PROTECTORATE RESISTANCE
# The backlash: when the throne pushes incorporation too hard (via .40's
# aggressive option), or when a proud/martial dependency chafes under
# tightening control, the native chieftain (土司 tusi) defies the throne.
# Historical anchors: the 1905 Batang Uprising (巴塘起義, triggered by Zhao
# Erfeng's coercive incorporation), the Jinchuan Wars (金川之戰, 1747-49 &
# 1771-76, where Gyalrong chieftains resisted Qing encroachment for decades),
# and the Revolt of the Three Feudatories (三藩之亂, 1673-81, the southern
# warlords' refusal to be absorbed).
#
# This event offers the throne real choices: conciliate (roll back some
# integration progress, restore autonomy), crush (military + tyranny cost),
# or co-opt the chieftain (via the affinity spine).
########################################################################
qing_integ.41 = {
	type = country_event
	title = "qing_integ.41.t"
	# [#task-20 2026-08-04] base text + an actor-status addendum naming the local amban / garrison when
	# present (resolved in the immediate by SUBJ_QING_resolve_integ_actors). Mutually exclusive on presence.
	desc = {
		desc = "qing_integ.41.desc"
		first_valid = {
			triggered_desc = { trigger = { has_variable = integ_amban_present  has_variable = integ_garrison_size  var:integ_garrison_size >= 1 }  desc = "qing_integ.41.desc_actors_both" }
			triggered_desc = { trigger = { has_variable = integ_amban_present }  desc = "qing_integ.41.desc_actors_amban" }
			triggered_desc = { trigger = { has_variable = integ_garrison_size  var:integ_garrison_size >= 1 }  desc = "qing_integ.41.desc_actors_garrison" }
			triggered_desc = { trigger = { always = yes }  desc = "qing_integ.41.desc_actors_none" }
		}
	}
	picture = looting

	left_portrait = root.current_ruler
	right_portrait = scope:target.current_ruler

	goto_location = scope:resistance_province


	# [#111/#112 review-fix] Re-validate at fire time (mirrors qing_integ.30's trigger).
	# qing_integ.40.a queues this at days{10 30} AND sets SUBJ_integration_progress=5,
	# which also arms the capstone pulse to fire qing_integ.30 (absorb) at days{3 10}.
	# If .30 resolves first it annexes the subject via SUBJ_QING_absorb_subject; a
	# queued .41 would then fire on an already-incorporated, landless subject — its
	# any_owned_province guard false, scope:resistance_province never saved, and
	# goto_location dangling — presenting a resistance crisis for a dependency that no
	# longer exists. This trigger discards the stale .41 in that race.
	trigger = {
		tag = CHI
		exists = scope:target
		scope:target = { is_subject_of = ROOT }
		scope:target = { any_owned_province = { count >= 1 } }
		scope:target = { has_variable = SUBJ_integration_active }
	}

	immediate = {
		LOG_line = { sys = SUBJ  msg = "EVENT qing_integ.41 (resistance) overlord=[ROOT.GetTag] subject=[scope:target.GetTag]" }
		# Anchor the event on the subject's most populous province.
		if = {
			limit = { scope:target = { any_owned_province = { count >= 1 } } }
			scope:target = {
				ordered_owned_province = {
					order_by = total_population
					max = 1
					save_scope_as = resistance_province
				}
			}
		}
		# [#task-8] resolve the two on-the-ground actors (resident amban + imperial garrison)
		# whose presence, loyalty, and relative strength gate the options below.
		SUBJ_QING_resolve_integ_actors = yes
	}

	# --- Conciliate: roll back the decree and restore autonomy -----------
	# The Kangxi model after the Three Feudatories: when Wu Sangui's grandson
	# surrendered (1681), the court granted him a lesser title and left him
	# in Yunnan rather than executing him outright. Restore face, slow the
	# process, keep the peace.
	option = {
		name = "qing_integ.41.a"
		custom_tooltip = "qing_integ.41.a.tt"
		# Roll back integration progress by 2 steps
		SUBJ_QING_reduce_integration = { steps = 2 }
		add_stability = 3
		add_legitimacy = -10
		# Grant the subject a temporary autonomy boost
		add_country_modifier = {
			name = qing_integ_conciliation_granted
			duration = 1095
		}
		# The subject's ruler gains loyalty and prominence
		if = {
			limit = { exists = scope:target.current_ruler }
			scope:target.current_ruler = {
				add_loyalty = loyalty_qing_delta_p15
				add_prominence = 10
			}
		}
		SUBJ_QING_clear_integ_actors = yes
	}

	# --- [#task-8] The resident AMBAN negotiates a settlement -------------
	# Shown only when a STAUNCH resident amban (loyalty > 70) sits in the dependency: he knows
	# the chieftain and can talk him down. A cheap de-escalation — a modest rollback, peace kept,
	# and the amban's standing rises for the service. Absent / wavering amban => option hidden
	# (and the "amban flees" flavour fires under the crush/fortress branches instead).
	option = {
		name = "qing_integ.41.d"
		custom_tooltip = "qing_integ.41.d.tt"
		trigger = {
			has_variable = integ_amban_present
			var:integ_amban_band = 2
			exists = scope:integ_amban
		}
		SUBJ_QING_reduce_integration = { steps = 1 }
		add_stability = 2
		# the amban spends his personal standing to broker the peace
		scope:integ_amban = {
			add_prominence = 10
			add_popularity = 5
		}
		if = {
			limit = { exists = scope:target.current_ruler }
			scope:target.current_ruler = { add_loyalty = loyalty_qing_delta_p10 }
		}
		LOG_line = { sys = SUBJ  msg = "resistance NEGOTIATED by resident amban subject=[scope:target.GetTag]" }
		SUBJ_QING_clear_integ_actors = yes
	}

	# --- [#task-20 garrison actor] The local GARRISON marches out and breaks the rising ----
	# Shown only when an imperial garrison strong enough to take the field (edge >= 1) already sits
	# on the dependency's soil — including the case of MULTIPLE garrisons stacking (e.g. the two in
	# Yili), which the resolver sums into the edge. The banner troops are ALREADY there, so this puts
	# the revolt down IN THE FIELD at once: no re-subjugation war to declare, no releasing the subject,
	# and far less tyranny/AE than the full expedition (.41.b) that a garrison-less throne must mount.
	# A DOMINANT garrison (edge 2) crushes it cleanly; rough parity (edge 1) still wins but bloodier.
	# The subject stays a subject (integration is NOT rolled back — the rising is simply broken), and
	# the garrison commander's standing rises for the service. This is the garrison's OWN action, the
	# counterpart to the amban's .41.d — distinct from .41.b (the from-scratch war when no garrison can
	# do the job).
	option = {
		name = "qing_integ.41.e"
		custom_tooltip = "qing_integ.41.e.tt"
		trigger = {
			has_variable = integ_garrison_edge
			var:integ_garrison_edge >= 1
		}
		add_tyranny = 3
		add_aggressive_expansion = 3
		# [cost 2026-08-04] no GOLD — like the expedition, the garrison crush is paid in STABILITY and
		# LEGITIMACY, but LIGHTER than .41.b: the troops are already on the spot, no war is declared,
		# and the subject is held. A dominant garrison breaks it cleanly (small toll); rough parity
		# wins but bloodier (larger toll).
		if = {
			limit = { var:integ_garrison_edge = 2 }
			add_stability = -1
			add_legitimacy = -3
			add_war_exhaustion = 1
			LOG_line = { sys = SUBJ  msg = "resistance BROKEN in the field by a dominant local garrison, subject=[scope:target.GetTag]" }
		}
		else = {
			add_stability = -3
			add_legitimacy = -6
			add_war_exhaustion = 2
			LOG_line = { sys = SUBJ  msg = "resistance broken in the field by the local garrison (hard-fought), subject=[scope:target.GetTag]" }
		}
		# the rising is put down where it stands; local unrest lingers but the subject is held.
		if = {
			limit = { exists = scope:resistance_province }
			scope:resistance_province = { add_province_modifier = { name = local_unrest_harsh  duration = 730 } }
		}
		# the subject resents the garrison's heavy hand, but the bond holds (a standing occupation-
		# style resentment, not the full re-subjugation aftermath of .41.b).
		add_country_modifier = { name = qing_integ_frontier_resentment  duration = 1095 }
		LOG_line = { sys = SUBJ  msg = "resistance CRUSHED-IN-PLACE by garrison (no war, subject held) subject=[scope:target.GetTag]" }
		SUBJ_QING_clear_integ_actors = yes
	}

	# --- Crush: let the revolt become open war, and break it by arms ---
	# [#task-7; flood-fix 2026-08-04] Now a REAL war via the VANILLA independence path: the subject
	# declares its war of independence on the throne (SUBJ_QING_crush_revolt_war → the subject-scope bare
	# declare_war_with_wargoal { independence_wargoal, target = ROOT }, upstream epigoni idiom). NO
	# release_subject and NO conquer war — the dependency stays a subject unless it WINS; the player
	# crushes the rising by winning the defence. (The prior release+conquer+reparent implementation
	# flooded the boot log 858k lines — removed.)
	# [#task-8] This is the from-scratch war for when no garrison marches out (that path is .41.e above).
	# Always available (no trigger) as the hard option. A wavering / absent amban abandons his post as
	# the fighting starts.
	option = {
		name = "qing_integ.41.b"
		custom_tooltip = "qing_integ.41.b.tt"
		# [#task-20] the open war of independence. The garrison-marches-out path is now its OWN option
		# (.41.e); this is the full field campaign the throne fights when the revolt flares into open war
		# (the subject declares independence; the player wins the defence to crush it). [cost 2026-08-04]
		# The crush costs no GOLD directly — its price is paid in STABILITY and LEGITIMACY (plus tyranny /
		# aggressive expansion / war exhaustion), the political toll of the war, not a treasury debit.
		add_war_exhaustion = 3
		add_tyranny = 10
		add_aggressive_expansion = 10
		add_stability = -5
		add_legitimacy = -15
		LOG_line = { sys = SUBJ  msg = "crush: a full field expedition is mounted (re-subjugation war), subject=[scope:target.GetTag]" }
		# a wavering / absent amban abandons his post when the shooting starts (flavour + a small ding)
		if = {
			limit = {
				OR = {
					NOT = { has_variable = integ_amban_present }
					var:integ_amban_band = 0
				}
			}
			add_legitimacy = -3
			LOG_line = { sys = SUBJ  msg = "crush: no steady amban on the spot — the resident flees his post, subject=[scope:target.GetTag]" }
		}
		# Local unrest in the rebel heartland (captured before the subject is freed).
		if = {
			limit = { exists = scope:resistance_province }
			scope:resistance_province = {
				add_province_modifier = {
					name = local_unrest_harsh
					duration = 1095
				}
			}
		}
		# [#task-7] the actual war: the subject declares its war of independence on the throne (no
		# release, no conquer — vanilla independence path; see SUBJ_QING_crush_revolt_war).
		SUBJ_QING_crush_revolt_war = yes
		LOG_line = { sys = SUBJ  msg = "resistance CRUSHED (real war declared) subject=[scope:target.GetTag]" }
		SUBJ_QING_clear_integ_actors = yes
	}

	# --- Co-opt: bribe the chieftain, marry into his family, grant titles -
	# The Yongzheng rou yuan (柔遠, "soothe the distant") model: instead of
	# force, co-opt the native elite via the affinity spine. This option
	# requires a viable courtier (the subject's ruler) and the gold to endow
	# him. If he is already loyal, this cements the bond; if disloyal, it
	# may turn him.
	option = {
		name = "qing_integ.41.c"
		custom_tooltip = "qing_integ.41.c.tt"
		trigger = {
			treasury >= 100
			exists = scope:target.current_ruler
			scope:target.current_ruler = { is_alive = yes }
		}
		add_treasury = -100
		# Cultivate the chieftain via the affinity spine: endow with gold,
		# honour his family with prestige, and (if he is not already loyal)
		# nudge his loyalty. The existing QING_char_* mutators are for
		# CHI courtiers; the subject ruler is not a CHI courtier, so we
		# apply the effects directly here.
		scope:target.current_ruler = {
			add_loyalty = loyalty_qing_delta_p20
			add_prominence = 15
			add_gold = 50
			# Give him a character modifier
			add_character_modifier = {
				name = qing_integ_coopted_chieftain
				duration = 1825
			}
		}
		# [#task-7] "marry into his family" made LITERAL: wed an imperial prince to the chieftain's
		# daughter. Fully guarded — if either house lacks an eligible child the co-opt still lands
		# on the gold/prestige above; only the wedding is skipped.
		SUBJ_QING_marry_into_chieftain = yes
		# The subject itself gains a relationship buff
		add_country_modifier = {
			name = qing_integ_imperial_favor
			duration = 1460
		}
		# Integration proceeds smoothly (no setback)
		LOG_line = { sys = SUBJ  msg = "resistance CO-OPTED subject=[scope:target.GetTag] chieftain=[scope:target.current_ruler.GetName]" }
		SUBJ_QING_clear_integ_actors = yes
	}
