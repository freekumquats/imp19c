########################################################################
# qing_greatgame_events.txt — the Zongli Yamen / Great Game events (#108).
#
# The DRAMA of 以夷制夷 ("use barbarians to control barbarians"): the Zongli holder
# mediates great-power overtures/threats, and the player plays the powers off each
# other. Gated on: tag=CHI + Zongli office existing + some GP tension present.
#
# These are WEIGHTED entries in the QING_frontier_flavour_roll dispatcher
# (se_QING_DECLINE.txt) — not dated, not on_action. They fire alongside the rest
# of the Qing pulse-driven flavour (reforms, embassies, rebellions) when the
# conditions align.
#
# The holder's skill/loyalty gates the OPTIONS: a capable holder opens the "play
# one power against another" finesse path; a weak / absent holder forces costly
# concessions. The affinity spine (QING_char_affinity / QING_char_bind) is used
# where the Zongli holder is a concrete character involved in the decision.
#
# Namespace = qing_greatgame
########################################################################

namespace = qing_greatgame

# ======================================================================
# .1 — A GREAT POWER OVERTURE / THREAT. One of the three powers (Britain / France
# / Russia) whose tension is elevated opens a diplomatic overture or presses a
# demand — a frontier concession, a trade privilege, a diplomatic slight redressed.
# The Zongli holder (if present + capable) can PLAY the power's two rivals against
# it (以夷制夷: placate one at the cost of affronting another); or the Qing can
# concede (lower tension with the pressing power) or stand firm (tension rises).
#
# The power is chosen RANDOMLY from those whose tension is >= 40 (the band where
# overtures / threats become plausible). If multiple are high, the highest is
# picked with a weight bias.
# ======================================================================
qing_greatgame.1 = {
	type = country_event
	title = "qing_greatgame.1.t"
	desc = "qing_greatgame.1.desc"
	picture = chinese_throne_room
	left_portrait = root.current_ruler
	right_portrait = scope:qing_gg_zongli

	trigger = {
		tag = CHI
		# at least one GP tension counter is elevated (the band where overtures begin)
		OR = {
			AND = { has_variable = qing_gp_tension_britain  var:qing_gp_tension_britain >= 40 }
			AND = { has_variable = qing_gp_tension_france   var:qing_gp_tension_france  >= 40 }
			AND = { has_variable = qing_gp_tension_russia   var:qing_gp_tension_russia  >= 40 }
		}
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_greatgame.1 (GP overture/threat)" }
		# Save the Zongli holder (if present) — gates the finesse path.
		if = {
			limit = {
				has_variable = qing_office_zongli_holder
				var:qing_office_zongli_holder = { is_alive = yes  employer = ROOT }
			}
			var:qing_office_zongli_holder = { save_scope_as = qing_gg_zongli }
		}
		# Pick the pressing power: the HIGHEST tension counter among those >= 40.
		# Weighted random if multiple are tied.
		if = {
			limit = {
				has_variable = qing_gp_tension_britain
				var:qing_gp_tension_britain >= 40
				OR = {
					NOT = { has_variable = qing_gp_tension_france }
					var:qing_gp_tension_britain >= qing_gp_tension_france_cmpsvalue
				}
				OR = {
					NOT = { has_variable = qing_gp_tension_russia }
					var:qing_gp_tension_britain >= qing_gp_tension_russia_cmpsvalue
				}
			}
			set_variable = { name = qing_gg_pressing_power  value = flag:britain }
		}
		else_if = {
			limit = {
				has_variable = qing_gp_tension_france
				var:qing_gp_tension_france >= 40
				OR = {
					NOT = { has_variable = qing_gp_tension_russia }
					var:qing_gp_tension_france >= qing_gp_tension_russia_cmpsvalue
				}
			}
			set_variable = { name = qing_gg_pressing_power  value = flag:france }
		}
		else = {
			set_variable = { name = qing_gg_pressing_power  value = flag:russia }
		}
	}

	# OPTION A — PLAY THE RIVALS (以夷制夷). Gate on the Zongli holder being
	# present + capable (finesse + charisma high, loyalty >= 50). Placate the
	# pressing power at the cost of affronting its two rivals (they see the Qing
	# taking a side). Cost: prestige (the court resents a compromise).
	option = {
		name = "qing_greatgame.1.a"
		custom_tooltip = "qing_greatgame.1.a.tt"
		trigger = {
			exists = scope:qing_gg_zongli
			scope:qing_gg_zongli = {
				loyalty >= 50
				finesse >= 6
				charisma >= 6
			}
		}
		current_ruler = { add_popularity = -8 }
		# Accommodate the pressing power (lower its tension) + improve relations.
		if = {
			limit = { var:qing_gg_pressing_power = flag:britain }
			QING_gp_accommodate = { power = britain  tag = GBR  amount = -12 }
			# Affront its two rivals (they bristle at the Qing's partisanship).
			QING_gp_rivals_bristle = { rival_a = france  rtag_a = FRA  rival_b = russia  rtag_b = RUS }
		}
		else_if = {
			limit = { var:qing_gg_pressing_power = flag:france }
			QING_gp_accommodate = { power = france  tag = FRA  amount = -12 }
			QING_gp_rivals_bristle = { rival_a = britain  rtag_a = GBR  rival_b = russia  rtag_b = RUS }
		}
		else = {
			QING_gp_accommodate = { power = russia  tag = RUS  amount = -12 }
			QING_gp_rivals_bristle = { rival_a = britain  rtag_a = GBR  rival_b = france  rtag_b = FRA }
		}
		# The Zongli holder's finesse is vindicated — reward him via the affinity
		# spine (the same idiom as QING_acc_reward).
		scope:qing_gg_zongli = {
			QING_char_promote_standing = { pop = 3  prom = 2 }
			QING_char_affinity = yes
			if = {
				limit = { var:qing_char_affinity >= 55 }
				QING_char_bind = yes
			}
		}
	}

	# OPTION B — CONCEDE. Lower tension with the pressing power at the cost of
	# treasury + prestige. The safe, low-risk path (the Qing buys peace).
	option = {
		name = "qing_greatgame.1.b"
		custom_tooltip = "qing_greatgame.1.b.tt"
		add_treasury = -310
		TREASURY_LOG_it = { amount = -310 }
		current_ruler = { add_popularity = -12 }
		if = {
			limit = { var:qing_gg_pressing_power = flag:britain }
			QING_gp_accommodate = { power = britain  tag = GBR  amount = -15 }
		}
		else_if = {
			limit = { var:qing_gg_pressing_power = flag:france }
			QING_gp_accommodate = { power = france  tag = FRA  amount = -15 }
		}
		else = {
			QING_gp_accommodate = { power = russia  tag = RUS  amount = -15 }
		}
	}

	# OPTION C — STAND FIRM. Refuse the demand; tension with the pressing power
	# rises. Gains prestige (the court applauds defiance) but risks a crisis.
	option = {
		name = "qing_greatgame.1.c"
		custom_tooltip = "qing_greatgame.1.c.tt"
		current_ruler = { add_popularity = 10 }
		if = {
			limit = { var:qing_gg_pressing_power = flag:britain }
			QING_DECLINE_nudge = { var = qing_gp_tension_britain  amount = 8 }
		}
		else_if = {
			limit = { var:qing_gg_pressing_power = flag:france }
			QING_DECLINE_nudge = { var = qing_gp_tension_france  amount = 8 }
		}
		else = {
			QING_DECLINE_nudge = { var = qing_gp_tension_russia  amount = 8 }
		}
	}
}

# ======================================================================
# .2 — THE POWERS COORDINATE AGAINST THE QING. When MULTIPLE tensions are high,
# the powers sense a common opportunity and press together — a coordinated threat
# that is harder to deflect. The Qing can play them off (if the Zongli holder is
# capable) or buy them off (expensive) or weather the storm (all tensions rise).
#
# Trigger: at least TWO of the three GP tensions >= 55 (the coordination band).
# ======================================================================
qing_greatgame.2 = {
	type = country_event
	title = "qing_greatgame.2.t"
	desc = "qing_greatgame.2.desc"
	picture = senate_debate
	left_portrait = root.current_ruler
	right_portrait = scope:qing_gg_zongli

	trigger = {
		tag = CHI
		# at least TWO tensions >= 55
		calc_true_if = {
			amount >= 2
			AND = { has_variable = qing_gp_tension_britain  var:qing_gp_tension_britain >= 55 }
			AND = { has_variable = qing_gp_tension_france   var:qing_gp_tension_france  >= 55 }
			AND = { has_variable = qing_gp_tension_russia   var:qing_gp_tension_russia  >= 55 }
		}
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_greatgame.2 (powers coordinate)" }
		if = {
			limit = {
				has_variable = qing_office_zongli_holder
				var:qing_office_zongli_holder = { is_alive = yes  employer = ROOT }
			}
			var:qing_office_zongli_holder = { save_scope_as = qing_gg_zongli }
		}
	}

	# OPTION A — DIVIDE THEM (以夷制夷 at full strength). Gate on the Zongli
	# holder being present + VERY capable (the coordination is harder to break).
	# Play two powers against the third: lower tension with two, raise with one.
	option = {
		name = "qing_greatgame.2.a"
		custom_tooltip = "qing_greatgame.2.a.tt"
		trigger = {
			exists = scope:qing_gg_zongli
			scope:qing_gg_zongli = {
				loyalty >= 60
				finesse >= 8
				charisma >= 8
			}
		}
		current_ruler = { add_popularity = -10 }
		# Placate Britain + France (send embassies); affront Russia (it is left out).
		QING_gp_accommodate = { power = britain  tag = GBR  amount = -10 }
		QING_gp_accommodate = { power = france   tag = FRA  amount = -10 }
		QING_DECLINE_nudge = { var = qing_gp_tension_russia  amount = 6 }
		# Reward the Zongli holder (a diplomatic coup).
		scope:qing_gg_zongli = {
			QING_char_promote_standing = { pop = 6  prom = 4 }
			QING_char_affinity = yes
			if = {
				limit = { var:qing_char_affinity >= 55 }
				QING_char_bind = yes
			}
		}
	}

	# OPTION B — BUY THEM OFF. Pay a heavy treasury cost to lower tension with all
	# three (a bribe / settlement that placates the coordinated threat).
	option = {
		name = "qing_greatgame.2.b"
		custom_tooltip = "qing_greatgame.2.b.tt"
		add_treasury = -440
		TREASURY_LOG_it = { amount = -440 }
		current_ruler = { add_popularity = -15 }
		QING_gp_accommodate = { power = britain  tag = GBR  amount = -12 }
		QING_gp_accommodate = { power = france   tag = FRA  amount = -12 }
		QING_gp_accommodate = { power = russia   tag = RUS  amount = -12 }
	}

	# OPTION C — WEATHER THE STORM. Refuse; all three tensions rise (the powers
	# press their coordinated advantage). Gains prestige but risks crisis.
	option = {
		name = "qing_greatgame.2.c"
		custom_tooltip = "qing_greatgame.2.c.tt"
		current_ruler = { add_popularity = 12 }
		QING_DECLINE_nudge = { var = qing_gp_tension_britain  amount = 6 }
		QING_DECLINE_nudge = { var = qing_gp_tension_france   amount = 6 }
		QING_DECLINE_nudge = { var = qing_gp_tension_russia   amount = 6 }
	}
}

# ======================================================================
# .3 — A ZONGLI HOLDER'S DIPLOMATIC COUP. When the Zongli holder is capable + loyal
# AND at least one GP tension is elevated, he engineers a breakthrough — a treaty /
# settlement that meaningfully lowers tension with one power. Reward event (low
# weight, high payoff). Gated on the holder's charisma + finesse.
# ======================================================================
qing_greatgame.3 = {
	type = country_event
	title = "qing_greatgame.3.t"
	desc = "qing_greatgame.3.desc"
	picture = senate
	left_portrait = root.current_ruler
	right_portrait = scope:qing_gg_zongli

	trigger = {
		tag = CHI
		# Zongli holder exists + is capable
		has_variable = qing_office_zongli_holder
		var:qing_office_zongli_holder = {
			is_alive = yes
			employer = ROOT
			loyalty >= 60
			finesse >= 8
			charisma >= 8
		}
		# at least one GP tension >= 35 (there is a frontier to settle)
		OR = {
			AND = { has_variable = qing_gp_tension_britain  var:qing_gp_tension_britain >= 35 }
			AND = { has_variable = qing_gp_tension_france   var:qing_gp_tension_france  >= 35 }
			AND = { has_variable = qing_gp_tension_russia   var:qing_gp_tension_russia  >= 35 }
		}
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_greatgame.3 (Zongli coup)" }
		var:qing_office_zongli_holder = { save_scope_as = qing_gg_zongli }
		# Pick the highest tension counter (the frontier the holder settles).
		if = {
			limit = {
				has_variable = qing_gp_tension_britain
				var:qing_gp_tension_britain >= 35
				OR = {
					NOT = { has_variable = qing_gp_tension_france }
					var:qing_gp_tension_britain >= qing_gp_tension_france_cmpsvalue
				}
				OR = {
					NOT = { has_variable = qing_gp_tension_russia }
					var:qing_gp_tension_britain >= qing_gp_tension_russia_cmpsvalue
				}
			}
			set_variable = { name = qing_gg_settled_power  value = flag:britain }
		}
		else_if = {
			limit = {
				has_variable = qing_gp_tension_france
				var:qing_gp_tension_france >= 35
				OR = {
					NOT = { has_variable = qing_gp_tension_russia }
					var:qing_gp_tension_france >= qing_gp_tension_russia_cmpsvalue
				}
			}
			set_variable = { name = qing_gg_settled_power  value = flag:france }
		}
		else = {
			set_variable = { name = qing_gg_settled_power  value = flag:russia }
		}
	}

	# OPTION A — APPLAUD THE COUP. The throne endorses the settlement; tension
	# drops meaningfully, and the Zongli holder is rewarded (prestige, standing,
	# loyalty). The high-payoff path.
	option = {
		name = "qing_greatgame.3.a"
		custom_tooltip = "qing_greatgame.3.a.tt"
		current_ruler = { add_popularity = 10 }
		if = {
			limit = { var:qing_gg_settled_power = flag:britain }
			QING_gp_accommodate = { power = britain  tag = GBR  amount = -20 }
		}
		else_if = {
			limit = { var:qing_gg_settled_power = flag:france }
			QING_gp_accommodate = { power = france  tag = FRA  amount = -20 }
		}
		else = {
			QING_gp_accommodate = { power = russia  tag = RUS  amount = -20 }
		}
		# Reward the holder (a major diplomatic triumph).
		scope:qing_gg_zongli = {
			# [BT-5/6 review-fix] CHARACTER scope (qing_gg_zongli = the Zongli holder) — reward must be
			# add_popularity, NOT add_legitimacy (country-only). The bulk sweep misclassified this bare line as
			# country-scope; add_legitimacy on a character errors/no-ops and the triumph reward was lost.
			add_popularity = 15
			QING_char_promote_standing = { pop = 8  prom = 5 }
			QING_char_affinity = yes
			QING_char_bind = yes
		}
	}

	# OPTION B — TEMPER THE COUP. The throne accepts the settlement but does not
	# publicly applaud (the court is split). Tension drops, but the holder gains
	# only modest reward.
	option = {
		name = "qing_greatgame.3.b"
		custom_tooltip = "qing_greatgame.3.b.tt"
		if = {
			limit = { var:qing_gg_settled_power = flag:britain }
			QING_gp_accommodate = { power = britain  tag = GBR  amount = -15 }
		}
		else_if = {
			limit = { var:qing_gg_settled_power = flag:france }
			QING_gp_accommodate = { power = france  tag = FRA  amount = -15 }
		}
		else = {
			QING_gp_accommodate = { power = russia  tag = RUS  amount = -15 }
		}
		scope:qing_gg_zongli = {
			QING_char_promote_standing = { pop = 4  prom = 2 }
		}
	}
}

# ======================================================================
# .4 — [P3 2026-07-24] EXCHANGE OF LEGATIONS (互派使節). The missing HOME of the
# orphaned QING_fbuild_embassy effect (se_QING_FOREIGNBUILD.txt): it plants a resident
# Qing legation (使館) in a great power's capital and grants the standing
# qing_embassy_representation modifier — but nothing called it. This event is that call.
#
# The Zongli Yamen's whole purpose after 1861 was to conduct RESIDENT diplomacy on the
# Westphalian model the Qing had refused at Macartney/Amherst. When tension with a power
# has been brought DOWN into a workable band (a détente the holder can build on), the court
# may exchange permanent legations — the concrete institution of the modern diplomacy the
# reformers wanted. This is a DÉTENTE payoff, so it fires when a pressing power's tension is
# moderate (25..55: cooled enough to talk, warm enough to matter) and the Zongli office is
# filled by a living holder. Picks that power, opens the legation, deepens the détente.
#
# Trigger: CHI, Zongli holder alive+employed, some power in the détente band, and we do NOT
# already hold the representation modifier (the legation is opened once — the modifier is a
# standing asset, not per-power, matching QING_fbuild_embassy's own single-grant guard).
# ======================================================================
qing_greatgame.4 = {
	type = country_event
	title = "qing_greatgame.4.t"
	desc = "qing_greatgame.4.desc"
	picture = senate
	left_portrait = root.current_ruler
	right_portrait = scope:qing_gg_zongli

	trigger = {
		tag = CHI
		has_variable = qing_office_zongli_holder
		var:qing_office_zongli_holder = { is_alive = yes  employer = ROOT }
		# the legation is a one-time standing asset — don't offer it if already opened.
		NOT = { has_country_modifier = qing_embassy_representation }
		# some power sits in the détente band (cooled enough to talk).
		OR = {
			AND = { has_variable = qing_gp_tension_britain  var:qing_gp_tension_britain >= 25  var:qing_gp_tension_britain <= 55  exists = c:GBR }
			AND = { has_variable = qing_gp_tension_france   var:qing_gp_tension_france  >= 25  var:qing_gp_tension_france  <= 55  exists = c:FRA }
			AND = { has_variable = qing_gp_tension_russia   var:qing_gp_tension_russia  >= 25  var:qing_gp_tension_russia  <= 55  exists = c:RUS }
		}
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_greatgame.4 (exchange of legations)" }
		var:qing_office_zongli_holder = { save_scope_as = qing_gg_zongli }
		# pick the détente-band power and save it BOTH as a flag (for tension routing) and as a
		# country scope (QING_fbuild_embassy needs a country scope with a capital). Britain first,
		# then France, then Russia — same priority order as the other greatgame pickers.
		if = {
			limit = { has_variable = qing_gp_tension_britain  var:qing_gp_tension_britain >= 25  var:qing_gp_tension_britain <= 55  exists = c:GBR }
			set_variable = { name = qing_gg_legation_power  value = flag:britain }
			c:GBR = { save_scope_as = qing_gg_legation_ctry }
		}
		else_if = {
			limit = { has_variable = qing_gp_tension_france  var:qing_gp_tension_france >= 25  var:qing_gp_tension_france <= 55  exists = c:FRA }
			set_variable = { name = qing_gg_legation_power  value = flag:france }
			c:FRA = { save_scope_as = qing_gg_legation_ctry }
		}
		else = {
			set_variable = { name = qing_gg_legation_power  value = flag:russia }
			c:RUS = { save_scope_as = qing_gg_legation_ctry }
		}
	}

	# OPTION A — EXCHANGE LEGATIONS (互派使節). Open the resident Qing legation in the power's
	# capital (QING_fbuild_embassy — the standing representation modifier + the on-map building),
	# and deepen the détente (tension down further). The modern-diplomacy path the reformers wanted:
	# it BREAKS the tribute worldview (a small reform-pressure rise, the conservative unease) but
	# wins a lasting diplomatic asset. Rewards the Zongli holder.
	option = {
		name = "qing_greatgame.4.a"
		custom_tooltip = "qing_greatgame.4.a.tt"
		# plant the resident legation in the chosen power's capital (the orphaned effect's home).
		QING_fbuild_embassy = { power = scope:qing_gg_legation_ctry }
		# deepen the détente with that power.
		if = {
			limit = { var:qing_gg_legation_power = flag:britain }
			QING_gp_accommodate = { power = britain  tag = GBR  amount = -12 }
		}
		else_if = {
			limit = { var:qing_gg_legation_power = flag:france }
			QING_gp_accommodate = { power = france  tag = FRA  amount = -12 }
		}
		else = {
			QING_gp_accommodate = { power = russia  tag = RUS  amount = -12 }
		}
		# accepting resident foreign diplomacy unsettles the tribute-world conservatives.
		QING_DECLINE_nudge = { var = qing_reform_pressure  amount = 3 }
		scope:qing_gg_zongli = {
			QING_char_promote_standing = { pop = 5  prom = 3 }
			QING_char_affinity = yes
		}
	}

	# OPTION B — HOLD TO THE TRIBUTE FORMS (守朝貢舊制). Decline the resident legation; the
	# court keeps the old ceremonial distance. No legation asset, but the conservatives are
	# vindicated (reform pressure eases, a little legitimacy on the Mandate's dignity), and the
	# détente does not advance (tension unchanged).
	option = {
		name = "qing_greatgame.4.b"
		custom_tooltip = "qing_greatgame.4.b.tt"
		QING_DECLINE_nudge = { var = qing_reform_pressure  amount = -3 }
		add_legitimacy = 6
	}
}

# .5 — [P4 2026-07-24] THE QING DICTATES TERMS (反客為主). The treaty port as an instrument the
# Qing WIELDS, not only suffers — the two-sided rework of the treaty-port model. When the Qing
# stands strong (high legitimacy + stability) and a great power's enmity is HIGH (tension >= 55,
# a power that has pressed the Qing), a moment arrives to turn the tables (反客為主 "the guest
# becomes the host"): force a commercial CONCESSION PORT onto THAT power's own coast. The Qing
# gains the standing concession advantage (qing_treaty_port_concession) and the power bears the
# humiliation — the mirror image of the unequal treaties, playable because the mechanic is now
# symmetric (QING_treaty_impose_concession).
#
# This is deliberately a HIGH BAR (a strong Qing humbling a hostile power is alternate-history,
# not the historical norm) and only surfaces where the target power actually has a coastal
# province to concede. Gated: CHI, legitimacy + stability high, a hostile power (tension>=55)
# that exists and holds an un-stamped coastal city.
# ======================================================================
qing_greatgame.5 = {
	type = country_event
	title = "qing_greatgame.5.t"
	desc = "qing_greatgame.5.desc"
	picture = chinese_throne_room
	left_portrait = root.current_ruler

	trigger = {
		tag = CHI
		# a strong throne dictating terms — both are engine country values (no var guard needed).
		legitimacy >= 60
		stability >= 50
		# a hostile power with a coastal province to concede.
		OR = {
			AND = { has_variable = qing_gp_tension_britain  var:qing_gp_tension_britain >= 55  exists = c:GBR  c:GBR = { any_owned_province = { is_coastal = yes  NOT = { has_building = qing_treaty_port_building } } } }
			AND = { has_variable = qing_gp_tension_france   var:qing_gp_tension_france  >= 55  exists = c:FRA  c:FRA = { any_owned_province = { is_coastal = yes  NOT = { has_building = qing_treaty_port_building } } } }
			AND = { has_variable = qing_gp_tension_russia   var:qing_gp_tension_russia  >= 55  exists = c:RUS  c:RUS = { any_owned_province = { is_coastal = yes  NOT = { has_building = qing_treaty_port_building } } } }
		}
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_greatgame.5 (Qing dictates terms)" }
		# pick the hostile power (highest-tension among the eligible), save it as a country scope,
		# and pull its highest-population un-stamped coastal city into a province scope.
		if = {
			limit = { has_variable = qing_gp_tension_britain  var:qing_gp_tension_britain >= 55  exists = c:GBR  c:GBR = { any_owned_province = { is_coastal = yes  NOT = { has_building = qing_treaty_port_building } } } }
			set_variable = { name = qing_gg_dictate_power  value = flag:britain }
			c:GBR = { save_scope_as = qing_gg_dictate_ctry }
		}
		else_if = {
			limit = { has_variable = qing_gp_tension_france  var:qing_gp_tension_france >= 55  exists = c:FRA  c:FRA = { any_owned_province = { is_coastal = yes  NOT = { has_building = qing_treaty_port_building } } } }
			set_variable = { name = qing_gg_dictate_power  value = flag:france }
			c:FRA = { save_scope_as = qing_gg_dictate_ctry }
		}
		else = {
			set_variable = { name = qing_gg_dictate_power  value = flag:russia }
			c:RUS = { save_scope_as = qing_gg_dictate_ctry }
		}
		# the concession city: the target's most-populous un-stamped coastal province.
		scope:qing_gg_dictate_ctry = {
			ordered_owned_province = {
				limit = { is_coastal = yes  NOT = { has_building = qing_treaty_port_building } }
				order_by = total_population
				max = 1
				save_scope_as = qing_gg_dictate_prov
			}
		}
	}

	# OPTION A — DICTATE THE CONCESSION (反客為主). Force the treaty port onto the power's coast:
	# the Qing takes the standing concession advantage, the power bears the humiliation, tension
	# with it SPIKES (a coerced power resents it), and the court exults at the reversal.
	option = {
		name = "qing_greatgame.5.a"
		custom_tooltip = "qing_greatgame.5.a.tt"
		trigger = { exists = scope:qing_gg_dictate_prov }
		QING_treaty_impose_concession = { victim = scope:qing_gg_dictate_ctry  prov = scope:qing_gg_dictate_prov }
		current_ruler = { add_popularity = 20 }
		add_legitimacy = 8
		# the coerced power's enmity spikes.
		if = {
			limit = { var:qing_gg_dictate_power = flag:britain }
			QING_DECLINE_nudge = { var = qing_gp_tension_britain  amount = 12 }
		}
		else_if = {
			limit = { var:qing_gg_dictate_power = flag:france }
			QING_DECLINE_nudge = { var = qing_gp_tension_france  amount = 12 }
		}
		else = {
			QING_DECLINE_nudge = { var = qing_gp_tension_russia  amount = 12 }
		}
	}

	# OPTION B — STAY THE HAND (以德服人). Forgo the coercion; win no concession but no fresh enmity.
	# The court styles it magnanimity (以德服人 "win men by virtue") — a little legitimacy, and the
	# power's tension eases slightly (restraint noticed).
	option = {
		name = "qing_greatgame.5.b"
		custom_tooltip = "qing_greatgame.5.b.tt"
		add_legitimacy = 4
		if = {
			limit = { var:qing_gg_dictate_power = flag:britain }
			QING_DECLINE_nudge = { var = qing_gp_tension_britain  amount = -6 }
		}
		else_if = {
			limit = { var:qing_gg_dictate_power = flag:france }
			QING_DECLINE_nudge = { var = qing_gp_tension_france  amount = -6 }
		}
		else = {
			QING_DECLINE_nudge = { var = qing_gp_tension_russia  amount = -6 }
		}
	}
}
