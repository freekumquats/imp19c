########################################################################
# qing_ideology_events.txt — [#61 2026-07-23] IDEOLOGY ADOPTION events (Design 3 — Religion → Ideology).
#
# The TRAMPOLINE for the layered ideology-adoption flow. QING_adopt_ideology (se_QING_IDEOLOGY.txt)
# runs set_country_religion + set_character_religion + a bounded set_pop_religion loop — a heavy-ish
# chain that must NEVER be compile-inlined into a scripted_gui button (the #443 compile-inline crash
# class). So a button/decision fires THIS is_triggered_only event (a runtime ref, never inlined), and
# the event runs the effect. Each option is one ideology; the option's trigger gates on era/reform-state
# so the creed cannot be adopted anachronistically at the 1763 start.
#
# All events hidden/is_triggered_only. Namespace = qing_ideology. ROOT = the adopting country (CHI).
# Sources: Design 3 digest (the thinker roster + adoption model, §A.3).
########################################################################

namespace = qing_ideology

########################################################################
# .1  ADOPT AN IDEOLOGY — the dispatcher. Fired from a decision/button (to be wired), which first
# stamps qing_pending_ideology (a flag var naming the chosen creed). This event reads it and runs the
# proven layered adoption effect for that creed. Kept option-free + hidden so it is a pure trampoline;
# the player's CHOICE happens upstream (the decision), not here.
########################################################################
qing_ideology.1 = {
	type = country_event
	hidden = yes

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_ideology.1 (adopt ideology dispatcher)" }
		# read the chosen creed from the flag var the upstream decision set, and run the layered adoption.
		if = {
			limit = { has_variable = qing_pending_ideology  var:qing_pending_ideology = flag:liberalism }
			QING_adopt_ideology = { ideology = liberalism }
		}
		else_if = {
			limit = { has_variable = qing_pending_ideology  var:qing_pending_ideology = flag:conservatism }
			QING_adopt_ideology = { ideology = conservatism }
		}
		else_if = {
			limit = { has_variable = qing_pending_ideology  var:qing_pending_ideology = flag:monarchism }
			QING_adopt_ideology = { ideology = monarchism }
		}
		else_if = {
			limit = { has_variable = qing_pending_ideology  var:qing_pending_ideology = flag:nationalism }
			QING_adopt_ideology = { ideology = nationalism }
		}
		else_if = {
			limit = { has_variable = qing_pending_ideology  var:qing_pending_ideology = flag:socialism }
			QING_adopt_ideology = { ideology = socialism }
		}
		else_if = {
			limit = { has_variable = qing_pending_ideology  var:qing_pending_ideology = flag:communism }
			QING_adopt_ideology = { ideology = communism }
		}
		else = {
			LOG_fail = { sys = QING  fn = "qing_ideology.1"  reason = "no valid qing_pending_ideology flag set - adoption skipped" }
		}
		# clear the pending marker so a re-fire can't re-convert on a stale value.
		remove_variable = qing_pending_ideology
	}
}

########################################################################
# .2  EMBRACE A POLITICAL CREED — the player's CHOICE event. Fired by the
# qing_embrace_political_creed decision (decisions/imp19c_mod_decisions/
# imp19c_ideology_decisions.txt). One option per creed; each option's own
# trigger gates it on the government/reform-state so a creed cannot be adopted
# anachronistically (e.g. communism only once the reform crisis has ripened).
# Choosing an option stamps qing_pending_ideology and dispatches the .1
# trampoline (which runs the heavy layered adoption effect). The CHOICE is here;
# the ADOPTION machinery is behind the trampoline (compile-inline-safe).
########################################################################
qing_ideology.2 = {
	type = country_event
	title = "qing_ideology.2.t"
	desc = "qing_ideology.2.desc"
	picture = senate_debate

	left_portrait = root.current_ruler


	# --- Liberalism (立憲自由) — available once modernising pressure is real. ----
	option = {
		name = "qing_ideology.2.a"
		trigger = { has_variable = qing_reform_pressure  var:qing_reform_pressure >= 40 }
		set_variable = { name = qing_pending_ideology  value = flag:liberalism }
		trigger_event = { id = qing_ideology.1  days = 1 }
	}

	# --- Conservatism (保守) — the reform-sceptic ordering; always available. ----
	option = {
		name = "qing_ideology.2.b"
		set_variable = { name = qing_pending_ideology  value = flag:conservatism }
		trigger_event = { id = qing_ideology.1  days = 1 }
	}

	# --- Monarchism (尊王) — the throne as keystone; available to a monarchy. ----
	option = {
		name = "qing_ideology.2.c"
		trigger = { is_monarchy = yes }
		set_variable = { name = qing_pending_ideology  value = flag:monarchism }
		trigger_event = { id = qing_ideology.1  days = 1 }
	}

	# --- Nationalism (民族主義) — a nation-people; needs the awakening under way. --
	option = {
		name = "qing_ideology.2.d"
		trigger = { has_variable = qing_reform_pressure  var:qing_reform_pressure >= 40 }
		set_variable = { name = qing_pending_ideology  value = flag:nationalism }
		trigger_event = { id = qing_ideology.1  days = 1 }
	}

	# --- Socialism (社會主義) — the cooperative order; ripe reform crisis only. ---
	option = {
		name = "qing_ideology.2.e"
		trigger = { has_variable = qing_reform_pressure  var:qing_reform_pressure >= 55 }
		set_variable = { name = qing_pending_ideology  value = flag:socialism }
		trigger_event = { id = qing_ideology.1  days = 1 }
	}

	# --- Communism (共產主義) — the most militant creed; a state in deep crisis. --
	option = {
		name = "qing_ideology.2.f"
		trigger = { has_variable = qing_reform_pressure  var:qing_reform_pressure >= 70 }
		set_variable = { name = qing_pending_ideology  value = flag:communism }
		trigger_event = { id = qing_ideology.1  days = 1 }
	}

	# --- Hold to the ancestral order — decline any creed. ----------------------
	option = {
		name = "qing_ideology.2.g"
	}
}

########################################################################
# .3  [P5 2026-07-24] THE REFORMERS MEMORIALIZE THE THRONE (公車上書). Design 3 shipped the
# adoption DECISION (qing_embrace_political_creed) but nothing MOTIVATED or dramatized it. This
# is the motivating arc: as the reform crisis ripens, a body of reform-minded officials and
# literati memorializes the throne to embrace a modern political creed (the historical 公車上書 /
# self-strengthening memorials). It fires from the flavour roll once reform pressure is real, and
# either opens the CHOICE (dispatches qing_ideology.2, the adoption event) or is rebuffed.
#
# Gated (trigger + roll): CHI, reform pressure >= 40 (the same threshold the decision uses), and
# NOT already on an ideology (nothing to memorialize for once converted).
########################################################################
qing_ideology.3 = {
	type = country_event
	title = "qing_ideology.3.t"
	desc = "qing_ideology.3.desc"
	picture = senate_debate
	left_portrait = root.current_ruler

	trigger = {
		tag = CHI
		has_variable = qing_reform_pressure
		var:qing_reform_pressure >= 40
		NOT = { is_ideology_religion = yes }
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_ideology.3 (reformers memorialize) [ROOT.GetTag]" }
	}

	# OPTION A — RECEIVE THE MEMORIAL. Hand the court the creed CHOICE — dispatch the adoption
	# event (qing_ideology.2), whose options are each era/reform-gated. The reformers are heartened
	# (a little reform pressure, the debate is now live) but nothing is decided here.
	option = {
		name = "qing_ideology.3.a"
		custom_tooltip = "qing_ideology.3.a.tt"
		trigger_event = { id = qing_ideology.2  days = 3 }
		current_ruler = { add_popularity = -5 }
	}

	# OPTION B — REBUFF THE MEMORIALISTS. The throne stands on the ancestral order; the reformers
	# are silenced for now. Reform pressure eases (the lid is held down) and legitimacy rises on the
	# Mandate's dignity — but the underlying crisis is only deferred.
	option = {
		name = "qing_ideology.3.b"
		custom_tooltip = "qing_ideology.3.b.tt"
		QING_DECLINE_nudge = { var = qing_reform_pressure  amount = -5 }
		add_legitimacy = 6
	}
}

########################################################################
# .4  [P5 2026-07-24] THE CONSERVATIVE BACKLASH (守舊反動). The reaction to a fresh ideological
# conversion: the old order resents the new creed (the qing_ideology_recent_convert modifier is
# on the country). The court's conservatives — princes, censors, the old degree-holding elite —
# push back. Fires from the flavour roll while the recent-convert modifier stands. The player
# chooses to face down the reaction (spend legitimacy to bed the creed in faster) or appease it
# (ease the backlash at the cost of the creed's momentum).
#
# Gated (trigger + roll): CHI, on an ideology, and the recent-convert churn still active.
########################################################################
qing_ideology.4 = {
	type = country_event
	title = "qing_ideology.4.t"
	desc = "qing_ideology.4.desc"
	picture = chinese_throne_room
	left_portrait = root.current_ruler

	trigger = {
		tag = CHI
		is_ideology_religion = yes
		has_country_modifier = qing_ideology_recent_convert
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_ideology.4 (conservative backlash) [ROOT.GetTag]" }
	}

	# OPTION A — FACE DOWN THE REACTION (乾綱獨斷). The throne imposes the new creed over the old
	# guard's objections: spend legitimacy, but the conversion beds in faster (accelerate the pop
	# conversion drift by re-seeding a capital vanguard) and the court is cowed (stability holds).
	option = {
		name = "qing_ideology.4.a"
		custom_tooltip = "qing_ideology.4.a.tt"
		add_legitimacy = -8
		add_stability = 1
		# re-seed a small capital vanguard onto the state creed (accelerates the drift). Bounded, guarded.
		if = {
			limit = { exists = capital_scope }
			capital_scope = {
				ordered_pops_in_province = {
					limit = { NOT = { pop_religion = ROOT.religion } }
					order_by = pop_hapiness
					max = 2
					set_pop_religion = ROOT.religion
				}
			}
		}
	}

	# OPTION B — APPEASE THE OLD GUARD (調和新舊). The throne softens the creed's edges to placate
	# the conservatives: the backlash eases (remove the recent-convert churn early) but the creed's
	# momentum is blunted (a step of reform pressure returns — the compromise satisfies no one fully).
	option = {
		name = "qing_ideology.4.b"
		custom_tooltip = "qing_ideology.4.b.tt"
		if = {
			limit = { has_country_modifier = qing_ideology_recent_convert }
			remove_country_modifier = qing_ideology_recent_convert
		}
		QING_DECLINE_nudge = { var = qing_reform_pressure  amount = 4 }
		current_ruler = { add_popularity = 6 }
	}
}
