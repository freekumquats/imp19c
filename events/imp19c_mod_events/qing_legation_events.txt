########################################################################
# qing_legation_events.txt — DIPLOMATIC MISSIONS & PERMANENT LEGATIONS
# (遣使 / 駐外使館). The proactive path for the legation engine
# (se_QING_LEGATIONS.txt): send the grand Burlingame embassy, then plant
# permanent legations in the capitals of the powers. CHI-only, player-driven.
# Extends the Zongli Yamen (總理衙門): the hub requires the Yamen to exist.
#
# .1  the legation hub (choose a capital; the Guo Songtao backlash rides along)
# .2  the Burlingame grand embassy (a one-off; opens several relations at once)
# .3  [#22] Foreign Friends via the Yamen — a repeatable beat once the 總理衙門 stands: a real
#     diplomat from the corps wins the goodwill of a foreign power; his skill sizes the reward
#
# Sources: se_QING_LEGATIONS.txt header.
########################################################################

namespace = qing_legation

# ======================================================================
# .1  PLANT A PERMANENT LEGATION — the hub. Opened from the mechanics panel verb
# (qing_action_open_legation). One option per capital, each shown only if that
# power exists on the map and no legation stands there yet.
# ======================================================================
qing_legation.1 = {
	type = country_event
	title = "qing_legation.1.t"
	desc = "qing_legation.1.desc"
	picture = council_country

	left_portrait = root.current_ruler


	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_legation.1 (legation hub) [ROOT.GetTag] count=[ROOT.MakeScope.GetVariable('qing_legation_count').GetValue|0]" }
		QING_legation_init = yes
	}

	# --- London (駐英使館 — Guo Songtao's post, 1877) ---------------------------
	option = {
		name = "qing_legation.1.britain"
		custom_tooltip = "qing_legation.1.britain.tt"
		trigger = {
			exists = c:GBR
			NOT = { has_variable = qing_legation_britain }
		}
		QING_legation_open = { power = britain  tag = GBR }
	}

	# --- Paris (駐法使館) -------------------------------------------------------
	option = {
		name = "qing_legation.1.france"
		custom_tooltip = "qing_legation.1.france.tt"
		trigger = {
			exists = c:FRA
			NOT = { has_variable = qing_legation_france }
		}
		QING_legation_open = { power = france  tag = FRA }
	}

	# --- St Petersburg (駐俄使館) ----------------------------------------------
	option = {
		name = "qing_legation.1.russia"
		custom_tooltip = "qing_legation.1.russia.tt"
		trigger = {
			exists = c:RUS
			NOT = { has_variable = qing_legation_russia }
		}
		QING_legation_open = { power = russia  tag = RUS }
	}

	# --- Tokyo (駐日使館 — the peer-power legation) -----------------------------
	option = {
		name = "qing_legation.1.japan"
		custom_tooltip = "qing_legation.1.japan.tt"
		trigger = {
			OR = { exists = c:JPN  exists = c:TKG }
			NOT = { has_variable = qing_legation_japan }
		}
		QING_legation_open = { power = japan  tag = JPN }
	}

	# --- Washington (駐美使館) -------------------------------------------------
	option = {
		name = "qing_legation.1.america"
		custom_tooltip = "qing_legation.1.america.tt"
		trigger = {
			exists = c:USA
			NOT = { has_variable = qing_legation_america }
		}
		QING_legation_open = { power = america  tag = USA }
	}

	# --- Hold — the Middle Kingdom is slow to send envoys abroad ----------------
	option = {
		name = "qing_legation.1.decline"
	}
}

# ======================================================================
# .2  THE BURLINGAME MISSION (蒲安臣使團, 1868) — the grand embassy that first
# carried China's case to the treaty powers, led by the retiring American
# minister Anson Burlingame in the dynasty's name. Offered once, while the Zongli
# Yamen exists but before the mission has been sent. Fired from the flavour roll.
# ======================================================================
qing_legation.2 = {
	type = country_event
	title = "qing_legation.2.t"
	desc = "qing_legation.2.desc"
	picture = council_country

	left_portrait = root.current_ruler


	trigger = {
		tag = CHI
		has_variable = qing_office_zongli_holder
		NOT = { has_variable = qing_burlingame_mission_sent }
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_legation.2 (Burlingame mission offered) [ROOT.GetTag]" }
	}

	# --- Send the grand embassy -------------------------------------------------
	option = {
		name = "qing_legation.2.a"
		custom_tooltip = "qing_legation.2.a.tt"
		trigger = { treasury >= 310 }
		add_treasury = -310
		QING_legation_burlingame_mission = yes
	}

	# --- Decline — no envoy shall bow in a barbarian court ----------------------
	option = {
		name = "qing_legation.2.b"
		custom_tooltip = "qing_legation.2.b.tt"
		QING_DECLINE_nudge = { var = qing_reform_pressure  amount = 3 }
	}
}

# ======================================================================
# .3  [#22] FOREIGN FRIENDS (協和萬邦) — the repeatable Zongli-Yamen version of the generic
# flavor_eve.18. Once the 總理衙門 stands (its director seat exists) and the diplomat corps
# holds at least one man, the court hears that one of its ministers abroad has won the goodwill
# of a foreign power. Unlike the generic once-only flavor_eve.18, THIS names the real envoy
# (drawn from qing_zongli_diplomats) and the power, scales the reward by the envoy's charisma
# (the Yamen's governing skill), and rewards both the envoy and the director — tying the corps
# to concrete outcomes. Fired from the Qing court-slot random_list (se_QING_DECLINE.txt),
# cooldown-throttled. ROOT = CHI. scope:qing_ff_envoy = the drawn diplomat; scope:qing_ff_power
# = the foreign power; scope:qing_ff_portrait = the right-portrait char (always valid).
# ======================================================================
qing_legation.3 = {
	type = country_event
	title = "qing_legation.3.t"
	desc = "qing_legation.3.desc"
	picture = council_country

	left_portrait = root.current_ruler
	right_portrait = scope:qing_ff_portrait

	# re-check in the event's own trigger (the roll's gate can go stale over the days delay).
	trigger = {
		tag = CHI
		has_variable = qing_office_zongli_holder
		has_variable = qing_zongli_diplomat_count
		var:qing_zongli_diplomat_count > 0
		any_country = {
			has_land = yes
			is_subject = no
			NOT = { alliance_with = root }
			NOT = { this = root }
			in_diplomatic_range = root
		}
	}

	immediate = {
		LOG_line = { sys = QING  msg = "EVENT qing_legation.3 (Foreign Friends via the Yamen) [ROOT.GetTag]" }
		# the ablest diplomat in the Yamen corps carries the day.
		ordered_in_list = {
			variable = qing_zongli_diplomats
			order_by = combined_stats_council_svalue
			check_range_bounds = no
			max = 1
			save_scope_as = qing_ff_envoy
		}
		# a foreign power in reach whose goodwill he has won.
		random_country = {
			limit = {
				has_land = yes
				is_subject = no
				NOT = { alliance_with = root }
				NOT = { this = root }
				in_diplomatic_range = root
			}
			save_scope_as = qing_ff_power
		}
		# right portrait = the envoy if drawn, else the foreign ruler (always a valid char scalar —
		# conditional portraits are unproven, so resolve to one saved scope here).
		if = {
			limit = { exists = scope:qing_ff_envoy }
			scope:qing_ff_envoy = { save_scope_as = qing_ff_portrait }
		}
		else = {
			scope:qing_ff_power = { current_ruler = { save_scope_as = qing_ff_portrait } }
		}
	}

	# --- Commend the envoy — his skilled diplomacy is rewarded --------------------------------
	option = {
		name = "qing_legation.3.a"
		custom_tooltip = "qing_legation.3.a.tt"
		if = {
			limit = { exists = scope:qing_ff_power }
			# a capable envoy (charisma-heavy — the Yamen's governing skill) wins a STRONGER opinion.
			if = {
				limit = { exists = scope:qing_ff_envoy  scope:qing_ff_envoy = { charisma >= 8 } }
				scope:qing_ff_power = { add_opinion = { modifier = qing_zongli_envoy_opinion  target = ROOT } }
			}
			else = {
				scope:qing_ff_power = { add_opinion = { modifier = foreign_friends_opinion  target = ROOT } }
			}
		}
		# reward the envoy's standing + a modest lift for the Yamen director who runs the corps.
		if = {
			limit = { exists = scope:qing_ff_envoy }
			scope:qing_ff_envoy = { QING_char_promote_standing = { pop = 3  prom = 3 } }
		}
		if = {
			limit = { has_variable = qing_office_zongli_holder  var:qing_office_zongli_holder = { is_alive = yes } }
			var:qing_office_zongli_holder = { QING_char_promote_standing = { pop = 2  prom = 1 } }
		}
		add_legitimacy = 3
	}

	# --- Note it — the court is pleased but sends no special favour -----------------------------
	option = {
		name = "qing_legation.3.b"
		custom_tooltip = "qing_legation.3.b.tt"
		if = {
			limit = { exists = scope:qing_ff_power }
			scope:qing_ff_power = { add_opinion = { modifier = foreign_friends_opinion  target = ROOT } }
		}
	}
}
