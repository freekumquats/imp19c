# [Phase 4 Part D — 2026-07-25] QING INTEGRATION-SCIENCE decisions (清朝整合學術).
#
# The arming half of the Qing integration-science invention branch (common/inventions/00_qing_inventions.txt
# Layer 2). Each decision is opened on an INTEGRATED culture of one of the empire's subject families and, once
# taken, PERMANENTLY sets an X_science country variable (set_variable days = -1) that unlocks the matching
# qing_tech_X_science invention for a Qing player whose OWN culture group is not that family.
#
# Idiom = Terra Indomita culture_decisions/integrated_foreign_researcher_inv.txt (patronize_greek_literature):
# potential requires the target culture is integrated + of the family + the science var not yet set; effect
# sets the permanent var. Trimmed to this mod's conventions (scope:target_culture, minor_cultural_decision_price,
# no create_character — the boot-crash rule; TI's version spawns a researcher character, which we deliberately
# OMIT to stay clear of the #90 grant-to-fresh-char class).
#
# The families mirror the invention gates: chinese_group(Han) / mongolic(Mongol) / bodish(Tibetan) /
# east_turkic(Uyghur). The jurchen(Manchu) family is CHI's OWN primary group, so manchu_science needs no
# decision (the invention unlocks via country_culture_group=jurchen directly) — no decision is defined for it.
#
# scope:target_culture = the country_culture the decision is opened on.

culture_decisions = {

	# ---- 儒家治術 patronise Han (Confucian) scholarship -> arms han_science ----
	qing_patronize_han_science = {
		potential = {
			trigger_if = {
				limit = { exists = scope:target_culture }
				tag = CHI
				NOT = { has_variable = han_science }
				scope:target_culture = {
					is_integrated = yes
					this.culture.culture_group = culture_group:chinese_group
				}
			}
		}
		allow = {
			can_pay_price = major_cultural_decision_price
			scope:target_culture = {
				custom_tooltip = {
					text = qing_patronize_science_req
					is_integrated = yes
				}
			}
		}
		effect = {
			pay_price = major_cultural_decision_price
			set_variable = {
				name = han_science
				days = -1
			}
			custom_tooltip = qing_han_science_unlocked_tt
			add_stability = 3
		}
		ai_will_do = { factor = 0 }
