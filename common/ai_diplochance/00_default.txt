#This file contains AI acceptance factors (actually terms), aka. "diplo chance" in code.

#Existing factors can be arbitrarily multiplied (scale) and added to (offset).
#In general, we prefer adding to base factor.
#Hardcoded factors can be removed by scaling by 0.
#Subject specific demanding become subject are located in common/subject_types diplo_chance blocks.

#In some cases these overlap some defines since this file is comparatively new,
#the intent is to add future constants here rather than new defines where possible.
#Since hardcoded factors are scaled by these and may change, make sure to double-check
#your scaling on a new version. We will try to export scaling to this file gradually.

#In the future we might add support for entirely custom scriptable factors like EU4 has,
#or otherwise make the factors freely exchangable. There may always be a degree of
#hardcoding due to performance however.

trade_access = {
	# [BT trade-route flood] imp19c supersedes vanilla engine trade routes with its own Trade
	# Agreement system (se_DIP_TRADE.txt), so the AI must NOT spam the vanilla "Request Trade
	# Route" action (loc ASKTRADEACCTITLE). The original imp19c change only zeroed the BASE offset
	# but left opinion scale = 0.25, so a positive opinion still pushed the chance above threshold.
	# This stayed quiet while almost no good carried a country{} trade-surplus bonus, but once
	# MG-3 (#148) gave all 56 goods a country{} bonus the vanilla import AI woke up and flooded
	# the player from turn 1. Hard-suppress: a large negative base offset swamps any positive term,
	# and opinion scale = 0 removes the term that was still driving it. (Header: "Hardcoded factors
	# can be removed by scaling by 0.") The mod's Trade Agreement diplomacy is unaffected.
	base = {
		offset = -1000 # Changed imp19c: was 0 — hard-disable vanilla trade-route requests
	}
	opinion = {
		scale = 0 # Changed imp19c: was 0.25 — opinion no longer drives the request
	}
	# [#219 2026-07-30] The base+opinion change ALONE did NOT stop the flood (boot-tested at 446b3f8).
	# Root cause: this file's idiom is that any hardcoded factor NOT listed in an action keeps its
	# native nonzero engine weight (header lines 3-5, "removed by scaling by 0"). The goods-VALUATION
	# factors — strategic_trade_good / good_for_us / bad_for_us (header order, lines 85/87/88) — read
	# the requested good's economic value, which MG-3 (#148) inflated by giving all 56 goods a country{}
	# surplus bonus. Those factors were never zeroed, so they kept driving the AI's propensity to ask,
	# and (scaled by the stock define TRADE_REQUEST_ACCEPTANCE=200) produced a positive term large enough
	# to survive the -1000 base offset. Zero all three so the vanilla import AI has no value signal left.
	strategic_trade_good = { scale = 0 }
	good_for_us = { scale = 0 }
	bad_for_us = { scale = 0 }
}
