qing_great_wall_building = {
	# 長城 — the Great Wall proper: the monumental defensive wonder, far grander than a restored
	# section. Strong defence + a real prestige/legitimacy + civilization mark. Built once, at the
	# northern frontier, by the great-project event (manpower + gold).
	# defence + a prestige/civilization mark. Prestige/legitimacy reward lives in the BUILD EVENT
	# (country-scope keys are not valid on a province building — the selfstr wonder proves buildings
	# carry only local_* keys); here the monument radiates civilization from the frontier.
	local_defensive = 0.25
	local_country_civilization_value = 0.03
	local_population_happiness = 0.05

	cost = 6000
	time = 3650

	# [#24 sweep 2026-08-01] CORRECTION: potential = { always = no } was WRONG — add_building_level
	# RESPECTS potential (proven at boot: the qing_mission_cathedral_building BT#6 fix note, "a
	# permanently-false gate HIDES the type"; and memory imp19c-add-building-level-respects-potential).
	# So the old always=no meant the seed (Shanhaiguan/Juyongguan) AND the Works-verb build BOTH
	# silently dropped — leaving the QING_works_build_great_wall verb repeatable for free legitimacy/
	# stability (the once-only NOT-has_building guard could never latch). FIX: a real, satisfiable POTENTIAL
	# gate — the historical Great-Wall belt (owner Chinese/Manchu) — so the monument lands.
	# [soft-blocker, direct user instruction 2026-08-17] allow = { always = no } REPLACED with a real,
	# satisfiable allow (matching the family's sufficient_job_slots idiom) + cost/time raised to an
	# astronomically high 6000/3650 — a hard always=no gate meant this monument could NEVER complete
	# through the real construction queue, which also meant it could never trigger the vanilla
	# Global-News construction-completion notification (that feed is fired by the engine's own queue-
	# completion pipeline, not by any scriptable mod effect — confirmed, no manual "post to news" effect
	# exists anywhere in this engine's scripting surface). A real queue path (even a prohibitively
	# expensive one) lets a player who really wants to ALSO see it in Global News queue it the normal
	# way; the QING_works_build_great_wall verb's own once-only NOT-has_building guard is unaffected
	# (add_building_level still bypasses allow/cost/time entirely) and remains the cheap, intended route
	# at its own fixed 620 treasury + 10 manpower price. NOTE: the once-per-realm invariant is no longer
	# allow-enforced for the menu path — a player could in principle queue this in multiple eligible
	# provinces; the astronomical cost is the ONLY deterrent now, by explicit design choice.
	potential = {
		owner = { OR = { country_culture_group = jurchen  country_culture_group = chinese_group } }
		OR = {
			is_in_region = Zhili
			is_in_region = Shanxi
			is_in_region = Shaanxi
			is_in_region = Gansu
			is_in_region = Liaoning
		}
	}
	allow = { sufficient_job_slots = yes }

	modification_display = {
		0 = local_defensive
		1 = local_country_civilization_value
		2 = local_population_happiness
	}
}
