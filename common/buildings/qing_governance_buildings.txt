	# CONCRETE face of the abstracted bureaucracy-CAPACITY meter: the governance
	# pulse builds a level of it in the empire's administrative centre when
	# capacity rises across a band, and removes one when capacity collapses.
	# Admin / tax / governance-throughput flavour.

	local_monthly_state_loyalty = 0.02
	local_pop_promotion_speed_modifier = 0.05
	local_monthly_civilization = 0.004

	cost = 60
	time = 180

	# earlier self-referential `potential = { NOT = { has_building = self } }` (UNATTESTED, risks
	# self-destruction when the building's own potential flips false after placement). [USER 2026-08-06]
	# max_amount = 1 does the CAP; yamens did NOT require cities (rural county seats too) — so NO
	# has_city_status gate. potential is a STABLE owner-culture gate (proven hanlin/guozijian pattern) — it
	# stays TRUE after placement (unlike the self-referential form) and keeps the type out of non-Qing build
	# menus. num_of_qing_yamen_building ∈ {0,1} per province → QING_bureau_reach counts COVERED PROVINCES and

		owner = { OR = { country_culture_group = jurchen  country_culture_group = chinese_group } }
	}

	allow = {
		owner = { invention = tech_central_administration }
		sufficient_job_slots = yes
	}

	modification_display = {
		0 = local_tax_modifier
		1 = local_monthly_state_loyalty
		2 = local_pop_promotion_speed_modifier
		3 = local_monthly_civilization
	}
}
