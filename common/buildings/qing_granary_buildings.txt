########################################################################
# qing_granary_buildings.txt — the ever-normal granary (常平倉) building [#91 item K / #435].
#
# A concrete on-map counterpart to the abstract qing_granary_stock counter
# (se_QING_DECLINE.txt:25,66): where the counter is the summary/AI layer, this is
# a real building the decline engine raises in the Yellow River (黃河) provinces
# in good years / when the stock is high, and tears down (remove_building_level)
# in famine as the reserves are eaten. Additive: the counter is KEPT, the building
# hangs off it.
#
# Schema follows the mod's own arsenal_building (00_military_buildings.txt:20) and
# the food-capacity + monthly-food shape of Terra-Indomita's population_building
# (#Granary). Event-spawned via add_building_level, so no macro_builder/config
# entry is needed; the potential/allow gate is for the player build menu only.
#
# BOM / UTF-8, matching its common/buildings/ siblings.
########################################################################

qing_granary_building = {
	# [#435] the EVER-NORMAL GRANARY (常平倉) — not a plain grain store but the distinctive Qing
	# PRICE-STABILISATION institution: it bought grain dear in glut and sold cheap in dearth, so
	# its teeth are (a) food storage + accumulation (the buffer against famine), AND (b) market
	# stabilisation — a settled grain market that steadies the countryside. Modelled with the
	# storage keys (food_capacity/food_modifier, ~ TI population_building #Granary), PLUS the
	# stabilisation character the plain granary lacked: a trade-route (market reach), a loyalty
	# lift (a fed, price-steadied populace is a quiet one — the famine-riot prevention the 常平倉
	# was built for), and happiness. All keys proven building-scope (local_state_trade_routes /
	# local_monthly_state_loyalty on the customs-house + dike buildings).
	local_food_capacity = 200
	local_monthly_food_modifier = 0.05
	local_state_trade_routes = 0.2
	local_monthly_state_loyalty = 0.02
	local_population_happiness = 0.03

	cost = 60
	time = 180

	# [#234] the ever-normal granary (常平倉) is a Chinese/Qing fiscal institution — gate it to a
	# jurchen/chinese realm (or a Qing subject) so it is not globally buildable, but do NOT
	# require city rank (granaries served county-wide, including rural grain markets). Build-menu
	# gate only; add_building_level does not enforce allow (job slots) or potential.
	potential = {
		OR = {
			owner = {
				OR = {
					country_culture_group = jurchen
					country_culture_group = chinese_group
				}
			}
			owner = {
				exists = overlord
				overlord = {
					OR = {
						country_culture_group = jurchen
						country_culture_group = chinese_group
					}
				}
			}
		}
	}
	allow = {
		sufficient_job_slots = yes
	}

	modification_display = {
		0 = local_food_capacity
		1 = local_monthly_food_modifier
		2 = local_state_trade_routes
		3 = local_monthly_state_loyalty
		4 = local_population_happiness
	}
}

qing_capital_granary_building = {
	# [#94 京倉 — CAPITAL GRAIN RESERVE, slice 2 of the concrete grain economy] the metropolitan
	# granary of the capital — 京倉 (the capital-city stores) / 通倉 (the Tongzhou 通州 terminus of
	# the Grand Canal, where the 漕運 tribute grain landed). Distinct from the provincial 常平倉
	# (qing_granary_building) and the rural 社倉: this is THE strategic reserve of the throne, the
	# store the canal's tribute-grain lifeline (漕運) fills and the capital + its banner garrisons
	# eat from. This slice places the CAPACITY ANCHOR only — a real local_food_capacity the later
	# canal→capital-state delivery slice fills via add_state_food (design/DESIGN_GRAIN_FOOD_VALUE_94_95.md §8c).
	#
	# local_food_capacity is a VANILLA-food-scale BALANCE knob (design §1.3), NOT a historical shi
	# figure; 400 = 2× the provincial 常平倉 (200), the metropolitan reserve being the largest single
	# store. DELIBERATELY NO local_monthly_food_modifier: this slice must add capacity WITHOUT
	# creating a food surplus, or the hardcoded positive_state_food_growth (+0.02 local_population_growth,
	# 00_hardcoded.txt) would run Beijing's pop away before any delivery is even wired. The two flavour
	# keys below are non-food (a provisioned capital is a quiet one) — they cannot make surplus.
	local_food_capacity = 400
	local_monthly_state_loyalty = 0.03
	local_population_happiness = 0.03

	cost = 80
	time = 200

	# Gate to the CAPITAL CORRIDOR (Zhili/順天府) + a jurchen/chinese realm. add_building_level RESPECTS
	# potential (memory imp19c-add-building-level-respects-potential + the #190 correction in
	# se_QING_BUILDINGS.txt), so this MUST be satisfiable at the seed site Beijing P8363 — which is
	# CHI-owned, city, region Zhili. is_in_region + culture is the proven seeding idiom (province_id in a
	# building potential is unattested). Build-menu gate only; the seed bypasses allow (job slots).
	potential = {
		is_capital = yes   # [#86] restrict the 京倉 to THE CAPITAL province (was is_in_region = Zhili, the whole metropolitan corridor). is_capital is a proven province trigger; still satisfied at the Beijing P8363 seed site (the CHI capital). One per realm, in its own capital.
		OR = {
			owner = {
				OR = {
					country_culture_group = jurchen
					country_culture_group = chinese_group
				}
			}
			owner = {
				exists = overlord
				overlord = {
					OR = {
						country_culture_group = jurchen
						country_culture_group = chinese_group
					}
				}
			}
		}
	}
	# allow = sufficient_job_slots (NOT always = no). This building's whole point is a FUNCTIONING
	# food-capacity modifier the later canal-delivery slice fills; a permanently-false allow HIDES the
	# building type at boot (proven: qing_mission_cathedral_building note + memory
	# imp19c-add-building-level-respects-potential — "visibility requires buildability"), which would
	# silently void the +400 capacity anchor. So it follows the FOOD-building family (granary/dike/depot,
	# all sufficient_job_slots) rather than the seed-only institution family (hanlin/guozijian, always=no).
	# No phantom build option results: a build-menu entry requires a macro_builder config include, and
	# this building has none — allow alone does not add it to any menu.
	allow = {
		sufficient_job_slots = yes
	}

	modification_display = {
		0 = local_food_capacity
		1 = local_monthly_state_loyalty
		2 = local_population_happiness
	}
}
