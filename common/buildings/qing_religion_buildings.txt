qing_temple_of_heaven_building = {
	# 天壇 — the Temple of Heaven: the altar-complex where the emperor performed the
	# winter-solstice sacrifice to Heaven, the supreme rite legitimising his rule as
	# Son of Heaven. The most important ritual site of the dynasty. UNIQUE — Beijing
	# only, seeded via SE_qing_religion_buildings (add_building_level bypasses the gates).
	local_country_civilization_value = 0.05
	local_monthly_state_loyalty = 0.04
	local_population_happiness = 0.05

	cost = 400
	time = 1200

	# [#24 sweep 2026-08-01] CORRECTION: always=no dropped the Beijing seed (add_building_level respects
	# potential — memory imp19c-add-building-level-respects-potential). Real POTENTIAL gate (region Zhili +
	# Chinese/Manchu owner) so the 天壇 lands; allow KEPT always=no (add_building_level bypasses allow → seed
	# plants, menu stays closed; a unique never-player-built rite site, no snowball).
	potential = {
		owner = { OR = { country_culture_group = jurchen  country_culture_group = chinese_group } }
		is_in_region = Zhili
	}
	allow = { always = no }

	modification_display = {
		0 = local_country_civilization_value
		1 = local_monthly_state_loyalty
		2 = local_population_happiness
	}
}
