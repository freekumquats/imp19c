EDU_university = {
	local_research_points_modifier = 0.01
	local_middle_strata_desired_pop_ratio = 0.05
	local_upper_strata_desired_pop_ratio = 0.05

	cost = 50
	time = 180

	allow = {
		owner = {invention = tech_education}
		# [review-fix 2026-08-25] bare has_city_status would gate-lock further expansion of
		# the 19 setup-seeded historic university towns (Cambridge, Padua, St Andrews, Yale,
		# etc. -- setup/main/00_default.txt) that are province_rank=settlement, not city --
		# `allow` is a one-time build-menu gate (never rechecks existing buildings, so no
		# existing seed is removed), but the "add level" button would show locked for them.
		# OR in has_building = EDU_university so an already-placed university can still be
		# expanded regardless of city rank; a NEW university still requires city status.
		OR = {
			has_city_status = yes
			has_building = EDU_university
		}
		sufficient_job_slots = yes
		sufficient_education_slots = {
			tier = t2
		}
	}

	modification_display = {

	}
}
