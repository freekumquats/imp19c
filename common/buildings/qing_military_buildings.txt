# LOGISTICS and COASTAL-DEFENCE plant of the Qing empire. Concrete on-map

# distinct from the generic arsenal_building / fortress_building and from the

#  • 駐防八旗城 Banner Garrison City — the provincial Eight-Banners garrison

#    player-buildable (invention-gated). Loyalty + defence, but Han friction.
#  • 御馬牧場 Imperial Horse Pasture — the Court of the Imperial Stud's steppe
#    remount pastures. SEEDED in the Mongol/Manchurian grasslands; buildable on

#  • 綠營汛塘 Green Standard Post — the sedentary provincial constabulary posts.
#    Cheap, broadly buildable, weaker (a decaying institution by the 1800s).
#  • 海防砲臺 Coastal Battery — the Humen-style coastal artillery fort.
#    Player-buildable on the coast. Littoral defence + naval support.
#  • 軍屯總局 Military Colony HQ — the tuntian soldier-farmer colony
#    headquarters of the Xinjiang/Manchurian frontier. Food + defence + settlement.
#

qing_banner_garrison_building = {
	# 駐防八旗城 — a provincial Eight-Banners garrison quarter, the walled "Manchu
	# city" (滿城) planted inside a great Han city (Xi'an, Jingzhou, Guangzhou,
	# Hangzhou, Chengdu, Fuzhou). Professional hereditary soldiery enforcing imperial
	# order — but a privileged, segregated caste the local Han resented. SEEDED in the
	# historical garrison cities; player-buildable to extend the system.
	local_defensive = 0.15
	local_monthly_state_loyalty = 0.05
	local_upper_strata_desired_pop_ratio = 0.03
	local_population_happiness = -0.04   # Han resentment of the privileged banner caste

	cost = 150
	time = 480

	potential = {
		has_city_status = yes
		owner = { country_culture_group = jurchen }   # the banner system is the Qing's own
	}
	allow = {
		has_city_status = yes
		owner = {
			country_culture_group = jurchen
			invention = tech_central_administration
		}
		sufficient_job_slots = yes
	}

	modification_display = {
		0 = local_defensive
		1 = local_monthly_state_loyalty
	}
}
