# qing_military_buildings.txt — [buildings-research 2026-07-27] the GARRISON,
# LOGISTICS and COASTAL-DEFENCE plant of the Qing empire. Concrete on-map
# counterparts to the banner / green-standard / frontier military layers,
# distinct from the generic arsenal_building / fortress_building and from the
# frontier colony/fort already in qing_foreign_buildings.txt.
#
#  • 駐防八旗城 Banner Garrison City — the provincial Eight-Banners garrison
#    quarter ("Manchu city"). SEEDED in the historical garrison capitals AND
#    player-buildable (invention-gated). Loyalty + defence, but Han friction.
#  • 御馬牧場 Imperial Horse Pasture — the Court of the Imperial Stud's steppe
#    remount pastures. SEEDED in the Mongol/Manchurian grasslands; buildable on
#    the pastoral frontier. Cavalry logistics.
#  • 綠營汛塘 Green Standard Post — the sedentary provincial constabulary posts.
#    Cheap, broadly buildable, weaker (a decaying institution by the 1800s).
#  • 海防砲臺 Coastal Battery — the Humen-style coastal artillery fort.
#    Player-buildable on the coast. Littoral defence + naval support.
#  • 軍屯總局 Military Colony HQ — the tuntian soldier-farmer colony
#    headquarters of the Xinjiang/Manchurian frontier. Food + defence + settlement.
#
# Schema mirrors 00_military_buildings.txt (arsenal). Modifier VALUES on the
# proven scale. All keys verified building-scope. BOM / UTF-8.

qing_banner_garrison_building = {
	# 駐防八旗城 — a provincial Eight-Banners garrison quarter, the walled "Manchu
	# city" (滿城) planted inside a great Han city (Xi'an, Jingzhou, Guangzhou,
	# Hangzhou, Chengdu, Fuzhou). Professional hereditary soldiery enforcing imperial
	# order — but a privileged, segregated caste the local Han resented. SEEDED in the
	# historical garrison cities; player-buildable to extend the system.
	local_defensive = 0.15
	local_monthly_state_loyalty = 0.05
	local_upper_strata_desired_pop_ratio = 0.03
	local_population_happiness = -0.04   # Han resentment of the privileged banner caste

	cost = 150
	time = 480

	potential = {
		owner = { country_culture_group = jurchen }   # the banner system is the Qing's own
	}
	allow = {
		owner = {
			country_culture_group = jurchen
			invention = tech_central_administration
		}
		sufficient_job_slots = yes
	}

	modification_display = {
		0 = local_defensive
		1 = local_monthly_state_loyalty
		2 = local_upper_strata_desired_pop_ratio
		3 = local_population_happiness
	}
}

qing_horse_pasture_building = {

	# breeding the war-horses and remounts of the banner cavalry across the Mongol and
	# Manchurian grasslands. SEEDED in the steppe; player-buildable on pastoral frontier.
	# No has_city_status — this is open-range pastoral, not urban.
	local_tribesmen_output = 0.40
	local_tribesmen_desired_pop_ratio = 0.10
	army_movement_speed = 0.05
	local_monthly_state_loyalty = 0.03

	cost = 90
	time = 300

	potential = {
		owner = { country_culture_group = jurchen }
		OR = {
			is_in_region = Mongolia
			is_in_region = Liaoning
			is_in_region = Turkestan
			is_in_region = Zhili          # [#24 sweep] the Inner-Mongol grasslands the map files under Zhili — Hohhot 呼和浩特 herds + the Mulan 木蘭圍場 / Rehe imperial hunting-preserve pastures

	}
	allow = {
		owner = { country_culture_group = jurchen }
		OR = {
			is_in_region = Mongolia
			is_in_region = Liaoning
			is_in_region = Turkestan
		}
		sufficient_job_slots = yes
	}

	modification_display = {
		0 = local_tribesmen_output
		1 = local_tribesmen_desired_pop_ratio
		2 = army_movement_speed
		3 = local_monthly_state_loyalty
	}
}

qing_military_colony_building = {
	# 軍屯總局 — the headquarters of a tuntian (屯田) military-agricultural colony: the
	# soldier-farmer settlements that held the Xinjiang and Manchurian frontier and fed
	# the frontier armies. Distinct from the generic frontier colony — this is the
	# administrative superstructure coordinating a network, so it carries food capacity
	# and settlement pull. Player-buildable on the settled frontier.
	local_defensive = 0.08
	local_population_growth = 0.05
	local_migration_attraction = 0.10
	local_lower_strata_output = 0.30
	local_food_capacity = 100
	# The tuntian (屯田) soldier-farms actually GREW grain to feed the frontier garrison. Previously this
	# building was pure storage (local_food_capacity with no production), so the Urumqi/Lhasa/Heilongjiang
	# garrisons drained their food_storage with nothing replenishing it and starved, even though the parent
	# state held food. Fix in two parts:
	#  - local_monthly_food (FLAT, base-independent): the real feed. These frontier provinces have almost no
	#    base food output, so a percentage modifier alone multiplies ~nothing. 2 food/mo mirrors the mod's own
	#    qing_grain_relief province modifier (local_monthly_food = 2), which exists for exactly this "localized
	#    shortage the per-governorship sim can't fix" case; at ~0.20-0.24 food/cohort it feeds ~8-10 cohorts.
	#  - local_monthly_food_modifier (%): a thematic bonus where some base food does exist; in line with the
	#    best agriculture building (0.10).
	local_monthly_food = 2
	local_monthly_food_modifier = 0.10

	cost = 100
	time = 360

	# [#garrison-food-fix 2026-08-03] Gate ONLY on owner-is-Qing-or-Qing-subject. The prior explicit
	# region allowlist (Turkestan/Mongolia/Liaoning/Gansu[/Tibet/Far_East]) was a recurring bug
	# generator: it had already been patched twice to add regions, and STILL disabled the building at
	# garrison provinces not on the list — a force-seeded tuntian in an unlisted region is present but
	# INERT (its local_monthly_food never applies), so Urumqi/Lhasa/Heilongjiang garrisons starved.
	# The owner-culture gate (Qing OR a Qing subject via the overlord branch — Lhasa 3819 is owned by
	# TIB, bodish, a protectorate) is the meaningful restriction and is self-maintaining: no frontier
	# province can ever silently drop again. The overlord branch is REQUIRED so it lands on subject soil.
	potential = {
		OR = {
			owner = {
				OR = {
					country_culture_group = jurchen
					country_culture_group = chinese_group
				}
			}
			owner = {
				exists = overlord
				overlord = {
					OR = {
						country_culture_group = jurchen
						country_culture_group = chinese_group
					}
				}
			}
		}
	}
	allow = {
		# Same owner-OR-overlord gate as potential (must match, or the building seeds but stays disabled
		# on subject soil). tech_construction was ALSO required here but is NOT held at the 1763 start
		# (only unlocked later via se_QING_EARLYINDUS/SELFSTR), so it disabled all five seeded tuntian;
		# the 屯田 soldier-farm is a historical starting fixture predating any civic-works tech — dropped.
		OR = {
			owner = {
				OR = {
					country_culture_group = jurchen
					country_culture_group = chinese_group
				}
			}
			owner = {
				exists = overlord
				overlord = {
					OR = {
						country_culture_group = jurchen
						country_culture_group = chinese_group
					}
				}
			}
		}
		sufficient_job_slots = yes
	}

	modification_display = {
		0 = local_defensive
		1 = local_population_growth
		2 = local_migration_attraction
		3 = local_lower_strata_output
		4 = local_food_capacity
		5 = local_monthly_food
		6 = local_monthly_food_modifier
	}
}
