########################################################################
# qing_cottage_buildings.txt — #101 Cottage Industry. Eight cheap, RURAL-ONLY
# raw-good buildings for the Qing player, mirroring qing_production_buildings.txt's
# exact shape (base_resources, strata local_output, cost, time, allow gated on
# culture group + trade_goods), differentiated from their factory-tier siblings
# (the six named Qing works, IND_resource_gathering_operation) by: (a) a rural
# gate (NOT has_city_status), (b) omitting sufficient_job_slots — the ONE shared
# capacity gate every existing raw-good building requires. Cottage buildings
# represent household/family labour OUTSIDE the formal job-slot economy (Philip
# Huang's "involutionary growth" — see design/DESIGN_101_COTTAGE_INDUSTRY_
# BUILDINGS.md), so they remain buildable in a province that has already spent
# every job slot on factories/administration/commerce.
#
# See design/DESIGN_101_COTTAGE_INDUSTRY_BUILDINGS.md (v5) for the full design,
# adversarial-review history, and the raw-good -> COTTAGEIND recipe reader lists
# each building below cites.
#
# BOM / UTF-8, matching common/buildings/ siblings.
########################################################################

qing_cottage_smithy_building = {
	# 鄉村鐵匠鋪 — the village smith, working iron for tools, fittings, and simple
	# arms alike at the household scale, outside any state works. Feeds
	# COTTAGEIND_raw_iron, read by construction_materials / early_artillery / steel.
	local_lower_strata_output = 0.2
	local_proletariat_output = 0.15
	base_resources = 1

	cost = 50
	time = 150

	allow = {
		owner = {
			OR = {
				country_culture_group = jurchen        # the Qing: Manchu primary culture is in the jurchen group, NOT chinese_group
				country_culture_group = chinese_group   # any Han-Chinese successor state
			}
		}
		NOT = { has_city_status = yes }
		trade_goods = iron
	}

	modification_display = {
		0 = local_lower_strata_output
		1 = local_proletariat_output
		2 = base_resources
	}
}
