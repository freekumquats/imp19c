# qing_cottage_buildings.txt — #101 Cottage Industry. Cheap, RURAL-ONLY raw-good
# buildings mirroring qing_production_buildings.txt's exact shape (base_resources,
# strata local_output, cost, time), differentiated from their factory-tier
# siblings (the six named Qing works, IND_resource_gathering_operation) by: (a) a
# rural gate (NOT has_city_status), (b) omitting sufficient_job_slots — the ONE
# shared capacity gate every existing raw-good building requires. Cottage
# buildings represent household/family labour OUTSIDE the formal job-slot economy
# (Philip Huang's "involutionary growth" — see design/DESIGN_101_COTTAGE_INDUSTRY_
# BUILDINGS.md), so they remain buildable in a province that has already spent
# every job slot on factories/administration/commerce.
#
# [fix, #58 2026-08-14, design/DESIGN_58_COTTAGE_CULTURE_GATING.md] the original 8
# buildings below were ALL culture-gated Qing-only, but a per-good research pass
# found ZERO of the 8 crafts (iron/lead/textile-fibres/silk/wood/vegetables/
# copper-tin/stone) clear the bar of a genuinely Qing-distinctive institution at
# HOUSEHOLD/COTTAGE scale — each is a near-universal pre-industrial village craft.
# All 8 dropped their culture gate (kept every other condition unchanged) and are
# now available to every culture, matching the mod's own stated principle: "there
# is nothing wrong with Qing-specific buildings so long as they are backed by
# academic research as actually being specific to Qing." The single merged
# non-Qing generic `row_cottage_workshop_building` (which pretended all 9 crafts
# were one interchangeable building — the identical defect, unnoticed until this
# review) is DELETED, not widened, superseded by these 8 now-generic buildings.
#
# The SAME research pass found TWO goods that DO clear the bar with a sharper,
# specifically-Chinese institutional citation — `qing_timber_lineage_building`
# (below) and `qing_cottage_sugarhouse_building` (below) — both stay Qing-only,
# gated on the SPECIFIC historic provinces the citation names, not a generic
# culture-group condition. 8 of the original 8 survive (culture-gate dropped) +
# 2 genuinely new buildings = 10 total; only the 2 new ones remain Qing-exclusive.
#
# See design/DESIGN_101_COTTAGE_INDUSTRY_BUILDINGS.md (v5) and design/DESIGN_58_
# COTTAGE_CULTURE_GATING.md (FINAL v8) for the full design, adversarial-review
# history, and the raw-good -> COTTAGEIND recipe reader lists each building cites.
#
# BOM / UTF-8, matching common/buildings/ siblings.
########################################################################

# [#58, design/DESIGN_58_COTTAGE_CULTURE_GATING.md] the TWO cottage-tier goods
# that DO clear the Qing-specificity bar -- gated on the SPECIFIC historic
# province IDs the citation names, not a generic culture-group condition, unlike

qing_timber_lineage_building = {
	# 力分/山分 (labour-share/mountain-share) lineage-tenure timber contracts —
	# Huizhou (徽州) and Fujian-highland landowning lineages (宗族) granted tenant
	# planters long-term rights to plant/tend timber against a divisible, tradable
	# share of the eventual harvest (Meng Zhang, "Timber and Forestry in Qing
	# China," 2021). A registered, tradable financial instrument over standing
	# timber, not an ordinary village woodlot — the institution is about the LAND
	# (hence the province-ID gate below, not `trade_goods = wood`; Huangshan is
	# currently tagged `tea`, not wood, and this does not need changing).
	local_lower_strata_output = 0.2
	local_proletariat_output = 0.15
	base_resources = 1.5                 # [ASSUMPTION, boot-tune] modestly better than the
	local_monthly_civilization = 0.01   #  generic woodlot's 1/0 -- a lineage/merchant trade
	                                      #  network kicker; no source gives a clean numeric
	                                      #  "% better than an ordinary woodlot."

	cost = 55
	time = 150

	allow = {
		owner = {
			OR = {
				country_culture_group = jurchen
				country_culture_group = chinese_group
			}
		}
		NOT = { has_city_status = yes }
		OR = {
			province_id = 4441   # Huangshan (黄山) -- historic Huizhou (徽州) prefecture
			# Fujian interior highlands -- the terrain-verified hill/low_mountain/forest
			# candidate list (design doc round-2, 14 provinces; the final historical-geography
			# subset within these 14 is unresolved by any in-repo research -- OVERNIGHT
			# DEFAULT: gate on all 14, per Rule 1a, rather than hold the building for an
			# unresolvable judgment call).
			province_id = 831    # Jianyang
			province_id = 1785   # Yanxi
			province_id = 2693   # Jian'ou
			province_id = 3302   # Anxi
			province_id = 3317   # Wuyishan
			province_id = 4247   # Hua'an
			province_id = 5218   # Zhouning
			province_id = 5822   # Pocheng
			province_id = 7470   # Minqing ("Minging" in the mod's own source comment)
			province_id = 7474   # Sanming
			province_id = 7862   # Liancheng
			province_id = 8172   # Shuibei
			province_id = 8498   # Zhenghe
			province_id = 9711   # Pinghe
		}
	}

	modification_display = {
		0 = local_lower_strata_output
		1 = local_proletariat_output
		2 = base_resources
		3 = local_monthly_civilization
	}
}

qing_cottage_sugarhouse_building = {
	# 糖廍 — the jointly-capitalized sugar house: cane-growing was smallholder, but
	# processing ran through pooled-household institutions distinct from both pure
	# cottage and full factory scale, documented for Taiwan and Guangdong (this
	# mod's own 1763 sugar geography is Guangdong/Fujian, not Taiwan -- no Taiwan
	# province carries `trade_goods = sugar` in the current setup). A genuine
	# pooled-household institution, over the generic ROW plantation baseline
	# (row_production_buildings.txt) every sugar province already qualifies for.
	local_lower_strata_output = 0.2
	local_proletariat_output = 0.15
	base_resources = 1.5                 # [ASSUMPTION, boot-tune] modestly better than the
	local_state_trade_routes = 0.03      #  generic plantation's raw yield -- the pooled-mill's
	                                      #  market-network kicker; no clean numeric source.

	cost = 55
	time = 150

	allow = {
		owner = {
			OR = {
				country_culture_group = jurchen
				country_culture_group = chinese_group
			}
		}
		NOT = { has_city_status = yes }
		trade_goods = sugar
		# the historical-geography judgment call (which sugar provinces represent genuine
		# 糖廍 pooled-mill organization vs. ordinary smallholder cane-growing) is unresolved by
		# any in-repo research -- OVERNIGHT DEFAULT: gate on all 6 sugar-tagged CHI provinces
		# confirmed by direct grep (design doc), per Rule 1a, rather than hold for an
		# unresolvable judgment call.
		OR = {
			province_id = 2218   # Haikou, Guangdong
			province_id = 2993   # Shantou, Guangdong
			province_id = 7842   # Chaonan, Guangdong
			province_id = 2693   # Jian'ou, Fujian
			province_id = 4244   # Nanping, Fujian
			province_id = 7470   # Minqing, Fujian ("Minging" in the mod's own source comment)
		}
	}

	modification_display = {
		0 = local_lower_strata_output
		1 = local_proletariat_output
		2 = base_resources
		3 = local_state_trade_routes
	}
}
