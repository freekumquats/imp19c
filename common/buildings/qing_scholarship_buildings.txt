qing_hanlin_academy_building = {
	# 翰林院 — the Hanlin Academy: the empire's apex scholarly body, staffed by the
	# top 進士 graduates who drafted edicts, compiled the histories and tutored princes.
	# The talent pool for the highest offices; a unique institution of the capital.
	# UNIQUE — Beijing only, seeded by SE_qing_scholarship_buildings via add_building_level.
	local_research_points_modifier = 0.20
	local_monthly_civilization = 0.1
	local_country_civilization_value = 0.05
	local_upper_strata_output = 0.15
	local_upper_strata_desired_pop_ratio = 0.03
	local_population_happiness = 0.05

	cost = 300
	time = 900

	# [#24 sweep 2026-08-01] CORRECTION: potential = { always = no } was WRONG — add_building_level
	# RESPECTS potential (proven: memory imp19c-add-building-level-respects-potential), so the Beijing seed
	# silently DROPPED and the Hanlin never appeared. FIX: a real, proven POTENTIAL gate (region Zhili +
	# Chinese/Manchu owner) so the seed lands. allow is KEPT always=no — add_building_level bypasses allow
	# (the codebase's uniform assumption: qing_foreign_buildings / se_QING_BUILDINGS), so the seed still
	# plants while the building stays OUT of the player build menu (a unique, never-player-built institution;
	# no region-wide menu-build snowball).
	potential = {
		owner = { OR = { country_culture_group = jurchen  country_culture_group = chinese_group } }
		is_in_region = Zhili
	}
	allow = { always = no }

	modification_display = {
	}
}
