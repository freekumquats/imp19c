# Scope: province

INDUSTRY_province_industry_capacity = { 
	# Scope: Province
	# Represent the province's capacity for industries to transport goods
	value = 1 # By default every province can support 1
	if = {
		limit = {
			OR = {
				terrain = extreme_mountain
				terrain = extreme_desert
				terrain = boreal_forest
			}
		}
		subtract = 4
	}
	if = {
		limit = {
			has_minor_river = yes
		}
		add = 1
	}
	if = {
		limit = {
			has_province_rank = city
		}
		add = 1
	}
	if = {
		limit = {
			has_province_rank = city_metropolis
		}
		add = 1
	}

	add = num_of_INF_canal
	add = num_of_INF_railway_upgrade
	add = num_of_port_building
	add = num_of_river_port_building

	# TODO - errors
	# add = INF_province_industry_capacity_bonus

multiply = owner.MODIFIER_industry_productivity
	min = 0 # No reason to have negative industry capacity
}

INDUSTRY_free_province_industry_slots = {
	value = INDUSTRY_province_industry_capacity
	subtract = num_of_IND_industrial_estate
}

INDUSTRY_governorship_used_industry_slots = {
	value = 0
	if = {
		limit = {
			has_global_variable = wealth_setup_done
			# [#flood-fix] sentinel guard: the 29 INDUSTRY_factories_assigned_* vars are all set
			# to 0 together in INDUSTRY_setup_all_factory_assignments, so a governorship has all or
			# none. When this svalue is evaluated for a governorship whose setup hasn't run yet
			# (boot ordering / newly-formed govs), the global flag is TRUE but the per-gov vars are
			# UNSET -> ~28-per-good unset-var reads/boot across all 29 goods. Guarding on the first
			# var gates the whole sum; unset -> skip -> value stays 0 (behaviour-preserving).
			has_variable = INDUSTRY_factories_assigned_clothing
		}
		# add every variable for assigning industry slots to industries
		add = var:INDUSTRY_factories_assigned_clothing
		add = var:INDUSTRY_factories_assigned_luxury_clothing
		add = var:INDUSTRY_factories_assigned_furniture
		add = var:INDUSTRY_factories_assigned_luxury_furniture
		add = var:INDUSTRY_factories_assigned_alcohol
		add = var:INDUSTRY_factories_assigned_glass
		add = var:INDUSTRY_factories_assigned_chemicals
		add = var:INDUSTRY_factories_assigned_rare_alloys
		add = var:INDUSTRY_factories_assigned_construction_materials
		add = var:INDUSTRY_factories_assigned_early_munitions
		add = var:INDUSTRY_factories_assigned_late_munitions
		add = var:INDUSTRY_factories_assigned_naval_supplies
		add = var:INDUSTRY_factories_assigned_steel_ships
		add = var:INDUSTRY_factories_assigned_wooden_ships
		add = var:INDUSTRY_factories_assigned_steel
		add = var:INDUSTRY_factories_assigned_bronze
		add = var:INDUSTRY_factories_assigned_machine_parts
		add = var:INDUSTRY_factories_assigned_early_artillery
		add = var:INDUSTRY_factories_assigned_late_artillery
		add = var:INDUSTRY_factories_assigned_electronics
		add = var:INDUSTRY_factories_assigned_pharmaceuticals
		add = var:INDUSTRY_factories_assigned_motors
		add = var:INDUSTRY_factories_assigned_processed_foods
		add = var:INDUSTRY_factories_assigned_petrochemicals

# ===== [#133 I5] Employment scaling (D4a) =====
# Factory output scales with how fully the mechanised sector's job slots are manned by
# the governorship's industrial workforce (proletariat + lower_strata pops). Mirrors how
# the cottage path multiplies output by COTTAGEIND_pops_output. Terms verified on-disk:
# denominator = INDUSTRY_governorship_used_industry_slots (all 24 goods' assigned factories);
# numerator = governorship_proletariat + governorship_lower_strata (ECON_svalues.txt).
INDUSTRY_workers_per_factory = {
	# [#133 I5] Tunable: industrial-worker pops needed to fully staff ONE assigned factory.
	# Best-guess = 5; a fully-staffed sector yields ratio 1 (no output change vs pre-I5).
	value = 5
}

INDUSTRY_employment_ratio_compute = {
	# Scope: Governorship. Clamp( workforce / (assigned factories x workers_per_factory), 0..1 ).
	# NOT an accumulator - a fresh clamp each quarter (no-restoring-drift-ratchet-rule).
	value = 1
	if = {
		limit = {
			INDUSTRY_governorship_used_industry_slots > 0
		}
		value = governorship_proletariat
		add = governorship_lower_strata
		divide = {
			value = INDUSTRY_governorship_used_industry_slots
			multiply = INDUSTRY_workers_per_factory
		}
		min = 0
		max = 1
	}
}

INDUSTRY_employment_ratio = {
	# Scope: Governorship. Read the per-quarter cache when present (perf #71 idiom), else
	# compute inline so a missing cache degrades to correctness, never to zeroed output.
	if = {
		limit = {
			has_variable = industry_employment_ratio_cached
		}
		value = var:industry_employment_ratio_cached
	}
	else = {
		value = INDUSTRY_employment_ratio_compute
	}
}
