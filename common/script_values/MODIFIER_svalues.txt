# MILITARY & NAVY

MODIFIER_military_supply_consumption = {
	# Scope: Country
	value = 1
	add = modifier:army_maintenance_cost
	# [fix, task #6 2026-09-03] army_maintenance_cost reduction stacks from many independent
	# sources (laws e.g. common/laws/00_army_laws.txt -0.75/-0.5, Self-Strengthening reforms,
	# subject types, military traditions, governance modifiers -- 70+ sites across the repo,
	# most -0.05 to -0.15 but several -0.5/-0.75). A well-developed empire can plausibly stack
	# past -1.0, driving this multiplier to 0 or negative -- and EVERY military-supply demand/
	# production term (DEMAND_*_from_military_base in DEMAND_svalues.txt) multiplies by this
	# at the end, so the whole "Military supplies" topbar tooltip zeroed simultaneously across
	# every good (confirmed via a real boot screenshot: production nonzero, but "Supplied by
	# good" and "Consumption by good" both exactly 0.00 for every category at once). Floored at
	# 0.1 -- never fully free, matching the identical "never truly zero" clamp shape already
	# used in this exact file family (DEMAND_elasticity_impact, DEMAND_svalues.txt:32,
	# "Maximum 90% malus to demand" -- max=1 then min=0.1).
	min = 0.1
}

MODIFIER_navy_supply_consumption = {
