########################

TECH_cottage_industry_overall_bonus = {
	value = 1
	if = {
		limit = {
			owner = { invention = tech_manufactories }
		}
		add = TECH_manufactories_bonus_cottage_industry
	}
	if = {
		limit = {
			owner = { invention = tech_manufactories }
		}
		add = TECH_mechanical_tools_bonus_cottage_industry
	}
	# [MO#12 2026-07-29] cottage_to_factory_transition_law bias: protect_cottage_industry
	# raises this multiplier, promote/full mechanisation lower it (tilting production toward
	# factories). Guarded read of a law-set country var (unset = untouched).
	if = {
		limit = {
			owner = { has_variable = INDUS_cottage_bias }
		}
		add = owner.var:INDUS_cottage_bias
	}
}

TECH_manufactories_bonus_cottage_industry = {
	value = 0.1
}

TECH_mechanical_tools_bonus_cottage_industry = {
	value = 0.1
}
