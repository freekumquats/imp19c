#NOTES: Every trade zone has a base value for transportation between EVERY other trade node. Those base values are in the other movement values file.
# Each connection between each zone has an individual svalue.
# Each zone also has a transportation value, this is how the movement cost reductions like roads and railroads are factored in. Each connecting svalue should subtract the transportation value of BOTH of the nodes it is connecting. 

# MOVEMENT BONUSES FROM INFRASTRUCTURE:

railway_bonus = { # Applied per level of railway in provinces on the route
	value = 1
	add = num_of_INF_railway_upgrade
	multiply = 0.2
}

port_bonus = { # Applied as total number of ports in each tradezone on the route
	value = num_of_port_building
	multiply = 0.2
}

canal_bonus = { # [#441] Applied per level of canal in provinces on the route. A canal is
	# an inter-regional transport PATH like a railway (gated on route connectivity, not
	# building-presence), so it lowers inter-zone transport cost too — not only intra-zone
	# trade throughput (TRADE_canal_bonus). Weighted 0.3, ABOVE rail/port (0.2): water
	# transport was historically the cheapest bulk carriage (a barge moved far more per
	# ton-mile than early rail), so a canal is the STRONGER artery.
	value = num_of_INF_canal
	multiply = 0.3
}

#Trade Zone: india
#---------------------------------------

india_transportation_svalue = {
    value = 0
    # [BOOT-SPIKE 2026-08-27, task #32, corrected 2026-08-27] every_<X>_TZ_region (the
    # engine-auto-generated iterator for the india_TZ_region scripted_list,
    # common/scripted_lists/TRADE_lists.txt) fails to parse inside a script_value:
    # "Unexpected token: every_india_TZ_region" at PARSE/LOAD time (pdx_persistent_reader.cpp:229
    # is the load-time script reader, not a runtime error -- corrected from an earlier,
    # inaccurate "at runtime" description). No precedent found anywhere in this codebase or
    # either oracle (Invictus, Terra Indomita) for a custom scripted_list's auto-generated
    # iterator working inside a script_value context. Replaced with the built-in `every_region`
    # iterator (proven working -- used inside scripted_effects in both oracle repos, e.g.
    # Invictus ai_trade_effects.txt -- but NOT yet confirmed working specifically inside a
    # script_value; this is the single spike for that confirmation) + the exact same
    # india_tradezone filter the scripted_list itself used (india_TZ_region's own definition:
    # `base = region, india_tradezone = { PROVINCE = no }`), so the semantics should be
    # identical if this parses correctly.
    # PERFORMANCE CAUTION (independent review, 2026-08-27): line 29's file-level comment says
    # every_region was DELIBERATELY avoided here for performance ("use lots of limits here
    # instead of using every_region to help get rid of the performance impact"). This spike
    # reintroduces exactly that pattern in a quarterly hot-path svalue. If the parse error is
    # confirmed fixed by boot, do NOT roll this pattern out to the other 21
    # *_transportation_svalue blocks without first re-checking the performance impact the
    # original author avoided -- a parse fix should not be allowed to silently reintroduce a
    # known perf regression.
    # ONLY this one svalue is converted; the other 21 *_transportation_svalue blocks in
    # this file are UNCHANGED pending boot confirmation that this actually resolves the
    # parse error -- do not copy this pattern to the rest until that's confirmed.
    every_region = {
        limit = {
            india_tradezone = { PROVINCE = no }
        }
        if = {
            limit = {
                any_region_province = {
                    OR = {
                        any_neighbor_province = { has_road_towards = PREV }
                        has_building = port_building
                    }
                }
            }
            every_region_province = {
                limit = {
                    any_neighbor_province = { has_road_towards = PREV }
                }
                add = railway_bonus
            }
            every_region_province = {
                limit = {
                    has_building = port_building
                }
                add = port_bonus
            }
            every_region_province = {
                limit = {
                    any_neighbor_province = { has_road_towards = PREV }
                }
                add = canal_bonus
            }
        }
    }
}
