		}

	multiply = owner.MODIFIER_agriculture_productivity
	# [#281] Arsenal small-arms output. rifles DEMAND is army/arsenal-driven (DEMAND_rifles),
	# so supply must ALSO come from arsenals or a large-army country with no dedicated
	# rifle-good province (e.g. the Qing, which owns none of the 15 gun towns) would run a
	# permanent phantom shortage. Reuse the munitions arsenal-infra term (cached, #71) so each
	# arsenal/depot yields rifles on the SAME basis as munitions -> supply and demand scale
	# together through arsenals. Added AFTER the agriculture multiply so arsenal output is not
	# scaled by farm productivity (mirrors GOODS_governorship_early_munitions_produced).
	# Gated on tech_rifles to MATCH the demand gate (DEMAND_rifles_base): the munitions-infra
	# term is internally tech_firearms-gated, but rifle DEMAND only appears at tech_rifles, so
	# gate arsenal RIFLE output the same way — no arsenal makes rifles before the army wants them.
	if = {
		limit = { owner = { invention = tech_rifles } }
		add = GOODS_governorship_munitions_infra_output
	}
}

	# Symmetric with the demand side, which already counts these same buildings
	# (DEMAND_svalues.txt:1586/1936/1961/1990). This is the scripted supply-production hook the
	# arsenal previously lacked (RESEARCH_MILITARY_LOGISTICS.md §A2/§E2).

	# every province here: this svalue is evaluated >1x/quarter (production pass + the
	# DEMAND_difference/trade/currency passes), and an inline province loop re-ran each

		value = GOODS_governorship_munitions_infra_output_compute

GOODS_governorship_munitions_infra_output_compute = {
	# Scope: Governorship
	# The actual arsenal/depot munitions-output walk (see the cached wrapper above).
	# Only firearms-capable owners field munitions demand, so gate to match the demand side.
	value = 0
	if = {
		limit = {

# Country level modifiers

GOODS_governorship_bonus_to_industrial_production_from_industrialisation = {
	# Scope: Governorship
	# Function: Multiplier to industrial goods output, based on the governorship's
	# average civilization (industrialisation) level.
	# PERF (#71): this is read once per industrial good (28x) per governorship per
	# quarter and each read does a double state-traversal. GOODS_cache_industrialisation_bonus
	# computes it ONCE into var:industrialisation_bonus_cached at the start of the
	# governorship's stockpile update. Read the cache if present; otherwise fall back
	# to computing inline so a missing cache can never zero out production.
	if = {
		limit = {
			has_variable = industrialisation_bonus_cached
		}
		value = var:industrialisation_bonus_cached
	}
	else = {
		value = GOODS_governorship_bonus_to_industrial_production_from_industrialisation_compute
	}
}

GOODS_governorship_bonus_to_industrial_production_from_industrialisation_compute = {
	# Scope: Governorship
	# The actual industrialisation-bonus math (see the cached wrapper above).
	# Guard: a governorship with zero states would divide-by-zero below
	# (the state-count denominator == 0) -> NaN multiplier corrupting all industrial
	# output. With no states there is no industrialisation bonus, so return 1 (no-op).
	if = {
		limit = {
			any_governorship_state = { count >= 1 }
		}
		every_governorship_state = {
			ordered_state_province = {
				order_by = civilization_value
				max = 1
				add = civilization_value
			}
		}
		divide = {
			every_governorship_state = {
				add = 1
			}
		}
		divide = 100
		add = 1
	}
	else = {
		value = 1
	}
}
