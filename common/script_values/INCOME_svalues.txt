## Script values used in the calculation of government income

# National summary

INCOME_national_total_quarterly = {
	# Scope: Country
	# This should be cached quarterly to minimise hot-loading it all the time as it is always visible in the ingame topbar
	# [fix, CRITICAL correction 2026-08-14] do NOT add INCOME_national_total_from_qing_revenue here.
	# This exact script_value is what INCOME_update_treasury_country's add_treasury reads
	# (se_INCOME.txt:62) -- it is the REAL PAYMENT, not a display-only figure. Salt/Canton/caravan
	# already pay the treasury directly via their own add_treasury calls in their respective pulses
	# (se_QING_SALT.txt:180, se_QING_CANTON.txt:210, se_QING_CARAVAN.txt:265); adding their cached
	# totals here as well double-credits them every quarter (caught by adversarial code review before
	# commit). The UI-visibility fix lives ENTIRELY in INCOME_national_total_quarterly_display below,
	# a separate value that is never read by add_treasury.
	value = INCOME_national_total_from_regions
	add = INCOME_national_total_from_subjects
	# Negative
	add = INCOME_cost_administrator_wages_country
	add = INCOME_cost_military_wages_country
	add = INCOME_cost_military_supplies_country
	add = INCOME_cost_navy_supplies_country
	add = CURRENCY_national_debt_interest_actual_wealth_cost
	add = INCOME_cost_poor_law_spending
}

# [fix] DISPLAY-ONLY total = the real payment (INCOME_national_total_quarterly) + the Qing revenue
# suite that pays the treasury through its OWN separate add_treasury calls. Read by the topbar's
# Plus/Minus/Change bindings and the Ministry of Revenue's Total line (gui/ingame_topbar.gui,
# gui/qing_revenue_ministry.gui) -- NEVER by add_treasury -- so it can show the true quarterly
# treasury movement without paying anything twice.
INCOME_national_total_quarterly_display = {
	# Scope: Country
	value = INCOME_national_total_quarterly
	add = INCOME_national_total_from_qing_revenue
	# [fix, task #79 2026-08-14] ALSO surface this quarter's one-shot CURRENCY_grant_country_wealth
	# payouts (marriage-pact dowries, treasure-fleet hauls, decline/keju/roster/frontier-sea events) --
	# every one of them pays add_treasury directly (se_CURRENCY.txt), same invisible-to-the-topbar
	# class of bug as salt/Canton/caravan/opium above, just event-driven instead of standing-quarterly.
	# Reset to 0 by INCOME_update_treasury_country AFTER this read (se_INCOME.txt), so each quarter's
	# display reflects only THAT quarter's one-shot grants, never a stale carry-forward.
	if = {
		limit = { has_variable = CURRENCY_oneshot_grants_this_quarter }
		add = var:CURRENCY_oneshot_grants_this_quarter
	}
}

# [fix] the STANDING Qing quarterly revenue streams that pay add_treasury directly outside the
# vanilla governorship-summed INCOME_governorship_total chain: salt gabelle (se_QING_SALT.txt,
# QING_salt_income, qing_salt_income_last), Canton customs (se_QING_CANTON.txt, QING_canton_pulse,
# qing_canton_last_state), caravan trade (se_QING_CARAVAN.txt, qing_caravan_income_last), and the
# opium tolerate-for-revenue take (se_QING_OPIUM.txt, qing_opium_income_last -- only nonzero under
# posture=1). All four are CHI-only mechanics (the vars only exist once their own pulse has fired
# for CHI at least once), so every non-Qing country reading this same shared total correctly gets 0.
# Tribute (se_SUBJECT_QING.txt/se_QING_TRIBUTE.txt) is deliberately NOT folded in here: it is an
# episodic per-subject payment, not a standing per-quarter line item like the three above, and has
# no persisted "_last" cache to read (it pays and is gone) -- closer in kind to INCOME_national_
# total_from_subjects, which already exists as its own line.
INCOME_national_total_from_qing_revenue = {
	# Scope: Country
	value = 0
	if = {
		limit = { has_variable = qing_salt_income_last }
		add = var:qing_salt_income_last
	}
	if = {
		limit = { has_variable = qing_canton_last_state }
		add = var:qing_canton_last_state
	}
	if = {
		limit = { has_variable = qing_caravan_income_last }
		add = var:qing_caravan_income_last
	}
	# [fix, task #63 2026-08-14] the Opium Commissioner's tolerate-for-revenue take
	# (se_QING_OPIUM.txt, qing_opium_income_last -- 0 outside posture=1, published unconditionally
	# per the same "even at 0" precedent the other three streams use). Folded in immediately so this
	# new stream never becomes a fifth hidden-income gap of the exact kind task #79 was opened to
	# close.
	if = {
		limit = { has_variable = qing_opium_income_last }
		add = var:qing_opium_income_last
	}
}

INCOME_national_total_from_subjects = {
	# Scope: Country
	value = 0
	if = {
		limit = {
			is_subject = no
		}
		every_subject = {
			add = INCOME_subject_overlord_duty
		}
	}
	else = {
		subtract = INCOME_subject_overlord_duty
		every_subject = {
			add = INCOME_subject_overlord_duty
		}
	}
}

# MILITARY_supplies_income_country: the good's Total demand reduced by its own shortage

# shortage_early_munitions) — the clean per-good sums below each use their OWN good's shortage.
