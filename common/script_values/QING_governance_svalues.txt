# [#34] KEJU DEGREE PRESTIGE — the soft-preference weight that makes the examination
# the SOURCE of the empire's civil servants. Added to every candidate-ranking value
# below, so a keju graduate outscores an equally-skilled commoner wherever the council,
# the ambanate, or a legation picks a man — WITHOUT a hard lock (a degreeless character
# can still be seated when the graduate pool is empty, so the realm never soft-locks for
# want of a laureate). The ladder mirrors the degree hierarchy (translate/academy floor
# → jinshi/hanlin apex), and is deliberately large relative to the 0..~9 skill axis so a
# degree is the dominant sort key: even a modest graduate heads the bench over a brilliant
# commoner, exactly as the historical bureaucracy privileged the metropolitan laureate.
# Evaluated in a CHARACTER scope. Conditional add-by-trait is the proven script-value
# idiom (cf. religionator_svalue, 00_governor_policies.txt). Degrees are mutually exclusive
# (opposites), so at most one branch adds.
qing_degree_prestige_svalue = {
	value = 0
	if = { limit = { has_trait = hanlin }       add = 30 }   # 翰林 — the Academy apex
	else_if = { limit = { has_trait = jinshi }       add = 25 }   # 進士 — the metropolitan laureate
	else_if = { limit = { has_trait = fanyi_jinshi } add = 25 }   # 繙譯進士 — the banner-track laureate (amban stock)
	else_if = { limit = { has_trait = gongshi }      add = 20 }   # 貢士 — passed the metropolitan, awaiting the palace
	else_if = { limit = { has_trait = juren }        add = 15 }   # 舉人 — the provincial laureate
	else_if = { limit = { has_trait = shengyuan }    add = 10 }   # 生員/秀才 — the licentiate
	else_if = { limit = { has_trait = jiansheng }    add = 5  }   # 監生 — the purchased/academy floor
}

# [#34] KEJU PASS CHANCE — the 0..100 probability a coming-of-age court character passes
# the examination (used as `chance = qing_keju_pass_chance_svalue` in QING_exam_sit_candidate).
# Evaluated in the CANDIDATE'S character scope. Blends the realm's metropolitan pass-rate
# (employer.var:qing_exam_pass_rate, itself ladder-health net of corruption/捐納 drag) with the
# candidate's own scholarly aptitude (finesse + charisma), so an able child of a vigorous realm
# passes readily while a dull one in a clogged ladder rarely does. Bounded 5..90 (the exam is
# never a certainty nor wholly closed). Reading a country var via the employer link is done
# through a guarded `if = { limit = { employer = { var:X >= N } } }` band (the proven form —
# a bare `add = employer.var:X` accessor is not attested), so the pass-rate enters as a coarse
# floor rather than a continuous term; the finesse/charisma terms then separate candidates.
qing_keju_pass_chance_svalue = {
	value = 20
	# scholarly aptitude: finesse is the examiner's chief measure, charisma the lesser.
	add = finesse
	add = finesse
	add = charisma
	# the realm's pass-rate enters as a coarse band (guarded employer read).
	if = { limit = { exists = employer  employer = { has_variable = qing_exam_pass_rate  var:qing_exam_pass_rate >= 75 } }  add = 40 }
	else_if = { limit = { exists = employer  employer = { has_variable = qing_exam_pass_rate  var:qing_exam_pass_rate >= 60 } }  add = 30 }
	else_if = { limit = { exists = employer  employer = { has_variable = qing_exam_pass_rate  var:qing_exam_pass_rate >= 45 } }  add = 20 }
	else_if = { limit = { exists = employer  employer = { has_variable = qing_exam_pass_rate  var:qing_exam_pass_rate >= 30 } }  add = 10 }
	# bound: never a certainty, never wholly closed.
	min = 5
	max = 90
}

# [#61 2026-08-08] AMBAN-specific candidate ranking = the diplomatic-broker skills only.
# An amban (駐紮大臣) is a persuasion/negotiation post, not a generalist councillor, so the
# picker + auto-draw rank on CHARISMA (weighted double, the chief broker measure) + FINESSE —
# matching qing_amban_negotiate_chance_svalue (25 + 2*charisma + finesse). NOT martial/zeal.
# Keeps the keju degree-prestige preference (a translation-examination laureate still heads the
# bench) and CARRIES the #47 lesser-taint deprioritization (a botched-negotiation amban ranks
# last), so switching the amban sites off combined_stats_council_svalue does not lose the taint.
# Evaluated in a character scope. Used ONLY by the two amban order_by sites (se_QING_AMBAN.txt);
# the generalist GC pickers keep combined_stats_council_svalue below.
qing_amban_rank_svalue = {
	value = 0
	add = charisma
	add = charisma
	add = finesse
	add = qing_degree_prestige_svalue
	if = {
		limit = { has_trait = disgraced }
		subtract = 100
	}
	# [#40 2026-08-09] RECALL cooldown penalty — a just-recalled amban (qing_amban_recalled_recently,
	# stamped ~2y in QING_amban_recall) is docked so every fresh bench laureate outranks him and he is
	# picked LAST, not instantly re-appointed (the reported bug). Same -100 shape as the disgraced dock
	# above (dwarfs the ~0-40 stat band without going unbounded). He stays eligible as a last resort;
	# auto-expires with the timed variable, after which he ranks on merit again.
	if = {
		limit = { has_variable = qing_amban_recalled_recently }
		subtract = 100
	}
}

# Combined 4-skill worth of a character, used to rank council/office candidates
# (order_by) and — historically — to weight the council's effectiveness. Mirrors
# combined_stats_olympics_svalue in 00_event_values.txt. Evaluated in a
# character scope. [#34] The keju degree prestige is folded in so graduates head
# the generalist (chancellor / regent / first-vacant) benches.
combined_stats_council_svalue = {
	value = 0
	add = martial
	add = finesse
	add = charisma
	add = zeal
	add = qing_degree_prestige_svalue
	# [#47 2026-08-07] LESSER-TAINT deprioritization. A man carrying disgraced_cmod ("Scandalously
	# Tainted" — a botched amban negotiation, etc.) stays ELIGIBLE (only completely_disgraced_cmod
	# hard-bars, via QING_char_hard_disgraced), but this order_by score docks him a large fixed
	# penalty so every untainted candidate outranks him and he is picked LAST. The taint auto-expires
	# (5y), after which the penalty lifts and he ranks on merit again. -100 dwarfs the ~0-40 stat band
	# so a tainted man never beats a clean one, without going negative-unbounded (single flat term).
	if = {
		limit = { has_trait = disgraced }
		subtract = 100
	}
}

# [meter-concretize #10 2026-08-06] Real oasis trade — sum of num_goods_produced across the Dzungaria +
# Tarim oasis provinces (the Western Regions caravan economy). SUBJECT-held ground (ILI/XNG), so iterate by
# AREA (the #91 owner-independent idiom) — every_owned_province returns 0 for CHI. area:+every_area_province
# inside a script_value is proven (00_mission_seleukid.txt:114); num_goods_produced bare province read is
# proven (GOODS_svalues.txt:1352). Read by QING_caravan_recompute_target via set_variable (the correct form —
# NOT prev.num_goods_produced in an effect, which is unproven per review MED-1).
QING_caravan_oasis_trade_svalue = {
	# [meter-concretize #10 / USER refine 2026-08-06] TRADE, not just production. num_goods_produced measures
	# PRODUCTION (which drives trade but isn't trade itself); a caravan entrepôt like Kashgar is prosperous
	# by RE-EXPORTING goods it doesn't make. So mirror the mod's own trade metric all_region_trade_svalue
	# (EE_svalues.txt:65): num_goods_produced PLUS the regional_center_of_trade_level_N bonus (+5..+25). Those
	# center-of-trade modifiers are LIVE MOD trade infrastructure (stamped by tradezone_setup_effect in
	# se_TRADE.txt, the mod tradezone sim — NOT obsolete vanilla trade-node artifacts; tracked in
	# list_of_trade_centers), so a real oasis trade hub is credited for its throughput, not only its output.
	# NOTE: state_commerce_income would be the TRUEST "trade itself" read, but it's STATE-scope and the oasis
	# states are subject-held (ILI/XNG) — unreachable from CHI's every_governorship_state; area-province +
	# center-of-trade is the best trade-not-production metric reachable by the mandatory area iteration.
	value = 0
	area:Dzungaria = {
		every_area_province = {
			add = num_goods_produced
			if = { limit = { has_province_modifier = regional_center_of_trade_level_1 } add = 5 }
			else_if = { limit = { has_province_modifier = regional_center_of_trade_level_2 } add = 10 }
			else_if = { limit = { has_province_modifier = regional_center_of_trade_level_3 } add = 15 }
			else_if = { limit = { has_province_modifier = regional_center_of_trade_level_4 } add = 20 }
			else_if = { limit = { has_province_modifier = regional_center_of_trade_level_5 } add = 25 }
		}
	}
	area:Tarim = {
		every_area_province = {
			add = num_goods_produced
			if = { limit = { has_province_modifier = regional_center_of_trade_level_1 } add = 5 }
			else_if = { limit = { has_province_modifier = regional_center_of_trade_level_2 } add = 10 }
			else_if = { limit = { has_province_modifier = regional_center_of_trade_level_3 } add = 15 }
			else_if = { limit = { has_province_modifier = regional_center_of_trade_level_4 } add = 20 }
			else_if = { limit = { has_province_modifier = regional_center_of_trade_level_5 } add = 25 }
		}
	}
}

# Denominator: coverable county seats = owned CITY provinces (mirrors the 1763 seed's own

	value = 0
	every_owned_province = {
		limit = { has_city_status = yes }
		add = 1
	}

# Bureau reach 0..100 = fraction of coverable county seats that have a 衙門. COUNTS COVERED
# PROVINCES, not building levels (max_amount=1 on the yamen makes num ∈ {0,1}/province, so
# `add = 1` under a has_building limit = a true coverage fraction — cf. FUNC_num_ports_in_country
# shape). div/0 floored at 1 (proven ADMIN_svalues idiom). Scope: Country.
QING_bureau_reach = {
	value = 0
	every_owned_province = {
		limit = { has_building = qing_yamen_building }
		add = 1
	}
	multiply = 100
	divide = { value = QING_city_province_count  min = 1 }
	min = 0
	max = 100
}

# The one unproven capability #112c rests on: a WIN-PROBABILITY read from TWO characters' relative
# attributes, rendered as competing percentages that sum to 100 (the amban svalue is a ONE-SIDED
# solo roll; §9b flagged this two-sided form as NET-NEW). Both operands read SAVED SCOPES
# (scope:qing_super, scope:qing_aqsaqal) — proven that a script_value can read scope:X.attr
# (00_event_values.txt:1807-1810 reads scope:target.martial/finesse/zeal/charisma).
#
# MODEL: a concession negotiation. The superintendent's hand = finesse (fiscal/administrative
# skill) + charisma (winning the begs & merchants). The aqsaqal's hand = finesse (his own
# bargaining) + zeal (his khoja/Islamic leverage, the resistance to a Confucian settlement).
# chance = 50 + 4×(super_hand − aqsaqal_hand), clamped 10..90 (never certain, never hopeless) —
# the same clamp shape as qing_amban_negotiate_chance_svalue. The complement (100 − chance) is
# the aqsaqal's odds; the two are computed in the SAME (CHI) scope so they always sum to 100.
# ======================================================================
qing_super_contest_chance_svalue = {
	value = 50
	# superintendent's hand (+): finesse + charisma, each weighted x2 (the amban-svalue idiom:
	# repeated `add` for a coefficient — QING_governance_svalues.txt:105-106).
	add = scope:qing_super.finesse
	add = scope:qing_super.finesse
	add = scope:qing_super.charisma
	add = scope:qing_super.charisma
	# aqsaqal's hand (−): finesse + zeal, same x2 weight (so the contest is symmetric)
	subtract = scope:qing_aqsaqal.finesse
	subtract = scope:qing_aqsaqal.finesse
	subtract = scope:qing_aqsaqal.zeal
	subtract = scope:qing_aqsaqal.zeal
	min = 10
	max = 90
}

# the aqsaqal's odds = 100 − the superintendent's, in the SAME scope so they sum to 100.
qing_super_contest_fail_svalue = {
	value = 100
	subtract = qing_super_contest_chance_svalue
}
