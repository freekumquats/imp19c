########################################################################
# qing_central_asia_missions.txt — THE WESTERN REGIONS (經略西域), #448.
#
# A dedicated expansion tree carrying the Qing PAST the Xinjiang consolidation
# (qing_xinjiang_missions.txt, #422, which pacifies Dzungaria + the Tarim) and out
# into Central Asia proper: the Kazakh steppe (the three Juz — GKH/ORT/KSH), and the
# khanates of the Ferghana and Transoxiana (Kokand KOK, Bukhara BUK, Khiva KHV).
#
# Historically the Qianlong conquest of Xinjiang (1755-59) brought the empire to the
# doorstep of the Kazakh hordes and the Central Asian khanates, who sent tribute to
# Beijing; this tree lets the player push that dominance further — the steppe hordes
# brought into the tributary system, the caravan cities of the Silk Road overawed, and
# (at the far reach) the Ferghana valley itself held.
#
# As with the Burma (#421) tree, the PLAYER does the real work: the task goals gate on
# the Qing genuinely bringing the steppe tags under the wing (is_subject_of) or
# CONTROLLING the ground (provinces in the Central Asian regions). What is scripted
# (se_QING_CENTRAL_ASIA.txt) is only the texture — the tributary embrace of the hordes
# and khanates (FUNC_make_subject) and the lasting dominion of the Western Regions.
#
# CAPSTONE — 西域大定 THE WESTERN REGIONS PACIFIED: the steppe and the caravan cities
# under the Son of Heaven; the lasting qing_western_regions dominion modifier sealed.
#
# Gating: CHI-only, player-only (ai_chance factor 0). Requires the Xinjiang tree's
# consolidation as the launch-point is unnecessary as a hard gate (the player may drive
# west by any road), but the potential requires at least one Central Asian neighbour to
# exist. Tasks cost treasury / political influence. Icons reuse stock test1/2/3.
#
# Sources: Peter Perdue, China Marches West: The Qing Conquest of Central Eurasia (2005);
#   James Millward, Beyond the Pass: Economy, Ethnicity and Empire in Qing Xinjiang (1998)
#   & Eurasian Crossroads: A History of Xinjiang (2007); Joseph Fletcher, "Ch'ing Inner
#   Asia" (Cambridge History of China v.10).
########################################################################

qing_central_asia_mission = {

	# Chine, la Tartarie Chinoise, et le Thibet" (1734), via Wikimedia Commons. Public domain.
	icon = qing_central_asia_mission
	header = mission_image_qing_central_asia

	repeatable = no

	chance = { }
	ai_chance = { factor = 0 }   # player-only

	potential = {
		tag = CHI
		is_ai = no
		# there must be a Central Asian power to reach for — a Kazakh horde or a khanate.
		OR = {
			exists = c:GKH
			exists = c:ORT
			exists = c:KSH
			exists = c:KOK
			exists = c:BUK
			exists = c:KHV
		}
	}

	abort = {
		NOT = { tag = CHI }
	}

	on_start = {
		LOG_line = { sys = QING  msg = "MISSION the Western Regions (經略西域) started for" }
	}

	on_completion = {
		LOG_line = { sys = QING  msg = "MISSION the Western Regions COMPLETED for" }
	}

	# ==================================================================
	# ROOT — THE ROAD BEYOND THE PASS (出關). Commit the frontier armies and the caravan
	# revenues to the westward push past Jiayuguan and the Ili valley — the point from
	# which the empire reaches into the steppe and the khanates.
	# ==================================================================
	qing_ca_beyond_pass = {
		icon = qing_ca_beyond_pass
		allow = {
			treasury >= 100
			political_influence >= 25
		}
		on_start = {
			add_treasury = -100
			add_political_influence = -25
		}
		on_completion = {
			custom_tooltip = qing_ca_beyond_pass_tt
			current_ruler = { add_popularity = 10 }
			LOG_line = { sys = QING  msg = "MISSION task Central Asia road beyond the pass for" }
		}
	}

	# ==================================================================
	# THE KAZAKH STEPPE (撫哈薩克). Bring the three Juz — the Great, Middle and Little
	# Hordes — into the tributary system, as they historically sent tribute to Qianlong
	# after the fall of the Dzungars. At least two of the three hordes under the wing.
	# ==================================================================
	qing_ca_kazakh = {
		icon = qing_ca_kazakh
		requires = { qing_ca_beyond_pass }
		allow = {
			political_influence >= 30
			custom_tooltip = {
				text = qing_ca_needs_kazakh_tt
				# at least one of the three Kazakh hordes must be on the map to be reached.
				OR = { exists = c:GKH  exists = c:ORT  exists = c:KSH }
			}
		}
		on_start = {
			add_political_influence = -30
		}
		on_completion = {
			custom_tooltip = qing_ca_kazakh_tt
			# bring every extant Kazakh horde into the tributary fold (the make-subject-by-fiat
			# idiom the southern-frontier tree uses; each guarded on existence + not-already-bound).
			QING_ca_embrace_kazakh = yes
			current_ruler = { add_popularity = 15 }
			LOG_line = { sys = QING  msg = "MISSION task Central Asia Kazakh hordes embraced for" }
		}
	}

	# ==================================================================
	# THE CARAVAN CITIES (懾服浩罕). Overawe the khanates of the Silk Road — Kokand,
	# Bukhara, Khiva — into the tributary system. At least one khanate under the wing.
	# ==================================================================
	qing_ca_khanates = {
		icon = qing_ca_khanates
		requires = { qing_ca_beyond_pass }
		allow = {
			political_influence >= 30
			custom_tooltip = {
				text = qing_ca_needs_khanates_tt
				# at least one of the Silk Road khanates must be on the map to be reached.
				OR = { exists = c:KOK  exists = c:BUK  exists = c:KHV }
			}
		}
		on_start = {
			add_political_influence = -30
		}
		on_completion = {
			custom_tooltip = qing_ca_khanates_tt
			# overawe every extant Silk Road khanate into the tributary fold (guarded per tag).
			QING_ca_embrace_khanates = yes
			# [#27 salvage 2026-08-02] press the imperial claim over the Ferghana oasis cities —
			# Kokand (p:110), Tashkent (p:9370) and the rest of the khanate heartland. This RE-HOMES
			# the Fergana claims the deleted colonization-tree qing_col_central_asia task used to grant
			# (its Turkestan/steppe claims are covered by qing_ca_begs' region sweep; only these
			# Fergana-region ones needed a new home). Region-swept + guarded, mirroring qing_ca_begs.
			every_province = {
				limit = { is_in_region = Fergana  NOT = { owner = ROOT } }
				add_claim = ROOT
			}
			current_ruler = { add_popularity = 15 }
			LOG_line = { sys = QING  msg = "MISSION task Central Asia khanates overawed for" }
		}
	}

			custom_tooltip = qing_ca_tuntian_tt

			current_ruler = { add_popularity = 10 }
			add_stability = 1
			# [#234] the tuntian made concrete: plant a military colony (屯田) on the most-
			# populous held Turkestan oasis — the soldier-farms that fed the far garrisons.
			if = {
				limit = { any_owned_province = { is_in_region = Turkestan  NOT = { has_building = qing_military_colony_building } } }
				ordered_owned_province = {
					order_by = total_population
					max = 1
					limit = { is_in_region = Turkestan  NOT = { has_building = qing_military_colony_building } }
					add_building_level = qing_military_colony_building
				}
			}
			LOG_line = { sys = QING  msg = "MISSION task Central Asia military colonies for" }
		}
	}

			treasury >= 40
		}
		on_start = { add_treasury = -40 }
		on_completion = {
			custom_tooltip = qing_ca_border_tt
			add_country_modifier = { name = qing_frontier_reclamation  duration = 7300 }
			current_ruler = { add_popularity = 10 }
			add_stability = 1
			# [#234] the karun (卡倫) picket line made concrete: raise a frontier fort on the
			# westernmost held frontier province (Fergana if held, else the Turkestan march).
			if = {
				limit = { any_owned_province = { is_in_region = Fergana  NOT = { has_building = qing_frontier_fort_building } } }
				ordered_owned_province = {
					order_by = total_population
					max = 1
					limit = { is_in_region = Fergana  NOT = { has_building = qing_frontier_fort_building } }
					add_building_level = qing_frontier_fort_building
				}
			}
			else_if = {
				limit = { any_owned_province = { is_in_region = Turkestan  NOT = { has_building = qing_frontier_fort_building } } }
				ordered_owned_province = {
					order_by = total_population
					max = 1
					limit = { is_in_region = Turkestan  NOT = { has_building = qing_frontier_fort_building } }
					add_building_level = qing_frontier_fort_building
				}
			}
			LOG_line = { sys = QING  msg = "MISSION task Central Asia frontier fixed for" }
		}
	}

	# ==================================================================
	# 安西都護府 THE PROTECTORATE-GENERAL OF THE PACIFIED WEST (relocated here from the
	# colonization tree, #27 P3b 2026-08-02). The Tang held the Silk Road through the
	# 安西都護府, seated at Kucha. Revived as an EIC-model FRONTIER MARCH: a self-governing
	# oasis march carved from the CONQUERED khanate ground (Ferghana / Bukhara / Khwarezm —
	# NOT the Kazakh steppe "Turkestan", and NOT already-Qing Tarim/Xinjiang: the map-region
	# trap of DESIGN §4.1), holding the OTHER conquered khanates as its own princely-state
	# subjects. Founded here in the tree that actually conquers the khanates (DESIGN §4.2),
	# off the Ferghana-hold task. Land march (no navy). Replaces the OLD backwards qing_col_anxi,
	# which carved the region as a fake tributary via QING_establish_protectorate.
	# ==================================================================
	qing_ca_anxi_march = {
		icon = qing_ca_ferghana
		# [#27 P3b-1 review-fix] require the CAPSTONE, not just qing_ca_ferghana. Founding the march
		# carves CHI's Fergana land + reparents KOK/BUK/KHV down to the march — which would BREAK the
		# capstone's allow (it needs CHI to own Fergana ground + hold >=2 khanate/horde subjects) if the
		# march were founded first. Gating the march AFTER the capstone (the Western Regions pacified) both
		# avoids the soft-lock and is narratively right: you pacify the region, THEN erect the 都護府 over it.
		requires = { qing_ca_capstone }
		allow = {
			political_influence >= 30
			custom_tooltip = {
				text = qing_ca_needs_oasis_land_tt
				# at least 3 held provinces across the conquered oasis-khanate regions.
				any_owned_province = {
					count >= 3
					OR = {
						is_in_region = Fergana
						is_in_region = Bukhara
						is_in_region = Khwarezm
					}
				}
			}
		}
		on_start = {
			add_political_influence = -30
			LOG_line = { sys = QING  msg = "MISSION task START Anxi march for" }
		}
		on_completion = {
			custom_tooltip = qing_ca_anxi_march_tt
			# carve the march from the CONQUERED oasis ground (held, non-capital, non-CHI-core).
			every_owned_province = {
				limit = {
					is_capital = no
					OR = {
						is_in_region = Fergana
						is_in_region = Bukhara
						is_in_region = Khwarezm
					}
				}
				add_to_list = qing_protectorate_list
			}
			# subordinate the OTHER conquered khanates (still CHI subjects) as the march's own
			# princely states. Guarded per tag inside QING_found_march (skips CHI/non-free/self).
			# every_country + free-OR-CHI-subject filter (NOT every_subject): robust whether the
			# khanate is still a CHI subject or was freed by war before founding. QING_found_march
			# reparents a CHI subject / takes a free one / skips a foreign-held one.
			every_country = {
				limit = {
					OR = { tag = KOK  tag = BUK  tag = KHV }
					OR = { is_subject_of = ROOT  is_subject = no }
				}
				add_to_list = qing_march_locals_list
			}
			QING_found_march = {
				list = qing_protectorate_list
				name_key = qing_protectorate_anxi_name
				adj_key = qing_protectorate_anxi_adj
				maritime = no
				log_tag = ANXI
				locallist = qing_march_locals_list
			}
			add_country_modifier = { name = qing_col_protectorate_west  duration = -1 }
			# an autonomous oasis march eases the strain of direct alien rule
			QING_DECLINE_nudge = { var = qing_ethnic_tension  amount = -5 }
			current_ruler = { add_popularity = 20 }
			LOG_line = { sys = QING  msg = "MISSION task Anxi march for" }
		}
	}

	# ==================================================================
	# 安北都護府 THE PROTECTORATE-GENERAL OF THE PACIFIED NORTH (relocated here from the
	# colonization tree, #27 P3b-2 2026-08-02). The Tang held the steppe through the
	# 安北都護府 on the Orkhon. Revived as an EIC-model FRONTIER MARCH of the KAZAKH STEPPE
	# — map region `Turkestan` (the Akmolinsk/Turgai nomad steppe, NOT the oasis khanates:
	# the map-region trap of DESIGN §4.1). Carved from the conquered steppe ground, holding
	# the three Kazakh hordes (GKH/ORT/KSH — made CHI tributaries by qing_ca_kazakh) as its
	# own princely-state subjects. Founded off the capstone (avoids the same soft-lock the
	# Anxi march would cause; see qing_ca_anxi_march). Land march (no navy). Replaces the OLD
	# backwards qing_col_anbei, which carved CHI-core Mongolia as a fake tributary.
	# ==================================================================
	qing_ca_anbei_march = {
		icon = qing_ca_kazakh
		requires = { qing_ca_capstone }
		allow = {
			political_influence >= 30
			custom_tooltip = {
				text = qing_ca_needs_steppe_land_tt
				# at least 3 held provinces across the conquered Kazakh steppe (region Turkestan).
				any_owned_province = { count >= 3  is_in_region = Turkestan }
			}
		}
		on_start = {
			add_political_influence = -30
			LOG_line = { sys = QING  msg = "MISSION task START Anbei march for" }
		}
		on_completion = {
			custom_tooltip = qing_ca_anbei_march_tt
			# carve the march from the CONQUERED steppe ground (held, non-capital, non-CHI-core).
			every_owned_province = {
				limit = { is_in_region = Turkestan  is_capital = no }
				add_to_list = qing_protectorate_list
			}
			# subordinate the three Kazakh hordes (still CHI subjects) as the march's princely states.
			# every_country + free-OR-CHI-subject filter (NOT every_subject): the capstone does NOT
			# guarantee the hordes are CHI subjects (qing_ca_kazakh is OFF the capstone-required path),
			# so a conquered-but-independent horde must still be captured. QING_found_march reparents a
			# CHI subject / takes a free one / skips a foreign-held one.
			every_country = {
				limit = {
					OR = { tag = GKH  tag = ORT  tag = KSH }
					OR = { is_subject_of = ROOT  is_subject = no }
				}
				add_to_list = qing_march_locals_list
			}
			QING_found_march = {
				list = qing_protectorate_list
				name_key = qing_protectorate_anbei_name
				adj_key = qing_protectorate_anbei_adj
				maritime = no
				log_tag = ANBEI
				locallist = qing_march_locals_list
			}
			add_country_modifier = { name = qing_col_protectorate_north  duration = -1 }
			current_ruler = { add_popularity = 20 }
			LOG_line = { sys = QING  msg = "MISSION task Anbei march for" }
		}
	}
