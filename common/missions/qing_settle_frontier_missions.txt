########################################################################
# qing_settle_frontier_missions.txt — SETTLE THE NOMADS (定牧墾邊), #451.
#
# The Qing consolidation of the great northern frontier — the Mongol steppe and the
# Manchurian homeland — from a belt of autonomous military governorships and banner
# pastures into settled, directly-held imperial ground. 定牧 "fix the pastures" (the
# banner/league system that pinned the once-mobile Mongols to fixed grazing) + 墾邊
# "reclaim the frontier" (the 屯田 military-agricultural colonies and the slow tide of
# Han settlement that broke the sod of Inner Mongolia and the Northeast).
#
# The 1763 board (setup/main/00_default.txt): the north is held for the Qing (CHI) not
# directly but through a tier of subjects — the Mongol military governorships (ULS
# Uliastai / KBD Kobdo, with MGA the Urga monastic state nested under ULS) and the
# Manchurian ones (MNC Manchuria / MKD Mukden / HLJ Heilongjiang), plus TNN (Tannu Tuva /
# the Uriankhai) as a tributary. This tree is the arc that INTEGRATES that tier — the
# player brings the governorships' ground under direct rule and settles it.
#
# As with the other Qing enterprise trees, the PLAYER does the real work: the task goals
# gate on the Qing genuinely OWNING ground in the frontier regions (Mongolia; Liaoning +
# the Far East), which the player reaches by integrating / annexing the autonomous
# governorships that hold it — and on the Tannu tie. What is scripted
# (se_QING_SETTLE_FRONTIER.txt) is only the texture: the reclamation colonies laid on the
# settled ground, and (a safety net) the tributary embrace of Tannu if it has broken away.
#
# It FORKS after the shared opening (the frontier policy set):
#   ARC N — 定漠北 THE MONGOL LANDS: integrate the Mongol governorships and settle the
#     steppe — hold ground across Mongolia, the banners pinned and the colonies planted.
#   ARC E — 實東北 THE MANCHURIAN HOMELAND: integrate the Northeast governorships and
#     settle the taiga — hold ground across Liaoning and the Far East, the willow palisade
#     opened to the plough.
#
# CAPSTONE — 定牧墾邊 THE FRONTIER SETTLED: the steppe and the taiga under direct rule and
# the plough, the northern march made imperial ground; the lasting qing_frontier_settled
# dominion modifier sealed.
#
# Gating: CHI-only, player-only (ai_chance factor 0). Tasks cost treasury / political
# influence and gate on the advancing OWNERSHIP of the frontier regions. Icons reuse stock
# test1/2/3 (the whole Qing suite's placeholder convention).
#
# Sources: Peter Perdue, China Marches West (2005); Owen Lattimore, Inner Asian Frontiers
#   of China (1940); Mark Elliott, The Manchu Way: The Eight Banners and Ethnic Identity in
#   Late Imperial China (2001); C. Patterson Giersch on Qing frontier settlement.
########################################################################

qing_settle_frontier_mission = {
	# Art source: Qing painted map of the Aihun (Aigun) frontier garrison on the Amur,
	# Kangxi era (1689-1722), via Wikimedia Commons. Public domain.
	icon = qing_settle_frontier_mission
	header = mission_image_qing_settle_frontier

	repeatable = no

	chance = { }
	ai_chance = { factor = 0 }   # player-only

	potential = {
		tag = CHI
		is_ai = no
	}

	abort = {
		NOT = { tag = CHI }
	}

	on_start = {
		LOG_line = { sys = QING  msg = "MISSION settle the nomads (定牧墾邊) started" }
	}

	on_completion = {
		LOG_line = { sys = QING  msg = "MISSION settle the nomads COMPLETED" }
	}

	# ==================================================================
	# ROOT — THE FRONTIER POLICY (定邊策). Set the policy of the settled frontier: fix the
	# banners to their pastures (定牧), open the willow palisade, and commit the treasury to
	# the military-agricultural colonies. The point of commitment for the northern march.
	# ==================================================================
	qing_settle_policy = {
		icon = qing_settle_policy
		allow = {
			treasury >= 120
			political_influence >= 25
		}
		on_start = {
			add_treasury = -120
			TREASURY_LOG_it = { amount = -120 }
			add_political_influence = -25
		}
		on_completion = {
			custom_tooltip = qing_settle_policy_tt
			# a safety net: if Tannu Tuva has slipped the leash, bring it back under the wing.
			QING_settle_embrace_tannu = yes
			current_ruler = { add_popularity = 10 }
			LOG_line = { sys = QING  msg = "MISSION task settle-frontier policy set" }
		}
	}

	# ==================================================================
	# ARC N — 定漠北 THE MONGOL LANDS. Integrate the Mongol military governorships (ULS/KBD,
	# and MGA at Urga) and settle the steppe: hold the ground of Mongolia directly, the
	# banners pinned to their leagues and the colonies planted among them.
	# ==================================================================
	qing_settle_mongolia = {
		icon = qing_settle_mongolia
		requires = { qing_settle_policy }
		allow = {
			treasury >= 80
			custom_tooltip = {
				text = qing_settle_needs_mongolia_tt
				# the Qing hold ground directly across the Mongol lands (reached by integrating
				# the autonomous governorships that hold them as CHI subjects in 1763).
				any_owned_province = { count >= 8  is_in_region = Mongolia }
			}
		}
		on_start = {
			add_treasury = -80
			TREASURY_LOG_it = { amount = -80 }
		}
		on_completion = {
			custom_tooltip = qing_settle_mongolia_tt
			# lay the reclamation colonies on the settled Mongol ground.
			QING_settle_reclaim_mongolia = yes
			# [P2] plant the industrial/garrison texture (屯田 colony + 駐防 fort) on the held Mongol ground.
			QING_settle_plant_works = { region = Mongolia }
			# [P1] forge the nation: naturalise the Mongol frontier peoples + raise the civic-identity ceiling.
			QING_settle_forge_nation = { group = mongolic }
			current_ruler = { add_popularity = 15 }
			add_stability = 1
			LOG_line = { sys = QING  msg = "MISSION task settle-frontier Mongol lands settled" }
		}
	}

	# ==================================================================
	# ARC E — 實東北 THE MANCHURIAN HOMELAND. Integrate the Northeast governorships
	# (MNC/MKD/HLJ) and settle the taiga: hold the ground of Liaoning and the Far East
	# directly, the willow palisade opened to the plough and the homeland peopled.
	# ==================================================================
	qing_settle_manchuria = {
		icon = qing_settle_manchuria
		requires = { qing_settle_policy }
		allow = {
			treasury >= 80
			custom_tooltip = {
				text = qing_settle_needs_manchuria_tt
				# the Qing hold ground directly across the Manchurian homeland (Liaoning + the
				# Far East), reached by integrating the autonomous governorships that hold them.
				any_owned_province = {
					count >= 8
					OR = { is_in_region = Liaoning  is_in_region = Far_East }
				}
			}
		}
		on_start = {
			add_treasury = -80
			TREASURY_LOG_it = { amount = -80 }
		}
		on_completion = {
			custom_tooltip = qing_settle_manchuria_tt
			# lay the reclamation colonies on the settled Manchurian ground.
			QING_settle_reclaim_manchuria = yes
			# [P2] plant the industrial/garrison texture on the held Manchurian ground (Liaoning + Far East).
			QING_settle_plant_works = { region = Liaoning }
			QING_settle_plant_works = { region = Far_East }
			# [P1] forge the nation: naturalise the jurchen/Tungusic frontier peoples + raise the civic ceiling.
			QING_settle_forge_nation = { group = jurchen }
			current_ruler = { add_popularity = 15 }
			add_stability = 1
			LOG_line = { sys = QING  msg = "MISSION task settle-frontier Manchurian homeland settled" }
		}
	}

	# ==================================================================
	# 改設郡縣 THE GOVERNORSHIPS INTEGRATED (改盟旗為郡縣). Bring the autonomous Mongol
	# governorships (ULS/KBD, MGA at Urga) fully into the direct provincial administration.
	# ==================================================================
	qing_settle_mongol_govs = {
		icon = qing_settle_mongol_govs
		requires = { qing_settle_mongolia }
		allow = {
			political_influence >= 30
			custom_tooltip = {
				text = qing_settle_needs_mongol_govs_tt
				NOR = {
					AND = { exists = c:ULS  NOT = { c:ULS = { is_subject_of = ROOT } } }
					AND = { exists = c:KBD  NOT = { c:KBD = { is_subject_of = ROOT } } }
				}
			}
		}
		on_start = { add_political_influence = -30 }
		on_completion = {
			custom_tooltip = qing_settle_mongol_govs_tt
			QING_settle_plant_works = { region = Mongolia }
			current_ruler = { add_popularity = 12 }
			add_stability = 1
			LOG_line = { sys = QING  msg = "MISSION task settle-frontier Mongol governorships integrated" }
		}
	}

	# ==================================================================
	# 移民實邊 PEOPLE THE MARCH (廣募墾民). The great settlement drive — Han farmers to the

	# ==================================================================

		requires = { qing_settle_mongol_govs  qing_settle_willow }

			treasury >= 70
		}

			LOG_line = { sys = QING  msg = "MISSION task settle-frontier migration drive" }
		}
	}

	# ==================================================================
	# 邊軍駐防 THE FRONTIER GARRISON (邊軍常駐). Plant a standing garrison across the settled
	# march — the banner forts and Green-Standard posts that hold the peopled frontier.
	# ==================================================================
	qing_settle_garrison = {
		icon = qing_settle_garrison
		requires = { qing_settle_migration }
		allow = {
			treasury >= 60
		}
		on_start = { add_treasury = -60  TREASURY_LOG_it = { amount = -60 } }
		on_completion = {
			custom_tooltip = qing_settle_garrison_tt
			# [#234] the standing frontier garrison made concrete: plant a banner garrison (駐防
			# 八旗) on the most-populous held frontier province — the other settle tasks plant the
			# 屯田 colony + 卡倫 fort via QING_settle_plant_works, but the standing garrison itself
			# had no building. Prefers the Mongol steppe, else the Manchurian homeland.
			if = {
				limit = { any_owned_province = { is_in_region = Mongolia  NOT = { has_building = qing_banner_garrison_building } } }
				ordered_owned_province = {
					order_by = total_population
					max = 1
					limit = { is_in_region = Mongolia  NOT = { has_building = qing_banner_garrison_building } }
					add_building_level = qing_banner_garrison_building
				}
			}
			else_if = {
				limit = { any_owned_province = { OR = { is_in_region = Liaoning  is_in_region = Far_East }  NOT = { has_building = qing_banner_garrison_building } } }
				ordered_owned_province = {
					order_by = total_population
					max = 1
					limit = { OR = { is_in_region = Liaoning  is_in_region = Far_East }  NOT = { has_building = qing_banner_garrison_building } }
					add_building_level = qing_banner_garrison_building
				}
			}
			current_ruler = { add_popularity = 12 }
			add_stability = 1
			LOG_line = { sys = QING  msg = "MISSION task settle-frontier garrison planted" }
		}
	}

	# ==================================================================
	# 蒙漢一體 THE FRONTIER FORGED (蒙漢共冶). Weld the frontier peoples — Mongol, Manchu,
	# Tungus, Han settler — into one national community of the northern march.
	# ==================================================================
	qing_settle_forge = {
		icon = qing_settle_forge
		requires = { qing_settle_garrison }
		allow = {
			political_influence >= 25
		}
		on_start = { add_political_influence = -25 }
		on_completion = {
			custom_tooltip = qing_settle_forge_tt
			QING_settle_forge_nation = { group = mongolic }

			current_ruler = { add_popularity = 16 }
			add_stability = 1
			LOG_line = { sys = QING  msg = "MISSION task settle-frontier peoples forged" }
		}
	}

	# ==================================================================
	# CAPSTONE — 定牧墾邊 THE FRONTIER SETTLED. The steppe and the taiga alike under direct
	# rule and the plough — the Mongol lands and the Manchurian homeland made imperial ground,
	# the autonomous governorships consolidated and the frontier peopled. The lasting
	# qing_frontier_settled dominion modifier is sealed: the northern march secured for good.
	# ==================================================================
	qing_settle_capstone = {
		icon = qing_settle_capstone
		requires = { qing_settle_ginseng qing_settle_kulun qing_settle_new_province qing_settle_railway_east }
		allow = {
			custom_tooltip = {
				text = qing_settle_needs_settled_tt
				# ground held directly across BOTH the Mongol lands and the Manchurian homeland.
				any_owned_province = { count >= 12  is_in_region = Mongolia }
				any_owned_province = {
					count >= 12
					OR = { is_in_region = Liaoning  is_in_region = Far_East }
				}
			}
		}
		on_completion = {
			custom_tooltip = qing_settle_capstone_tt
			add_country_modifier = { name = qing_frontier_settled  duration = -1 }
			# [P1] the frontier fully forged into the nation — the standing national-awakening modifier
			# (Design-4 hook, previously unused by any mission): the mass mobilisation of a coherent nation
			# out of the settled march. Guarded not-already-held (idempotent if a later arc re-grants).
			if = {
				limit = { NOT = { has_country_modifier = qing_national_awakening } }
				add_country_modifier = { name = qing_national_awakening  duration = -1 }
			}
			current_ruler = { add_popularity = 40 }
			add_stability = 1
			LOG_line = { sys = QING  msg = "MISSION settle the nomads CAPSTONE (定牧墾邊)" }
		}
	}

	# [task-D stub-fix] made concrete: gated on actually holding ground in the real

	# plough" — plants the same 屯田 frontier-colony building QING_settle_plant_works
	# uses, on up to 2 held Heilongjiang-area provinces that lack it (ordered by

		on_start = { add_treasury = -60  TREASURY_LOG_it = { amount = -60 } }
		on_completion = {
			custom_tooltip = qing_settle_heilongjiang_tt
			if = {
				limit = { any_owned_province = { is_in_area = Heilongjiang  NOT = { has_building = qing_frontier_colony_building } } }

					order_by = total_population
					max = 2
					limit = { is_in_area = Heilongjiang  NOT = { has_building = qing_frontier_colony_building } }

				}
			}

			LOG_line = { sys = QING  msg = "MISSION task Heilongjiang thrown open to the plough (frontier colonies planted)" }
