########################################################################
# qing_mexico_missions.txt — THE MANILA GALLEON REBORN & THE MEXICAN
# ADVENTURE (馬尼拉大帆船 / 墨西哥). #27 (lifted out of
# qing_colonization_missions.txt, 2026-08-02 — see
# design/DESIGN_COLONIZATION_SPLIT.md §C).
#
# For two and a half centuries (1565-1815) the Manila galleon carried Mexican
# silver west to Canton and Chinese silk east to Acapulco — the one sustained
# artery between the Middle Kingdom and the New World, and the pump that drew
# the world's silver into China. The line closed in 1815, the very year the
# 1815 start opens; here the dynasty may re-open it, and follow the silver road
# to its far terminus. History that grounds the arc: France's Second Empire, in
# the American Civil War's shadow, landed at Veracruz (1862) and installed the
# Habsburg archduke Maximilian as Emperor of Mexico (1864-67) — a European
# throne planted in the Americas over the objection of the United States. A
# Qing that has learned the sea can insert itself into that same contest.
#
# The mission FRONT for the Mexican-empire machinery in se_QING_MEXICO.txt /
# design/DESIGN_MEXICO.md (unchanged; only the mission tasks relocate here). The
# qing_col_mexican_empire task legitimately releases a client Empire of Mexico via
# QING_establish_protectorate — that is a real sinosphere-tributary client-state
# release (NOT one of the seven EIC-style frontier marches), so it is left intact.
#
# Gating: CHI-only, player-only (ai_chance factor 0), same self-strengthening /
# High-Qing-era gate as the parent colonization tree. Art reuses the parent
# colonization card + banner (the Pacific enterprise, of which this is the
# American terminus). Icons/effects/vars resolve against the shared codebase.
#
# Sources: Schurz, The Manila Galleon (1939); Cunningham, The French in Mexico
#   (on the Maximilian adventure); see design/DESIGN_MEXICO.md for the arc.
########################################################################

qing_mexico_mission = {
	# Art reuse: the Great Pacific Enterprise card + banner (this arc is that
	# enterprise's American / Gulf terminus). Both DDS present in gfx/interface/.
	icon = qing_colonization_mission
	header = mission_image_qing_colonization

	repeatable = no

	chance = { }
	ai_chance = { factor = 0 }   # player-only

	potential = {
		tag = CHI
		is_ai = no
		# same two-road gate as the parent colonization tree: [develop] a self-
		# strengthening industrial dynasty (progress >= 40), OR [bookmark-1763] the
		# High-Qing zenith under Qianlong (qing_high_qing_era).
		OR = {
			AND = {
				has_variable = qing_selfstr_progress
				var:qing_selfstr_progress >= 40
			}
			has_variable = qing_high_qing_era
		}
	}

	abort = {
		NOT = { tag = CHI }
	}

	on_start = {
		LOG_line = { sys = QING  msg = "MISSION Manila galleon & Mexican adventure started for" }
	}

	on_completion = {
		LOG_line = { sys = QING  msg = "MISSION Manila galleon & Mexican adventure COMPLETED for" }
	}

	# --- 馬尼拉大帆船 THE GALLEON REBORN — re-open the Acapulco silver road ---
	qing_col_galleon = {
		icon = qing_col_galleon
		allow = {
			treasury >= 150
			custom_tooltip = {
				text = qing_col_needs_pacific_reach_tt
				OR = {
					has_country_modifier = qing_selfstr_beiyang_fleet
					has_country_modifier = qing_selfstr_nanyang_fleet
					owns = 1493
				}
			}
		}
		on_start = {
			add_treasury = -150
			LOG_line = { sys = QING  msg = "MISSION task START Manila galleon reborn (spending treasury) for" }
		}
		on_completion = {
			custom_tooltip = qing_col_galleon_tt
			# Acapulco (p:1800) — the galleon's American terminus, the far anchor of the silver road
			if = { limit = { exists = p:1800  NOT = { owns = 1800 } }  p:1800 = { add_claim = ROOT } }
			QING_COLON_frontier_pull = { province = 1800 }
			add_country_modifier = { name = qing_col_silver_road  duration = -1 }
			CURRENCY_grant_country_wealth = { thousands = 80 }
			current_ruler = { add_popularity = 20 }
			# reviving the Pacific bullion trade nettles Britain, the reigning master
			# of world commerce — UNLESS the Daoguang entente has made the Pacific a
			# shared American-Qing lake rather than a contested one.
			if = {
				limit = { NOT = { has_country_modifier = qing_daoguang_doctrine } }
				QING_gp_provoke_britain = { severity = 6 }
			}
			LOG_line = { sys = QING  msg = "MISSION task Manila galleon reborn for" }
		}
	}

	# --- 韋拉克魯斯 VERACRUZ — a foothold on the Gulf, the door to Mexico ---
	qing_col_veracruz = {
		icon = qing_col_veracruz
		requires = { qing_col_galleon }
		allow = {
			treasury >= 170
			custom_tooltip = {
				text = qing_col_needs_acapulco_tt
				OR = {
					owns = 1800
					has_country_modifier = qing_col_silver_road
				}
			}
		}
		on_start = {
			add_treasury = -170
			LOG_line = { sys = QING  msg = "MISSION task START Veracruz foothold (spending treasury) for" }
		}
		on_completion = {
			custom_tooltip = qing_col_veracruz_tt
			# Veracruz (p:2069) — the real 1862 French landing point, the Gulf gate to
			# the Mexican interior. Press it as a play against France, whose own
			# Mexican adventure contests the same ground.
			if = {
				limit = { exists = p:2069  NOT = { owns = 2069 } }
				p:2069 = { add_claim = ROOT }
				QING_gp_frontier_play = { tag = FRA  province = 2069  goal = flag:get_territory }
			}
			add_country_modifier = { name = qing_col_gulf_gate  duration = -1 }
			CURRENCY_grant_country_wealth = { thousands = 70 }
			QING_gp_provoke_france = { severity = 8 }
			LOG_line = { sys = QING  msg = "MISSION task Veracruz foothold for" }
		}
	}

	# --- 墨西哥帝業 THE MEXICAN ADVENTURE — the fork over the throne of Mexico ---
	# Two roads to a crown in the Americas. If the Emperor Emeritus (Napoleon)
	# sits at the Qing court, the dynasty makes common cause with his nephew's
	# Second Empire: a Franco-Qing condominium props up the Habsburg Maximilian,
	# and the enterprise DELIGHTS Napoleon (his family's imperial project served,
	# Britain affronted, his revenge advanced). If he is absent, the Qing plays
	# the adventure for itself — muscling into the New World over the head of the
	# very France that would otherwise own it.
	qing_col_maximilian = {
		icon = qing_col_maximilian
		requires = { qing_col_veracruz }
		allow = {
			treasury >= 200
			political_influence >= 20
		}
		on_start = {
			add_treasury = -200
			add_political_influence = -20
			LOG_line = { sys = QING  msg = "MISSION task START Mexican adventure (spending treasury+influence) for" }
		}
		on_completion = {
			custom_tooltip = qing_col_maximilian_tt
			# Mexico City (p:8516) — the seat of the throne being contested
			if = { limit = { exists = p:8516  NOT = { owns = 8516 } }  p:8516 = { add_claim = ROOT } }
			add_country_modifier = { name = qing_col_mexican_crown  duration = -1 }
			# [#96-fix m5] decide ONCE whether this run pays the additive package (wealth /
			# prestige / GP-tension provokes). The Qing has THREE prop entry points that share
			# ROOT = CHI (this mission, qing_col_mexican_empire, and qing_mexico_adventure.1);
			# qing_col_maximilian is a prerequisite of qing_col_mexican_empire, so both mission
			# on_completions would otherwise stack the package. qing_mexico_propped is the shared
			# first-wins flag; qing_mexico_pkg_this is this run's local decision (the always-run
			# arc hook sits between the grants, so a single contiguous guard won't do). Name-
			# unique crown modifiers don't stack and stay outside the guard.
			set_variable = { name = qing_mexico_pkg_this  value = 0 }
			if = {
				limit = { NOT = { has_variable = qing_mexico_propped } }
				set_variable = qing_mexico_propped
				set_variable = { name = qing_mexico_pkg_this  value = 1 }
				CURRENCY_grant_country_wealth = { thousands = 90 }
				current_ruler = { add_popularity = 30 }
			}
			# [#96 REFACTOR — behavioural-equivalence-scrutinized change to shipped #69]
			# HOOK INTO THE MEXICO ARC (se_MEXICO.txt is the source of truth). Instead of
			# conjuring the adventure in a vacuum, react to whatever the Mexico arc has
			# actually done to Mexico by now:
			#   • If the Second Empire ALREADY stands (the arc's MEX_install_empire ran and
			#     set global_var:mex_empire_country), the Qing PROPS it — set mex_foreign_
			#     patron, the flag MEX_empire_fall_check honours to keep the Empire alive
			#     past 1867 (the alt-history divergence this #69 arc exists to produce).
			#   • If NOT yet (Mexico still a republic), the Qing ACCELERATES the road to a
			#     throne — nudge the arc's foreign-debt meter toward the intervention
			#     trigger and, if a live AI Mexico exists, poke its climax beat.
			# The old Napoleon-fork / Daoguang-betrayal flavour below is preserved verbatim.
			if = {
				limit = { exists = global_var:mex_empire_country }
				global_var:mex_empire_country = { set_variable = mex_foreign_patron }
				LOG_line = { sys = QING  msg = "MISSION Mexican adventure PROPS the existing Second Empire (mex_foreign_patron set) for" }
			}
			else = {
				if = {
					limit = { exists = c:MEX  c:MEX = { has_variable = mex_foreign_debt } }
					c:MEX = {
						MEX_nudge = { var = mex_foreign_debt  amount = 20 }
						if = {
							limit = { NOT = { has_variable = mex_empire_installed } }
							trigger_event = { id = mex_instability.12  days = { 30 120 } }
						}
					}
					LOG_line = { sys = QING  msg = "MISSION Mexican adventure ACCELERATES the arc (debt+20, climax poked) toward a throne for" }
				}
				else = {
					LOG_fail = { sys = QING  fn = qing_col_maximilian  reason = "no MEX on the map to hook into; Qing adventure proceeds on its own flavour only for" }
				}
			}
			# FORK on Napoleon's presence at court. [#96-fix m5] the GP-tension provokes below are
			# additive one-shots too, so they only fire on the run that won the package decision.
			if = {
				limit = { var:qing_mexico_pkg_this = 1  has_variable = qing_napoleon_present }
				# The condominium with the Second Empire. Provoke Britain (the Monroe
				# Doctrine's tacit backer resents a European throne in the Americas),
				# and warm relations with France, whose adventure this now shares.
				QING_gp_provoke_britain = { severity = 10 }
				if = {
					limit = { exists = c:FRA }
					c:FRA = { add_opinion = { modifier = qing_gp_accommodation_opinion  target = ROOT } }
					ROOT = { add_opinion = { modifier = qing_gp_accommodation_opinion  target = c:FRA } }
				}
				# the enterprise serves the Emperor Emeritus's own imperial house: he is gratified
				if = {
					limit = { has_variable = qing_napoleon_char  var:qing_napoleon_char = { is_alive = yes } }
					var:qing_napoleon_char = { add_loyalty = loyalty_qing_congenial }
				}
				LOG_line = { sys = QING  msg = "MISSION task Mexican adventure: FRANCO-QING condominium (Napoleon present) for" }
			}
			else_if = {
				limit = { var:qing_mexico_pkg_this = 1 }
				# no French partner — the Qing grabs the adventure alone, and France,
				# whose New-World project this pre-empts, is affronted.
				QING_gp_provoke_france = { severity = 12 }
				LOG_line = { sys = QING  msg = "MISSION task Mexican adventure: SOLO Qing intervention (no Napoleon) for" }
			}
			# Either way, planting a monarchy in the Americas violates the Pacific
			# entente: the American republic feels betrayed by its erstwhile partner.
			if = {
				limit = { var:qing_mexico_pkg_this = 1  exists = c:USA  has_country_modifier = qing_daoguang_doctrine }
				c:USA = { add_opinion = { modifier = qing_gp_rivalry_opinion  target = ROOT } }
				LOG_line = { sys = QING  msg = "MISSION task Mexican adventure BETRAYED the Daoguang entente; USA affronted for" }
			}
			LOG_line = { sys = QING  msg = "MISSION task Mexican adventure for" }
			remove_variable = qing_mexico_pkg_this
		}
	}

	# --- 墨西哥帝國 THE EMPIRE OF MEXICO — enthrone a client emperor ---
	# Release the Mexican interior as a client Empire — an imperial_monarchy bound
	# as a sinosphere tributary, the Son of Heaven's most distant vassal-throne.
	# Pacific_Mexico (which contains Mexico City, p:8516, the throne's seat) belongs
	# HERE, with the Empire of Mexico — not with Anxin, which is the Pacific-Northwest
	# and California coast. The two releases stay disjoint (Anxin = Alaska/BC/Cascadia/
	# California; the Empire = all four Mexican regions), so they never contest ground.
	qing_col_mexican_empire = {
		icon = qing_col_mexican_empire
		requires = { qing_col_maximilian }
		allow = {
			political_influence >= 30
			custom_tooltip = {
				text = qing_col_needs_mexico_land_tt
				any_owned_province = {
					count >= 3
					OR = {
						is_in_region = Pacific_Mexico
						is_in_region = Eastern_Mexico
						is_in_region = Northern_Mexico
						is_in_region = Central_America
					}
				}
			}
		}
		on_start = {
			add_political_influence = -30
			LOG_line = { sys = QING  msg = "MISSION task START Empire of Mexico (spending influence) for" }
		}
		on_completion = {
			custom_tooltip = qing_col_mexican_empire_tt
			# [#96 REFACTOR — behavioural-equivalence-scrutinized change to shipped #69]
			# RECONCILE WITH THE MEXICO ARC so the Qing never double-releases a rival
			# throne against the arc's own Second Empire (se_MEXICO.txt MEX_install_empire).
			# If that Empire already stands (mex_empire_installed / global_var:mex_empire_
			# country), the Qing does NOT stand up a duplicate client Empire on the same
			# ground — it becomes its PATRON instead (set mex_foreign_patron, the flag
			# MEX_empire_fall_check honours). Otherwise the original Qing client-Empire
			# release runs verbatim, as before this refactor.
			if = {
				limit = { exists = global_var:mex_empire_country }
				global_var:mex_empire_country = { set_variable = mex_foreign_patron }
				add_country_modifier = { name = qing_col_mexican_empire_mod  duration = -1 }
				# [#96-fix m5] the additive prestige pays out only if no earlier prop path already
				# did (qing_col_maximilian is this task's prerequisite, so its on_completion has
				# usually already fired the package). Shared first-wins flag qing_mexico_propped.
				if = {
					limit = { NOT = { has_variable = qing_mexico_propped } }
					set_variable = qing_mexico_propped
					current_ruler = { add_popularity = 30 }
				}
				LOG_line = { sys = QING  msg = "MISSION Empire of Mexico: the arc's Second Empire already stands — Qing becomes its PATRON, no duplicate release, for" }
			}
			else = {
				every_owned_province = {
					limit = {
						is_capital = no
						OR = {
							is_in_region = Pacific_Mexico
							is_in_region = Eastern_Mexico
							is_in_region = Northern_Mexico
							is_in_region = Central_America
						}
					}
					add_to_list = qing_protectorate_list
				}
				QING_establish_protectorate = {
					list = qing_protectorate_list
					name_key = qing_protectorate_mexico_name
					adj_key = qing_protectorate_mexico_adj
					government = imperial_monarchy
					log_tag = MEXICO
				}
				add_country_modifier = { name = qing_col_mexican_empire_mod  duration = -1 }
				current_ruler = { add_popularity = 30 }
				LOG_line = { sys = QING  msg = "MISSION task Empire of Mexico established (no pre-existing arc Empire) for" }
			}
		}
	}
}
