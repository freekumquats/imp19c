########################################################################
# qing_burma_war_missions.txt — THE INVASION OF BURMA (清緬戰爭), #421.
#
# A dedicated CONQUEST tree for the Sino-Burmese War (1765-69), Qianlong's "disguised
# defeat" against the Konbaung dynasty. Unlike the tributary-focused Burma task in the
# southern-frontier tree (qing_himalaya_seasia_missions.txt, which makes BUR a
# sinosphere_tributary by fiat), this tree makes the PLAYER actually WAGE THE WAR:
#
#   • The task goals gate on the Qing genuinely CONTROLLING the Burmese heartland
#     provinces (p:X = { controller = ROOT }) — Kyaikkhami/Namtu (the border), then
#     Mandalay (the road to the capital), then Rangoon (the capital) itself. The
#     player marches the armies and wins the battles; the mission only recognises the
#     ground once it is actually taken. NO scripted battle outcomes.
#   • What IS scripted (se_QING_BURMA.txt) is the campaign's texture: the MIASMA (瘴疫)
#     — the malarial lowlands that historically destroyed the banner columns, laid on
#     the heartland as heavy hostile attrition — and the KONBAUNG RALLY (a fresh
#     defending host mustered for BUR at the war's height). So the war is difficult and
#     bloody by design, but the difficulty is felt on the map, not decided by a die.
#
# Provoking the Konbaung realm on Britain's Indian doorstep needles the Raj
# (QING_gp_provoke_britain). The capstone — the whole Irrawaddy under the Son of
# Heaven — grants the lasting qing_burma_conquered dominion modifier.
#
# Gating: CHI-only, player-only (ai_chance factor 0). Tasks cost treasury/PI and gate
# on the advancing control of the heartland. Icons reuse stock test1/2/3 (the whole
# Qing suite's placeholder convention).
#
# Sources: Dai Yingcong, "A Disguised Defeat: The Myanmar Campaign of the Qing
#   Dynasty" (Modern Asian Studies, 2004); Charney, Southeast Asian Warfare 1300-1900
#   (2004); Cambridge History of China v.9 on the Qianlong frontier wars.
########################################################################

qing_burma_war_mission = {
	# Art source: Wikimedia Commons, "万国来朝图 / Myanmar (缅甸国) delegates in Peking in 1761"
	# (Qing court painting, public domain). File: 万国来朝图_Myanmar_(缅甸国)_delegates_in_Peking_in_1761.jpg
	icon = qing_burma_war_mission
	header = mission_image_qing_burma_war

	repeatable = no

	chance = { }
	ai_chance = { factor = 0 }   # player-only

	potential = {
		tag = CHI
		is_ai = no
		# there must be a Konbaung realm to fight.
		exists = c:BUR
	}

	abort = {
		NOT = { tag = CHI }
	}

	on_start = {
		LOG_line = { sys = QING  msg = "MISSION invasion of Burma (清緬戰爭) started" }
	}

	on_completion = {
		LOG_line = { sys = QING  msg = "MISSION invasion of Burma COMPLETED" }
	}

	# ==================================================================
	# ROOT — MOBILISE THE YUNNAN FRONTIER (雲南備邊). The Green Standard and banner
	# armies of the southwest are put on a war footing and the campaign opens: the
	# miasma settles over the lowlands ahead and the treasury takes on the strain of
	# provisioning an army beyond the passes. This is the point of no easy return.
	# ==================================================================
	qing_burma_mobilise = {
		icon = qing_burma_mobilise
		allow = {
			treasury >= 120
		}
		on_start = {
			add_treasury = -120
			TREASURY_LOG_it = { amount = -120 }
		}
		on_completion = {
			custom_tooltip = qing_burma_mobilise_tt
			# provoke the Raj (Burma sits on Britain's Indian frontier) and open the campaign
			# (lay the miasma over the heartland + take on the campaign strain).
			QING_gp_provoke_britain = { severity = 8 }
			QING_burma_open_campaign = yes
			LOG_line = { sys = QING  msg = "MISSION task Burma mobilise" }
		}
	}

	# ==================================================================
	# CROSS THE FRONTIER (度緬關) — take the border marches. The player must actually
	# seize and hold the Shan-border ground (Namtu 9052 / Kyaikkhami 7657). The
	# Konbaung rally is mustered here: the defenders do not fold at the first push.
	# ==================================================================
	qing_burma_frontier = {
		icon = qing_burma_frontier
		requires = { qing_burma_mobilise }
		allow = {
			custom_tooltip = {
				text = qing_burma_needs_frontier_tt
				# the border ground actually held by the Qing (either province suffices —
				# the frontier is breached).
				OR = {
					AND = { exists = p:9052  p:9052 = { controller = ROOT } }
					AND = { exists = p:7657  p:7657 = { controller = ROOT } }
				}
			}
		}
		on_completion = {
			custom_tooltip = qing_burma_frontier_tt
			# the defenders rally a fresh host — the war is far from won.
			QING_burma_rally_konbaung = yes
			current_ruler = { add_popularity = 10 }
			LOG_line = { sys = QING  msg = "MISSION task Burma frontier breached" }
		}
	}

	# ==================================================================
	# THE ROAD TO AVA (取阿瓦) — drive on the royal city. Mandalay (6562) must be
	# taken and held. This is the deep push into the fever country; a claim on the
	# ground is granted so the peace can convert control into cession.
	# ==================================================================
	qing_burma_mandalay = {
		icon = qing_burma_mandalay
		requires = { qing_burma_frontier }
		allow = {
			treasury >= 80
			custom_tooltip = {
				text = qing_burma_needs_mandalay_tt
				exists = p:6562
				p:6562 = { controller = ROOT }
			}
		}
		on_start = {
			add_treasury = -80
			TREASURY_LOG_it = { amount = -80 }
		}
		on_completion = {
			custom_tooltip = qing_burma_mandalay_tt
			if = { limit = { exists = p:6562  NOT = { owns = 6562 } }  p:6562 = { add_claim = ROOT } }
			current_ruler = { add_popularity = 15 }
			LOG_line = { sys = QING  msg = "MISSION task Burma Mandalay taken" }
		}
	}

	# ==================================================================
	# THE FALL OF THE CAPITAL (下緬京) — Rangoon / the Konbaung capital (7675) must be
	# under Qing control. With the royal city held, the Konbaung state is broken.
	# ==================================================================
	qing_burma_capital = {
		icon = qing_burma_capital
		requires = { qing_burma_mandalay }
		allow = {
			custom_tooltip = {
				text = qing_burma_needs_capital_tt
				exists = p:7675
				p:7675 = { controller = ROOT }
			}
		}
		on_completion = {
			custom_tooltip = qing_burma_capital_tt
			if = { limit = { exists = p:7675  NOT = { owns = 7675 } }  p:7675 = { add_claim = ROOT } }
			current_ruler = { add_popularity = 20 }
			add_stability = 1
			LOG_line = { sys = QING  msg = "MISSION task Burma capital fallen" }
		}
	}

	# ==================================================================
	# THE YUNNAN BASE (經理雲南) — build the forward base in Yunnan that supplies the Irrawaddy
	# campaign: the muster depots, the mule roads, and the frontier markets of the southwest.
	# ==================================================================
	qing_burma_yunnan = {
		icon = qing_burma_yunnan
		requires = { qing_burma_mobilise }
		allow = {
			treasury >= 70
		}
		on_start = { add_treasury = -70  TREASURY_LOG_it = { amount = -70 } }
		on_completion = {
			custom_tooltip = qing_burma_yunnan_tt
			# [#234] the forward base made concrete: raise a frontier fort (the muster depot) on the
			# most-populous held Yunnan province — the supply base for the Irrawaddy campaign.
			if = {
				limit = { any_owned_province = { is_in_region = Yunnan  NOT = { has_building = qing_frontier_fort_building } } }
				ordered_owned_province = {
					order_by = total_population
					max = 1
					limit = { is_in_region = Yunnan  NOT = { has_building = qing_frontier_fort_building } }
					add_building_level = qing_frontier_fort_building
				}
			}
			current_ruler = { add_popularity = 10 }
			add_stability = 1
			LOG_line = { sys = QING  msg = "MISSION task Burma Yunnan base" }
		}
	}

	# ==================================================================
	# THE SHAN STATES (羈縻撣邦) — bring the Shan sawbwa chieftaincies of the uplands under
	# the imperial warrant, the buffer belt between Yunnan and the Irrawaddy plain.
	# ==================================================================
	qing_burma_shan = {
		icon = qing_burma_shan
		requires = { qing_burma_frontier }
		allow = {
			political_influence >= 25
		}
		on_start = { add_political_influence = -25 }
		on_completion = {
			custom_tooltip = qing_burma_shan_tt
			current_ruler = { add_popularity = 12 }
			add_stability = 1
			LOG_line = { sys = QING  msg = "MISSION task Burma Shan states" }
		}
	}

	# ==================================================================
	# THE KONBAUNG TRIBUTE (緬甸入貢) — with the royal city broken, compel the Konbaung court
	# to resume the decennial tribute mission to Beijing (the historical 1790 settlement).
	# ==================================================================
	qing_burma_tribute = {
		icon = qing_burma_tribute
		requires = { qing_burma_capital }
		allow = {
			custom_tooltip = {
				text = qing_burma_needs_konbaung_tt
				exists = c:BUR
			}
		}
		on_completion = {
			custom_tooltip = qing_burma_tribute_tt
			if = { limit = { exists = c:BUR  NOT = { c:BUR = { is_subject_of = ROOT } } }  FUNC_make_subject = { overlord = ROOT  target = c:BUR  type = sinosphere_tributary } }
			current_ruler = { add_popularity = 15 }
			LOG_line = { sys = QING  msg = "MISSION task Burma tribute resumed" }
		}
	}

	# ==================================================================
	# THE TEAK & RUBY TRADE (通商采玉) — open the Irrawaddy trade: the teak forests, the
	# Mogok ruby mines, and the river route down to the sea — a new revenue for the empire.
	# ==================================================================
	qing_burma_trade = {
		icon = qing_burma_trade
		requires = { qing_burma_tribute }
		allow = {
			treasury >= 50
		}
		on_start = { add_treasury = -50  TREASURY_LOG_it = { amount = -50 } }
		on_completion = {
			custom_tooltip = qing_burma_trade_tt
			# [#234] the teak trade made concrete: turn a held Burma province to hardwood (teak) —
			# the Irrawaddy forests opened to imperial revenue. Only re-tools a subsistence good.
			if = {
				limit = { any_owned_province = { is_in_region = Burma  OR = { trade_goods = livestock  trade_goods = vegetables } } }
				ordered_owned_province = {
					order_by = total_population
					max = 1
					limit = { is_in_region = Burma  OR = { trade_goods = livestock  trade_goods = vegetables } }
					set_trade_goods = hardwood
				}
			}
			current_ruler = { add_popularity = 10 }
			LOG_line = { sys = QING  msg = "MISSION task Burma trade opened" }
		}
	}

	# ==================================================================
	# THE FEVER CONQUERED (瘴癘既平) — master the miasma that ruined four campaigns: the
	# quinine bark, the dry-season timing, the acclimatised garrisons. The strain lifts early.
	# ==================================================================
	qing_burma_fever = {
		icon = qing_burma_fever
		requires = { qing_burma_mandalay }
		allow = {
			treasury >= 40
		}
		on_start = { add_treasury = -40  TREASURY_LOG_it = { amount = -40 } }
		on_completion = {
			custom_tooltip = qing_burma_fever_tt
			if = { limit = { has_country_modifier = qing_burma_miasma }  remove_country_modifier = qing_burma_miasma }
			current_ruler = { add_popularity = 12 }
			add_stability = 1
			LOG_line = { sys = QING  msg = "MISSION task Burma fever conquered" }
		}
	}

	# ==================================================================
	# CAPSTONE — THE PACIFICATION OF BURMA (平定緬甸). The Irrawaddy heartland is held
	# by the Son of Heaven: the royal city AND the road to it are under Qing control.
	# The southern dominion is sealed with the lasting qing_burma_conquered modifier;
	# the ruinous campaign strain finally lifts.
	# ==================================================================
	qing_burma_capstone = {
		icon = qing_burma_capstone
		requires = { qing_burma_capital qing_burma_jade_mines qing_burma_monsoon qing_burma_pacify qing_burma_teak }
		allow = {
			custom_tooltip = {
				text = qing_burma_needs_conquest_tt
				# the heartland genuinely subdued — the capital AND Mandalay both held.
				exists = p:7675
				p:7675 = { controller = ROOT }
				exists = p:6562
				p:6562 = { controller = ROOT }
			}
		}
		on_completion = {
			custom_tooltip = qing_burma_capstone_tt
			add_country_modifier = { name = qing_burma_conquered  duration = -1 }
			# the campaign is won — the strain of the southern war lifts.
			QING_burma_close_campaign = yes
			# [#423 tie-in] the pacified Irrawaddy is annexed to the empire and the Shan uplands
			# (the Golden Triangle) turn to poppy, adding home-grown opium that offsets the import
			# silver drain.
			QING_burma_annex_and_plant_poppy = yes
			current_ruler = { add_popularity = 40 }
			add_stability = 1
			LOG_line = { sys = QING  msg = "MISSION invasion of Burma CAPSTONE (Burma pacified)" }
		}
	}

	# ==================================================================
	# 安南都護府 THE PROTECTORATE-GENERAL OF THE PACIFIED SOUTH (relocated here from the
	# colonization tree, #27 P3b-3 2026-08-02). For three centuries the Tang held the Red
	# River delta through the 安南都護府 at Hanoi — the name outlasting the institution to
	# become Vietnam's own. Revived here NOT as northern Vietnam (which is the CHI ritual
	# tributary TRH after the §4.3 fidelity fix) but as an EIC-model FRONTIER MARCH of
	# INDEPENDENT MAINLAND SE-ASIA (DESIGN §4.1): carved from the conquered ground of the
	# Burma-war theatre (regions Burma/Siam/Laos/Vietnam), holding the subjugated mainland
	# polities — Konbaung Burma (BUR, made a CHI tributary by qing_burma_tribute), Siam
	# (SIA), the freed Nguyễn south (VIE) — as its own princely-state subjects. Founded off
	# the Burma capstone (avoids the capstone soft-lock; land march, no navy). Replaces the
	# OLD backwards qing_col_annan, which carved CHI-core 'Vietnam' as a fake tributary.
	# ==================================================================
	qing_burma_annan_march = {
		icon = qing_burma_capstone
		requires = { qing_burma_capstone }
		allow = {
			political_influence >= 30
			custom_tooltip = {
				text = qing_burma_needs_seasia_land_tt
				# at least 3 held provinces across the conquered mainland-SE-Asia theatre.
				any_owned_province = {
					count >= 3
					OR = {
						is_in_region = Burma
						is_in_region = Siam
						is_in_region = South_Siam
						is_in_region = Laos
						is_in_region = Vietnam
						is_in_region = Cambodia
					}
				}
			}
		}
		on_start = {
			add_political_influence = -30
			LOG_line = { sys = QING  msg = "MISSION task START Annan march" }
		}
		on_completion = {
			custom_tooltip = qing_burma_annan_march_tt
			# carve the march from the CONQUERED mainland-SE-Asia ground (held, non-capital, non-CHI-core).
			every_owned_province = {
				limit = {
					is_capital = no
					OR = {
						is_in_region = Burma
						is_in_region = Siam
						is_in_region = South_Siam
						is_in_region = Laos
						is_in_region = Vietnam
						is_in_region = Cambodia
					}
				}
				add_to_list = qing_protectorate_list
			}
			# subordinate the mainland polities as the march's own princely states. Use every_country
			# with an explicit free-OR-CHI-subject filter (NOT every_subject): unlike the CA tree — where
			# the capstone's required chain provably makes the khanates/hordes CHI subjects first — the
			# Burma capstone does NOT guarantee BUR/SIA/VIE are subjugated (qing_burma_tribute is optional
			# and off the capstone path; SIA/VIE are never made CHI subjects in this tree). So a conquered-
			# but-independent Burma/Siam/Vietnam must still be captured here. QING_found_march reparents a
			# CHI subject / takes a free one / skips one under a foreign overlord.
			every_country = {
				limit = {
					OR = { tag = BUR  tag = SIA  tag = VIE }
					OR = { is_subject_of = ROOT  is_subject = no }
				}
				add_to_list = qing_march_locals_list
			}
			QING_found_march = {
				list = qing_protectorate_list
				name_key = qing_protectorate_annan_name
				adj_key = qing_protectorate_annan_adj
				maritime = no
				log_tag = ANNAN
				locallist = qing_march_locals_list
			}
			add_country_modifier = { name = qing_col_protectorate_south  duration = -1 }
			QING_DECLINE_nudge = { var = qing_ethnic_tension  amount = -4 }
			current_ruler = { add_popularity = 20 }
			LOG_line = { sys = QING  msg = "MISSION task Annan march" }
		}
	}
