########################################################################
# qing_colonization_missions.txt — the GREAT PACIFIC ENTERPRISE (大洋洲事業):
# an anachronistic Qing overseas-colonization mission tree. CHI-only, player-only,
# and gated so it opens ONLY to a genuinely powerful, self-strengthened dynasty —
# a naval, industrial Qing that reaches north into Siberia, across to Russian
# America (Alaska), and on to the Pacific coast of the New World, disrupting and
# potentially supplanting Russia's own thin colonial effort.
#
# History that grounds the fantasy: the Qing and Romanov empires met on the Amur
# in the 17th c. (Treaties of Nerchinsk 尼布楚條約 1689, Kyakhta 恰克圖條約 1727,
# later Aigun 璦琿條約 1858). Russian America was governed by the tiny, chronically
# under-supplied Russian-American Company from Novoarkhangelsk (Sitka); its
# Pacific posts leaned on the Canton tea trade and on Californian grain (Fort
# Ross, 1812-41) to survive. A Qing that had mastered steam and finance could, in
# an alternate history, have contested all of it. That is the premise.
#
# The tree fans out: secure the Amur (the northern frontier) → project across the
# Pacific to Alaska → contest the fur coast of Canada → plant a colony in
# California → the capstone, a Qing dominion spanning the Pacific.
#
# [bookmark-1763] TWO ROADS INTO THE ENTERPRISE. This tree was written for a self-
# strengthened, industrial Qing (the 1860s/develop path). But the 1763 start catches
# the empire at its HIGH-QING TERRITORIAL ZENITH under Qianlong (乾隆): Xinjiang had
# JUST been conquered (準噶爾/回部, 1755-59), the Ten Great Campaigns (十全武功) were
# live, the tributary system was at its peak, and the treasury was full. Such a Qing
# did not NEED steam and arsenals to expand — it had the men, the silver, and the
# momentum. So at a 1763 start the SAME tree opens on the High-Qing era flag
# (qing_high_qing_era, se_QING_DECLINE #304) instead of self-strengthening progress,
# and the maritime springboard (the Maritime Bureau) opens on the era + treasury
# rather than on the anachronistic Beiyang/Nanyang fleets. The land branches (Amur,
# Xinjiang, Central Asia, the tributary marches 安北/安東/安西/安南) are HISTORICAL for
# the 1760s, not fantasy; only the trans-Pacific reach (Alaska/Canada/California) stays
# the alt-history flourish. At the 1815 develop start the era flag is never set, so
# ONLY the self-strengthening road opens the tree — develop behaviour is unchanged.
#
# Gating: potential requires tag CHI + is_ai=no + (self-strengthening >= 40 OR the
# High-Qing era). Individual tasks gate on treasury and ownership/claims. Rewards call
# add_building_level, add_claim, FUNC_make_subject (sinosphere_tributary),
# CURRENCY_grant_country_wealth, and the QING_* effects, all logged to sys = QING.
#
# NOTE (cosmetic): task icons reuse test1/2/3, header mission_image_test — see the
# note in qing_selfstrengthening_missions.txt.
########################################################################

qing_colonization_mission = {
	# Art source: Abraham Ortelius, "Maris Pacifici" (1589), Princeton University Library
	# digital maps collection (map-pacific-ortelius-1589). Public domain.
	icon = qing_colonization_mission
	header = mission_image_qing_colonization

	repeatable = no

	chance = { }
	ai_chance = { factor = 0 }

	potential = {
		tag = CHI
		is_ai = no
		# opens to a capable Qing by EITHER road: [develop] a self-strengthening, industrial
		# dynasty (progress >= 40), OR [bookmark-1763] the High-Qing zenith under Qianlong
		# (the qing_high_qing_era flag, set only at a pre-1772 start by QING_DECLINE_init).
		# At the 1815 develop start the era flag is never set, so only the self-strengthening
		# road opens the tree there — develop behaviour byte-for-byte unchanged.
		OR = {
			AND = {
				has_variable = qing_selfstr_progress
				var:qing_selfstr_progress >= 40
			}
			has_variable = qing_high_qing_era
		}
	}

	abort = {
		NOT = { tag = CHI }
	}

	on_start = {
		LOG_line = { sys = QING  msg = "MISSION colonization (Pacific enterprise) started for" }
	}

	on_completion = {
		LOG_line = { sys = QING  msg = "MISSION colonization COMPLETED for" }
	}

	# ==================================================================
	# ROOT — the Maritime Bureau: resolve to look outward across the ocean.
	# ==================================================================
	qing_col_bureau = {
		icon = qing_col_bureau
		allow = {
			treasury >= 100
			custom_tooltip = {
				text = qing_col_needs_fleet_tt
				# [develop] a modern self-strengthening fleet/yard, OR [bookmark-1763] the High-Qing
				# zenith — the Qianlong state commanded the Fujian junk fleets + the manpower to mount a
				# maritime enterprise without steam. The era flag opens the springboard for a 1763 start;
				# the modern-fleet requirement stands for develop (era flag never set at 1815).
				OR = {
					has_country_modifier = qing_selfstr_beiyang_fleet
					has_country_modifier = qing_selfstr_nanyang_fleet
					has_country_modifier = qing_selfstr_fuzhou_yard
					# [#439 tie-in] the revived New Treasure Fleet (新寶船隊) is itself a springboard for
					# overseas enterprise — completing that tree (its 下西洋 capstone grants this modifier)
					# opens the Great Pacific Enterprise, the treasure-fleet path to colonisation.
					has_country_modifier = qing_treasure_fleet_glory
					has_variable = qing_high_qing_era
				}
			}
		}
		on_start = {
			add_treasury = -100
			LOG_line = { sys = QING  msg = "MISSION task START maritime bureau (spending treasury) for" }
		}
		on_completion = {
			custom_tooltip = qing_col_bureau_tt
			add_country_modifier = { name = qing_col_maritime_bureau  duration = -1 }
			LOG_line = { sys = QING  msg = "MISSION task maritime bureau for" }
		}
	}

	# ==================================================================
	# TAIWAN — make the island prefecture a true maritime province (臺灣).
	# History: Taiwan was a prefecture of Fujian from 1684 and became its own
	# province only in 1885 under Liu Mingchuan (劉銘傳), who built its first
	# railway and telegraph. Here the Qing develops it early as the springboard
	# of the whole Pacific enterprise.
	# ==================================================================
	qing_col_taiwan = {
		icon = qing_col_taiwan
		requires = { qing_col_bureau }
		allow = {
			treasury >= 90
			custom_tooltip = {
				text = qing_col_needs_taiwan_tt
				OR = { owns = 6799  owns = 6781 }
			}
		}
		on_start = {
			add_treasury = -90
			LOG_line = { sys = QING  msg = "MISSION task START Taiwan (spending treasury) for" }
		}
		on_completion = {
			custom_tooltip = qing_col_taiwan_tt
			# Liu Mingchuan's modernisation: a port, commerce, a customs house, and rail on Taibei.
			# [#27 colonial-buildings] the qing_customs_house (海關) is the on-map Qing institution of the
			# newly-provincialised island (Taiwan became a province 1885 under Liu). Guarded owns + NOT-present.
			if = {
				limit = { owns = 6799 }
				p:6799 = {
					set_city_status = city
					add_building_level = port_building
					add_building_level = URB_commerce_district
					if = { limit = { NOT = { has_building = qing_customs_house_building } }  add_building_level = qing_customs_house_building }
				}
			}
			else_if = {
				limit = { owns = 6781 }
				p:6781 = {
					set_city_status = city
					add_building_level = port_building
					if = { limit = { NOT = { has_building = qing_customs_house_building } }  add_building_level = qing_customs_house_building }
				}
			}
			# claim the Pescadores (Penghu) if not held — the strait's key
			if = { limit = { exists = p:10423  NOT = { owns = 10423 } }  p:10423 = { add_claim = ROOT } }
			# make the developed ports a settlement magnet: mainland surplus crosses the
			# strait of its own accord via the migration pulse. See se_QING_COLON.txt.
			QING_COLON_frontier_pull = { province = 6799 }
			QING_COLON_frontier_pull = { province = 6781 }
			add_country_modifier = { name = qing_col_taiwan_province  duration = -1 }
			LOG_line = { sys = QING  msg = "MISSION task Taiwan for" }
		}
	}

	# ==================================================================
	# THE PACIFIC ISLANDS — the sea road between the two shores (太平洋群島).
	# Ryukyu (琉球, a Qing tributary until Japan annexed it in 1879), the
	# Marianas/Guam, the Philippines (呂宋), and on to Hawaii (檀香山, "Sandalwood
	# Mountains" to the Chinese who traded there) — the island chain that links
	# the China coast to the New World.
	# ==================================================================
	qing_col_pacific_isles = {
		icon = qing_col_pacific_isles
		requires = { qing_col_taiwan }
		allow = {
			treasury >= 140
			custom_tooltip = {
				text = qing_col_needs_strong_fleet_tt
				# [#315 review-fix] added the High-Qing era escape here to match the ROOT (qing_col_bureau)
				# and the new southern branch: without it the whole maritime chain (isles -> New Holland ->
				# New Guinea) was UNREACHABLE at a 1763 start, since the two fleet modifiers are granted only
				# by the 19th-c. Self-Strengthening missions. The era flag is never set at the 1815 develop
				# start, so develop still requires a modern fleet here — behaviour unchanged there.
				OR = {
					has_country_modifier = qing_selfstr_beiyang_fleet
					has_country_modifier = qing_selfstr_nanyang_fleet
					has_variable = qing_high_qing_era
				}
			}
		}
		on_start = {
			add_treasury = -140
			LOG_line = { sys = QING  msg = "MISSION task START Pacific isles (spending treasury) for" }
		}
		on_completion = {
			custom_tooltip = qing_col_pacific_isles_tt
			# reassert suzerainty over Ryukyu; claim the island stepping-stones
			if = { limit = { exists = p:524   NOT = { owns = 524 } }  p:524 = { add_claim = ROOT } }
			if = { limit = { exists = p:2901  NOT = { owns = 2901 } }  p:2901 = { add_claim = ROOT } }
			if = { limit = { exists = p:7994  NOT = { owns = 7994 } }  p:7994 = { add_claim = ROOT } }
			if = { limit = { exists = p:6151  NOT = { owns = 6151 } }  p:6151 = { add_claim = ROOT } }
			add_country_modifier = { name = qing_col_island_road  duration = -1 }
			# [logfix 2026-08-02] wrapped in hidden_effect: the reward-preview tooltip-eval of
			# CURRENCY_grant_country_wealth floods error.log (~1000 lines when the tree is opened)
			# because its reserve-ratio svalue chain resolves to 'none' in preview context. hidden_effect
			# is excluded from tooltip generation, so the flood stops while the grant still executes.
			hidden_effect = { CURRENCY_grant_country_wealth = { thousands = 50 } }
			LOG_line = { sys = QING  msg = "MISSION task Pacific isles for" }
		}
	}

	# ==================================================================
	# [#315] THE SOUTHERN OCEAN — New Holland (Australia) + New Guinea (南方大陸事業).
	# Alt-history maritime reach into the southern hemisphere. In our timeline Britain
	# claimed the Australian east coast in 1770 (Cook) and settled Botany Bay/Sydney in
	# 1788; New Guinea's coasts were contested by the Dutch, Germans and British through
	# the 19th c. In a 1763 divergence a High-Qing maritime state — Fujian junk fleets,
	# full treasury, the Ten Great Campaigns momentum — reaches south first, planting
	# 新南威爾士 (New South Wales) settlements and contesting the Papuan coast. Racing
	# Britain to New Holland is the sharpest Pacific provocation of the Great Game, so
	# both tasks provoke GBR. Runs off the Maritime Bureau springboard, like the isles.
	# Anchors on real map provinces claimed below: Sydney (p:895), Melbourne (p:2107),
	# Adelaide (p:6001), Perth (p:2441), Hobart/Tasmania (p:5734); New Guinea: Wau (p:50),
	# Sarawaget (p:318), Barapasi (p:1076). (Sydney is p:895 — p:633 is Sydney, Nova Scotia.)
	# ==================================================================
	qing_col_new_holland = {
		icon = qing_col_new_holland
		requires = { qing_col_pacific_isles }
		allow = {
			treasury >= 150
			custom_tooltip = {
				text = qing_col_needs_strong_fleet_tt
				# [develop] a modern fleet, OR [bookmark-1763] the High-Qing zenith commanded the
				# manpower + silver to mount a southern-ocean enterprise without steam.
				OR = {
					has_country_modifier = qing_selfstr_beiyang_fleet
					has_country_modifier = qing_selfstr_nanyang_fleet
					has_variable = qing_high_qing_era
				}
			}
		}
		on_start = {
			add_treasury = -150
			LOG_line = { sys = QING  msg = "MISSION task START New Holland (spending treasury) for" }
		}
		on_completion = {
			custom_tooltip = qing_col_new_holland_tt
			# plant the first settlement at the great natural harbour (Sydney), claim the
			# temperate coast; the empire's surplus migrates south via the migration pull.
			if = {
				limit = { exists = p:895  NOT = { owns = 895 } }
				p:895 = { add_claim = ROOT }   # Sydney (Port Jackson), Australia
				# beating Britain to New Holland is pressed as a diplomatic play (#53)
				QING_gp_frontier_play = { tag = GBR  province = 895  goal = flag:get_territory }
			}
			if = { limit = { exists = p:2107  NOT = { owns = 2107 } }  p:2107 = { add_claim = ROOT } }  # Melbourne
			if = { limit = { exists = p:6001  NOT = { owns = 6001 } }  p:6001 = { add_claim = ROOT } }  # Adelaide
			if = { limit = { exists = p:2441  NOT = { owns = 2441 } }  p:2441 = { add_claim = ROOT } }  # Perth
			# migration-pull the harbour settlement (state incentivises, doesn't teleport pops)
			QING_COLON_frontier_pull = { province = 895 }
			add_country_modifier = { name = qing_col_new_holland_settled  duration = -1 }
			# [logfix 2026-08-02] wrapped in hidden_effect: the reward-preview tooltip-eval of
			# CURRENCY_grant_country_wealth floods error.log (~1000 lines when the tree is opened)
			# because its reserve-ratio svalue chain resolves to 'none' in preview context. hidden_effect
			# is excluded from tooltip generation, so the flood stops while the grant still executes.
			hidden_effect = { CURRENCY_grant_country_wealth = { thousands = 60 } }
			# contesting the southern ocean alarms Britain (she claimed this coast in 1770)
			QING_gp_provoke_britain = { severity = 12 }
			LOG_line = { sys = QING  msg = "MISSION task New Holland for" }
		}
	}

	# --- New Guinea + Tasmania: complete the southern arc (巴布亞 + 塔斯馬尼亞) ---
	qing_col_new_guinea = {
		icon = qing_col_new_guinea
		requires = { qing_col_new_holland }
		allow = {
			treasury >= 130
			custom_tooltip = {
				text = qing_col_needs_new_holland_tt
				has_country_modifier = qing_col_new_holland_settled
			}
		}
		on_start = {
			add_treasury = -130
			LOG_line = { sys = QING  msg = "MISSION task START New Guinea (spending treasury) for" }
		}
		on_completion = {
			custom_tooltip = qing_col_new_guinea_tt
			# claim the Papuan coast trading posts + the Tasman island
			if = { limit = { exists = p:50    NOT = { owns = 50 } }    p:50 = { add_claim = ROOT } }    # Wau
			if = { limit = { exists = p:318   NOT = { owns = 318 } }   p:318 = { add_claim = ROOT } }   # Sarawaget
			if = { limit = { exists = p:1076  NOT = { owns = 1076 } }  p:1076 = { add_claim = ROOT } }  # Barapasi
			if = { limit = { exists = p:5734  NOT = { owns = 5734 } }  p:5734 = { add_claim = ROOT } }  # Hobart (Tasmania)
			# migration-pull the first Papuan trading post
			QING_COLON_frontier_pull = { province = 50 }
			add_country_modifier = { name = qing_col_south_seas_dominion  duration = -1 }
			# [logfix 2026-08-02] wrapped in hidden_effect: the reward-preview tooltip-eval of
			# CURRENCY_grant_country_wealth floods error.log (~1000 lines when the tree is opened)
			# because its reserve-ratio svalue chain resolves to 'none' in preview context. hidden_effect
			# is excluded from tooltip generation, so the flood stops while the grant still executes.
			hidden_effect = { CURRENCY_grant_country_wealth = { thousands = 50 } }
			# rounding out the southern ocean is a further affront to British Pacific ambition
			QING_gp_provoke_britain = { severity = 8 }
			LOG_line = { sys = QING  msg = "MISSION task New Guinea for" }
		}
	}

	# ##################################################################
	# THE PROTECTORATES-GENERAL (都護府) — revive the Tang institution of
	# indirect frontier rule (羈縻, "loose rein"). Where the dynasty holds enough
	# territory in a frontier region, it may erect a Protectorate there: that
	# territory is spun off as a NEW COUNTRY and bound at once as a
	# sinosphere_tributary — an autonomous march that pays tribute and shelters
	# under Qing protection rather than being ruled directly from Beijing. Each
	# task builds the province list for its region and calls
	# QING_establish_protectorate (se_QING_PROTECTORATE.txt).
	#
	# History: the Tang governed its marches through Protectors-General (都護)
	# over 安北/安東/安西/安南 — the Pacified North, East, West and South. Two
	# invented marches, 安海 (Pacified Seas) and 安新 (Pacified New World), extend
	# the scheme to the oceans and the Americas; and the Lanfang kongsi (蘭芳公司)
	# of western Borneo — whose tributary overtures the historical Qing rejected —
	# is here taken under protection. Sources: Beckwith, Empires of the Silk Road
	# (2009); Skaff, Sui-Tang China and its Turko-Mongol Neighbors (2012); on
	# Lanfang, Heidhues, Golddiggers, Farmers, and Traders (2003).
	# ##################################################################

	# --- 安海 ĀNHǍI, Pacified Seas — the invented Pacific-island march ---
	# The oceanic protectorate: the scattered island holdings of the enterprise
	# (the Ryukyus, Marianas, Philippines, Hawaii and the microstates between)
	# gathered under a single maritime march.
	# [#48] ANHAI is the OCEANIA tree's centerpiece: found the Pacified-Seas maritime march from the
	# tree's OPEN-PACIFIC + southern-continent holdings, then let it EXPAND autonomously across those
	# same regions via the maritime expansion pulse (#39, se_QING_MARCH_PULSE.txt Roads 3/4, which scan
	# by region-membership). Theatre = the open Pacific (Hawaii, Micronesia, Melanesia, Polynesia) + the
	# southern continent (Australia, New Guinea, NZ) — NOT the Philippines/Okinawa/Borneo/Indies, which
	# are the NANYANG tree's South-Seas theatre (Lanfang march + the Severed Indies).
	qing_col_anhai = {
		icon = qing_col_anhai
		requires = { qing_col_pacific_isles }
		allow = {
			political_influence >= 30
			custom_tooltip = {
				text = qing_col_needs_isles_land_tt
				any_owned_province = {
					count >= 3
					OR = {
						is_in_region = Hawaii
						is_in_region = Mariana_Islands
						is_in_region = Guam
						is_in_region = Palau
						is_in_region = Caroline_Islands
						is_in_region = Fiji
						is_in_region = Nouvelle-Caledonie
						is_in_region = Samoa
						is_in_region = Tahiti
						is_in_region = Queensland
						is_in_region = South_Australia
						is_in_region = Western_Australia
						is_in_region = New_Guinea
					}
				}
			}
		}
		on_start = {
			add_political_influence = -30
			LOG_line = { sys = QING  msg = "MISSION task START Anhai protectorate for" }
		}
		on_completion = {
			custom_tooltip = qing_col_anhai_tt
			every_owned_province = {
				limit = {
					is_capital = no
					OR = {
						is_in_region = Hawaii
						is_in_region = Mariana_Islands
						is_in_region = Guam
						is_in_region = Palau
						is_in_region = Caroline_Islands
						is_in_region = Fiji
						is_in_region = Nouvelle-Caledonie
						is_in_region = Samoa
						is_in_region = Tahiti
						is_in_region = Queensland
						is_in_region = South_Australia
						is_in_region = Western_Australia
						is_in_region = New_Guinea
					}
				}
				add_to_list = qing_protectorate_list
			}
			# [#27 P3 / #48] the EIC-model march founding (se_QING_MARCH.txt). The maritime Pacific march is
			# carved from the player's COLONIZED Oceania island/coastal holdings (overseas frontier land, NOT
			# CHI core). It then EXPANDS across those regions on its own via the maritime pulse (#39 Roads 3/4:
			# region-membership conquest + colonise). maritime = yes (Anhai gets a navy, DESIGN §3.3).
			QING_found_march = {
				list = qing_protectorate_list
				name_key = qing_protectorate_anhai_name
				adj_key = qing_protectorate_anhai_adj
				maritime = yes
				log_tag = ANHAI
				locallist = qing_march_locals_list
			}
			add_country_modifier = { name = qing_col_protectorate_seas  duration = -1 }
			current_ruler = { add_popularity = 25 }
			LOG_line = { sys = QING  msg = "MISSION task Anhai protectorate for" }
		}
	}

	# [#48] LANFANG relocated to the Nanyang tree (qing_nanyang_missions.txt) — Lanfang (LAF)
	# is a LIVE nominal-vassal on the 1763 map, so the old "release Borneo as a new kongsi"
	# task duplicated it. The South Seas / Borneo theatre now belongs wholly to the Nanyang
	# tree; Lanfang is upgraded to a frontier_protectorate march THERE. See #49.

	# ==================================================================
	# [#43] THE ISLAND ROAD DETAILED (太平洋群島之路) — the expanded body of the
	# enterprise: the mid-Pacific hub of Hawaii, the Micronesian and Melanesian and
	# Polynesian scatters, temperate New Zealand, and the two great chokepoints of the
	# Nanyang — the Straits of Malacca and the Dutch spice islands. These hang off the
	# Pacific-isles / New-Holland springboards and feed a South-Sea dominion. The
	# European-held ground (the Straits under Johor, the Indies under the Dutch) presses
	# as diplomatic plays against those holders; the open Pacific isles are simple claims.
	# The maritime tasks all carry the same High-Qing-era escape as the sibling isles
	# task, so the whole chain is reachable at a 1763 start as well as the develop path.
	# ==================================================================

	# --- 檀香山 HAWAII — the Sandalwood Mountains, the great mid-Pacific hub where the
	# # Chinese sandalwood traders gave the islands their Cantonese name. The pivot of the sea road.
	qing_col_hawaii = {
		icon = qing_col_pacific_isles
		requires = { qing_col_pacific_isles }
		allow = {
			treasury >= 130
			custom_tooltip = { text = qing_col_needs_island_road_tt  has_country_modifier = qing_col_island_road }
		}
		on_start = {
			add_treasury = -130
			LOG_line = { sys = QING  msg = "MISSION task START qing_col_hawaii for" }
		}
		on_completion = {
			custom_tooltip = qing_col_hawaii_tt
			if = { limit = { exists = p:7603  NOT = { owns = 7603 } }  p:7603 = { add_claim = ROOT } }  # Hilo
			if = { limit = { exists = p:8305  NOT = { owns = 8305 } }  p:8305 = { add_claim = ROOT } }  # Kailua-Kona
			if = { limit = { owns = 6151 }  QING_COLON_frontier_pull = { province = 6151 } }
			add_country_modifier = { name = qing_oc_hawaii  duration = -1 }
			# [logfix] reward wealth grant hidden from tooltip preview-eval (see the shipped tasks' note).
			hidden_effect = { CURRENCY_grant_country_wealth = { thousands = 60 } }
			LOG_line = { sys = QING  msg = "MISSION task qing_col_hawaii for" }
		}
	}

	# --- 密克羅尼西亞 MICRONESIA — Palau and the Caroline scatter, the western stepping-stones
