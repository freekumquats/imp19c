########################################################################
# qing_africa_missions.txt — THE TREASURE VOYAGES REBORN (鄭和下西洋復興):
# the SCRAMBLE FOR AFRICA. #27 (lifted out of qing_colonization_missions.txt,
# 2026-08-02 — see design/DESIGN_COLONIZATION_SPLIT.md §B).
#
# A second, southern maritime enterprise mirroring the Pacific one. Where the
# Pacific arc looks east across the ocean the Ming treasure fleets never crossed,
# this arc looks WEST across the ocean they DID — reviving the seven voyages of the
# eunuch-admiral Zheng He (鄭和, 1371-1433), whose great fleets under the Yongle
# and Xuande emperors reached the Swahili coast of East Africa (Malindi, Mombasa,
# Kilwa) between 1405 and 1433, and bore a live giraffe back to the Ming court as
# a tribute-qilin. The historical dynasty then burned the shipyards and turned
# inward; here a self-strengthened Qing does the opposite — it sails again, and
# arrives in Africa in the very decades the European powers are carving it up.
#
# The arc is deliberately a contest with BRITAIN and FRANCE (the paramount
# powers of the 19th-c. Scramble), not Russia — so each task routes through
# QING_gp_provoke_britain / _france, which by the 以夷制夷 logic DELIGHTS Russia
# (Britain's and France's rival) rather than provoking her. NOTE (deliberate): the
# Daoguang Doctrine is a PACIFIC-BASIN entente closing the Americas and the Pacific
# — it does NOT extend to Africa or the Indian Ocean, so unlike the Canada/
# California tasks these do NOT check for it: the European friction of the Scramble
# stands regardless of the American entente.
#
# The capstone MARCH — 安非 Anfei, the Protectorate-General of the Pacified Africa
# — is founded here (design/DESIGN_PROTECTORATES_GENERAL.md §4.2), the tree that
# actually conquers its African theatre.
#
# Gating: CHI-only, player-only (ai_chance factor 0), same self-strengthening /
# High-Qing-era gate as the parent colonization tree. Icons/header reuse the
# existing qing_treasure_fleet_mission art (an ocean-going treasure fleet — fitting).
#
# Sources: Dreyer, Zheng He: China and the Oceans in the Early Ming (2007);
#   Wade, "The Zheng He Voyages: A Reassessment" (2005); Pakenham, The Scramble for
#   Africa (1991).
########################################################################

qing_africa_mission = {
	# Art reuse: the Treasure Fleet card + banner (an ocean-going Ming fleet — the
	# vehicle of this southern enterprise). Both DDS present in gfx/interface/.
	icon = qing_treasure_fleet_mission
	header = mission_image_qing_treasure_fleet

	repeatable = no

	chance = { }
	ai_chance = { factor = 0 }   # player-only

	potential = {
		tag = CHI
		is_ai = no
		# same two-road gate as the parent colonization tree: [develop] a self-
		# strengthening industrial dynasty (progress >= 40), OR [bookmark-1763] the
		# High-Qing zenith under Qianlong (qing_high_qing_era).
		OR = {
			AND = {
				has_variable = qing_selfstr_progress
				var:qing_selfstr_progress >= 40
			}
			has_variable = qing_high_qing_era
		}
	}

	abort = {
		NOT = { tag = CHI }
	}

	on_start = {
		LOG_line = { sys = QING  msg = "MISSION Scramble for Africa (treasure voyages reborn) started for" }
	}

	on_completion = {
		LOG_line = { sys = QING  msg = "MISSION Scramble for Africa COMPLETED for" }
	}

	# --- 鄭和下西洋 — revive the treasure voyages, land on the Swahili coast ---
	# ROOT of this tree (re-rooted 2026-08-02: was `requires = qing_col_bureau` in the
	# colonization tree; now the standalone Africa tree's own root, so it opens on the
	# tree's shared gate rather than on a cross-tree prerequisite).
	qing_col_zheng_he = {
		icon = qing_col_zheng_he
		allow = {
			treasury >= 150
			custom_tooltip = {
				text = qing_col_needs_strong_fleet_tt
				OR = {
					has_country_modifier = qing_selfstr_beiyang_fleet
					has_country_modifier = qing_selfstr_nanyang_fleet
					# [#439 tie-in] the New Treasure Fleet (新寶船隊) is the grand ocean-going fleet these
					# treasure voyages demand — its 下西洋 glory satisfies the strong-fleet requirement.
					has_country_modifier = qing_treasure_fleet_glory
					# [#315-parity, added on split 2026-08-02] the High-Qing zenith commanded the Fujian
					# junk fleets + the manpower to mount the southern enterprise without steam. REQUIRED
					# now that this task is the tree ROOT: the three fleet modifiers above are all
					# 19th-c. Self-Strengthening grants that DO NOT EXIST at a 1763 start, so without this
					# era escape the tree would open (its potential admits qing_high_qing_era) yet its root
					# task could never be completed at the 1763 bookmark. Mirrors the identical escape the
					# sibling maritime tasks (qing_col_pacific_isles/_new_holland) already carry (#315), and
					# the parent colonization ROOT (qing_col_bureau). Never set at the 1815 develop start.
					has_variable = qing_high_qing_era
				}
			}
		}
		on_start = {
			add_treasury = -150
			LOG_line = { sys = QING  msg = "MISSION task START Zheng He treasure voyages (spending treasury) for" }
		}
		on_completion = {
			custom_tooltip = qing_col_zheng_he_tt
			# the Swahili coast the Ming fleets actually reached — Zanzibar and Kilwa
			if = { limit = { exists = p:1489  NOT = { owns = 1489 } }  p:1489 = { add_claim = ROOT } }  # Zanzibar (Kiunguja)
			if = { limit = { exists = p:1715  NOT = { owns = 1715 } }  p:1715 = { add_claim = ROOT } }  # Kilwa (Mohoro)
			# a treaty-port foothold draws the overseas Chinese of the Nanyang onward down
			# the monsoon route, of their own accord, via the migration pulse.
			if = { limit = { owns = 1489 }  QING_COLON_frontier_pull = { province = 1489 } }
			add_country_modifier = { name = qing_col_treasure_fleet  duration = -1 }
			# [logfix 2026-08-02] wrapped in hidden_effect: the reward-preview tooltip-eval of
			# CURRENCY_grant_country_wealth floods error.log (~1000 lines when the tree is opened)
			# because its reserve-ratio svalue chain resolves to 'none' in preview context. hidden_effect
			# is excluded from tooltip generation, so the flood stops while the grant still executes.
			hidden_effect = { CURRENCY_grant_country_wealth = { thousands = 70 } }
			current_ruler = { add_popularity = 25 }
			# a Chinese squadron in the western Indian Ocean crosses the Royal Navy's lake
			QING_gp_provoke_britain = { severity = 8 }
			LOG_line = { sys = QING  msg = "MISSION task Zheng He treasure voyages for" }
		}
	}

	# --- 好望角 — the Cape of Good Hope, the sea-lane's southern hinge ---------
	# History: the Cape was Dutch (VOC, 1652), then British from 1806 — the pivot of
	# every route between the Atlantic and the Indian Ocean. A Qing that would hold
	# the southern sea must contest it with the Royal Navy that took it from the Dutch.
	qing_col_cape = {
		icon = qing_col_cape
		requires = { qing_col_zheng_he }
		allow = {
			treasury >= 170
		}
		on_start = {
			add_treasury = -170
			LOG_line = { sys = QING  msg = "MISSION task START Cape of Good Hope (spending treasury) for" }
		}
		on_completion = {
			custom_tooltip = qing_col_cape_tt
			if = {
				limit = { exists = p:132  NOT = { owns = 132 } }
				p:132 = { add_claim = ROOT }  # Clanwilliam, the Western Cape
				# wresting the Cape from the Royal Navy is pressed as a diplomatic play (#53)
				QING_gp_frontier_play = { tag = GBR  province = 132  goal = flag:get_territory }
			}
			add_country_modifier = { name = qing_col_cape_route  duration = -1 }
			# [logfix 2026-08-02] wrapped in hidden_effect: the reward-preview tooltip-eval of
			# CURRENCY_grant_country_wealth floods error.log (~1000 lines when the tree is opened)
			# because its reserve-ratio svalue chain resolves to 'none' in preview context. hidden_effect
			# is excluded from tooltip generation, so the flood stops while the grant still executes.
			hidden_effect = { CURRENCY_grant_country_wealth = { thousands = 80 } }
			# seizing the hinge of the southern sea-lanes is a sharp affront to Britain
			QING_gp_provoke_britain = { severity = 12 }
			LOG_line = { sys = QING  msg = "MISSION task Cape of Good Hope for" }
		}
	}

	# --- 蘇伊士 — the Suez passage / Red Sea gate, the short road home ---------
	# History: the Suez Canal opened in 1869; Egypt's finances fell under an
	# Anglo-French condominium, and Britain occupied Egypt outright in 1882. The
	# short route between the oceans is contested by BOTH powers at once.
	qing_col_suez = {
		icon = qing_col_suez
		requires = { qing_col_zheng_he }
		allow = {
			treasury >= 160
		}
		on_start = {
			add_treasury = -160
			LOG_line = { sys = QING  msg = "MISSION task START Suez passage (spending treasury) for" }
		}
		on_completion = {
			custom_tooltip = qing_col_suez_tt
			if = {
				limit = { exists = p:413  NOT = { owns = 413 } }
				p:413 = { add_claim = ROOT }  # Suez
				QING_gp_frontier_play = { tag = GBR  province = 413  goal = flag:get_territory }
			}
			if = { limit = { exists = p:429  NOT = { owns = 429 } }  p:429 = { add_claim = ROOT } }  # Bani Suwayf, Cairo
			add_country_modifier = { name = qing_col_suez_passage  duration = -1 }
			# [logfix 2026-08-02] wrapped in hidden_effect: the reward-preview tooltip-eval of
			# CURRENCY_grant_country_wealth floods error.log (~1000 lines when the tree is opened)
			# because its reserve-ratio svalue chain resolves to 'none' in preview context. hidden_effect
			# is excluded from tooltip generation, so the flood stops while the grant still executes.
			hidden_effect = { CURRENCY_grant_country_wealth = { thousands = 70 } }
			# the short road home offends the two powers who condominated Egypt
			QING_gp_provoke_britain = { severity = 10 }
			QING_gp_provoke_france = { severity = 8 }
			LOG_line = { sys = QING  msg = "MISSION task Suez passage for" }
		}
	}

	# --- 剛果 — the Congo basin, the heart of the interior scramble -----------
	# History: the Congo was the very crucible of the Scramble — Leopold's Congo
	# Free State and the Berlin Conference (1884-85) that partitioned the continent.
	# France held the north bank (Brazzaville). Here the Qing thrusts into the interior.
	qing_col_congo = {
		icon = qing_col_congo
		requires = { qing_col_cape }
		allow = {
			treasury >= 180
		}
		on_start = {
			add_treasury = -180
			LOG_line = { sys = QING  msg = "MISSION task START Congo interior (spending treasury) for" }
		}
		on_completion = {
			custom_tooltip = qing_col_congo_tt
			if = { limit = { exists = p:2652  NOT = { owns = 2652 } }  p:2652 = { add_claim = ROOT } }  # the Congo (Kongo, Equatorial Africa — in the Congo_Basin region)
			if = { limit = { exists = p:3526  NOT = { owns = 3526 } }  p:3526 = { add_claim = ROOT } }  # Libreville, the Gulf-of-Guinea coast (Gabon)
			add_country_modifier = { name = qing_col_congo_interior  duration = -1 }
			# [logfix 2026-08-02] wrapped in hidden_effect: the reward-preview tooltip-eval of
			# CURRENCY_grant_country_wealth floods error.log (~1000 lines when the tree is opened)
			# because its reserve-ratio svalue chain resolves to 'none' in preview context. hidden_effect
			# is excluded from tooltip generation, so the flood stops while the grant still executes.
			hidden_effect = { CURRENCY_grant_country_wealth = { thousands = 90 } }
			# carving the interior sets the Qing directly against Britain and France, the
			# paramount partitioners of the Berlin Conference
			QING_gp_provoke_britain = { severity = 8 }
			QING_gp_provoke_france = { severity = 10 }
			LOG_line = { sys = QING  msg = "MISSION task Congo interior for" }
		}
	}

	# --- 安非 ĀNFĒI, Pacified Africa — the invented African march (都護府) ------
	# The African protectorate: the dynasty's holdings across the continent —
	# the Swahili coast, the Cape, the Red Sea gate, the Congo — gathered into a
	# single march under a Protector-General, extending the Tang institution to the
	# ocean Zheng He last crossed. Wholly an invention of this Qing.
	# NOTE [#27]: the march-founding mechanic is reworked to the EIC model in a later
	# chunk (P3, design/DESIGN_PROTECTORATES_GENERAL.md §4.2); this SPLIT chunk
	# relocates the task VERBATIM so the lift is a pure, boot-safe relocation.
	qing_col_anfei = {
		icon = qing_col_anfei
		requires = { qing_col_cape }
		allow = {
			political_influence >= 30
			custom_tooltip = {
				text = qing_col_needs_africa_land_tt
				any_owned_province = {
					count >= 3
					OR = {
						is_in_region = Lake_Victoria
						is_in_region = South_Africa
						is_in_region = Egypt
						is_in_region = Congo_Basin
						is_in_region = Mozambique
						is_in_region = Madagascar
						is_in_region = Horn_of_Africa
						is_in_region = Gulf_of_Guinea
						is_in_region = Coastal_West_Africa
						is_in_region = Angola
					}
				}
			}
		}
		on_start = {
			add_political_influence = -30
			LOG_line = { sys = QING  msg = "MISSION task START Anfei protectorate for" }
		}
		on_completion = {
			custom_tooltip = qing_col_anfei_tt
			every_owned_province = {
				limit = {
					is_capital = no
					OR = {
						is_in_region = Lake_Victoria
						is_in_region = South_Africa
						is_in_region = Egypt
						is_in_region = Congo_Basin
						is_in_region = Mozambique
						is_in_region = Madagascar
						is_in_region = Horn_of_Africa
						is_in_region = Gulf_of_Guinea
						is_in_region = Coastal_West_Africa
						is_in_region = Angola
					}
				}
				add_to_list = qing_protectorate_list
			}
			# [#27 P3] the EIC-model march founding (se_QING_MARCH.txt). Carved from the player's
			# COLONIZED African holdings (overseas frontier land, NOT CHI core). locallist empty for now
			# (grows via the P5 expansion pulse). maritime = yes (Anfei gets a navy, DESIGN §3.3).
			QING_found_march = {
				list = qing_protectorate_list
				name_key = qing_protectorate_anfei_name
				adj_key = qing_protectorate_anfei_adj
				maritime = yes
				log_tag = ANFEI
				locallist = qing_march_locals_list
			}
			add_country_modifier = { name = qing_col_protectorate_africa  duration = -1 }
			current_ruler = { add_popularity = 30 }
			LOG_line = { sys = QING  msg = "MISSION task Anfei protectorate for" }
		}
	}
}
