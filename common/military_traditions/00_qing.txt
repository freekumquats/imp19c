# 00_qing.txt — QING CULTURE/SUBJECT MILITARY TRADITIONS (Phase 6).
#
# Beyond the commemorative Ten Great Campaigns (00_manchu.txt) and the Napoleon-at-court
# alt-history (00_napoleon.txt), these are the STANDING military institutions of the Qing
# multi-ethnic army — the banner/Green-Standard/Mongol/frontier/tributary organisation the
# user asked for ("Qing-specific military trees tied to cultures and subjects").
#
# Each tree gates on the manchu_shiquan idiom (began_with_tradition_group) for its OWN culture
# group, PLUS — the Ministry-of-War tie-in (Part G, the user's brief) — a trigger_else_if branch
# that unlocks the tree once the 兵部 is STAFFED (has_variable = qing_office_war_holder). The
# integration branches (Green-Standard/Mongol/frontier) additionally unlock via the X_science
# variables the empire earns by integrating its subject peoples. A FAIL-CLOSED trigger_else
# re-asserts the full OR of unlock conditions so a country matching NEITHER branch is LOCKED
# (without it a trigger_if-only allow evaluates TRUE for everyone — the M1 review finding). This
# is the proven Invictus 00_greek_2.txt allow idiom (trigger_if / trigger_else_if has_variable /
# trigger_else custom_tooltip) — has_variable in a tradition allow is proven engine behaviour.
#
# FLAT trees (arabic/indian/manchu pattern: no path-container level, rewards folded into nodes).
# Placeholder .dds images/icons (existing files) so nothing renders blank. Modifier keys are
# ONLY those already used in the mod's populated trees + proven terrain/unit keys. TAB indent.
#
# Sources: [[imp19c-qing-frontier-garrisons]], [[imp19c-qing-history-and-mechanics]]; the banner
# (八旗) / Green-Standard (綠營) / Lifan-Yuan Mongol-banner / 藩部 frontier-garrison / 朝貢 levy
# structure of the High-Qing military.

# 八旗 — THE EIGHT BANNERS. The hereditary Manchu-Mongol-Han banner garrison
# system, the martial core of the dynasty. Gated on the jurchen group (CHI's own)
# OR a staffed Ministry of War.
# ======================================================================
qing_eight_banners_tradition = {
	color = hsv { 0.08 0.9 0.85 }
	image = arabic_philosophy_start_bonus   # placeholder (existing .dds)

	allow = {
		trigger_if = {
			limit = { country_culture_group = jurchen }
			custom_tooltip = { text = "began_with_tradition_group" }

		}
		trigger_else_if = {
			limit = { has_variable = qing_office_war_holder }
			custom_tooltip = {
				text = "qing_tradition_unlock_war_ministry"

			}
		}
		# fail-closed: a country matching neither branch above is LOCKED (this OR is false here).
		trigger_else = {
			custom_tooltip = {
				text = "qing_tradition_locked_banner"
				OR = {
					country_culture_group = jurchen
					has_variable = qing_office_war_holder

			}
		}
	}

	# START — the banner establishment: a standing reserve of hereditary soldiers.
	qing_banner_start = {
		icon = indian_philosophy_start_bonus   # placeholder
		ai_will_do = { modifier = { trigger = { country_culture_group = jurchen } add = { value = 5 } } }
		modifier = {
			global_manpower_modifier = 0.05
			land_morale_modifier = 0.05
		}
	}

	qing_banner_cavalry = {
		icon = arabic_peninsula_path_1   # placeholder
		requires = { qing_banner_start }

		modifier = {
			light_cavalry_offensive = 0.1
