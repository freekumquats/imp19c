# [BT 2026-07-25] Hanlin waiting-scholar degree title (科舉功名). Character-scope custom loc mapping
# a pool scholar's exam-degree trait to its title, so the Hanlin roster shows a TITLE under each
# scholar's name the same way the Grand Council sub-position rosters (and the harem 位號) do.
# Referenced in gui/qing_hanlin.gui via [Character.Custom('qing_scholar_degree_name')]. Idiom mirrors
# qing_consort_rank_name (00_qing_harem_loc.txt): text{ localization_key + trigger }, first_valid top-down
# (highest degree first). The degree traits are the #34 ladder (00_imp19c.txt); loc keys reuse the
# existing trait-name loc so the title reads the same as the trait.
qing_scholar_degree_name = {
	type = character
	first_valid = yes

	text = { localization_key = hanlin       trigger = { has_trait = hanlin } }
	text = { localization_key = fanyi_jinshi trigger = { has_trait = fanyi_jinshi } }
	text = { localization_key = jinshi       trigger = { has_trait = jinshi } }
	text = { localization_key = gongshi      trigger = { has_trait = gongshi } }
	text = { localization_key = juren        trigger = { has_trait = juren } }
	text = { localization_key = shengyuan    trigger = { has_trait = shengyuan } }
	text = { localization_key = jiansheng    trigger = { has_trait = jiansheng } }
	# a waiting laureate with no degree trait (should not happen; safe default).
	text = { localization_key = qing_scholar_degree_none  trigger = { always = yes } }
}
