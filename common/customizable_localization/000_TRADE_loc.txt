# [fix task #5 2026-09-04] the Diplomatic Play card title (DIPLO_play_title,
# imp19c_interface_l_english.yml) was one fixed territory-style string reused for every
# play_goal -- for flag:marriage_proposal specifically, play_target_area is the PROPOSER's
# OWN capital (se_MARRIAGE_PLAY.txt:19-22 -- a marriage play has no territorial locus, so
# the capital just stands in to satisfy the "a play needs an area" requirement), so the
# title read "Qing play for Beijing against Russia" for a marriage offer. This dispatcher
# branches per play_goal (proven text/trigger idiom, see DIPLO_toggle_start_diplomatic_play_button
# below) to a goal-appropriate title key; type = province matches trade_center_title_loc
# above (has_variable/var: read directly off the root, no MakeScope needed on the trigger
# side -- only the LOCALIZED TEXT itself still needs MakeScope, same as every other loc key
# in this file). Called as [Scope.GetProvince.Custom('DIPLO_play_title_custom')] from the
# two diplomatic-play card templates (gui/shared/gui_templates.gui).
DIPLO_play_title_custom = {
	type = province

	text = {
		localization_key = DIPLO_play_title_marriage
		trigger = { var:play_goal = flag:marriage_proposal }
	}
	text = {
		localization_key = DIPLO_play_title_colonise
		trigger = { var:play_goal = flag:colonise_area }
	}
	# Fallback: every territorial goal (get_territory, annex_state, purchase_state,
	# subjugate_nation, liberate_state) plus the unhandled-goal default all read sensibly
	# with the original territory-style wording.
	text = {
		localization_key = DIPLO_play_title
		trigger = { always = yes }
	}
}

DIPLO_toggle_start_diplomatic_play_button = {
	type = country

	# [#210] STOP variant first: if the player already runs a play here, the button ends it.
	text = {
		localization_key = STOP_DIPLOMATIC_PLAY_BUTTON

		trigger = {
            any_in_list = {
                variable = local_area_diplomatic_plays

                var:play_target_country = ROOT.var:selected_province.owner
                var:play_instigator = ROOT
            }
		}
	}

	# [#210] START variants: reflect WHICH play the player has SELECTED with the type
	# buttons (var:pending_play_goal), so the button previews what pressing it will begin.
	# One text per goal, then a bare fallback when no type has been chosen yet.
	text = {
		localization_key = START_DIPLOMATIC_PLAY_BUTTON_GET_TERRITORY
		trigger = { ROOT = { has_variable = pending_play_goal  var:pending_play_goal = flag:get_territory } }
	}
	text = {
		localization_key = START_DIPLOMATIC_PLAY_BUTTON_ANNEX_STATE
		trigger = { ROOT = { has_variable = pending_play_goal  var:pending_play_goal = flag:annex_state } }
	}
	text = {
		localization_key = START_DIPLOMATIC_PLAY_BUTTON_PURCHASE_STATE
		trigger = { ROOT = { has_variable = pending_play_goal  var:pending_play_goal = flag:purchase_state } }
	}
	text = {
		localization_key = START_DIPLOMATIC_PLAY_BUTTON_SUBJUGATE
		trigger = { ROOT = { has_variable = pending_play_goal  var:pending_play_goal = flag:subjugate_nation } }
	}
	text = {
		localization_key = START_DIPLOMATIC_PLAY_BUTTON_LIBERATE
		trigger = { ROOT = { has_variable = pending_play_goal  var:pending_play_goal = flag:liberate_state } }
	}
	text = {
		localization_key = START_DIPLOMATIC_PLAY_BUTTON_COLONISE
		trigger = { ROOT = { has_variable = pending_play_goal  var:pending_play_goal = flag:colonise_area } }
	}
	# Fallback: no goal picked yet (defaults to get_territory when pressed).
	text = {
		localization_key = START_DIPLOMATIC_PLAY_BUTTON
		trigger = { always = yes }
	}
}

DIPLO_toggle_start_diplomatic_play_button_tt = {
	type = country

	text = {
		localization_key = START_DIPLOMATIC_PLAY_BUTTON_TT

		trigger = {
			NOT = {
                any_in_list = {
                    variable = local_area_diplomatic_plays

                    var:play_target_country = ROOT.var:selected_province.owner
                    var:play_instigator = ROOT
                }
			}
		}
	}

	text = {
		localization_key = STOP_DIPLOMATIC_PLAY_BUTTON_TT

		trigger = {
            any_in_list = {
                variable = local_area_diplomatic_plays

                var:play_target_country = ROOT.var:selected_province.owner
                var:play_instigator = ROOT
            }
		}
	}
}
