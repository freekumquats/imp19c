# se_QING_MARCH.txt — the frontier PROTECTORATES-GENERAL (都護府) as "Qing EICs".
# #27 rework (2026-08-02). Design: design/DESIGN_PROTECTORATES_GENERAL.md;
# build log/decisions: overnight/OVERNIGHT_PROTECTORATE.md.
#
# The seven marches (安北/安東/安西/安南/安海/安新/安非) are self-governing FRONTIER
# SUBJECTS of the Son of Heaven, modelled on the mod's own East India Company:
#   CHI --(frontier_protectorate)--> MARCH (govt megacorporation) --> local princely states
# A march is carved from CONQUERED LOCAL-POLITY land (NEVER from CHI's own core),
# holds the theatre's other local polities as ITS subjects, is funded by Beijing in
# REVERSE tribute (協餉 xiéxiǎng subsidy: money + a manpower levy on army raises), and
# expands by its own wars + colonisation. See the design doc for the full rationale
# and the failure history of the OLD (backwards) QING_establish_protectorate mechanic
# — which is KEPT, unchanged, for the Lanfang kongsi + the Mexican client-Empire
# releases (those are genuine sinosphere_tributary client states, NOT marches).
#
# WHY LAND_release_from_list, not create_country->change_country_tag: the design doc
# (§5/§8.0) flags the predefined-tag mint + change_country_tag-as-a-unit as UNPROVEN
# (no repo precedent), needing an isolation boot-spike. This build takes the doc's
# documented FALLBACK as the primary path: a DYNAMIC-tag release via the proven
# LAND_release_from_list (the same verb the old effect + Lanfang + Mexico use), with a
# scripted country name. Cost of the fallback: the march is a generated tag (no fixed
# 3-letter code / predefined COA) — acceptable and boot-safe. See OVERNIGHT R2.
#
# UNIT SIZING: COHORT_SIZE = 500 men (common/defines/00_defines.txt:355), so the
# subsidy tiers low/med/high = 5,000/10,000/15,000 men = 10/20/30 cohorts.
#
# Proven idioms only:
#  - LAND_release_from_list (se_LAND.txt:654) — spins a new country off a province list,
#    saves scope:new_country_scope, handles econ init (fires ECON_events.7).
#  - FUNC_make_subject (se_FUNC.txt) — overlord can be ANY tag (so a march is overlord
#    of its locals); type = frontier_protectorate binds the march to CHI.
#  - bare create_unit in a COUNTRY scope + `location = capital_scope` (se_QING_BURMA
#    QING_burma_rally_konbaung raises c:BUR's OWN host; TI galatian_invasion) — raises
#    the MARCH's own army in the MARCH scope.
#  - add_manpower = -N on CHI (se_QING_WORKS.txt:449) — the manpower-on-raise dock.
#  - ncohorts (00_republic_triggers.txt) — read a country's current cohort count for the
#    army-maintenance reconcile.
#  - is_subject_type = X, every_subject { limit } (se_SUBJECT_QING QING_subject_collect_
#    tribute) — the quarterly per-subject sweep this mirrors, flow reversed.
#
# PREFIX: QING_march_ / QING_found_march. sys = QING (LOG_* module, se_LOG.txt).

# ======================================================================
# QING_found_march — establish one march from a pre-built province list of
# CONQUERED LOCAL-POLITY territory. Scope: country (ROOT = the Qing, c:CHI).
# Args:
#   list      : province list of the just-conquered LOCAL land the march is carved from
#               (built by the caller; guarded so CHI core is never included — see §6.1).
#   name_key  : loc key for the march's name
#   adj_key   : loc key for the march's adjective
#   maritime  : yes|no — a maritime march also gets a navy from its subsidy (P9); the
#               flag is stamped on the march for the subsidy pulse to read.
#   locallist : (optional) a country list of the theatre's OTHER local polities to
#               subordinate as the march's OWN subjects (its princely states). May be
#               empty — the caller builds it; guarded per tag.
#   log_tag   : short identifier for the log line (e.g. ANFEI).
# ======================================================================
QING_found_march = {
	LOG_enter = { sys = QING  fn = "QING_found_march" }

	if = {
		limit = { any_in_list = { list = $list$  count >= 1 } }

		LOG_line = { sys = QING  msg = "MARCH $log_tag$ founding from conquered local land held by" }

		# 1. Spin the conquered LOCAL land off as a new country — the march. Released
		#    INDEPENDENT (as_subject_type = flag:independent) so we bind the exact
		#    frontier_protectorate type below rather than the release's default colony.
		#    Dynamic tag + name/adj loc keys (the proven fallback, see header). Capital =
		#    the most-populous province of the conquered land; culture/religion inherited
		#    from that seat (the march is a LOCAL conqueror-state under a Qing charter).
		LAND_release_from_list = {
			releaser = ROOT
			list = $list$
			capital_location = flag:by_population
			primary_culture = scope:new_capital_scope.dominant_province_culture
			country_religion = scope:new_capital_scope.dominant_province_religion
			as_subject_type = flag:independent
			government_type = flag:dynamic
			give_overlord_hos = no
			country_flag = flag:dynamic
			country_name = $name_key$
			country_adj = $adj_key$
			country_color = flag:dynamic
		}

		if = {
			limit = { exists = scope:new_country_scope }

			# 2. Government = megacorporation (the EIC's government; the ruler slot IS
			#    the Governor-General, has_co_ruler = no). Then bind the march to CHI as
			#    frontier_protectorate (LAND_release fired ECON_events.7 already, so the
			#    economy is initialised — do not re-init).
			scope:new_country_scope = { change_government = megacorporation }
			FUNC_make_subject = {
				overlord = ROOT
				target = scope:new_country_scope
				type = frontier_protectorate
			}
			# stamp the maritime flag for the subsidy pulse (navies, P9) + a founding marker.
			scope:new_country_scope = {
				set_variable = { name = qing_march  value = 1 }
				if = { limit = { $maritime$ = yes }  set_variable = { name = qing_march_maritime  value = 1 } }
			}

			# 3. GOVERNOR-GENERAL appointed by the Lifan Yuan (理藩院). Only if a Lifan Yuan
			#    holder is seated (the appointment is an act of that office); else the march
			#    keeps the engine-generated ruler and the yearly pulse re-imposes a GG later
			#    (QING_march_reappoint_gg). Create the Manchu administrator in CHI scope, THEN
			#    install as the march's ruler in a SEPARATE step (never grant a role inside the
			#    create_character block — the #90 boot-crash gotcha; no HEALTH trait on the char).
			QING_march_appoint_gg = { march = scope:new_country_scope }

			# 4. Subordinate the theatre's OTHER local polities as the MARCH's own subjects
			#    (its princely states — they answer to the march, not to Beijing). Guarded per
			#    tag: skip any that no longer exists, is the march itself, or is CHI. A polity is
			#    taken ONLY if it is EITHER free (is_subject=no) OR already a subject of CHI
			#    (is_subject_of=ROOT) — the latter a deliberate REPARENT hand-down (CHI's conquered
			#    tributaries devolved to the march, the EIC model; the LAND marches' khanates/daimyo
			#    are ALREADY CHI subjects when the founding task runs, so this branch is what actually
			#    populates them). A polity under a FOREIGN overlord is skipped (never yank a third
			#    party's subject). FUNC_make_subject re-parents in place; overlord = the march scope.
			if = {
				limit = { any_in_list = { list = $locallist$  count >= 1 } }
				every_in_list = {
					list = $locallist$
					limit = {
						exists = this
						NOT = { this = scope:new_country_scope }
						NOT = { this = ROOT }
						# take only free polities OR CHI's own subjects (a deliberate reparent
						# hand-down); never yank a polity from a FOREIGN overlord.
						OR = {
							is_subject = no
							is_subject_of = ROOT
						}
					}
					save_temporary_scope_as = local_princely_state
						# RELEASE-then-rebind: if the polity is already a CHI subject, release it from CHI
						# FIRST, then bind it to the march. make_subject on an ALREADY-bound subject is not
						# safely reparenting in this engine — the proven idiom (se_FUNC.txt:349-351 +
						# se_FUNC.txt:1411-1415) releases before re-subjugating. A free polity skips the release.
						if = {
							limit = { scope:local_princely_state = { is_subject_of = ROOT } }
							ROOT = { release_subject = scope:local_princely_state }
						}
						FUNC_make_subject = {
							overlord = scope:new_country_scope
							target = scope:local_princely_state
							type = protectorate
						}
					}
				}

			# 5. Seed the march's initial army sized to whatever subsidy tier the founding
			#    task set (default small if none set). The subsidy pulse (§3.3) maintains it
			#    thereafter. Raise in the MARCH scope so the troops are the MARCH's own.
			#    [#39-review-fix] QING_march_size_army's else-branch targets 0 cohorts when
			#    qing_subsidy_tier is unset, so a march founded before any support law is enacted
			#    was born ARMYLESS despite the "default small" comment. Stamp the small tier here.
			scope:new_country_scope = {
				if = { limit = { NOT = { has_variable = qing_subsidy_tier } }  set_variable = { name = qing_subsidy_tier  value = flag:small } }
			}
			QING_march_size_army = { march = scope:new_country_scope }

			LOG_line = { sys = QING  msg = "MARCH $log_tag$ established as a frontier_protectorate of" }
		}
		else = {
			LOG_fail = { sys = QING  fn = "QING_found_march"  reason = "release produced no new_country_scope" }
		}

		LOG_exit = { sys = QING  fn = "QING_found_march"  result = OK }
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_found_march"  reason = "no conquered local provinces in list to found the march" }
	}

	# Clean the working lists so a later march task starts from empty.
	every_in_list = { list = $list$  remove_from_list = $list$ }
	every_in_list = { list = $locallist$  remove_from_list = $locallist$ }
}

# ======================================================================
# [#49] QING_convert_to_march — turn an EXISTING country (already on the map) into a
# frontier_protectorate march, with the SAME march-hood as QING_found_march but WITHOUT a
# release (the country already exists; we don't spin a new tag from a province list). Built
# for Lanfang (LAF) — a live Hakka kongsi republic that is CHI's nominal/sinosphere vassal at
# start; the Nanyang tree elevates it from a firm tributary to a full march. Scope: ROOT =
# c:CHI. Args: target = <the existing country scope>; maritime = yes|no.
# Steps mirror QING_found_march 2-3/5: release-then-rebind as frontier_protectorate,
# megacorporation government, march flags, Lifan-Yuan Governor-General, seed the subsidy army.
# ======================================================================
QING_convert_to_march = {
	LOG_enter = { sys = QING  fn = "QING_convert_to_march" }
	if = {
		limit = { exists = $target$ }
		# 1. RELEASE-then-REBIND: FUNC_make_subject on an already-bound subject does not safely
		#    reparent in this engine (the same idiom the local-princely-state loop uses). LAF is a
		#    CHI subject after the embrace, so release it from CHI first, then bind as the march type.
		if = {
			limit = { $target$ = { is_subject_of = ROOT } }
			ROOT = { release_subject = $target$ }
		}
		# 2. Government = megacorporation (the EIC's government; the ruler slot IS the Governor-
		#    General). Then bind to CHI as frontier_protectorate. (An existing country already has its
		#    economy initialised — nothing to re-init, unlike a fresh LAND_release.)
		$target$ = { change_government = megacorporation }
		FUNC_make_subject = {
			overlord = ROOT
			target = $target$
			type = frontier_protectorate
		}
		# 3. Stamp the march flags (subsidy pulse + expansion pulse key off these).
		$target$ = {
			set_variable = { name = qing_march  value = 1 }
			if = { limit = { $maritime$ = yes }  set_variable = { name = qing_march_maritime  value = 1 } }
		}
		# 4. Governor-General appointed by the Lifan Yuan (no-op if no holder seated; the yearly
		#    pulse re-imposes one later). Same call QING_found_march uses.
		QING_march_appoint_gg = { march = $target$ }
		# 5. Seed the march's initial army to its subsidy tier (default small if none set); the
		#    subsidy pulse maintains it thereafter. [#39-review-fix] stamp the small tier if unset,
		#    else QING_march_size_army targets 0 cohorts and the converted march is born armyless.
		$target$ = {
			if = { limit = { NOT = { has_variable = qing_subsidy_tier } }  set_variable = { name = qing_subsidy_tier  value = flag:small } }
		}
		QING_march_size_army = { march = $target$ }
		LOG_line = { sys = QING  msg = "MARCH established by CONVERTING an existing country to a frontier_protectorate" }
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_convert_to_march"  reason = "target country does not exist; conversion skipped" }
	}
	LOG_exit = { sys = QING  fn = "QING_convert_to_march"  result = OK }
}

# ======================================================================
# QING_march_appoint_gg — install a Lifan-Yuan-appointed Manchu Governor-General as
# the march's ruler. Scope: country (ROOT = c:CHI). Arg: march = <the march scope>.
# Gated on a seated Lifan Yuan holder (the appointment is an act of that office — the
# same gate the amban sweep uses). If no holder is seated, no-op (the march keeps its
# generated local ruler; the yearly re-appointment pulse re-checks).
#
# IDIOM (proven — MEX_install_empire, se_MEXICO.txt:398): enter the MARCH scope, create
# the character THERE, and set_as_ruler THERE — the create-in-target-then-set_as_ruler
# pattern. This deliberately AVOIDS the unproven cross-country create-in-CHI-then-move-
# and-set_ruler primitive the design doc (§3.1/M5) flagged for a boot-spike I can't run:
# creating the GG directly in the march scope is functionally identical (he is a Manchu
# Lifan-Yuan appointee — marked by culture manchu + fanyi_jinshi + the qing_march_gg var
# — regardless of which scope's create_character minted him) and uses only proven verbs.
# #90 gotcha honoured: the role (set_as_ruler) is a SEPARATE statement AFTER the
# create_character block closes, never inside it; no HEALTH trait on the created char.
# ======================================================================
QING_march_appoint_gg = {
	if = {
		limit = {
			exists = $march$
			has_variable = qing_office_lifanyuan_holder
			var:qing_office_lifanyuan_holder = { is_alive = yes  employer = ROOT }
		}
		$march$ = {
			# [#39 review-fix #2] clear the OUTGOING ruler's GG markers before seating the new one, so a
			# displaced Governor-General is not left with a stale qing_march_gg=1 (and no tenure timer) —
			# which would make him eligible for the overmighty/turnover events again if succession ever
			# re-elevated him. Guarded on the current ruler actually being a GG.
			if = {
				limit = { exists = current_ruler  current_ruler = { has_variable = qing_march_gg } }
				current_ruler = {
					remove_variable = qing_march_gg
					if = { limit = { has_variable = qing_march_gg_tenure_pending }  remove_variable = qing_march_gg_tenure_pending }
				}
			}
			# create the Manchu conquest-elite Governor-General in the MARCH's own scope.
			create_character = {
				age = { 40 55 }
				culture = manchu
				religion = vajrayana
				female = no
				add_martial = { 4 6 }
				add_finesse = { 4 6 }
				add_charisma = { 3 5 }
				add_zeal = { 2 4 }
				add_trait = fanyi_jinshi   # a translation-examination laureate, as the ambans are
				save_scope_as = qing_new_gg
			}
			# SEPARATE step (post-creation, outside the create block — #90 gotcha): mark him a
			# Lifan-Yuan appointee and seat him as the march's ruler.
			scope:qing_new_gg = { set_variable = { name = qing_march_gg  value = 1 } }
			set_as_ruler = scope:qing_new_gg
			# [#39] tenure timer, mirroring the amban's qing_amban_tenure_pending: a self-expiring
			# flag (days=2920 ≈ 8y). While it EXISTS the GG is freshly-appointed; once it has expired
			# (no var) he has served long enough for the "overmighty / gone native" event (qing_march.4)
			# to consider him. `age`/`health` remain readable directly for the turnover event (.5).
			scope:qing_new_gg = { set_variable = { name = qing_march_gg_tenure_pending  value = 1  days = 2920 } }
		}
		LOG_line = { sys = QING  msg = "MARCH governor-general appointed by the Lifan Yuan" }
	}
	else = {
		LOG_line = { sys = QING  msg = "MARCH no Lifan Yuan holder seated; march keeps its local ruler for now" }
	}
}

# ======================================================================
# QING_march_size_army — reconcile ONE march's standing host to its subsidy-tier
# target. Scope: country (ROOT = c:CHI). Arg: march = <the march scope>.
# Target men by tier: none=0, small=5000, medium=10000, high=15000 (COHORT_SIZE 500 →
# 0/10/20/30 cohorts), PLUS any temporary relief bonus (qing_march_relief_cohorts, P8).
# H3 MODEL: CHI pays manpower ONLY for the cohorts actually RAISED toward the target
# (target - current), never a flat quarterly upkeep. A march at target costs 0 manpower.
# Lifecycle symmetry: if over target (tier lowered), disband down toward it.
# ======================================================================
QING_march_size_army = {
	if = {
		limit = { exists = $march$ }
		$march$ = {
			# derive target COHORTS from the tier var (men/500). Relief bonus (P8) adds cohorts.
			if =      { limit = { var:qing_subsidy_tier = flag:small }   set_variable = { name = qing_march_target_cohorts  value = 10 } }
			else_if = { limit = { var:qing_subsidy_tier = flag:medium }  set_variable = { name = qing_march_target_cohorts  value = 20 } }
			else_if = { limit = { var:qing_subsidy_tier = flag:high }    set_variable = { name = qing_march_target_cohorts  value = 30 } }
			else =    { set_variable = { name = qing_march_target_cohorts  value = 0 } }
			# temporary relief bonus (P8 war-relief event) — ADDS to the target while live, then DECAYS.
			# The relief event stamps a timed qing_march_relief_decay flag (auto-expires ~2y); once that
			# flag is gone but the bonus var remains, clear the bonus so the host reverts to the tier
			# baseline (the down-step then stands the extra cohorts down).
			if = {
				limit = {
					has_variable = qing_march_relief_cohorts
					NOT = { has_variable = qing_march_relief_decay }
				}
				remove_variable = qing_march_relief_cohorts
			}
			if = {
				limit = { has_variable = qing_march_relief_cohorts }
				change_variable = { name = qing_march_target_cohorts  add = var:qing_march_relief_cohorts }
			}
		}
		# In the MARCH scope: compute the cohort shortfall (target - current) and, if positive,
		# raise exactly that many cohorts as one host at the capital (the march's OWN troops).
		# The comparison uses a NAMED svalue on the RHS (RHS-operator rule); the raise `count`
		# uses a var directly (proven `count = var:X`, se_QING_AMBAN:270).
		# In the MARCH scope: shortfall = target - current cohorts.
		# CREATE_UNIT SEMANTICS (imp19c_effects_legion_setup.txt:60): a create_unit with a base
		# `sub_unit` PLUS `while = { count = N add_subunit }` raises N+1 cohorts (total = count + 1,
		# the base + N added). So to raise `shortfall` cohorts total, the loop count is shortfall-1
		# (and shortfall must be >= 1). Failing to subtract 1 overshoots by one every time, so the
		# host never converges and thrashes (destroy+rebuild) every peacetime quarter — the HIGH bug.
		$march$ = {
			save_scope_as = march_being_sized
			set_variable = { name = qing_march_shortfall  value = var:qing_march_target_cohorts }
			change_variable = { name = qing_march_shortfall  subtract = qing_march_ncohorts_svalue }
			# the loop count = shortfall - 1 (base sub_unit is the +1), stored in its own var.
			set_variable = { name = qing_march_raise_count  value = var:qing_march_shortfall }
			change_variable = { name = qing_march_raise_count  subtract = 1 }
			if = {
				# raise only when at least one full cohort short (shortfall >= 1 → count >= 0).
				limit = {
					exists = capital_scope
					var:qing_march_shortfall >= 1
				}
				create_unit = {
					navy = no
					name = "Protectorate Host"
					location = capital_scope
					sub_unit = regular_infantry
					# count = shortfall-1 additional on top of the 1 base = shortfall total.
					while = { count = var:qing_march_raise_count  add_subunit = regular_infantry }
				}
				# CHI pays the 協餉 manpower levy for exactly the cohorts RAISED (H3): shortfall total
				# cohorts * COHORT_SIZE(500) men. add_manpower deduct on CHI is proven (se_QING_WORKS:449);
				# inline { value multiply } negation is the proven effect form (tribute model). ROOT = c:CHI
				# throughout (all three callers enter in CHI scope). Bills the TRUE raised count (shortfall),
				# not the loop count, so the levy matches the men actually added.
				ROOT = { add_manpower = { value = scope:march_being_sized.var:qing_march_shortfall  multiply = -500 } }
				LOG_line = { sys = QING  msg = "MARCH host raised toward subsidy target; CHI paid the manpower levy" }
			}
			# lifecycle symmetry: if OVER target by at least a full cohort (shortfall <= -1 — tier
			# lowered / relief expired), stand the hosts down so a de-funded march shrinks. The <= -1
			# guard (not < 0) means a mere rounding never trips this — only a genuine over-strength host,
			# so a march at target sits STABLE (no quarterly thrash). destroy_unit=yes on every_army is
			# the PROVEN disband verb (SE_qing_navy_disband: every_navy = { destroy_unit = yes });
			# ordered_army/disband_unit are NOT attested here. Disband all + re-raise to the lower target
			# keeps only proven verbs. Guard war=no: never strip a march mid-campaign. The re-raise
			# is a SHRINK (target < current), so it docks NO new manpower (H3 — down-steps are free).
			else_if = {
				limit = {
					var:qing_march_shortfall <= -1
					war = no
					any_army = { count >= 1 }
				}
				every_army = { destroy_unit = yes }
				if = {
					limit = {
						exists = capital_scope
						var:qing_march_target_cohorts >= 1
					}
					# count = target-1 additional + 1 base = target total (reuse the raise-count var).
					set_variable = { name = qing_march_raise_count  value = var:qing_march_target_cohorts }
					change_variable = { name = qing_march_raise_count  subtract = 1 }
					create_unit = {
						navy = no
						name = "Protectorate Host"
						location = capital_scope
						sub_unit = regular_infantry
						while = { count = var:qing_march_raise_count  add_subunit = regular_infantry }
					}
				}
				LOG_line = { sys = QING  msg = "MARCH over subsidy target; host stood down and re-raised to the lower target" }
			}
			remove_variable = qing_march_shortfall
			remove_variable = qing_march_raise_count

			# --- MARITIME marches also get a NAVY scaled to the subsidy tier (DESIGN §3.3/H2, P9) ---
			# Only a maritime march (qing_march_maritime flag, set at founding) that HOLDS a port. Navy
			# target = half the army tier in ships (small/med/high = 5/10/15 ships). Issued BARE in the
			# MARCH's own scope at a march-held port — the PROVEN navy-raise form (SE_qing_navy_guangdong /
			# TI rel_flavor_kemetic.7: create_unit navy=yes location=<owned is_port> in the country's own
			# scope). This is NOT the H2-flagged re-scoped raise (which re-resolves location wrong) — the
			# march IS `this` here, so location resolves to the march's port. Simple top-up: raise the navy
			# ONLY if the march currently has none (avoids per-quarter stacking); a lost fleet is re-raised.
			if = {
				limit = {
					has_variable = qing_march_maritime
					var:qing_march_target_cohorts >= 1
					# "march has NO fleet" — `any_navy = { count = 0 }` is ALWAYS TRUE (count = N means
					# >= N in this engine), which would stack a new fleet every quarter. The attested
					# absence idiom is NOT { any_navy count >= 1 } (SE_qing_navy_disband uses this form).
					NOT = { any_navy = { count >= 1 } }
					# a SEA port: is_port = yes is the PROVEN sea-port predicate (SE_qing_navy_guangdong /
					# se_QING_TREASURE_FLEET use is_port ALONE and explicitly warn NOT is_coastal — is_coastal
					# is the LOOSER predicate that also matches river/lake ports, which create_unit navy=yes
					# rejects). A maritime march founded from overseas/coastal conquest holds real sea ports.
					any_owned_province = { is_port = yes }
				}
				# ship count by tier (target cohorts 10/20/30 -> 5/10/15 ships; -1 for the base sub_unit).
				if =      { limit = { var:qing_march_target_cohorts >= 30 }  set_variable = { name = qing_march_ships  value = 15 } }
				else_if = { limit = { var:qing_march_target_cohorts >= 20 }  set_variable = { name = qing_march_ships  value = 10 } }
				else =    { set_variable = { name = qing_march_ships  value = 5 } }
				change_variable = { name = qing_march_ships  subtract = 1 }   # base sub_unit is the +1
				ordered_owned_province = {
					limit = { is_port = yes }
					order_by = total_population
					max = 1
					save_scope_as = qing_march_berth
				}
				create_unit = {
					navy = yes
					name = "Protectorate Fleet"
					location = scope:qing_march_berth
					sub_unit = brig
					while = { count = var:qing_march_ships  add_subunit = brig }
				}
				remove_variable = qing_march_ships
				LOG_line = { sys = QING  msg = "MARITIME MARCH raises a fleet scaled to its subsidy tier" }
			}
		}
	}
}

# ======================================================================
# QING_march_pay_subsidy — the 協餉 REVERSE-TRIBUTE quarterly pulse. Scope: country
# (ROOT = c:CHI), called from the Qing quarterly pulse (qing_mechanics_pulse_on_action).
# For every march (frontier_protectorate) carrying a subsidy tier: transfer GOLD from
# CHI to the march scaled by tier (clamped to CHI's own treasury — a broke Beijing can
# not fund beyond what it holds, modelling 19th-c. frontier abandonment), and reconcile
# the march's army to the tier target (QING_march_size_army, which handles the manpower
# levy on raise, H3). Mirror of QING_subject_collect_tribute, flow REVERSED.
# ======================================================================
QING_march_pay_subsidy = {
	LOG_enter = { sys = QING  fn = "QING_march_pay_subsidy" }
	every_subject = {
		limit = {
			is_subject_type = frontier_protectorate
			# a funded march (has a tier) OR a march carrying a live/expired RELIEF bonus. The latter
			# is essential: an UNFUNDED march that received a war-relief army (qing_march_relief_cohorts)
			# must still be routed through QING_march_size_army so its relief host DECAYS + stands down
			# after the bonus expires (the decay-clear + over-target stand-down live only in the sizer).
			# Without this a tier-less relief march would keep its swollen host forever.
			OR = {
				has_variable = qing_subsidy_tier
				has_variable = qing_march_relief_cohorts
			}
		}
		save_temporary_scope_as = subsidy_march
		# quarterly gold stipend by tier (a flat per-march stipend; small/med/high rising),
		# stored on the MARCH (this scope) so the cross-scope transfer below can read it.
		if =      { limit = { var:qing_subsidy_tier = flag:small }   set_variable = { name = qing_subsidy_amt  value = 15 } }
		else_if = { limit = { var:qing_subsidy_tier = flag:medium }  set_variable = { name = qing_subsidy_amt  value = 30 } }
		else_if = { limit = { var:qing_subsidy_tier = flag:high }    set_variable = { name = qing_subsidy_amt  value = 45 } }
		else =    { set_variable = { name = qing_subsidy_amt  value = 0 } }
		# clamp the stipend to what CHI actually holds. A broke Beijing pays only what it has
		# (the tribute-model clamp: when the owed stipend exceeds CHI's treasury, cut it to the
		# treasury). Evaluated in the MARCH scope, where qing_subsidy_amt lives; the comparison RHS
		# reads CHI's treasury via the NAMED svalue chi_treasury_svalue (RHS-operator rule).
		if = {
			limit = { var:qing_subsidy_amt > chi_treasury_svalue }
			# pay CHI's whole remaining treasury (0 if CHI is in deficit — set_variable to
			# ROOT.treasury; the > 0 transfer guard below drops a non-positive result).
			set_variable = { name = qing_subsidy_amt  value = ROOT.treasury }
		}
		# transfer: CHI -= amt ; march += amt (the reverse of the tribute haul), only if > 0.
		# Proven inline forms (QING_subject_collect_tribute): { value multiply -1 } to spend,
		# and add_treasury = scope:X.var:Y to credit across scopes.
		if = {
			limit = { var:qing_subsidy_amt > 0 }
			ROOT = {
				add_treasury = { value = scope:subsidy_march.var:qing_subsidy_amt  multiply = -1 }
				# [#108 fix 2026-08-15] stage into a var -- a compound amount= block trips TREASURY_LOG_it's
				# macro-arg validator (see common/scripted_effects/se_STATE_INVESTMENT.txt for the full note).
				set_variable = { name = TREASURY_LOG_amt_tmp  value = { value = scope:subsidy_march.var:qing_subsidy_amt  multiply = -1 } }
				TREASURY_LOG_it = { amount = var:TREASURY_LOG_amt_tmp }
				remove_variable = TREASURY_LOG_amt_tmp
			}
			add_treasury = var:qing_subsidy_amt
			TREASURY_LOG_it = { amount = var:qing_subsidy_amt }
		}
		remove_variable = qing_subsidy_amt
		# reconcile this march's army to its tier target (manpower levy on raise, H3).
		ROOT = { QING_march_size_army = { march = scope:subsidy_march } }
	}
	LOG_exit = { sys = QING  fn = "QING_march_pay_subsidy"  result = OK }
}

# ======================================================================
# QING_march_set_subsidy — the low/med/high toggle effect (mirrors QING_subject_set_
# tribute). Scope: country (ROOT = c:CHI). Arg: tier = small|medium|high. Swaps an
# exclusive tier modifier + the qing_subsidy_tier var on scope:target (the march), which
# the subsidy pulse + army-sizer read. Bound to the Subject-tab S/M/L buttons (P6 GUI).
# ======================================================================
QING_march_set_subsidy = {
	LOG_enter = { sys = QING  fn = "QING_march_set_subsidy -> $tier$" }
	scope:target = {
		# strip any existing tier first (exclusive selection).
		if = { limit = { has_country_modifier = qing_subsidy_small }   remove_country_modifier = qing_subsidy_small }
		if = { limit = { has_country_modifier = qing_subsidy_medium }  remove_country_modifier = qing_subsidy_medium }
		if = { limit = { has_country_modifier = qing_subsidy_high }    remove_country_modifier = qing_subsidy_high }
		add_country_modifier = { name = qing_subsidy_$tier$  duration = -1 }
		set_variable = { name = qing_subsidy_tier  value = flag:$tier$ }
	}
	# resize the march's army to the new tier immediately (so the toggle has instant effect).
	QING_march_size_army = { march = scope:target }
	LOG_exit = { sys = QING  fn = "QING_march_set_subsidy"  result = OK }
}

# ======================================================================
# [#46] QING_march_apply_support_standard — set EVERY march to the empire-wide subsidy
# STANDARD chosen by the Protectorates-General SUPPORT law (qing_protectorate_support_law).
# Scope: ROOT = c:CHI (the law's on_enact runs in the enacting country's scope). Reads the
# law-backing var qing_march_support_standard (flag:small/medium/high) and re-tiers each
# frontier_protectorate through the proven QING_march_set_subsidy (which strips the old tier
# modifier, sets the new one + qing_subsidy_tier, and resizes the army). This is the ONE
# empire-wide lever; the per-march GUI toggle still overrides an individual march afterward.
# ======================================================================
QING_march_apply_support_standard = {
	LOG_enter = { sys = QING  fn = "QING_march_apply_support_standard" }
	if = {
		limit = { has_variable = qing_march_support_standard }
		every_subject = {
			limit = { is_subject_type = frontier_protectorate }
			save_scope_as = target
			# re-tier this march to the empire standard. QING_march_set_subsidy operates on the saved
			# scope:target (this march) and the absolute ROOT (=CHI) internally, so it runs correctly in
			# this subject scope; dispatch by the CHI-held standard flag (ROOT.var, read from here).
			if =      { limit = { ROOT.var:qing_march_support_standard = flag:small }   QING_march_set_subsidy = { tier = small } }
			else_if = { limit = { ROOT.var:qing_march_support_standard = flag:medium }  QING_march_set_subsidy = { tier = medium } }
			else_if = { limit = { ROOT.var:qing_march_support_standard = flag:high }    QING_march_set_subsidy = { tier = high } }
		}
	}
	LOG_exit = { sys = QING  fn = "QING_march_apply_support_standard"  result = OK }
}

# ======================================================================
# [#39] QING_march_evaluate — the Protectors-General lifecycle pulse. Scope: ROOT = c:CHI
# (called from QING_GOV_pulse, right after QING_amban_evaluate). Modelled on QING_amban_evaluate:
# iterate the marches (subject countries carrying qing_march=1 — the SAME non-recursive
# every_subject over CHI's DIRECT subjects the amban roster uses; it does NOT descend into the
# marches' princely-state sub-subjects), (re)score each march's Governor-General (its current_ruler,
# a Lifan appointee) for affinity, and roll AT MOST ONE mutually-exclusive lifecycle event per march
# per quarter by priority — turnover (.5) > overmighty (.4) > unrest (.2) > petition (.1) > commend
# (.3). All modal events share the court-event slot (qing_gc_event_slot_used, cleared each Qing pulse,
# the GC-event-throttle rule) so several marches in one pulse cannot dogpile the court. CHI player-only
# (guarded tag=CHI + not-AI via the pulse's own gate). Per-quarter fire chance is LOW (~10%, below the
# amban ~15%) so amban + march events rarely fire together, and the meter-fold (term h) is half-weight.
# ======================================================================
QING_march_evaluate = {
	LOG_enter = { sys = QING  fn = "QING_march_evaluate" }
	# CHI player-only — the whole Protectors-General arc is a player mechanic (like the amban arc).
	if = {
		limit = { tag = CHI  is_ai = no }
		every_subject = {
			limit = {
				has_variable = qing_march
				# the GG must exist and actually rule this march (set_as_ruler'd Lifan appointee).
				exists = current_ruler
				current_ruler = { has_variable = qing_march_gg }
			}
			save_scope_as = qing_march_here
			current_ruler = { save_scope_as = qing_march_gg_here }
			# (re)score the GG's affinity to the throne each pulse (as the amban evaluate does for
			# residents), so a GG whose loyalty/culture/stats drift is judged on current standing.
			scope:qing_march_gg_here = { QING_char_affinity = yes }
			# roll one event, mutually exclusive by priority; each guarded on its own condition AND
			# on the shared court-event slot being free (claim it on the FIRST fire this pulse).
			# SCOPE-LIFETIME NOTE: qing_march_here / qing_march_gg_here are re-saved every loop
			# iteration, so by the time a DEFERRED event (days 5-15) runs they would hold the LAST
			# march iterated, not the triggering one. So each fire branch re-saves DEDICATED event
			# scopes (qing_march_evt_march / qing_march_evt_gg) at the moment of firing; only ONE
			# branch fires per pulse (the slot claim), so a single pair of names is safe, and the
			# events themselves re-validate/re-pick defensively (per the amban-arc idiom).
			ROOT = {
				random = {
					# ~10% per quarter that this march surfaces a lifecycle beat at all (below the amban
					# rate; keeps the two arcs from firing together and the meter from over-swinging).
					chance = 10
					if = {
						# (.5) TURNOVER — the GG is old or failing; he must be replaced.
						limit = {
							NOT = { has_variable = qing_gc_event_slot_used }
							scope:qing_march_gg_here = { OR = { age >= 65  health < 0.4 } }
						}
						scope:qing_march_here = { save_scope_as = qing_march_evt_march }
						scope:qing_march_gg_here = { save_scope_as = qing_march_evt_gg }
						trigger_event = { id = qing_march.5  days = { 5 15 } }
						set_variable = { name = qing_gc_event_slot_used  value = 1 }
					}
					else_if = {
						# (.4) OVERMIGHTY — a long-tenured (tenure timer expired), estranged (affinity < 40) GG.
						limit = {
							NOT = { has_variable = qing_gc_event_slot_used }
							scope:qing_march_gg_here = {
								NOT = { has_variable = qing_march_gg_tenure_pending }
								has_variable = qing_char_affinity
								var:qing_char_affinity < 40
							}
						}
						scope:qing_march_here = { save_scope_as = qing_march_evt_march }
						scope:qing_march_gg_here = { save_scope_as = qing_march_evt_gg }
						trigger_event = { id = qing_march.4  days = { 5 15 } }
						set_variable = { name = qing_gc_event_slot_used  value = 1 }
					}
					else_if = {
						# (.2) UNREST — a rising among the march's own princely-state sub-subjects or its provinces.
						limit = {
							NOT = { has_variable = qing_gc_event_slot_used }
							scope:qing_march_here = {
								OR = {
									any_owned_province = { count >= 1  province_unrest >= 3 }
									any_subject = { any_owned_province = { count >= 1  province_unrest >= 3 } }
								}
							}
						}
						scope:qing_march_here = { save_scope_as = qing_march_evt_march }
						scope:qing_march_gg_here = { save_scope_as = qing_march_evt_gg }
						trigger_event = { id = qing_march.2  days = { 5 15 } }
						set_variable = { name = qing_gc_event_slot_used  value = 1 }
					}
					else_if = {
						# (.1) PETITION — the GG asks for more silver and/or troops. Gate: the march is
						# not yet at the top subsidy tier (so there is a tier to grant). A march with no
						# subsidy_tier var yet (founded before any support law) also qualifies — it reads
						# as "not high". Kept to the tier signal only; comparing live vs establishment
						# cohorts would need a march-scope scratch shortfall (RHS-operator rule bars a
						# var-vs-var compare) and the tier gate already captures "wants more support".
						limit = {
							NOT = { has_variable = qing_gc_event_slot_used }
							scope:qing_march_here = { NOT = { var:qing_subsidy_tier = flag:high } }
						}
						scope:qing_march_here = { save_scope_as = qing_march_evt_march }
						scope:qing_march_gg_here = { save_scope_as = qing_march_evt_gg }
						trigger_event = { id = qing_march.1  days = { 5 15 } }
						set_variable = { name = qing_gc_event_slot_used  value = 1 }
					}
					else_if = {
						# (.3) COMMEND — an able GG (affinity >= 70) and a prospering march (no unrest). Reward.
						limit = {
							NOT = { has_variable = qing_gc_event_slot_used }
							scope:qing_march_gg_here = { has_variable = qing_char_affinity  var:qing_char_affinity >= 70 }
							scope:qing_march_here = { NOT = { any_owned_province = { count >= 1  province_unrest >= 3 } } }
						}
						scope:qing_march_here = { save_scope_as = qing_march_evt_march }
						scope:qing_march_gg_here = { save_scope_as = qing_march_evt_gg }
						trigger_event = { id = qing_march.3  days = { 5 15 } }
						set_variable = { name = qing_gc_event_slot_used  value = 1 }
					}
				}
			}
		}
	}
	LOG_exit = { sys = QING  fn = "QING_march_evaluate"  result = OK }
}
