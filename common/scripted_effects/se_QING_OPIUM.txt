# LAYER, DON'T DUPLICATE. This suite owns NO new currency-stress meter and NO new band
# machinery. The mod already models the SYMPTOM — a `qing_currency_stress` 0..100 meter
# (se_QING_DECLINE.txt, term A) that — since meter-concretize #3 (2026-08-06) — is a DERIVED
# level: clamp(reserve-backing base + opium flow + event residual). #366 supplies the CAUSE for
# the OPIUM-FLOW term: a quarterly balance-of-trade assessment that computes a net silver flow
# (tea/silk/porcelain export INFLOW minus an opium-import OUTFLOW) and STORES it (signed) in
# qing_opium_stress_flow, which QING_DECLINE_recompute_currency_stress adds to the reserve base
# each pulse. The existing QING_DECLINE_apply_currency_band then bands the level for free; the
# existing qing_treaty.1 Nanjing-moment (gated qing_currency_stress >= 40) fires off it unchanged.
#
# THE GRAND-COUNCIL FOLD IS ALREADY WIRED. QING_ministry_recompute_perf_revenue
# (se_QING_MINISTRY.txt term d) subtracts qing_currency_stress/6 from the Board of Revenue
# minister's performance meter, which folds into his Grand Council standing. So the opium
# drain reaches the office-holder TRANSITIVELY: opium net-drain -> qing_currency_stress ->
# qing_min_perf_revenue (d) -> qing_council_eff_target. #366 adds NO new fold; it makes the
# existing silver-crisis-drag term reflect a real trade balance. The 戶部 owned the historical
# response to the 19thC silver crisis (銀荒/銀貴錢賤), so this is the correct office.
#
# PROBE-DRIVEN DESIGN (both oracle probes returned NOT-PROVEN; both take the low-risk path):
#   • cross-country trade-flow matrix DOES NOT EXIST -> opium-import intensity is a historical
#     TIME RAMP (imports grew exponentially 1763->1838 as British Bengal "country trade", a
#     process external to the Qing frontier spheres), MODULATED by treaty ports, player posture,
#     and the domestic-production toggle. Export inflow uses the PROVEN, REAL country svalues
#     GOODS_national_production_{tea,silk,porcelain}.
#   • per-pop variables DO NOT EXIST -> addiction is an AGGREGATE 0..100 counter
#     (qing_opium_addicted_share) + PROVINCE-MODIFIER bands (the proven corruption-band pattern).
#
# Calibrated to Dermigny/Hsü: 1763 net ~+19.5M taels (healthy), 1838 net ~+5M then negative
# (crisis); baseline inflow ~20M, peak drain ~10M (40k chests x ~250 taels). Lin Zexu 1839 =
# 20k chests; legalization = 1858 Tianjin; Yunnan/Sichuan domestic opium rises 1850s+.
#
#   ROOT / this = CHI (guarded). Prefix = QING_opium · sys tag = QING.
#   Counters (all 0..100 unless noted):
#     qing_opium_import_index      the historical opium-import intensity ramp (0=1763 trickle, 100=1838 flood)
#     qing_opium_net_flow_tmp      scratch: this-quarter net silver flow (signed, taels/M units)
#     qing_opium_addicted_share    aggregate addiction share of the population
#     qing_opium_posture           policy: 0 = prohibit (default), 1 = tolerate-for-revenue, 2 = treaty-legalized
#     qing_opium_domestic_unlocked flag var: Yunnan/Sichuan domestic opium production is on
#     qing_lin_zexu_appointed      flag var: an Imperial Commissioner is enforcing interdiction
#   [#97] the Commissioner is now a REAL character actor (欽差大臣 / Lin Zexu), following the
#   Hoppo/Caravan/Salt office template (se_QING_FRONTIER_PICKER.txt) — not just the bare flag above,
#   which STAYS (the interdiction-suppression logic elsewhere in this file reads it unchanged):
#     qing_opium_commissioner_holder   country var: the character serving as Commissioner, if any
#     qing_opium_commissioner_marker   character var: the per-char post marker (offices custom_loc)

# ---- generic clamped-counter nudge (mirror of QING_DECLINE_nudge, sys=QING) --
# Self-inits the counter to 0. Scope: country. $var$ = counter, $amount$ = signed int.
QING_opium_nudge = {
	if = {
		limit = { NOT = { has_variable = $var$ } }
		set_variable = { name = $var$  value = 0 }
	}
	change_variable = { name = $var$  add = $amount$ }
	if = { limit = { var:$var$ > 100 }  set_variable = { name = $var$  value = 100 } }
	if = { limit = { var:$var$ < 0 }    set_variable = { name = $var$  value = 0 } }
	LOG_line = { sys = QING  msg = "opium counter $var$ nudged by $amount$ ->" }
}

# ======================================================================
# INIT — seed the counters once on game start for CHI. High Qing (pre-1775) starts
# at a near-zero import index (opium imports were a negligible <1k chests/yr trickle);
# a post-1815 start seeds the index to a mid value (the drain already well underway).
# ======================================================================
QING_opium_init = {
	LOG_enter = { sys = QING  fn = "QING_opium_init" }
	if = { limit = { NOT = { has_variable = qing_opium_import_index } }
		if = {
			limit = { current_date < 1800.1.1 }
			set_variable = { name = qing_opium_import_index  value = 3 }    # 1763 trickle
		}
		else_if = {
			limit = { current_date < 1830.1.1 }
			set_variable = { name = qing_opium_import_index  value = 30 }   # ~1815 the drain is on
		}
		else = {
			set_variable = { name = qing_opium_import_index  value = 55 }   # a late start: crisis approaching
		}
	}
	if = { limit = { NOT = { has_variable = qing_opium_addicted_share } } set_variable = { name = qing_opium_addicted_share  value = 0 } }
	if = { limit = { NOT = { has_variable = qing_opium_posture } }        set_variable = { name = qing_opium_posture  value = 0 } }  # prohibit
	# [fix, task #63 2026-08-14, design/DESIGN_63_OPIUM_COMMISSIONER_REVENUE_SQUEEZE.md] the
	# Commissioner's squeeze meter -- NOT graft on a revenue stream (this office has none; it is
	# an interdiction post, not a revenue-farming one, per this file's own header). Squeeze here
	# represents UNDER-ENFORCEMENT: how far the appointee's own corruption blunts the -4 suppression
	# bonus below. Seeded at 30, matching Canton's Hoppo baseline exactly (the same "moderate but not
	# zero" starting point the other 3 offices use).
	if = { limit = { NOT = { has_variable = qing_opium_commissioner_squeeze } } set_variable = { name = qing_opium_commissioner_squeeze  value = 30 } }
	LOG_exit = { sys = QING  fn = "QING_opium_init"  result = OK }
}

# ======================================================================
# THE IMPORT-INDEX RAMP — the historical growth of opium imports toward the 1838 flood.
# The index climbs toward a date-keyed CEILING (the British Bengal "country trade" expanded
# on its own schedule, external to any Qing lever), then policy MODULATES the climb rate:
#   • prohibition with a Lin Zexu commissioner suppresses the climb (interdiction);
#   • tolerate-for-revenue lets it run;
#   • treaty legalization removes all restraint (the flood is now legal);
#   • domestic production substitutes for imports (the import index falls, but addiction
#     keeps rising — see the addiction pulse).
# Scope: country (CHI). O(1).
# ======================================================================
QING_opium_ramp_import_index = {
	QING_opium_init = yes
	# date-keyed ceiling: 1763 ~a trickle, 1800 rising, 1820 heavy, 1838+ the flood.
	set_variable = { name = qing_opium_ceiling_tmp  value = 5 }
	if = { limit = { current_date >= 1800.1.1 } set_variable = { name = qing_opium_ceiling_tmp  value = 25 } }
	if = { limit = { current_date >= 1815.1.1 } set_variable = { name = qing_opium_ceiling_tmp  value = 45 } }
	if = { limit = { current_date >= 1830.1.1 } set_variable = { name = qing_opium_ceiling_tmp  value = 75 } }
	if = { limit = { current_date >= 1838.1.1 } set_variable = { name = qing_opium_ceiling_tmp  value = 95 } }
	# treaty legalization lifts the ceiling to the rails (the trade is now unrestrained + legal).
	if = { limit = { var:qing_opium_posture = 2 } set_variable = { name = qing_opium_ceiling_tmp  value = 100 } }
	# climb rate toward the ceiling, modulated by posture. Default +3/quarter.
	set_variable = { name = qing_opium_climb_tmp  value = 3 }
	if = {
		# a Lin Zexu commissioner + active prohibition suppresses the climb hard (interdiction).
		limit = { var:qing_opium_posture = 0  has_variable = qing_lin_zexu_appointed }
		set_variable = { name = qing_opium_climb_tmp  value = -4 }
		# [fix, task #63 2026-08-14] attenuate the suppression CONTINUOUSLY toward 0 as the seated
		# Commissioner's own squeeze (under-enforcement) rises -- never a hard threshold/cliff (this
		# codebase already tried and retired that shape twice: se_QING_CANTON.txt #111a, and
		# DESIGN_59's own round 1). At squeeze=0 the full -4 suppression applies; at squeeze=100 the
		# suppression is fully blunted (0, the trade climbs at its natural rate); squeeze=50 -> -2.
		if = {
			limit = { has_variable = qing_opium_commissioner_squeeze }
			change_variable = { name = qing_opium_climb_tmp  multiply = { value = 100  subtract = var:qing_opium_commissioner_squeeze } }
			change_variable = { name = qing_opium_climb_tmp  divide = 100 }
		}
	}
	else_if = {
		# tolerate-for-revenue: the trade runs faster (customs winks at it).
		limit = { var:qing_opium_posture = 1 }
		set_variable = { name = qing_opium_climb_tmp  value = 5 }
	}
	else_if = {
		# treaty-legalized: fastest of all.
		limit = { var:qing_opium_posture = 2 }
		set_variable = { name = qing_opium_climb_tmp  value = 6 }
	}
	# domestic Yunnan/Sichuan production substitutes for IMPORTS -> pull the import index down
	# (home-grown opium is not a silver drain), even as the epidemic keeps spreading.
	if = { limit = { has_variable = qing_opium_domestic_unlocked } change_variable = { name = qing_opium_climb_tmp  subtract = 5 } }
	# drift toward the ceiling: climb only if below it, and never push past it on a positive climb.
	if = {
		limit = { var:qing_opium_import_index < qing_opium_ceiling_tmp_cmpsvalue }
		QING_opium_nudge = { var = qing_opium_import_index  amount = var:qing_opium_climb_tmp }
		if = {
			limit = { var:qing_opium_import_index > qing_opium_ceiling_tmp_cmpsvalue }
			set_variable = { name = qing_opium_import_index  value = var:qing_opium_ceiling_tmp }
		}
	}
	else = {
		# already at/above the ceiling: only a negative climb (interdiction/domestic-substitution) moves it.
		if = {
			limit = { var:qing_opium_climb_tmp < 0 }
			QING_opium_nudge = { var = qing_opium_import_index  amount = var:qing_opium_climb_tmp }
		}
	}
	remove_variable = qing_opium_ceiling_tmp
	remove_variable = qing_opium_climb_tmp
}

# ======================================================================
# THE TRADE-BALANCE ASSESSMENT — the quarterly heart of the model.
# Computes net_silver_flow = export_inflow - opium_outflow, then NUDGES the existing
# qing_currency_stress meter: a surplus bleeds stress DOWN (silver flowing in backs the
# currency), a drain pushes it UP. Called from QING_DECLINE_pulse BEFORE
# QING_DECLINE_update_currency_stress, so the trade-flow nudge and the reserve-ratio drift
# compound in the same pulse. Scope: country (CHI). O(governorships) via the national svalues.
# ======================================================================
QING_opium_assess_trade_balance = {
	LOG_enter = { sys = QING  fn = "QING_opium_assess_trade_balance" }
	QING_opium_init = yes
	# [#423] ramp the import index (still drives the addiction epidemic + the Lin Zexu / treaty
	# event chain + the opium panel read-out; it no longer sizes the silver outflow — that is now
	# the concrete DEMAND_country_opium - GOODS_national_production_opium gap below).
	QING_opium_ramp_import_index = yes

	# --- EXPORT INFLOW: the real tea/silk/porcelain export economy (proven country svalues).
	# Staged into a scratch var (the proven set_variable=value=<svalue> read). Divided down to
	# the model's small-integer scale (production numbers are large); the constant is a scale
	# factor, not a magic tael figure.
	set_variable = { name = qing_opium_inflow_tmp  value = 0 }
	change_variable = { name = qing_opium_inflow_tmp  add = GOODS_national_production_tea }
	change_variable = { name = qing_opium_inflow_tmp  add = GOODS_national_production_silk }
	change_variable = { name = qing_opium_inflow_tmp  add = GOODS_national_production_porcelain }
	# scale to a 0..~40 export-strength figure (High-Qing China is the world's workshop).
	change_variable = { name = qing_opium_inflow_tmp  divide = 4 }
	if = { limit = { var:qing_opium_inflow_tmp > 40 } set_variable = { name = qing_opium_inflow_tmp  value = 40 } }

	# --- OPIUM OUTFLOW: [#423] the CONCRETE import gap. The silver drain is the opium the
	# empire CONSUMES but does not GROW — i.e. what must be imported (and paid for in silver).
	# Bengal opium paid in silver drains; Yunnan/Sichuan home-grown opium does not. So:
	#     imported opium  =  national consumption  -  domestic production
	#                     =  DEMAND_country_opium   -  GOODS_national_production_opium
	# both proven country-scope aggregates (DEMAND_svalues.txt:2687 / GOODS_svalues.txt:195).
	# Read into a scratch var, floored at 0 (you cannot import a negative amount — a net
	# exporter simply has no drain), scaled into the model's 0..50 outflow band and clamped.
	# This REPLACES the old date-ramp (qing_opium_import_index) as the outflow driver — the
	# index var is KEPT (the Lin Zexu / treaty-posture event chain + the opium panel still read
	# it), it just no longer sizes the silver drain. Now a player who plants domestic poppy
	# (raising production via QING_opium_unlock_domestic + the poppy-farm building) concretely
	# shrinks the drain, and a richer, more-addicted empire (higher consumption) deepens it.
	# [#423 calibration] DIVISOR 8 brings the raw (demand-production) gap into the ~0..50 band;
	# this is a scale factor to tune against a live economy, not a tael figure. Clamp 0..50.
	set_variable = { name = qing_opium_outflow_tmp  value = DEMAND_country_opium }
	change_variable = { name = qing_opium_outflow_tmp  subtract = GOODS_national_production_opium }
	if = { limit = { var:qing_opium_outflow_tmp < 0 } set_variable = { name = qing_opium_outflow_tmp  value = 0 } }
	change_variable = { name = qing_opium_outflow_tmp  divide = 8 }
	if = { limit = { var:qing_opium_outflow_tmp > 50 } set_variable = { name = qing_opium_outflow_tmp  value = 50 } }
	# addiction amplifier: + up to a third of the addicted share (deeper habit, larger imports).
	set_variable = { name = qing_opium_addamp_tmp  value = var:qing_opium_addicted_share }
	change_variable = { name = qing_opium_addamp_tmp  divide = 3 }
	change_variable = { name = qing_opium_outflow_tmp  add = var:qing_opium_addamp_tmp }
	remove_variable = qing_opium_addamp_tmp

	# --- NET FLOW (signed): inflow - outflow. Positive = silver in, negative = silver drained.
	set_variable = { name = qing_opium_net_flow_tmp  value = var:qing_opium_inflow_tmp }
	change_variable = { name = qing_opium_net_flow_tmp  subtract = var:qing_opium_outflow_tmp }

	# --- FEED THE CURRENCY-STRESS METER as a CONCRETE FLOW TERM. A surplus bleeds stress down; a drain
	# pushes it up. [meter-concretize #3 2026-08-06] STORE the flow (signed) in qing_opium_stress_flow
	# instead of nudging the level directly — the two-store recompute (QING_DECLINE_recompute_currency_
	# stress) ADDS this stored flow to the reserve-derived base each pulse. This keeps the concrete
	# balance-of-trade computation (§3, preserved verbatim above) but makes it a STANDING term of the
	# derived level, not an accumulator: recomputed from scratch each pulse from the real trade goods, so
	# it can't integrate to the rails. flow = net/-5 (a +20 surplus => -4 stress; a -20 drain => +4).
	set_variable = { name = qing_opium_stress_flow  value = var:qing_opium_net_flow_tmp }
	change_variable = { name = qing_opium_stress_flow  divide = 5 }
	# invert: surplus (positive net) -> negative stress contribution (eases the famine).
	change_variable = { name = qing_opium_stress_flow  multiply = -1 }
	LOG_line = { sys = QING  msg = "opium: trade balance assessed; net silver flow stored as currency-stress flow term" }

	# [fix, task #63 2026-08-14, design/DESIGN_63_OPIUM_COMMISSIONER_REVENUE_SQUEEZE.md] TOLERATE-
	# FOR-REVENUE (弛禁徵稅) recurring customs take. qing_opium.1's tolerate option already grants a
	# one-shot add_treasury=20 on adopting posture=1 -- this generalizes that REAL, sourced mechanism
	# into a recurring quarterly credit instead of inventing a new one. MUST live here: qing_opium_
	# outflow_tmp is a pure scratch var, removed at the end of this same function every pulse, so no
	# other function could ever read it. Published unconditionally (even at 0) matching qing_caravan_
	# income_last's own precedent, so a future Revenue-perf-rollup fold can read it the same way.
	if = {
		limit = { var:qing_opium_posture = 1 }
		set_variable = { name = qing_opium_tolerate_revenue_tmp  value = var:qing_opium_outflow_tmp }
		change_variable = { name = qing_opium_tolerate_revenue_tmp  divide = 5 }   # [ASSUMPTION] scale factor, boot-tune
		set_variable = { name = qing_opium_income_last  value = var:qing_opium_tolerate_revenue_tmp }
		add_treasury = var:qing_opium_tolerate_revenue_tmp
		TREASURY_LOG_it = { amount = var:qing_opium_tolerate_revenue_tmp }
		remove_variable = qing_opium_tolerate_revenue_tmp
	}
	else = {
		set_variable = { name = qing_opium_income_last  value = 0 }
	}

	remove_variable = qing_opium_inflow_tmp
	remove_variable = qing_opium_outflow_tmp
	remove_variable = qing_opium_net_flow_tmp   # [PERF-AUDIT 2026-07-18 L9] was leaked into every save (its 3 sibling scratch vars are removed; this one wasn't)
	LOG_exit = { sys = QING  fn = "QING_opium_assess_trade_balance"  result = OK }
}

# ======================================================================
# THE ADDICTION PULSE — the aggregate epidemic. The addicted share climbs with the
# opium availability (import index OR domestic production), decays slowly if the supply is
# choked (Lin Zexu interdiction). Then bands the province-level social drag. Scope: country.
# ======================================================================
QING_opium_addiction_pulse = {
	LOG_enter = { sys = QING  fn = "QING_opium_addiction_pulse" }
	QING_opium_init = yes
	# supply pressure = the import index; domestic production adds a home-grown floor of supply.
	set_variable = { name = qing_opium_supply_tmp  value = var:qing_opium_import_index }
	if = { limit = { has_variable = qing_opium_domestic_unlocked } change_variable = { name = qing_opium_supply_tmp  add = 30 } }
	# addiction drifts toward supply/2 (a heavy but not total-population ceiling), climbing +1/quarter
	# while below, decaying -1 while above (habits fade when the drug is scarce).
	set_variable = { name = qing_opium_addtarget_tmp  value = var:qing_opium_supply_tmp }
	change_variable = { name = qing_opium_addtarget_tmp  divide = 2 }
	if = { limit = { var:qing_opium_addtarget_tmp > 100 } set_variable = { name = qing_opium_addtarget_tmp  value = 100 } }
	if = {
		limit = { var:qing_opium_addicted_share < qing_opium_addtarget_tmp_cmpsvalue }
		QING_opium_nudge = { var = qing_opium_addicted_share  amount = 1 }
	}
	else_if = {
		limit = { var:qing_opium_addicted_share > qing_opium_addtarget_tmp_cmpsvalue }
		QING_opium_nudge = { var = qing_opium_addicted_share  amount = -1 }
	}
	remove_variable = qing_opium_supply_tmp
	remove_variable = qing_opium_addtarget_tmp
	# --- ADDICTION FEEDS THE SECT PRESSURE (White Lotus / Taiping recruited among the addicted
	# and the silver-crisis destitute). A high epidemic nudges qing_sect_pressure up gently.
	if = {
		limit = { var:qing_opium_addicted_share >= 40 }
		QING_DECLINE_nudge = { var = qing_sect_pressure  amount = 1 }
	}
	# [MO-ENH #5] ADDICTION FEEDS MALTHUSIAN POP PRESSURE. Opium was a demographic multiplier —
	# silver drain + health/productivity loss thickened the involution the pop-pressure meter models
	# (se_QING_DECLINE.txt qing_pop_pressure). A severe epidemic (>=50) adds a gentle quarterly drip
	# to the pressure, so the addiction has a downstream Malthusian consequence beyond the silver flow.
	# Additive to the existing derives; guarded + clamped by QING_DECLINE_nudge. Historical: the SW/
	# Yangzi opium belts saw real productivity collapse feeding the mid-century population crises.
	if = {
		limit = {
			var:qing_opium_addicted_share >= 50
			has_variable = qing_pop_pressure
		}
		QING_DECLINE_nudge = { var = qing_pop_pressure  amount = 1 }
	}
	QING_opium_apply_epidemic_band = yes
	LOG_exit = { sys = QING  fn = "QING_opium_addiction_pulse"  result = OK }
}

# ---- province-modifier bands for the local social drag (proven corruption-band pattern) ----
# The aggregate share maps to a country-wide epidemic modifier (0 = none, low >=30, high >=60).
# Country-scope modifier keeps it O(1) — no per-province sweep needed (the drag is empire-wide
# productivity loss + unrest, which a country modifier models cleanly). Scope: country.
QING_opium_apply_epidemic_band = {
	remove_country_modifier = qing_opium_epidemic_low
	remove_country_modifier = qing_opium_epidemic_high
	if = {
		limit = { var:qing_opium_addicted_share >= 60 }
		add_country_modifier = { name = qing_opium_epidemic_high  duration = -1 }
		LOG_line = { sys = QING  msg = "opium: epidemic band = HIGH for (addicted share=)" }
	}
	else_if = {
		limit = { var:qing_opium_addicted_share >= 30 }
		add_country_modifier = { name = qing_opium_epidemic_low  duration = -1 }
		LOG_line = { sys = QING  msg = "opium: epidemic band = LOW for (addicted share=)" }
	}
}

# --- [cr97 fix] a dead Commissioner otherwise leaves qing_opium_commissioner_holder pointing at a
# corpse forever (the card's is_shown gates on has_variable, not is_alive; the ghost qing_lin_zexu_appointed
# would keep suppressing the import ramp with no one actually enforcing it). BACKFILL a fresh appointee
# immediately, matching the other 3 frontier offices' "the state keeps this post staffed" model
# (interdiction is meant to be an ongoing policy, not something that lapses the moment one commissioner
# dies). [fix, task #63 2026-08-14] the old "no double-book/engine-role check needed... this office
# carries no squeeze/graft meter" reasoning is now FALSE the moment the squeeze meter below ships --
# added the same double-book relief Salt/Canton/Caravan all carry, PLUS the squeeze mirror, sequenced
# (a) relief -> (b) backfill -> (c) mirror, matching their real ordering exactly (mirror strictly AFTER
# backfill, so a freshly-seated holder's corruption is mirrored the SAME pulse he's seated, not left
# stale for a quarter). Country scope (ROOT = CHI).
QING_opium_commissioner_reconcile = {
	# (a) double-book relief -- mirrors se_QING_SALT.txt's own relief block, adapted to Opium's holder/
	# marker var names. A commissioner who took a command/governorship, or was drawn into another
	# courtier post, loses this one.
	# [fix, task #2 2026-08-17] added has_variable = qing_office_held so a sitting Opium Commissioner
	# PROMOTED into a Grand Council seat also self-heals here. Opium is the ONE frontier post that never
	# calls QING_post_stamp at its seat effect, so it is not a qing_current_post family and gets no #118
	# chokepoint vacate when a new post (gc_office) is stamped on him (se_QING_COUNCIL.txt:1723). Without
	# this he would keep BOTH qing_opium_commissioner_office AND the seat's qing_officeholder -- post-#2
	# those are the salary (monthly_character_wealth), so +10/mo phantom in his wealth tooltip plus a
	# double quarterly treasury tally, and the opium post frozen. This relief strips the opium marker +
	# salary modifier + holder var, then (b) backfills a replacement -- healing all three within a pulse.
	if = {
		limit = {
			has_variable = qing_opium_commissioner_holder
			var:qing_opium_commissioner_holder = { is_alive = yes }
			var:qing_opium_commissioner_holder = {
				OR = {
					is_general = yes
					is_admiral = yes
					is_governor = yes
					has_variable = qing_officer_marker
					has_variable = qing_customs_ig_marker
					has_variable = qing_court_artist
					has_variable = qing_office_held
					# [1:1 vanilla-role audit 2026-08-24, review-fix] this relief block had every OTHER
					# vanilla disqualifier but was missing the Researcher/base-game-office pair the rest
					# of the 1:1 system now checks -- a seated commissioner given a vanilla office or made
					# a Researcher after appointment kept both titles indefinitely.
					has_tech_office = yes
					has_any_office = yes
					# [fix task #2 2026-08-29] self-correct a pre-gate/law-regressed sex-ineligible holder,
					# same as the Hoppo (se_QING_CANTON.txt).
					AND = {
						is_female = yes
						NOT = { is_close_relative = ROOT.current_ruler }
						NOT = {
							ROOT = {
								OR = {
									has_law = women_limited_legal_rights
									has_law = women_equal_legal_rights
									has_law = womens_suffrage_law
								}
							}
						}
					}
				}
			}
		}
		var:qing_opium_commissioner_holder = {
			if = { limit = { has_variable = qing_opium_commissioner_marker }  remove_variable = qing_opium_commissioner_marker }
			# [fix, task #101-follow-up 2026-08-15] strip the salary too, matching the marker strip.
			remove_character_modifier = qing_opium_commissioner_office
		}
		remove_variable = qing_opium_commissioner_holder
		LOG_line = { sys = QING  msg = "opium: relieved a double-booked commissioner (also a serving commander/officer/other courtier post)" }
	}
	# (b) backfill -- fires on a NATURAL DEATH (var exists, dead) OR after the relief above just
	# cleared the holder var entirely (var absent). The old limit only covered the first case, which
	# would have left the post PERMANENTLY vacant after any double-book relief (no other lever
	# recovers it) -- fixed with the same OR-absent clause Salt's own backfill uses.
	if = {
		limit = {
			OR = {
				NOT = { has_variable = qing_opium_commissioner_holder }
				var:qing_opium_commissioner_holder = { is_alive = no }
			}
		}
		remove_variable = qing_opium_commissioner_holder
		QING_opium_commissioner_appoint = yes
		LOG_line = { sys = QING  msg = "opium: the Commissioner post was vacant or died in office, a replacement was appointed" }
	}
	# (c) mirror + squeeze -- AFTER backfill, so a freshly-seated holder is mirrored the same pulse.
	if = {
		limit = { has_variable = qing_opium_commissioner_holder  var:qing_opium_commissioner_holder = { is_alive = yes } }
		var:qing_opium_commissioner_holder = { save_scope_as = qing_opium_commissioner_seated }
		set_variable = { name = qing_opium_commissioner_squeeze  value = scope:qing_opium_commissioner_seated.corruption }
	}
}

# --- [#97] seat THIS character (char scope) as the Commissioner: holder var + post marker +

# Called by the auto-pick appoint (below, for event/AI callers that cannot open a GUI) AND the

	save_scope_as = qing_opium_commissioner_pick

	# the one-shot guard below, the appoint-button is_shown, the qing_opium.2 Humen trigger — evaluates

# ======================================================================
# [#423/#432] QING_opium_spread_poppy_country — turn the empire's poppy country to opium.
# Scope: COUNTRY (CHI). HISTORICAL GEOGRAPHY (#432): Yunnan was the earliest and largest
# domestic poppy zone (by late Qing effectively the whole province grew it — Bello, Opium
# and the Limits of Empire), and Sichuan's Red Basin became the single biggest opium producer
# in all China by ~1900. So the WHOLE Yunnan + Sichuan (Chengdu) set is real poppy country —
# not a token 3. Cap 8 (leaving the food-capital + a few rice provinces be), preferring the
# most crowded marginal land (poppy displaced subsistence crops on the hills first). The crop
# filter is broadened — poppy displaced grain/veg/livestock/wood AND the fibre/fruit smallholdings —
# so the conversion is not artificially starved of candidates. Idempotent (skips existing opium).
# set_trade_goods + is_in_area proven (se_QING_COLON, se_QING_XINJIANG). Called at the opium
# panel's domestic-unlock. A Chinese-conquered Burma's Shan hills are turned to poppy separately
# by the #421 capstone (QING_burma_annex_and_plant_poppy) — the two sources follow the real map.
# ======================================================================
QING_opium_spread_poppy_country = {
	ordered_owned_province = {
		limit = {
			is_capital = no
			NOT = { trade_goods = opium }
			OR = { is_in_area = Yunnan  is_in_area = Chengdu }
			OR = { trade_goods = grain  trade_goods = livestock  trade_goods = wood  trade_goods = vegetables  trade_goods = textile_fibres  trade_goods = temperate_fruit  trade_goods = tropical_fruit }
		}
		order_by = total_population
		max = 8
		set_trade_goods = opium
		LOG_line = { sys = QING  msg = "opium: a Yunnan/Sichuan hill province turned to poppy (domestic cultivation)" }
	}
}

# ======================================================================
# THE PULSE — one throttled call from QING_DECLINE_pulse. The trade balance is assessed
# separately (called earlier in the decline pulse, before update_currency_stress); this
# runs the addiction epidemic + offers the opium event chain. Scope: country (CHI).
# ======================================================================
QING_opium_pulse = {
	LOG_enter = { sys = QING  fn = "QING_opium_pulse" }
	QING_opium_commissioner_reconcile = yes
	QING_opium_addiction_pulse = yes
	# --- offer the pre-crisis trade-balance WARNING once, when the drain first bites but the
	# treaty order has not yet been forced (the memorial that a wise court heeds). The event
	# re-checks its own state; a modest chance keeps it eventful, not spammy.
	if = {
		limit = {
			var:qing_opium_import_index >= 40
			var:qing_opium_posture = 0
			NOT = { has_variable = qing_lin_zexu_appointed }
			NOT = { has_variable = qing_opium_warning_seen }
			is_ai = no
		}
		# [#366-R] the once-only flag is set in qing_opium.1's own immediate (only when it
		# actually fires), so a re-check failure during the delay never strands it set.
		trigger_event = { id = qing_opium.1  days = { 5 20 } }
	}
	# --- offer the Lin Zexu crackdown decision when the drain is severe + a commissioner is in
	# place under prohibition (the 1839 Guangzhou moment). Player-only; event re-checks state.
	else_if = {
		limit = {
			var:qing_opium_import_index >= 60
			var:qing_opium_posture = 0
			has_variable = qing_lin_zexu_appointed
			NOT = { has_variable = qing_opium_crackdown_resolved }
			is_ai = no
		}
		random = {
			chance = 40
			trigger_event = { id = qing_opium.2  days = { 5 20 } }
		}
	}
	# --- the FORCED treaty-legalization demand: once the unequal-treaty order is imposed and opium
	# is not yet legal, the victors demand the tariffed trade (the 1858 Tianjin settlement). Offered
	# once; the event re-checks its own state. A separate offer (not part of the prohibition chain).
	if = {
		limit = {
			has_variable = qing_treaty_system_imposed
			NOT = { var:qing_opium_posture = 2 }
			NOT = { has_variable = qing_opium_legalize_offered }
			is_ai = no
		}
		# [#366-R] the once-only flag is set in qing_opium.3's own immediate (see .1 note).
		trigger_event = { id = qing_opium.3  days = { 5 20 } }
	}
	# --- the EPIDEMIC crisis: when the addicted share is severe, the throne faces the social ruin
	# itself. Offered once; the event re-checks its own state.
	if = {
		limit = {
			var:qing_opium_addicted_share >= 55
			NOT = { has_variable = qing_opium_epidemic_offered }
			is_ai = no
		}
		set_variable = { name = qing_opium_epidemic_offered  value = 1 }
		trigger_event = { id = qing_opium.4  days = { 5 20 } }
	}
	LOG_exit = { sys = QING  fn = "QING_opium_pulse"  result = OK }
}
