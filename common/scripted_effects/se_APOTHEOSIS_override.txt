# [#203 BT trade-route-log follow-up] Override the vanilla naval_apotheosis_effect.
#
# The base-game effect (common/scripted_effects/00_apotheosis.txt, called by many deity files
# as `naval_apotheosis_effect = yes`, e.g. common/deities/01_hinduism.txt:148) does
# `add_subunit = octere` x2 into a random navy. imp19c is a Victorian TC that DELETED the ancient
# galley subunits (navy_8_octere.txt etc. are now empty 3-byte stubs), so `octere` is not in the
# subunit database and the engine logged 2x "add_subunit: target: Value 'octere' is invalid: Not
# found in database class CSubUnitDefinitionDataBase" whenever a naval-religion ruler was deified.
#
# This redefinition loads after base-game and wins. Body is the vanilla effect VERBATIM except
# octere -> screw_frigate (the modern heavy warship, matching the #185 culture-navy mapping
# tetrere->screw_frigate). Keeps the same "reinforce a random fleet on apotheosis" behaviour with
# a real, TC-appropriate hull.
naval_apotheosis_effect = {
	if = {
		limit = { exists = scope:deified_ruler }
		if = {
			limit = {
				any_navy = {
					unit_size > 0
				}
			}
			custom_tooltip = naval_apotheosis_effect_ships
			random_navy = {
				add_subunit = screw_frigate
				add_subunit = screw_frigate
			}
		}
		else = {
			custom_tooltip = apotheosis_military_experience_text
			add_military_experience = {
				value = scope:deified_ruler.martial
				multiply = 0.5
				min = 2
				max = 5
			}
		}
	}
}
