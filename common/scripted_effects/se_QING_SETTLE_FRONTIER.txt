# se_QING_SETTLE_FRONTIER.txt — the scripted half of SETTLE THE NOMADS
# (定牧墾邊, the Qing consolidation of the Mongol and Manchurian frontier),
# #451.
#
# The mission tree (common/missions/qing_settle_frontier_missions.txt) makes the
# PLAYER do the real work: task goals gate on the Qing actually HOLDING ground in
# Mongolia / Liaoning / the Far East (reached by integrating the autonomous military
# governorships — ULS/KBD/MGA in the Mongol lands, MNC/MKD/HLJ in Manchuria — that
# hold those regions as CHI subjects in 1763), and on the Tannu tributary tie. This
# file scripts only what the player cannot express through a control goal:
#
#   • THE RECLAMATION (墾荒) — the settlement of the steppe and the taiga: military-
#     agricultural colonies (屯田), fixed pasture and broken sod. Stamped as the
#     qing_frontier_reclamation province modifier on the Qing-held frontier provinces
#     when a settlement task completes (a food + tax texture). Finite duration, so it
#     lapses as the colonists take root; re-stampable, guarded against double-stamp.
#   • EMBRACE TANNU — take TNN (Tannu Tuva / Uriankhai) firmly under the wing as a
#     sinosphere tributary if it has slipped the leash. FUNC_make_subject, guarded.
#
# PREFIX: QING_settle_ / sys = QING. CHI-only, driven by the mission tasks (no pulse).
# Proven idioms only: add_province_modifier { duration } (self-expiring, se_QING_BURMA /
# se_QING_OPEN_JAPAN), FUNC_make_subject (se_FUNC / himalaya_seasia tree), guarded if.
#
# NO BOM (common/scripted_effects/*.txt convention here).
#
#   ROOT / this = the Qing country (CHI, the player).
########################################################################

# ======================================================================
# QING_settle_reclaim_mongolia — lay the reclamation texture (military-agricultural colonies)
# over the Qing-held ground of the Mongol lands. Scope: COUNTRY (CHI). Stamps every province
# the Qing OWNS in the Mongolia region (guarded against double-stamp); finite duration so it
# lapses as the colonists take root (the burma-miasma self-expiring idiom). Called by the
# Mongol-settlement task.
# ======================================================================
QING_settle_reclaim_mongolia = {
	LOG_enter = { sys = QING  fn = "QING_settle_reclaim_mongolia" }
	every_owned_province = {
		limit = {
			is_in_region = Mongolia
			NOT = { has_province_modifier = qing_frontier_reclamation }
		}
		add_province_modifier = { name = qing_frontier_reclamation  duration = 3650 }
	}
	LOG_line = { sys = QING  msg = "settle-frontier: reclamation (墾荒) laid on the Mongol lands for [ROOT.GetTag]" }
	LOG_exit = { sys = QING  fn = "QING_settle_reclaim_mongolia"  result = OK }
}

# ======================================================================
# QING_settle_reclaim_manchuria — lay the reclamation texture over the Qing-held ground of the
# Manchurian homeland (Liaoning + the Far East). Scope: COUNTRY (CHI). Guarded, self-expiring.
# Called by the Manchurian-settlement task.

QING_settle_reclaim_manchuria = {

			NOT = { has_province_modifier = qing_frontier_reclamation }
		}
		add_province_modifier = { name = qing_frontier_reclamation  duration = 3650 }
	}
	LOG_line = { sys = QING  msg = "settle-frontier: reclamation (墾荒) laid on the Manchurian homeland for [ROOT.GetTag]" }
	LOG_exit = { sys = QING  fn = "QING_settle_reclaim_manchuria"  result = OK }

# ======================================================================
# QING_settle_embrace_tannu — take Tannu Tuva (TNN, the Uriankhai) firmly under the wing as a
# sinosphere tributary if it has slipped the leash. Scope: COUNTRY (CHI). TNN starts a CHI
# tributary in 1763, so this is only a safety net (guarded on NOT-already-subject) for a game
# in which it has broken away. The make-subject-by-fiat idiom (FUNC_make_subject).
# ======================================================================
QING_settle_embrace_tannu = {
	LOG_enter = { sys = QING  fn = "QING_settle_embrace_tannu" }
	if = {
		limit = { exists = c:TNN  NOT = { c:TNN = { is_subject_of = ROOT } } }
		FUNC_make_subject = { overlord = ROOT  target = c:TNN  type = sinosphere_tributary }
		LOG_line = { sys = QING  msg = "settle-frontier: Tannu Tuva (唐努烏梁海) brought back under the wing for [ROOT.GetTag]" }
	}
	LOG_exit = { sys = QING  fn = "QING_settle_embrace_tannu"  result = OK }
}

# ======================================================================
# [P1 2026-07-24] QING_settle_forge_nation — the nationalism half of "settle the frontier":
# settling the land is also FORGING THE NATION out of its peoples. Turns the Design-4
# nationalism hooks — which existed but were untouched by any mission — into the payoff of the
# frontier arcs. Scope: COUNTRY (CHI). $group$ = a culture-group key (mongolic / jurchen) whose
# frontier peoples this arc has brought under direct rule.
#
# Two mechanisms, chosen deliberately over a bare civic-identity nudge:
#   (1) NATURALISE the frontier culture group — integrate_country_culture on every un-integrated
#       country_culture of the group actually present in the realm. This is PERMANENT engine state
#       (a culture, once integrated, stays integrated): it survives the decline pulse's restoring
#       drift, where a one-shot QING_DECLINE_nudge on qing_civic_identity would simply decay back
#       toward its self-strengthening-derived target (the no-restoring-drift ratchet rule — a nudge
#       on a target-seeking meter is transient). This is exactly the civic-assimilation bridge's
#       proven idiom (QING_DECLINE_civic_assimilation): any_country_culture{ is_integrated=no
#       is_culture=scope } enumerates only cultures that EXIST in the realm, so a displaced/absent
#       frontier people is correctly skipped.
#   (2) LIFT THE CIVIC-IDENTITY TARGET, not the meter — qing_civic_identity_settle_bonus is added
#       into the target at the drift site so the gain PERSISTS (see the drift rework in
#       se_QING_DECLINE.txt). +8 per arc; the meter then drifts up to meet the raised ceiling.
# ======================================================================
QING_settle_forge_nation = {
	LOG_enter = { sys = QING  fn = "QING_settle_forge_nation" }
	if = { limit = { NOT = { has_variable = qing_civic_identity_settle_bonus } }  set_variable = { name = qing_civic_identity_settle_bonus  value = 0 } }
	# naturalise every un-integrated culture of the arc's group that is actually present.
	set_variable = { name = qing_settle_naturalise_done  value = 0 }
	every_country_culture = {
		limit = {
			is_integrated = no
			# proven country_culture -> culture -> culture_group path (se_CULTURE.txt uses
			# scope.culture; a culture scope carries .culture_group). $group$ = mongolic / jurchen.
			culture.culture_group = culture_group:$group$
		}
		save_scope_as = qing_settle_nat_c
		ROOT = {
			integrate_country_culture = scope:qing_settle_nat_c
			change_variable = { name = qing_settle_naturalise_done  add = 1 }
		}
		LOG_line = { sys = QING  msg = "settle-frontier: naturalised a $group$ frontier culture for" }
	}
	# raise the civic-identity CEILING (target), not the meter — a persisting nation-building gain.
	change_variable = { name = qing_civic_identity_settle_bonus  add = 8 }
	if = { limit = { var:qing_civic_identity_settle_bonus > 40 }  set_variable = { name = qing_civic_identity_settle_bonus  value = 40 } }
	if = {
		limit = { var:qing_settle_naturalise_done = 0 }
		LOG_fail = { sys = QING  fn = "QING_settle_forge_nation"  reason = "no un-integrated $group$ culture present (all already citizens / none in realm) - civic ceiling still raised" }
	}
	remove_variable = qing_settle_naturalise_done
	LOG_exit = { sys = QING  fn = "QING_settle_forge_nation"  result = OK }
}

# ======================================================================
# [P2 2026-07-24] QING_settle_plant_works — plant the industrial/garrison texture on the
# directly-held frontier ground of a region: a frontier colony (屯田) + fort (駐防) on the Qing-
# owned provinces of the region that do not yet carry them. Scope: COUNTRY (CHI). $region$ = a
# region key. These buildings carry ONLY LOCAL modifiers (no country-modifier upkeep sweep), so
# planting them directly on directly-ruled ground is safe and needs no subject wrapper — unlike
# QING_fbuild_subject_works, which is owner=$subject$-scoped and does NOT fit direct-rule ground.
# add_building_level is the proven province verb (colonization missions grant these same two
# buildings the identical way). Guarded against double-plant; capped at 4 provinces per call via a
# REAL engine cap (ordered_owned_province max=4), NOT a counter-in-limit — an every_ limit is
# evaluated against a candidate list built ONCE up front and cannot see a mid-iteration counter
# update (the documented rule at se_QING_DECLINE.txt:610-611, QING_DECLINE_civic_assimilation),
# so a counter-in-limit would blanket the whole region. order_by total_population = plant on the
# biggest settled centres first.
# ======================================================================
QING_settle_plant_works = {
	LOG_enter = { sys = QING  fn = "QING_settle_plant_works" }
	ordered_owned_province = {
		limit = {
			is_in_region = $region$
			is_sea = no
			OR = {
				NOT = { has_building = qing_frontier_colony_building }
				NOT = { has_building = qing_frontier_fort_building }
			}
		}
		order_by = total_population
		max = 4
		if = { limit = { NOT = { has_building = qing_frontier_colony_building } }  add_building_level = qing_frontier_colony_building }
		if = { limit = { NOT = { has_building = qing_frontier_fort_building } }    add_building_level = qing_frontier_fort_building }
	}
	LOG_line = { sys = QING  msg = "settle-frontier: planted colony/fort works on the held $region$ ground for" }
	LOG_exit = { sys = QING  fn = "QING_settle_plant_works"  result = OK }
}
