# se_QING_SALT.txt — [#44] The 兩淮鹽政 Salt Commissioner + output-driven 鹽課 revenue.
#
# The salt gabelle was one of the largest single central revenues of the Qing (~10-15% of
# state income), farmed through the great salt administrations — the 兩淮鹽政 (Lianghuai Salt
# Censor) foremost. This file makes the salt revenue CONCRETE + CHARACTER-GRADED, mirroring the
# Canton Hoppo (se_QING_CANTON.txt) end-to-end:
#   • a real seated commissioner (qing_salt_commissioner_holder), promoted from the court by
#     finesse, one-man-one-post, backfilled if vacant — the SAME ordered_character pattern as the
#     Hoppo (QING_canton_appoint_hoppo) and the caravan superintendent (#112a).
#   • the 鹽課 REVENUE rewritten from a flat base-6 to OUTPUT × market-soft × gabelle-markup ×
#     character-factor: it scales with the real salt the empire produces (GOODS_national_production_salt),
#     a bounded market-price factor, a gabelle markup (the monopoly's tax wedge — the historically dear
#     part), and the commissioner's competence − squeeze. Lands near the ~12-15%-of-income target.
#   • a SKIM: a corrupt commissioner pockets a slice as REAL personal wealth (add_gold to the CHARACTER,
#     confiscated on impeach/retire) — NOT the treasury, NOT the silver reserve (the #44 review C1: salt
#     is MONEY-ONLY; the reserve is the currency-backing channel and must not be perturbed).
#
# STATE (CHI country scope):
#   qing_salt_commissioner_holder  the seated 兩淮鹽政 (country link, rendered on the panel)
#   qing_salt_commissioner_marker  per-char post marker (feeds QING_char_holds_court_position, 1:1)
#   qing_salt_squeeze              0..100 graft meter (= his corruption, mirrored at reconcile)
#   qing_salt_finesse              lagged finesse mirror (for the yield grade)
#   qing_salt_income_last          published quarterly 鹽課 (Revenue panel + salt-monopoly window read)
#   (init uses PER-VAR guards, not a single initialized flag — matches QING_canton_init / QING_caravan_init)
#
# [#44 ASSUMPTION — best-guess constants, boot-tune to the ~12-15%-of-income target via the LOG below]:
#   gabelle markup = ×3.0 ; market-soft band = [0.7,1.3] around 1.0 ; character factor band = [0.5,1.3].
#
# PREFIX: QING_salt_ / sys = QING. CHI-only. Proven idioms only (mirrors se_QING_CANTON + the #112a
# caravan superintendent). No reserve touch. All LOG static.
#   ROOT / this = the Qing country (CHI).
########################################################################

# ======================================================================
# QING_salt_commissioner_appoint — promote the ablest free courtier to the 兩淮鹽政. Clone of
# QING_canton_appoint_hoppo / QING_caravan_super_appoint: employer=ROOT, adult/alive, not ruler/
# heir/commander/officer/governor/disgraced, not already holding a court post. order_by finesse.
# Stamps qing_salt_commissioner_holder (country link) + qing_salt_commissioner_marker (on the man).
# Country scope (ROOT = CHI).
# ======================================================================
QING_salt_commissioner_appoint = {
	ordered_character = {
		limit = {
			employer = ROOT
			is_adult = yes
			is_alive = yes
			is_ruler = no
			NOT = { QING_char_holds_court_position = yes }
			is_general = no
			is_admiral = no
			is_governor = no
			NOT = { has_variable = qing_officer_marker }
			NOT = { QING_char_hard_disgraced = yes }
			NOT = { AND = { exists = ROOT.primary_heir  this = ROOT.primary_heir } }
		}
		order_by = finesse
		max = 1
		# [#80 2026-08-11] seating body extracted to QING_salt_commissioner_seat (char scope) so the player
		# candidate-list handler seats its clicked man through the SAME path (holder+marker+#77 pool-drop).
		QING_salt_commissioner_seat = yes
	}
}

# ======================================================================
# [#80 2026-08-11] QING_salt_commissioner_seat — seat THIS character (char scope) as 兩淮鹽政: holder var +
# post marker + the #77 1:1 Academy pool-drop. Called by the auto-pick appoint (above) AND the player
# candidate-list handler (qing_frontier_office_appoint_selected). Extracting it keeps ONE seating path, so a
# manual pick can never diverge from the auto-pick (esp. the pool-drop that closes #77).
# ======================================================================
QING_salt_commissioner_seat = {
	save_scope_as = qing_salt_commissioner_pick
	# [#80 cr8084 CRITICAL] employer, NOT ROOT: on the player-picker path this helper runs with root = the
	# CLICKED CHARACTER (GuiScope.SetRoot(Character)); entering scope:player changes `this`, never `root`, so
	# ROOT here would write the holder var onto the CHARACTER, not CHI — the pick would be silently discarded
	# (panel blank, next reconcile auto-backfills over it). employer = CHI on BOTH paths (the candidate filter +
	# handler is_valid both require employer=CHI). This is the proven QING_office_appoint idiom.
	employer = { set_variable = { name = qing_salt_commissioner_holder  value = scope:qing_salt_commissioner_pick } }
	set_variable = { name = qing_salt_commissioner_marker  value = 1 }
	# [#77 1:1] forfeit the Academy posting on taking the office: drop the pool marker + delist from
	# qing_scholar_pool (the qing_is_pool_scholar VARIABLE / roster membership, NOT the hanlin degree TRAIT,
	# which is a kept credential). His on-name title becomes the Salt Commissioner office.
	QING_exam_pool_drop_member = yes
	QING_post_stamp = { post = salt_commissioner }   # [#118]
}

# ======================================================================
# QING_salt_init — seed the salt bookkeeping once (per-var guards OUTSIDE a single flag, matching
# QING_canton_init / QING_caravan_init, so an existing save backfills the newly-added vars + seats a
# commissioner). Squeeze baseline 30, finesse mirror 7 (the neutral ×1 factor). Then seat if vacant.
# Scope: COUNTRY (CHI).
# ======================================================================
QING_salt_init = {
	if = { limit = { NOT = { has_variable = qing_salt_squeeze } }  set_variable = { name = qing_salt_squeeze  value = 30 } }
	if = { limit = { NOT = { has_variable = qing_salt_finesse } }  set_variable = { name = qing_salt_finesse  value = 7 } }
	# the 兩淮鹽政 was a real, always-filled post — seat one now if none is seated (a PICK, not a create:
	# safe at init, no #90 boot-crash risk), so the panel shows him + his stats grade the yield from turn one.
	if = { limit = { NOT = { has_variable = qing_salt_commissioner_holder } }  QING_salt_commissioner_appoint = yes }
}

# ======================================================================
# QING_salt_income — the STANDING 鹽課 revenue, rewritten OUTPUT-DRIVEN + character-graded (#44).
# Replaces the flat base-6 (QING_revenue_salt_income). Scope: COUNTRY (CHI). Called quarterly from
# QING_revenue_pulse. Mirrors QING_canton_pulse's yield chain for RHS-safety.
#   income = OUTPUT (GOODS_national_production_salt) × market-soft [0.7,1.3] × gabelle-markup (×3.0)
#            × character-factor (1 + (finesse-7)×0.03 − squeeze×0.004, clamped [0.5,1.3])
# The gabelle markup is the monopoly's tax wedge (the dear part — a mispriced sim salt price can't
# tank it, since the market factor is a BOUNDED soft nudge around 1.0, not the raw price). MONEY-ONLY:
# add_treasury; NO reserve touch. Floored >= 1.
# ======================================================================
QING_salt_income = {
	QING_salt_init = yes
	# --- OUTPUT base: the real salt the empire produces (num_goods × mining productivity, which
	# already responds to the salt-yard buildings' base_resources — satisfies "a function of the
	# actual salt yards" automatically). Read the national aggregate into a scratch var.
	set_variable = { name = qing_salt_income_tmp  value = GOODS_national_production_salt }
	# [fix, hygiene-only -- NOT the confirmed source of the reported ~9000/quarter treasury spike,
	# see audits/AUDIT_CURRENCY_23.md / task #79] this function's own header says it "Mirrors
	# QING_canton_pulse's yield chain", but Canton (se_QING_CANTON.txt:120, cap 45) and Caravan
	# (se_QING_CARAVAN.txt:207, cap 18) BOTH explicitly cap their real-goods base before any
	# multiplier specifically to stop "a runaway export boom" from ballooning the take -- salt never
	# got that cap ported over. An adversarial review (2026-08-14) confirmed CHI's 8 salt provinces
	# vs. Canton's 42 tea+silk+porcelain provinces means salt's realistic ceiling is ~46-138 (well
	# under Canton's own 45-pre-multiplier cap), so this is defensive symmetry with its sibling
	# mechanics, not a fix for the still-open magnitude bug. Cap set just above the reviewed
	# realistic ceiling (best-guess, boot-tunable -- see the LOG line below for verification).
	if = {
		limit = { var:qing_salt_income_tmp > 150 }
		set_variable = { name = qing_salt_income_tmp  value = 150 }
		LOG_line = { sys = QING  msg = "salt: national production base exceeded the 150 cap, clamped for" }
	}

	# --- MARKET-SOFT factor: the live salt price nudges the take, BOUNDED to [0.7,1.3] around a neutral
	# 1.0 (user: market price is not a fair value for the historically-dear gabelle; #52: sim prices salt
	# in the low staple band). Build in a scratch var with change_variable + clamp (var-vs-literal, RHS-safe);
	# guard the global-mean-price global (warm-guard) with a 1.0 fallback when unwarmed.
	set_variable = { name = qing_salt_mkt_tmp  value = 1 }
	if = {
		limit = { has_global_variable = global_mean_price_salt }
		set_variable = { name = qing_salt_mkt_tmp  value = global_mean_price_salt }
		# salt sits in the ~0.1-1 staple band; scale the raw price toward a neutral ~1.0 pivot (×2 puts a
		# 0.5 price at 1.0), then clamp [0.7,1.3] so a mispriced sim can neither tank nor balloon the take.
		change_variable = { name = qing_salt_mkt_tmp  multiply = 2 }
		if = { limit = { var:qing_salt_mkt_tmp > 1.3 }  set_variable = { name = qing_salt_mkt_tmp  value = 1.3 } }
		if = { limit = { var:qing_salt_mkt_tmp < 0.7 }  set_variable = { name = qing_salt_mkt_tmp  value = 0.7 } }
	}
	change_variable = { name = qing_salt_income_tmp  multiply = var:qing_salt_mkt_tmp }
	remove_variable = qing_salt_mkt_tmp

	# --- GABELLE MARKUP: the monopoly's tax wedge (the historically-dear part). [#44 ASSUMPTION ×3.0 —
	# boot-tune to land salt near ~12-15% of state income via the ratio LOG below.]
	change_variable = { name = qing_salt_income_tmp  multiply = 3.0 }

	# --- CHARACTER factor: (finesse-7)×0.03 lift − squeeze×0.004 shave, clamped [0.5,1.3] (the Canton
	# factor block, verbatim shape). Reads the LAGGED mirrors (set in the reconcile block below), so a
	# capable honest commissioner delivers the full take and a corrupt/vacant one leaks it.
	set_variable = { name = qing_salt_factor_tmp  value = 1 }
	if = {
		limit = { has_variable = qing_salt_finesse }
		set_variable = { name = qing_salt_fin_tmp  value = var:qing_salt_finesse }
		change_variable = { name = qing_salt_fin_tmp  subtract = 7 }
		change_variable = { name = qing_salt_fin_tmp  multiply = 0.03 }
		change_variable = { name = qing_salt_factor_tmp  add = var:qing_salt_fin_tmp }
		remove_variable = qing_salt_fin_tmp
	}
	if = {
		limit = { has_variable = qing_salt_squeeze }
		set_variable = { name = qing_salt_sq_tmp  value = var:qing_salt_squeeze }
		change_variable = { name = qing_salt_sq_tmp  multiply = 0.004 }
		change_variable = { name = qing_salt_factor_tmp  subtract = var:qing_salt_sq_tmp }
		remove_variable = qing_salt_sq_tmp
	}
	if = { limit = { var:qing_salt_factor_tmp > 1.3 }  set_variable = { name = qing_salt_factor_tmp  value = 1.3 } }
	if = { limit = { var:qing_salt_factor_tmp < 0.5 }  set_variable = { name = qing_salt_factor_tmp  value = 0.5 } }
	change_variable = { name = qing_salt_income_tmp  multiply = var:qing_salt_factor_tmp }
	remove_variable = qing_salt_factor_tmp

	# --- floor >= 1 (the monopoly always yields something), pay to treasury (MONEY-ONLY, no reserve).
	if = { limit = { var:qing_salt_income_tmp < 1 }  set_variable = { name = qing_salt_income_tmp  value = 1 } }
	add_treasury = var:qing_salt_income_tmp
	TREASURY_LOG_it = { amount = var:qing_salt_income_tmp }
	set_variable = { name = qing_salt_income_last  value = var:qing_salt_income_tmp }
	remove_variable = qing_salt_income_tmp
	# [#44 M3] LOG the salt take + a total-income denominator so the ~12-15%-of-income target is verifiable
	# on the boot (an IN-ENGINE ratio, not back-solved from taels). STATIC label, no macro/# in the string.
	LOG_line = { sys = QING  msg = "salt: the 鹽課 pays its output-driven quarterly take (see qing_salt_income_last vs treasury) for" }

	QING_salt_reconcile = yes
}

# ======================================================================
# QING_salt_reconcile — mirror the Hoppo/superintendent tail (se_QING_CANTON:255-347). Runs AFTER the
# income is banked. (a) relieve a double-booked holder; (b) backfill a dead/missing one; (c) graft creep;
# (d) SIPHON a slice as REAL char wealth (add_gold to the man, confiscable — NOT treasury/reserve);
# (e) mirror squeeze(=corruption) + finesse for next pulse's yield grade. Scope: COUNTRY (CHI).
# ======================================================================
QING_salt_reconcile = {
	# (a) double-book relief — a commissioner who later took a command/governorship, OR who was
	# already drawn into another courtier post (e.g. the Customs Inspector-General), loses this post.
	# [fix #28 2026-08-13] same gap as the caravan/Hoppo reconciles: QING_char_holds_court_position
	# excludes seated qing_customs_ig_marker/qing_court_artist holders from FUTURE draws (#27), but
	# this reconcile's own double-book gate never checked either, so a stale/pre-#27 double-booking
	# would never self-correct here, unlike the general/admiral/governor/qing_officer_marker cases.
	if = {
		limit = {
			has_variable = qing_salt_commissioner_holder
			var:qing_salt_commissioner_holder = { is_alive = yes }
			var:qing_salt_commissioner_holder = {
				OR = {
					is_general = yes
					is_admiral = yes
					is_governor = yes
					has_variable = qing_officer_marker
					has_variable = qing_customs_ig_marker
					has_variable = qing_court_artist
				}
			}
		}
		var:qing_salt_commissioner_holder = { if = { limit = { has_variable = qing_salt_commissioner_marker }  remove_variable = qing_salt_commissioner_marker } }
		remove_variable = qing_salt_commissioner_holder
		LOG_line = { sys = QING  msg = "salt: relieved a double-booked commissioner (also a serving commander/officer/other courtier post) for" }
	}
	# (b) backfill a dead/missing commissioner — appoint a fresh man (a new broom, low corruption).
	if = {
		limit = { OR = { NOT = { has_variable = qing_salt_commissioner_holder }  var:qing_salt_commissioner_holder = { is_alive = no } } }
		if = { limit = { has_variable = qing_salt_commissioner_holder }  remove_variable = qing_salt_commissioner_holder }
		QING_salt_commissioner_appoint = yes
		LOG_line = { sys = QING  msg = "salt: the commissioner's post fell vacant and was backfilled for" }
	}
	if = {
		limit = { has_variable = qing_salt_commissioner_holder  var:qing_salt_commissioner_holder = { is_alive = yes } }
		# (c) graft creep — corruption climbs quarterly (the source of the squeeze), capped < 90.
		var:qing_salt_commissioner_holder = { if = { limit = { corruption < 90 }  add_corruption = 1 } }
		# (d) SIPHON — a slice scaled by his corruption as REAL personal wealth, extracted from the salt
		# merchants' over-charge (does NOT reduce the banked 鹽課 — the throne's reduced take at high squeeze
		# is already the yield shave above). siphon = income_last × corruption / 200 — the LARGE-STREAM divisor
		# (like the Canton Hoppo, se_QING_CANTON.txt), NOT the caravan's /20: 鹽課 is a major revenue pillar
		# (output × 3.0 gabelle markup), so /20 would pocket multiples of the take (a Heshen-scale hoard per
		# quarter) and dump an outsized 抄家 windfall into the treasury on impeachment; /200 keeps the squeeze a
		# fraction of the take (≈0.45× at the corruption-90 cap), matching the intended ~Heshen ceiling.
		# add_gold to the CHARACTER only — NOT treasury/reserve (the #44 C1 rule; confiscated on impeach/retire).
		var:qing_salt_commissioner_holder = { save_scope_as = qing_salt_seated }
		if = {
			limit = { has_variable = qing_salt_income_last  var:qing_salt_income_last > 0 }
			set_variable = { name = qing_salt_siphon_tmp  value = var:qing_salt_income_last }
			change_variable = { name = qing_salt_siphon_tmp  multiply = scope:qing_salt_seated.corruption }
			change_variable = { name = qing_salt_siphon_tmp  divide = 200 }
			if = { limit = { var:qing_salt_siphon_tmp > 0 }  scope:qing_salt_seated = { add_gold = { value = ROOT.var:qing_salt_siphon_tmp } } }
			remove_variable = qing_salt_siphon_tmp
		}
		# (e) mirror the squeeze (= his corruption) + the lagged finesse for next pulse's yield grade.
		set_variable = { name = qing_salt_squeeze  value = scope:qing_salt_seated.corruption }
		set_variable = { name = qing_salt_finesse  value = scope:qing_salt_seated.finesse }
	}
}

# ======================================================================
# QING_salt_commissioner_rotate — the player's proactive lever on the squeeze (mirrors
# QING_canton_rotate_hoppo / QING_caravan_super_rotate). Relieve any prior holder, seat a fresh
# accountable man; a new broom eases the squeeze, a high-finesse man eases it more. Country scope
# (ROOT = CHI). Called from the Salt Monopoly window button.
# ======================================================================
QING_salt_commissioner_rotate = {
	LOG_enter = { sys = QING  fn = "QING_salt_commissioner_rotate" }
	if = {
		limit = { has_variable = qing_salt_commissioner_holder }
		if = {
			limit = { var:qing_salt_commissioner_holder = { is_alive = yes } }
			var:qing_salt_commissioner_holder = { if = { limit = { has_variable = qing_salt_commissioner_marker }  remove_variable = qing_salt_commissioner_marker } }
		}
		remove_variable = qing_salt_commissioner_holder
	}
	QING_salt_commissioner_appoint = yes
	# [#81 2026-08-11] ease the squeeze by the NEW commissioner's CORRUPTION (a clean man eases more; a venal one
	# eases nothing), replacing the old flat -12 / -6-if-finesse. One-quarter bridge (the pulse re-mirrors squeeze
	# = holder corruption next quarter — the durable lever is picking a low-corruption man via the list).
	QING_frontier_office_ease_squeeze = { squeeze = qing_salt_squeeze  holder = qing_salt_commissioner_holder }
	LOG_line = { sys = QING  msg = "salt: a fresh commissioner (兩淮鹽政) is sent to the salt yards for" }
	LOG_exit = { sys = QING  fn = "QING_salt_commissioner_rotate"  result = OK }
}

# ======================================================================
# [#46 2026-08-13] QING_salt_assess_fitness — the Salt Commissioner's own fitness for the salt-
# gabelle reform dilemma (qing_revenue.1). Mirrors QING_revenue_assess_fitness's proven shape
# (se_QING_REVENUE.txt) exactly, but reads the SEATED COMMISSIONER (qing_salt_commissioner_holder),
# not the Grand Minister of Revenue — the gabelle is HIS domain (兩淮鹽政), not the 戶部's own. This
# event predates the #44 Salt Commissioner subsystem (introduced weeks later) and was never
# retargeted onto it; it kept scoring the Revenue minister's generic fitness even though a real,
# always-seated commissioner with his own corruption/finesse/squeeze meter exists specifically for
# this domain. Writes qing_salt_fitness (0..100, same clamp shape). Scope: country.
# ======================================================================
QING_salt_assess_fitness = {
	LOG_enter = { sys = QING  fn = "QING_salt_assess_fitness" }
	set_variable = { name = qing_salt_fitness  value = 50 }

	if = {
		limit = {
			has_variable = qing_salt_commissioner_holder
			var:qing_salt_commissioner_holder = { is_alive = yes  employer = ROOT }
		}
		var:qing_salt_commissioner_holder = {
			# FINESSE is the core skill (the fiscal administrator's gift) — same weight as the Revenue helper.
			ROOT = {
				change_variable = { name = qing_salt_fitness  add = prev.finesse }
				change_variable = { name = qing_salt_fitness  subtract = 5 }
			}
			# HONESTY vs CORRUPTION — same trait weights as QING_revenue_assess_fitness.
			if = {
				limit = { OR = { has_trait = honest  has_trait = just } }
				ROOT = { change_variable = { name = qing_salt_fitness  add = 15 } }
			}
			if = {
				limit = { has_trait = corrupt }
				ROOT = { change_variable = { name = qing_salt_fitness  subtract = 20 } }
			}
			# NUMERIC corruption stat — reuses the commissioner's OWN squeeze source (his corruption
			# stat, the same value QING_salt_reconcile mirrors into qing_salt_squeeze) rather than the
			# generic qing_corruption_level the Revenue helper reads — the gabelle's own graft is the
			# relevant signal for THIS man's fitness to reform it.
			if = {
				limit = { corruption >= 40 }
				ROOT = { change_variable = { name = qing_salt_fitness  subtract = 15 } }
			}
			else_if = {
				limit = { corruption >= 20 }
				ROOT = { change_variable = { name = qing_salt_fitness  subtract = 8 } }
			}
			# LOYALTY — a disloyal commissioner won't act for the throne against his own patronage web.
			if = {
				limit = { loyalty < 40 }
				ROOT = { change_variable = { name = qing_salt_fitness  subtract = 20 } }
			}
			else_if = {
				limit = { loyalty >= 70 }
				ROOT = { change_variable = { name = qing_salt_fitness  add = 10 } }
			}
		}
	}
	# clamp 0..100
	if = { limit = { var:qing_salt_fitness > 100 }  set_variable = { name = qing_salt_fitness  value = 100 } }
	if = { limit = { var:qing_salt_fitness < 0 }    set_variable = { name = qing_salt_fitness  value = 0 } }

	LOG_line = { sys = QING  msg = "salt commissioner fitness (gabelle reform) = for" }
	LOG_exit = { sys = QING  fn = "QING_salt_assess_fitness"  result = OK }
}
