########################################################################
# se_SUBJECT_QING.txt — Player-facing per-subject interactions
#
# Scope of this work: interactions the PLAYER can perform on a single named
# subject. Built and tuned for a Qing (CHI) playthrough, whose starting
# subjects span exactly four types. AI usage is intentionally NOT handled.
#
# Two systems live here:
#   1. Change subject type (promotion = tighter control, demotion = looser),
#      stepping one rung along the Qing control ladder.
#   2. Gradual integration: spend influence to accrue integration progress on
#      one subject; at the threshold, absorb its provinces into the overlord.
#
# The change-type primitive is vanilla make_subject { target type }, wrapped
# by FUNC_make_subject (se_FUNC.txt). Province absorption uses
# LAND_transfer_provinces (se_LAND.txt), which correctly handles governorship
# wealth/stockpile splitting and foreign-influence recalculation.
#
# All effects are instrumented with the LOG_* module (se_LOG.txt). Launch the
# game with -debug_mode and the trace lands in game.log under [IMP19C][SUBJ].
#
# Convention for these effects:
#   ROOT / this  = the overlord (player, Qing)
#   scope:target = the single subject being acted upon
#
# Prefix = SUBJ_QING
########################################################################

########################################################################
# CONTROL LADDER
# Ordered loosest -> tightest for the four types a Qing player holds:
#   tributary < nominal_vassal < feudatory < autonomous_governorship
# "Promote" = move one rung tighter, "Demote" = one rung looser.
# Rungs are encoded as an integer on the subject (SUBJ_ladder_rung) so the
# UI and triggers can reason about position without a chain of is_subject_type
# checks. The rung is (re)derived from the live subject type on every use, so
# it stays correct even if the type is changed by other code or at game start.
########################################################################

# Derive the ladder rung integer from the subject's current type.
# Scope: the subject (call inside scope:target = { ... })
# Sets SUBJ_ladder_rung: 1..4, or 0 if the type is off-ladder (unsupported).
SUBJ_QING_derive_rung = {
	if = {
		# [1763-hist-fix 2026-07-17] rung 1 (bottom) = the sinosphere_tributary (朝貢國), the ladder's
		# new base type; the vanilla tributary also maps to rung 1 for any legacy plain-tributary subject.
		limit = { OR = { is_subject_type = sinosphere_tributary  is_subject_type = tributary } }
		set_variable = { name = SUBJ_ladder_rung  value = 1 }
	}
	else_if = {
		limit = { is_subject_type = nominal_vassal }
		set_variable = { name = SUBJ_ladder_rung  value = 2 }
	}
	else_if = {
		limit = { is_subject_type = feudatory }
		set_variable = { name = SUBJ_ladder_rung  value = 3 }
	}
	else_if = {
		limit = { is_subject_type = autonomous_governorship }
		set_variable = { name = SUBJ_ladder_rung  value = 4 }
	}
	else = {
		set_variable = { name = SUBJ_ladder_rung  value = 0 }
	}
}

########################################################################
# CHANGE SUBJECT TYPE
# Called from the scripted GUIs. Expects ROOT = overlord, scope:target = subject.
# $direction$ = promote | demote  (text, used only for logging clarity)
# $new_type$  = the subject_type to switch the subject to.
# The GUI is responsible for gating (influence cost, valid rung, cooldown);
# these effects assume validity but still log loudly and guard the transition.
########################################################################

SUBJ_QING_change_type = {
	# Scope: country (overlord / ROOT)
	LOG_enter = { sys = SUBJ  fn = "SUBJ_QING_change_type -> $new_type$ ($direction$)" }

	if = {
		limit = { exists = scope:target }

		scope:target = { SUBJ_QING_derive_rung = yes }

		LOG_line = {
			sys = SUBJ
			msg = "overlord= subject= from_rung= action=$direction$ new_type=$new_type$"
		}

		# Perform the type change.
		# [#17 fix 2026-08-07] make_subject HARD-ERRORS + no-ops on a country that ALREADY has an overlord
		# ("<X> already has an overlord!" — the same class the capstone authorize path hit). The engine has no
		# re-type-in-place command (change_subject_type does not exist), so the proven idiom is RELEASE-THEN-
		# REBIND (SUBJ_QING_authorize_integration; se_DIPLOMACY.txt subjugate transfer). Before this fix the
		# GUI promote/demote buttons were a SILENT NO-OP.
		#
		# CRUCIAL DIFFERENCE from the authorize path: there the intermediary is ANNEXED (destroyed), so its
		# sub-subjects are reparented UP to CHI. HERE the subject SURVIVES and must KEEP its own sub-subjects
		# (e.g. ILI holding XNG/KML). release_subject severs the DIRECT CHI->subject bond; whether it ALSO
		# cascades to the grandchildren is unverified engine behaviour — so we snapshot the subject's children
		# and their types BEFORE the release, and AFTER the rebind re-assert only those that actually detached
		# (guarded on is_subject_of, so it is a pure no-op if the release did NOT cascade). Correct either way,
		# never orphaning a grandchild, never double-binding an intact one.
		scope:target = {
			save_scope_as = QING_ct_parent
			every_subject = { add_to_list = QING_ct_children }
		}
		# record each child's current type while the subject->child bond still stands (guarded var writes;
		# mirrors SUBJ_QING_reparent_sub_subjects). has_variable guards keep the later reads off the var-flood.
		every_in_list = {
			list = QING_ct_children
			save_scope_as = QING_ct_child
			if = { limit = { is_subject_type = tributary }                     set_variable = { name = QING_ct_type  value = flag:tributary } }
			else_if = { limit = { is_subject_type = nominal_vassal }           set_variable = { name = QING_ct_type  value = flag:nominal_vassal } }
			else_if = { limit = { is_subject_type = feudatory }               set_variable = { name = QING_ct_type  value = flag:feudatory } }
			else_if = { limit = { is_subject_type = semi_autonomous_governorship } set_variable = { name = QING_ct_type  value = flag:semi_autonomous_governorship } }
			else_if = { limit = { is_subject_type = autonomous_governorship }  set_variable = { name = QING_ct_type  value = flag:autonomous_governorship } }
			else_if = { limit = { is_subject_type = client_state }            set_variable = { name = QING_ct_type  value = flag:client_state } }
			else_if = { limit = { is_subject_type = sinosphere_tributary }     set_variable = { name = QING_ct_type  value = flag:sinosphere_tributary } }
			else_if = { limit = { is_subject_type = protectorate }            set_variable = { name = QING_ct_type  value = flag:protectorate } }
			else = { set_variable = { name = QING_ct_type  value = flag:feudatory }  LOG_line = { sys = SUBJ  msg = "change_type: sub-subject had an off-ladder type; will re-bind as feudatory if detached" } }
		}

		# release from CHI, then re-bind at the new rung.
		# [review 2026-08-07] STATE-SAFETY of release-then-rebind on a SURVIVING subject (vs authorize, where
		# the subject is annexed): verified PROVEN — Terra Indomita's 000_country_switch_effect.txt does exactly
		# this (release PREV then re-make_subject at the SAME type, per type, on kept subjects) with NO
		# opinion/liberty_desire/loyalty restoration afterward, so the transient release+rebind is an accepted,
		# state-safe upstream idiom. The Qing subject system also tracks its state (integ progress, amban band)
		# in COUNTRY VARS, which survive release (the country is not destroyed), and the GUI's
		# qing_recently_changed_type cooldown var likewise persists.
		ROOT = { release_subject = scope:target }
		FUNC_make_subject = {
			overlord = ROOT
			target = scope:target
			type = $new_type$
		}
		# DEFENSIVE ASSERTION (mirrors the authorize path): if the rebind failed, the subject went PERMANENTLY
		# independent — a silent, unrecoverable loss. Log loudly so a regression is VISIBLE.
		if = {
			limit = { NOT = { scope:target = { is_subject_of = ROOT } } }
			LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_change_type"  reason = "CRITICAL: rebind FAILED — subject went independent after release (make_subject rejected the new type). Subject=" }
		}

		# re-assert any grandchild the release cascaded loose (pure no-op if the direct bond survived).
		# NOTE: release_subject almost certainly severs only the one named bond (CHI->target), so this branch
		# is expected to be UNEXERCISED — it exists purely as a safety net. Its rebind targets scope:QING_ct_parent
		# (a country that is itself CHI's subject), an overlord-is-a-subject make_subject the proven reparent path
		# never does (reparent always rebinds UP to CHI). Untested but only reachable on the unlikely cascade.
		every_in_list = {
			list = QING_ct_children
			save_scope_as = QING_ct_child
			if = {
				limit = { NOT = { is_subject_of = scope:QING_ct_parent } }
				if =      { limit = { has_variable = QING_ct_type  var:QING_ct_type = flag:tributary }                     FUNC_make_subject = { overlord = scope:QING_ct_parent  target = scope:QING_ct_child  type = tributary } }
				else_if = { limit = { has_variable = QING_ct_type  var:QING_ct_type = flag:nominal_vassal }               FUNC_make_subject = { overlord = scope:QING_ct_parent  target = scope:QING_ct_child  type = nominal_vassal } }
				else_if = { limit = { has_variable = QING_ct_type  var:QING_ct_type = flag:feudatory }                   FUNC_make_subject = { overlord = scope:QING_ct_parent  target = scope:QING_ct_child  type = feudatory } }
				else_if = { limit = { has_variable = QING_ct_type  var:QING_ct_type = flag:semi_autonomous_governorship } FUNC_make_subject = { overlord = scope:QING_ct_parent  target = scope:QING_ct_child  type = semi_autonomous_governorship } }
				else_if = { limit = { has_variable = QING_ct_type  var:QING_ct_type = flag:autonomous_governorship }     FUNC_make_subject = { overlord = scope:QING_ct_parent  target = scope:QING_ct_child  type = autonomous_governorship } }
				else_if = { limit = { has_variable = QING_ct_type  var:QING_ct_type = flag:client_state }                FUNC_make_subject = { overlord = scope:QING_ct_parent  target = scope:QING_ct_child  type = client_state } }
				else_if = { limit = { has_variable = QING_ct_type  var:QING_ct_type = flag:sinosphere_tributary }        FUNC_make_subject = { overlord = scope:QING_ct_parent  target = scope:QING_ct_child  type = sinosphere_tributary } }
				else_if = { limit = { has_variable = QING_ct_type  var:QING_ct_type = flag:protectorate }               FUNC_make_subject = { overlord = scope:QING_ct_parent  target = scope:QING_ct_child  type = protectorate } }
				else =    { FUNC_make_subject = { overlord = scope:QING_ct_parent  target = scope:QING_ct_child  type = feudatory } }
				LOG_line = { sys = SUBJ  msg = "change_type: re-asserted a sub-subject that the release cascaded loose, back under the surviving intermediary" }
			}
			if = { limit = { has_variable = QING_ct_type }  remove_variable = QING_ct_type }
			remove_from_list = QING_ct_children
		}

		# Re-derive the rung from the now-current type so state stays truthful.
		scope:target = { SUBJ_QING_derive_rung = yes }

		# Integration only runs on autonomous_governorship. If a demotion (or any
		# type change) drops the subject off that rung, wipe stale integration
		# state so the GUI bar and the pulse don't act on a frozen half-progress.
		if = {
			limit = {
				scope:target = {
					NOT = { is_subject_type = autonomous_governorship }
					OR = {
						has_variable = SUBJ_integration_progress
						has_variable = SUBJ_integration_active
						has_variable = SUBJ_integration_suspended
					}
				}
			}
			scope:target = { SUBJ_QING_clear_integration_state = yes }
		}

		LOG_exit = { sys = SUBJ  fn = "SUBJ_QING_change_type"  result = OK }
	}
	else = {
		LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_change_type"  reason = "scope:target does not exist" }
	}
}

########################################################################
# INCORPORATE A PROTECTORATE ONTO THE LADDER
# A protectorate (e.g. Tibet under the 1815 amban arrangement) sits OFF the
# Qing control ladder — can_be_integrated = no, so the promote/integrate verbs
# never apply to it. This is the historical loose-rein (羈縻) settlement.
#
# This effect is the one-way "formally incorporate" act modelled on Zhao Erfeng's
# (趙爾豐) forced incorporation of Kham (川邊改土歸流, 1904-1911): it converts a
# protectorate straight to an autonomous_governorship, dropping it onto rung 4 so
# the ordinary integration process can then absorb it. It is expensive and
# coercive by design (the prestige + influence price is deducted in the calling
# GUI's effect block; its is_valid only checks affordability), and it is
# NOT date-restricted — the player may attempt it from the 1815 start, at the
# same price the late historical attempt would have cost the court.
#
# One-way: the demote ladder runs autonomous_governorship -> feudatory -> ... ->
# tributary and never returns to protectorate, so there is no reverse path.
#
# Called from qing_subject_incorporate_button. ROOT = overlord, scope:target =
# the protectorate subject.
########################################################################
SUBJ_QING_incorporate_protectorate = {
	LOG_enter = { sys = SUBJ  fn = "SUBJ_QING_incorporate_protectorate" }

	if = {
		limit = {
			exists = scope:target
			scope:target = { is_subject_of = ROOT }
			scope:target = { is_subject_type = protectorate }
		}

		LOG_line = {
			sys = SUBJ
			msg = "incorporate overlord= protectorate= -> autonomous_governorship"
		}

		# The coercion abrogates a standing autonomy compact: a tyranny + AE bite,
		# mirroring the other coercive Qing verbs (audit/opium). The prestige and
		# influence price is deducted in the calling GUI's effect block, not here.
		add_tyranny = 3
		add_aggressive_expansion = 5

		FUNC_make_subject = {
			overlord = ROOT
			target = scope:target
			type = autonomous_governorship
		}

		# Land it truthfully on rung 4 so the UI and integrate verb pick it up.
		scope:target = { SUBJ_QING_derive_rung = yes }

		LOG_exit = { sys = SUBJ  fn = "SUBJ_QING_incorporate_protectorate"  result = OK }
	}
	else = {
		LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_incorporate_protectorate"  reason = "scope:target missing, not a subject of ROOT, or not a protectorate" }
	}
}

########################################################################
# INTEGRATION
# Integration is a STATEFUL PROCESS, not a silent counter. Progress is stored
# on the SUBJECT as SUBJ_integration_progress (0..threshold). The player drives
# it through the qing_integ.* event chain (events/imp19c_mod_events/
# qing_subject_integration.txt), which can be helped or hindered by positive
# and negative "reaction" events grounded in real Qing frontier history
# (the Revolt of the Three Feudatories, gaitu guiliu, the Heshen scandal,
# the White Lotus / Miao / Panthay / Dungan risings, tuntian harvests, and
# imperial investiture festivals). At the threshold the subject is absorbed:
# all its provinces transfer to the overlord and the subject relationship ends.
#
# State variables on the SUBJECT:
#   SUBJ_integration_progress   0..SUBJ_integration_threshold
#   SUBJ_integration_active     set once the process is under way (drives the
#                               ambient reaction pulse)
#   SUBJ_integration_suspended  set when the player suspends; the pulse skips it
########################################################################

# Tunables (kept inline; promote to script_values later if balancing needs it).
# threshold = 5 progress to absorb. Keep in sync with:
#   - the "N / 5" progress text and the 0.2 progressbar scale in diplomatic_view.gui
#   - the 5-step wording in the QING_SUBJECT_INTEGRATE_TT localisation

# Advance integration progress by $steps$ (scope: country / ROOT = overlord,
# scope:target = subject). Marks the process active, clears any suspension,
# and absorbs the subject once progress reaches the threshold.
SUBJ_QING_advance_integration = {
	LOG_enter = { sys = SUBJ  fn = "SUBJ_QING_advance_integration +$steps$" }

	if = {
		# [#19 2026-08-09 BUGFIX] LADDER GATE. Only a subject that is actually ON the integration ladder may
		# accrue integration progress. A can_be_integrated=no subject that sits OFF the ladder — a PROTECTORATE
		# (Tibet, the 羈縻 loose-rein settlement), a tributary/feudatory, or a sub-subject (e.g. Kobdo under
		# Uliastay) — must be brought ONTO the ladder FIRST (a protectorate via SUBJ_QING_incorporate_protectorate,
		# the priced+coercive Zhao-Erfeng 改土歸流 act; a lesser subject via the promote ladder). Without this
		# gate, any progress-driver (frontier/Xinjiang/caravan events, SUB_QING interactions — not all of which
		# gated on the type) could fill an off-ladder subject's meter and the threshold SAFETY-NET below would
		# DIRECTLY absorb it, bypassing both the engine handoff and the incorporate step (the reported bug:
		# Tibet + Kobdo annexed off-ladder). Gating at the SINK closes the bypass for every caller at once.
		# The autonomous_governorship (rung 4) and the transient integrating_governorship are the only
		# ladder-legit types; everything else is refused here with a loud, greppable LOG_fail.
		limit = {
			exists = scope:target
			OR = {
				scope:target = { is_subject_type = autonomous_governorship }
				scope:target = { is_subject_type = integrating_governorship }
			}
		}

		scope:target = {
			if = {
				limit = { NOT = { has_variable = SUBJ_integration_progress } }
				set_variable = { name = SUBJ_integration_progress  value = 0 }
			}
			set_variable = { name = SUBJ_integration_active  value = 1 }
			if = {
				limit = { has_variable = SUBJ_integration_suspended }
				remove_variable = SUBJ_integration_suspended
			}
			change_variable = { name = SUBJ_integration_progress  add = $steps$ }
			# Clamp to the threshold so display and absorption stay consistent.
			if = {
				limit = { var:SUBJ_integration_progress > 5 }
				set_variable = { name = SUBJ_integration_progress  value = 5 }
			}

			LOG_line = {
				sys = SUBJ
				msg = "integration overlord= subject= +$steps$ progress=/5"
			}

			# [#111/#112 SOLUTION A] At the threshold, DO NOT silently absorb. Fire the
			# 改土歸流 capstone (qing_integ.30) so the player chooses the MANNER of
			# incorporation (solemn / bureaucratic / coercive). [#subj-integ-rework 2026-07-22]
			# Each capstone option now calls SUBJ_QING_authorize_integration (NOT the old inline
			# SUBJ_QING_absorb_subject): the capstone AUTHORIZES the engine handoff and the engine
			# absorbs the provinces gradually — the meter (this progress var, filled by the push
			# button + reaction/frontier events) is the political LEAD-UP to that decision. The
			# event's own trigger re-checks progress>=5 so a queued .30 cannot double-resolve (once
			# the first .30 authorizes, it converts the type off autonomous_governorship and clears
			# the progress var, so any queued .30 discards silently).
			if = {
				limit = {
					var:SUBJ_integration_progress >= 5
					this = { is_subject_type = autonomous_governorship }
				}
				this = { save_scope_as = target }
				root = { trigger_event = { id = qing_integ.30  days = { 1 5 } } }
			}
				else_if = {
				# [#19 BUGFIX] SAFETY NET — now genuinely unreachable: the entry ladder-gate above admits
				# only autonomous_governorship / integrating_governorship, and the integrating type at
				# threshold is mid-engine-absorb (not a stuck meter). A subject at threshold that is NOT an
				# autonomous_governorship therefore should never reach here. The old branch DIRECT-absorbed
				# it (SUBJ_QING_absorb_subject), which was the very ladder-bypass the entry gate now closes —
				# so absorbing here is REMOVED. Log loudly if it ever fires so a regression in the gate surfaces
				# in debug.log instead of silently annexing an off-ladder subject.
				limit = { var:SUBJ_integration_progress >= 5 }
				root = { LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_advance_integration"  reason = "CRITICAL: threshold reached on a non-autonomous_governorship subject — the ladder entry-gate should have blocked this; NOT absorbing (would bypass the incorporate step, the no. 19 bug)" } }
				}
			}

			LOG_exit = { sys = SUBJ  fn = "SUBJ_QING_advance_integration"  result = OK }
		}
		else = {
			# [#19 BUGFIX] refused: either scope:target is missing, OR it is an off-ladder can_be_integrated=no
			# subject (protectorate / tributary / feudatory / sub-subject) that must be incorporated onto the
			# ladder (autonomous_governorship) before it can accrue integration progress. A no-op is correct —
			# the caller's push simply does nothing on such a subject, as intended.
			LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_advance_integration"  reason = "target missing OR not on the integration ladder (needs incorporate to autonomous_governorship first) — progress refused" }
		}
	}

# Set back integration progress by $steps$ (scope: country / ROOT), clamped at 0.
# Used by negative reaction events (unrest, scandal, riot) to punish a botched push.
SUBJ_QING_reduce_integration = {
	if = {
		limit = { exists = scope:target }
		scope:target = {
			if = {
				limit = { NOT = { has_variable = SUBJ_integration_progress } }
				set_variable = { name = SUBJ_integration_progress  value = 0 }
			}
			change_variable = { name = SUBJ_integration_progress  subtract = $steps$ }
			if = {
				limit = { var:SUBJ_integration_progress < 0 }
				set_variable = { name = SUBJ_integration_progress  value = 0 }
			}
			LOG_line = {
				sys = SUBJ
				msg = "integration SETBACK subject= -$steps$ progress=/5"
			}
		}
	}
	else = {
		# audit-fix (#79): trace the guard-miss instead of silently no-op'ing
		LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_reduce_integration"  reason = "scope:target does not exist" }
	}
}

# Suspend the process (scope: country, scope:target = subject).
SUBJ_QING_suspend_integration = {
	if = {
		limit = { exists = scope:target }
		scope:target = {
			set_variable = { name = SUBJ_integration_suspended  value = 1 }
			LOG_line = { sys = SUBJ  msg = "integration SUSPENDED subject=" }
		}
	}
	else = {
		# audit-fix (#79): trace the guard-miss instead of silently no-op'ing
		LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_suspend_integration"  reason = "scope:target does not exist" }
	}
}

# Resume a suspended process (scope: country, scope:target = subject).
# Progress is preserved; only the suspension flag is lifted. Called from the
# resume button so the player can restart without paying the push cost again.
SUBJ_QING_resume_integration = {
	if = {
		limit = { exists = scope:target }
		scope:target = {
			if = {
				limit = { has_variable = SUBJ_integration_suspended }
				remove_variable = SUBJ_integration_suspended
			}
			set_variable = { name = SUBJ_integration_active  value = 1 }
			LOG_line = { sys = SUBJ  msg = "integration RESUMED subject=" }
		}
	}
	else = {
		# audit-fix (#79): trace the guard-miss instead of silently no-op'ing
		LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_resume_integration"  reason = "scope:target does not exist" }
	}
}

# Clear all integration state from a subject. Scope: the subject
# (call inside scope:target = { ... }). Used when the subject is absorbed
# (no longer exists as a relationship) or drops off the integration-eligible
# rung via a demotion, so no stale progress/active/suspended flags linger to
# confuse the GUI progress bar or the ambient pulse.
SUBJ_QING_clear_integration_state = {
	if = { limit = { has_variable = SUBJ_integration_progress }   remove_variable = SUBJ_integration_progress }
	if = { limit = { has_variable = SUBJ_integration_active }     remove_variable = SUBJ_integration_active }
	if = { limit = { has_variable = SUBJ_integration_suspended }  remove_variable = SUBJ_integration_suspended }
	LOG_line = { sys = SUBJ  msg = "integration STATE CLEARED subject=" }
}

# Kept for backward compatibility: a single steady push (+1).
SUBJ_QING_add_integration = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = SUBJ  fn = "SUBJ_QING_add_integration" }
	SUBJ_QING_advance_integration = { steps = 1 }
	LOG_exit = { sys = SUBJ  fn = "SUBJ_QING_add_integration"  result = OK }
}

# [#subj-integ-rework 2026-07-22] AUTHORIZE VANILLA INTEGRATION
# The 改土歸流 capstone (qing_integ.30) no longer instant-annexes. Instead it
# AUTHORIZES the final absorption by handing off to the engine's native subject-
# integration: the subject (an autonomous_governorship at threshold) is converted
# to the transient `integrating_governorship` type (can_be_integrated = yes), which
# lights up the engine's hardcoded "Integrate" diplomatic action. The player then
# clicks it and the ENGINE absorbs the subject GRADUALLY. Completion is caught by
# on_diplomatic_annex (00_specific_from_code.txt), which runs the sub-subject
# reparenting + character cleanup the old instant absorb did inline.
#
# WHY the flag flip, not a "start integration" verb: neither vanilla, TI, nor
# Invictus exposes a script effect to START engine subject-integration (confirmed
# by both oracles — start_integrating_subject / integrate_subject do NOT exist). The
# ONLY interface is the can_be_integrated subject-type flag + the engine button +
# the on_diplomatic_annex completion hook. So "authorize" = flip the type; the player
# finishes via the now-visible native action.
#
# Scope: country (overlord / ROOT). scope:target = the subject at the capstone.
########################################################################
SUBJ_QING_authorize_integration = {
	LOG_enter = { sys = SUBJ  fn = "SUBJ_QING_authorize_integration" }

	if = {
		limit = {
			exists = scope:target
			scope:target = { is_subject_of = ROOT }
			scope:target = { any_owned_province = { count >= 1 } }
		}

		# SUB-SUBJECT REPARENTING happens HERE, at authorize time — NOT on completion.
		# Rationale: (a) scoping is correct — ROOT = overlord, scope:target = subject, exactly
		# what SUBJ_QING_reparent_sub_subjects expects (the same scope the old inline absorb used);
		# (b) the engine deletes the subject when IT completes annexation, taking its sub-subject
		# links with it, so there is no safe post-annex window to read them. Reparenting now (while
		# the intermediary still exists) hands e.g. Tibet's sub-subjects straight to CHI with no
		# orphan gap, and the intermediary is then absorbed gradually by the engine with no children
		# left to strand. No-op when the subject holds no sub-subjects (the common case).
		SUBJ_QING_reparent_sub_subjects = yes

		# Convert to the integrable transient type (lights up the engine's native Integrate action).
		# [BOOT-FIX 2026-08-06] make_subject HARD-ERRORS + no-ops on a country that ALREADY has an
		# overlord ("<X> already has an overlord!" in error.log — the subject stayed autonomous_governorship,
		# so the capstone appeared to do nothing). The engine has no re-type-in-place command
		# (change_subject_type does not exist); the proven idiom is RELEASE-THEN-REBIND
		# (se_DIPLOMACY.txt:1082-1092 subjugate transfer / :1415). Sub-subjects were already reparented to
		# ROOT above (BEFORE this release), so the brief release strands no children. Release from ROOT,
		# then re-bind as integrating_governorship.
		ROOT = { release_subject = scope:target }
		FUNC_make_subject = { overlord = ROOT  target = scope:target  type = integrating_governorship }
		# [review 2026-08-06] DEFENSIVE ASSERTION: the rebind relies on scripted make_subject bypassing the
		# type's allow={always=no} (proven by the reparent loop rebinding to autonomous_governorship/allow=no
		# + QING_convert_to_march to frontier_protectorate/allow=no, both erroring nowhere). If that ever fails,
		# the released subject would go PERMANENTLY INDEPENDENT — a silent, unrecoverable loss. Log loudly so a
		# regression is VISIBLE in error/debug rather than silently losing the player's subject.
		if = {
			limit = { NOT = { scope:target = { is_subject_of = ROOT } } }
			LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_authorize_integration"  reason = "CRITICAL: rebind FAILED — subject went independent after release (make_subject rejected integrating_governorship). Subject=" }
		}
		scope:target = { SUBJ_QING_derive_rung = yes }

		# The custom political lead-up is complete; the engine now owns the absorption.
		# Clear the custom 5-step progress/active/suspended state so the old progress bar
		# and the ambient reaction pulse stop acting on it (both also gate on the type, which
		# is no longer autonomous_governorship — belt and braces). The diplomatic-view label
		# reads the type itself (is_subject_type = integrating_governorship), so no marker var
		# is needed; the subject is deleted by the engine when it completes annexation anyway.
		scope:target = {
			SUBJ_QING_clear_integration_state = yes
		}

		LOG_line = { sys = SUBJ  msg = "AUTHORIZE integration (handoff to engine) overlord= subject= converted to integrating_governorship" }
		LOG_exit = { sys = SUBJ  fn = "SUBJ_QING_authorize_integration"  result = OK }
	}
	else = {
		LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_authorize_integration"  reason = "target missing / not our subject / no provinces" }
	}
}

########################################################################
# REACTION EVENTS
# After a push (and, at lower odds, on the ambient pulse) a positive or
# negative reaction event may fire on the overlord about this subject. The
# weighting reads live game state so a corrupt court, a rebellious populace,
# or a prosperous frontier each tilt the odds toward the matching history.
########################################################################

# Roll one reaction event for scope:target. Scope: country (ROOT = overlord).
# $ambient$ = yes on the pulse (adds a "nothing happens" weight so quiet years
# are common); omit / no when called right after a deliberate push.
SUBJ_QING_roll_reaction = {
	# audit-fix (#79, subject rework): guard the reaction chokepoint against a
	# ZOMBIE subject. When integration reaches 5, SUBJ_QING_advance_integration
	# calls SUBJ_QING_absorb_subject inline (transferring away all provinces), and
	# the integrate button THEN calls roll_reaction guarded only by is_subject_of.
	# In the same tick the just-emptied subject can still test is_subject_of=ROOT
	# before the engine deletes the landless country, so a reaction would schedule
	# qing_integ.* on a dying tag (whose .a option re-advances integration →
	# stuck loop / broken portrait). Requiring a live province closes that window;
	# it also protects any future caller (the ambient pulse already filters on
	# SUBJ_integration_active, which absorb clears, so the pulse was already safe).
	if = {
		limit = {
			exists = scope:target
			scope:target = { is_subject_of = ROOT }
			scope:target = { any_owned_province = { count >= 1 } }
		}
	LOG_line = {
		sys = SUBJ
		msg = "roll_reaction overlord= subject= ambient=$ambient$"
	}
	random_list = {
		# --- Quiet: only meaningful on the ambient pulse ---
		30 = {
			trigger = { always = $ambient$ }
			LOG_line = { sys = SUBJ  msg = "reaction=QUIET subject=" }
		}
		# --- NEGATIVE: popular unrest (Three Feudatories / Taiping echo) ---
		20 = {
			modifier = {
				# A restive frontier is likelier to rise.
				factor = 2
				scope:target = { any_owned_province = { total_population >= 20 } }
			}
			LOG_line = { sys = SUBJ  msg = "reaction=UNREST(qing_integ.10) subject=" }
			trigger_event = { id = qing_integ.10  days = { 3 10 } }
		}
		# --- NEGATIVE: amban graft scandal (the resident skims the integration-silver) ---
		# [#22 2026-08-07] Only roll this when a live resident amban is actually posted to the subject —
		# the scandal pins the diverted integration-silver on HIM (qing_integ.11 retargeted from a random
		# courtier to scope:integ_amban). Gate on the RAW qing_amban_here predicate (resolve doesn't run
		# until the event's immediate, so integ_amban_present isn't available here). No amban → no scandal
		# (this branch's weight simply doesn't apply; another reaction is rolled).
		15 = {
			trigger = {
				scope:target = { has_variable = qing_amban_here  var:qing_amban_here = { is_alive = yes  employer = scope:target } }
			}
			modifier = {
				# A corrupt court breeds scandal.
				factor = 3
				current_ruler = { corruption >= 5 }
			}
			LOG_line = { sys = SUBJ  msg = "reaction=SCANDAL(qing_integ.11) subject=" }
			trigger_event = { id = qing_integ.11  days = { 3 10 } }
		}
		# --- NEGATIVE: ethnic / "race" riot (Miao, Panthay, Dungan) ---
		15 = {
			modifier = {
				# Culturally alien subjects are likelier to riot against Han settlers.
				factor = 2
				scope:target = { NOT = { primary_culture = root.culture } }
			}
			LOG_line = { sys = SUBJ  msg = "reaction=STRIFE(qing_integ.12) subject=" }
			trigger_event = { id = qing_integ.12  days = { 3 10 } }
		}
		# --- POSITIVE: bountiful harvest on the frontier colonies (tuntian) ---
		12 = {
			LOG_line = { sys = SUBJ  msg = "reaction=HARVEST(qing_integ.20) subject=" }
			trigger_event = { id = qing_integ.20  days = { 3 10 } }
		}
		# --- POSITIVE: joyous festival / imperial investiture ---
		12 = {
			LOG_line = { sys = SUBJ  msg = "reaction=FESTIVAL(qing_integ.21) subject=" }
			trigger_event = { id = qing_integ.21  days = { 3 10 } }
		}
	}
	}
	else = {
		LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_roll_reaction"  reason = "target missing, no longer a subject, or landless (post-absorb) — reaction suppressed" }
	}
}

# Ambient pulse: for the Qing player, every subject with an active, unsuspended
# integration has a chance of a reaction event. Scope: country (CHI, human).
SUBJ_QING_integration_pulse = {
	LOG_enter = { sys = SUBJ  fn = "SUBJ_QING_integration_pulse" }
	every_subject = {
		limit = {
			has_variable = SUBJ_integration_active
			NOT = { has_variable = SUBJ_integration_suspended }
			is_subject_type = autonomous_governorship
		}
		save_scope_as = target
		root = {
			random = {
				chance = 40
				SUBJ_QING_roll_reaction = { ambient = yes }
			}
		}
	}
	LOG_exit = { sys = SUBJ  fn = "SUBJ_QING_integration_pulse"  result = OK }
}

SUBJ_QING_absorb_subject = {
	# Scope: country (overlord / ROOT). scope:target = subject to be absorbed.
	LOG_enter = { sys = SUBJ  fn = "SUBJ_QING_absorb_subject" }

	if = {
		limit = {
			exists = scope:target
			scope:target = { any_owned_province = { count >= 1 } }
		}

		LOG_state = { sys = SUBJ  note = "before absorbing" }

		# [#170 follow-up] SUB-SUBJECT RE-PARENTING. If the subject being absorbed
		# itself holds subjects (e.g. Tibet -> LTG/BTG), those sub-subjects must NOT be
		# orphaned or freed when their overlord vanishes: the integrating power inherits
		# them as its OWN direct subjects, preserving each one's current type. Run this
		# BEFORE the province transfer, while scope:target still exists as their overlord.
		# Uses the proven release-then-make_subject idiom (se_DIPLOMACY.txt:1066-1092,
		# se_FUNC.txt:349-352): a sub-subject bound to a soon-to-be-landless intermediary
		# would otherwise leave a stale double-overlord bond, so the release must run first.
		SUBJ_QING_reparent_sub_subjects = yes

		# Build the list of the subject's provinces, then transfer them to us.
		scope:target = {
			every_owned_province = {
				add_to_list = SUBJ_provinces_to_absorb
			}
		}

		LAND_transfer_provinces = {
			target_provinces = SUBJ_provinces_to_absorb
			grantee = ROOT
		}

		# Clean up the working list and integration state.
		every_in_list = {
			list = SUBJ_provinces_to_absorb
			remove_from_list = SUBJ_provinces_to_absorb
		}

		# Wipe the integration flags off the (now landless) subject so nothing
		# stale drives the pulse or the GUI progress bar if the tag persists.
		if = {
			limit = { exists = scope:target }
			scope:target = { SUBJ_QING_clear_integration_state = yes }
		}

		LOG_line = { sys = SUBJ  msg = "ABSORBED subject= into overlord=" }
		LOG_exit = { sys = SUBJ  fn = "SUBJ_QING_absorb_subject"  result = OK }
	}
	else = {
		LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_absorb_subject"  reason = "target missing or has no provinces to absorb" }
	}
}

# ======================================================================
# RE-PARENT SUB-SUBJECTS — when absorbing scope:target, transfer any subjects
# scope:target itself holds up to ROOT (the integrating overlord) as ROOT's own
# direct subjects, preserving each sub-subject's current type. Scope: country
# (ROOT = overlord). scope:target = the subject being absorbed.
#
# Order matters:
#   1. Snapshot scope:target's subjects into a list (iterating + mutating the live
#      subject list in one loop is unsafe).
#   2. For each, record its CURRENT type as a flag BEFORE any mutation — once it is
#      released it is independent and is_subject_type no longer applies.
#   3. Release it from scope:target (its overlord), then FUNC_make_subject it to ROOT
#      with the recorded type. The release-first order is the proven idiom
#      (se_DIPLOMACY.txt:1066-1092): make_subject on an already-subordinate country
#      leaves a stale double-overlord bond, so the old bond must be dropped first.
# ======================================================================
SUBJ_QING_reparent_sub_subjects = {
	LOG_enter = { sys = SUBJ  fn = "SUBJ_QING_reparent_sub_subjects" }
	if = {
		limit = {
			exists = scope:target
			scope:target = { any_subject = { count >= 1 } }
		}
		# 1. snapshot the intermediary's subjects
		scope:target = {
			every_subject = { add_to_list = SUBJ_sub_subjects_to_reparent }
		}
		# 2/3. record type, release from the vanishing overlord, rebind to ROOT
		every_in_list = {
			list = SUBJ_sub_subjects_to_reparent
			save_scope_as = SUBJ_reparent_child
			# record current type as a flag while still a subject of scope:target
			if = { limit = { is_subject_type = tributary }                    set_variable = { name = SUBJ_reparent_type  value = flag:tributary } }
			else_if = { limit = { is_subject_type = nominal_vassal }           set_variable = { name = SUBJ_reparent_type  value = flag:nominal_vassal } }
			else_if = { limit = { is_subject_type = feudatory }               set_variable = { name = SUBJ_reparent_type  value = flag:feudatory } }
			else_if = { limit = { is_subject_type = semi_autonomous_governorship } set_variable = { name = SUBJ_reparent_type  value = flag:semi_autonomous_governorship } }
			else_if = { limit = { is_subject_type = autonomous_governorship }  set_variable = { name = SUBJ_reparent_type  value = flag:autonomous_governorship } }
			else_if = { limit = { is_subject_type = client_state }            set_variable = { name = SUBJ_reparent_type  value = flag:client_state } }
			else_if = { limit = { is_subject_type = sinosphere_tributary }     set_variable = { name = SUBJ_reparent_type  value = flag:sinosphere_tributary } }
			else_if = { limit = { is_subject_type = protectorate }            set_variable = { name = SUBJ_reparent_type  value = flag:protectorate } }
			# fallback: any remaining / off-ladder subject type re-binds as feudatory (the
			# closest Qing-ladder fit for an Inner-Asian dependency); logged so an
			# unexpected type is visible. The Qing-relevant types above (incl.
			# sinosphere_tributary, the canonical 朝貢 tie, and protectorate) are
			# preserved exactly; only truly off-ladder types (colony/territory/etc.) fall here.
			else = {
				set_variable = { name = SUBJ_reparent_type  value = flag:feudatory }
				LOG_line = { sys = SUBJ  msg = "reparent: had an off-ladder type; rebinding as feudatory under" }
			}
			# free from the vanishing intermediary
			overlord = { release_subject = scope:SUBJ_reparent_child }
			# rebind to ROOT, preserving the recorded type
			if = {
				limit = { has_variable = SUBJ_reparent_type  var:SUBJ_reparent_type = flag:tributary }
				FUNC_make_subject = { overlord = ROOT  target = scope:SUBJ_reparent_child  type = tributary }
			}
			else_if = {
				limit = { has_variable = SUBJ_reparent_type  var:SUBJ_reparent_type = flag:nominal_vassal }
				FUNC_make_subject = { overlord = ROOT  target = scope:SUBJ_reparent_child  type = nominal_vassal }
			}
			else_if = {
				limit = { has_variable = SUBJ_reparent_type  var:SUBJ_reparent_type = flag:semi_autonomous_governorship }
				FUNC_make_subject = { overlord = ROOT  target = scope:SUBJ_reparent_child  type = semi_autonomous_governorship }
			}
			else_if = {
				limit = { has_variable = SUBJ_reparent_type  var:SUBJ_reparent_type = flag:autonomous_governorship }
				FUNC_make_subject = { overlord = ROOT  target = scope:SUBJ_reparent_child  type = autonomous_governorship }
			}
			else_if = {
				limit = { has_variable = SUBJ_reparent_type  var:SUBJ_reparent_type = flag:client_state }
				FUNC_make_subject = { overlord = ROOT  target = scope:SUBJ_reparent_child  type = client_state }
			}
			else_if = {
				limit = { has_variable = SUBJ_reparent_type  var:SUBJ_reparent_type = flag:sinosphere_tributary }
				FUNC_make_subject = { overlord = ROOT  target = scope:SUBJ_reparent_child  type = sinosphere_tributary }
			}
			else_if = {
				limit = { has_variable = SUBJ_reparent_type  var:SUBJ_reparent_type = flag:protectorate }
				FUNC_make_subject = { overlord = ROOT  target = scope:SUBJ_reparent_child  type = protectorate }
			}
			else = {
				FUNC_make_subject = { overlord = ROOT  target = scope:SUBJ_reparent_child  type = feudatory }
			}
			remove_variable = SUBJ_reparent_type
			LOG_line = { sys = SUBJ  msg = "reparent: transferred from absorbed to overlord" }
		}
		# clean up the working list
		every_in_list = {
			list = SUBJ_sub_subjects_to_reparent
			remove_from_list = SUBJ_sub_subjects_to_reparent
		}
		LOG_exit = { sys = SUBJ  fn = "SUBJ_QING_reparent_sub_subjects"  result = OK }
	}
	else = {
		# no sub-subjects to inherit — the common case; trace and no-op
		LOG_exit = { sys = SUBJ  fn = "SUBJ_QING_reparent_sub_subjects"  result = "NONE (subject held no sub-subjects)" }
	}
}

# ======================================================================
# [#task-7/#task-8 2026-08-04] SUBJ_QING_resolve_integ_actors — resolve the two on-the-ground
# actors (the resident Amban + the imperial garrison) whose presence, loyalty, and relative
# strength gate the subject-integration event choices. Called from each integration event's
# immediate at COUNTRY scope (ROOT = CHI); expects scope:target = the subject governorship.
# Reuses the EXISTING per-subject bookkeeping — qing_amban_here (se_QING_AMBAN.txt, a var on the
# subject holding the resident amban's CHARACTER scope) and the CHI garrison standing on the
# subject's soil (the same every_army + unit_location.owner walk QING_fgar_scan uses in
# se_QING_FRONTIER.txt) — so NO new persistent state is introduced. Writes only SCRATCH country
# vars on ROOT, cleared by SUBJ_QING_clear_integ_actors at the event tail. All comparisons use a
# LITERAL RHS (the RHS-comparison rule); the amban char scope is reached via the proven
# var-holding-a-char-scope idiom (qing_amban_here, se_QING_AMBAN.txt:223).
#
# Outputs (all on ROOT):
#   integ_amban_present   1 if a live resident amban sits in scope:target (else var absent)
#   scope:integ_amban     saved character scope of that amban (only when present)
#   integ_amban_band      0 wavering (loyalty<40) / 1 steady (40-70) / 2 staunch (>70)
#   integ_garrison_size   summed unit_size of CHI armies standing on scope:target's soil (0 = none)
#   scope:integ_garrison_cmd  commander of the LARGEST garrison army (only when a garrison w/ a live cmd)
#   integ_subject_size    the subject's manpower ÷ 500 (cohort-equivalents of its potential rising)
#   integ_garrison_edge   0 subject outweighs / no garrison, 1 rough parity (2g>=s), 2 dominant (g>=s)
# ======================================================================
SUBJ_QING_resolve_integ_actors = {
	LOG_enter = { sys = SUBJ  fn = "SUBJ_QING_resolve_integ_actors" }
	# start clean each resolve (idempotent — an event may re-run the immediate on refire)
	SUBJ_QING_clear_integ_actors = yes
	# init the band var UNCONDITIONALLY so every option trigger/effect can read it without a
	# missing-var fetch: 0 also cleanly means "no amban present" (present flag distinguishes an
	# ABSENT amban from a WAVERING one, but band 0 covers both for the force-branch flavour).
	set_variable = { name = integ_amban_band  value = 0 }

	# --- AMBAN: present + loyalty band --------------------------------------------------
	if = {
		limit = {
			exists = scope:target
			scope:target = {
				has_variable = qing_amban_here
				# [#task-20 BUGFIX 2026-08-04] the resident amban's employer is the SUBJECT, not ROOT/CHI
				# (QING_amban_wire does move_country = $subject$, se_QING_AMBAN.txt:167/189). The old
				# guard tested employer = ROOT, NEVER true for a posted amban, so integ_amban_present was
				# never set and the amban-gated options (.10.d/.12.d/.41.d) never showed even with a
				# resident present (the reported Mukden case). Test employer = the subject instead.
				var:qing_amban_here = { is_alive = yes  employer = scope:target }
			}
		}
		scope:target.var:qing_amban_here = { save_scope_as = integ_amban }
		set_variable = { name = integ_amban_present  value = 1 }
		# [#49 2026-08-07] stash the resident's live negotiate odds ON the amban character, so the
		# skill-check option tooltips (.10.d/.12.d/.40.c/.41.d) render REAL percentages that sum to 100
		# via the PROVEN loc form [integ_amban.MakeScope.GetVariable('X').GetValue] (the granary-tooltip
		# idiom, qing_integ.20.a.tt). Computed from the amban's OWN scope so charisma/finesse are his.
		# (Rendering a ScriptValue on a bare saved scope via GuiScope.SetRoot is NOT proven in this
		# codebase — a stored var read is, so we stash then read.) Recomputed each time the actors
		# resolve, so the shown odds always match the roll (qing_amban_negotiate_chance_svalue).
		scope:integ_amban = {
			set_variable = { name = qing_amban_negotiate_chance_shown  value = qing_amban_negotiate_chance_svalue }
			set_variable = { name = qing_amban_negotiate_fail_shown    value = qing_amban_negotiate_fail_svalue }
		}
		if = {
			limit = { scope:integ_amban = { loyalty > 70 } }
			set_variable = { name = integ_amban_band  value = 2 }
		}
		else_if = {
			limit = { scope:integ_amban = { loyalty >= 40 } }
			set_variable = { name = integ_amban_band  value = 1 }
		}
		# else leaves band = 0 (wavering resident)
		LOG_line = { sys = SUBJ  msg = "integ actors: resident amban present, loyalty band above, for subject" }
	}

	# --- GARRISON strength: sum CHI army unit_size standing on the subject's soil -------
	set_variable = { name = integ_garrison_size  value = 0 }
	every_army = {
		limit = {
			is_moving = no
			exists = unit_location
			unit_location = { exists = owner  owner = scope:target }
		}
		# accumulate into the outer country var via the proven <iterated> = { ROOT = { add = prev.X } }
		# idiom (se_QING_COUNCIL.txt:591); prev inside ROOT resolves back to the iterated army here.
		ROOT = { change_variable = { name = integ_garrison_size  add = prev.unit_size } }
	}

	# --- [#task-27 R3/req-10] GARRISON COMMANDER: save the commander of the LARGEST garrison army on
	# the subject's soil, so the .41.e march-out roll (and its War-Ministry perf coupling) can read his
	# MARTIAL + ZEAL for the clean-vs-chaotic outcome. ordered_army order_by unit_size max 1 picks the
	# principal garrison; guarded on has_commander + a live commander so a leaderless garrison saves
	# nothing (the .41.e outcome then falls back to the edge-only split — see the option). The commander
	# scope on an army is the proven read (se_QING_MINISTRY.txt:105). Skipped cleanly when no garrison.
	if = {
		limit = { var:integ_garrison_size >= 1 }
		ordered_army = {
			limit = {
				is_moving = no
				exists = unit_location
				unit_location = { exists = owner  owner = scope:target }
				has_commander = yes
				commander = { is_alive = yes }
			}
			order_by = unit_size
			max = 1
			commander = { save_scope_as = integ_garrison_cmd }
		}
	}

	# --- SUBJECT strength: the subject's POTENTIAL RISING, measured by its MANPOWER pool -----
	# [#task-20 refine 2026-08-04] was the subject's STANDING-army unit_size, which read ~0 for a
	# frontier governorate that fields no peacetime army — so a subject that levies a rising the moment
	# it revolts (e.g. Yili, which raised 500 the instant war was declared) looked defenceless and the
	# garrison always read dominant. Potential LEVY strength is the right measure but is NOT script-
	# readable (levy_size exists only as a modifier/law key in vanilla + both oracle repos, never as a
	# value/trigger). So use MANPOWER as the accessible proxy for how large a rising the subject can
	# field. Garrison strength is in COHORTS (unit_size); manpower is a men-scale pool, and 1 cohort =
	# 500 manpower in this measure, so convert the subject's manpower to cohort-equivalents (÷500) to
	# compare on one axis and reuse the edge bands below. Read scope:target.manpower directly in the
	# value field (the proven scope:X.<property> idiom, se_QING_MINISTRY.txt:201 etc.). manpower is
	# always readable on a country scope; default 0.
	set_variable = { name = integ_subject_size  value = scope:target.manpower }
	change_variable = { name = integ_subject_size  divide = 500 }

	# --- GARRISON EDGE band: garrison cohorts vs the subject's manpower-in-cohorts. Compare via a
	#     scratch delta so the RHS stays a literal (the RHS-comparison rule bars a var-ref on a
	#     comparison RHS). [#task-27 R4 2026-08-04] RECALIBRATED off the old fixed ±4/-3 thresholds,
	#     which were tuned to Yili's 500-manpower special case (size 1) and misfired at scale: any
	#     subject with a few thousand manpower gave a subject_size in the teens, so a real banner
	#     garrison could never reach subject+4 and the garrison options hid everywhere but Yili. The
	#     bands are now RATIO-based and scale-robust:
	#       dominant (edge 2) iff garrison_cohorts >= subject_size          (delta  = g − s   >= 0)
	#       parity   (edge 1) iff garrison_cohorts*2 >= subject_size        (delta2 = 2g − s  >= 0)
	#       else 0 (the subject's potential rising clearly outweighs the garrison)
	#     i.e. a garrison at least HALF the subject's manpower-pool-in-cohorts counts as parity, and
	#     one matching it counts as dominant — robust across the real 1763 manpower distribution.
	set_variable = { name = integ_garrison_edge  value = 0 }
	if = {
		limit = { var:integ_garrison_size >= 1 }
		# delta = garrison - subject; >= 0 means the garrison matches or outweighs the potential rising.
		set_variable = { name = integ_edge_delta  value = var:integ_garrison_size }
		change_variable = { name = integ_edge_delta  subtract = var:integ_subject_size }
		# delta2 = 2*garrison - subject; >= 0 means the garrison is at least half the potential rising.
		set_variable = { name = integ_edge_delta2  value = var:integ_garrison_size }
		change_variable = { name = integ_edge_delta2  multiply = 2 }
		change_variable = { name = integ_edge_delta2  subtract = var:integ_subject_size }
		if = {
			limit = { var:integ_edge_delta >= 0 }
			set_variable = { name = integ_garrison_edge  value = 2 }   # garrison matches/outweighs => dominant
		}
		else_if = {
			limit = { var:integ_edge_delta2 >= 0 }
			set_variable = { name = integ_garrison_edge  value = 1 }   # garrison >= half => rough parity
		}
		# else stays 0 (subject clearly outweighs the garrison)
		remove_variable = integ_edge_delta
		remove_variable = integ_edge_delta2
	}
	# [#49 2026-08-07; review-fix D] stash the .41.e clean/chaotic crush odds on ROOT (the country),
	# for the tooltip to render real percentages. On ROOT — NOT the commander character — because the
	# .41.e option shows on garrison EDGE alone (no commander guard) and clear_integ_actors leaves saved
	# scopes to fall out of use, so a char-var read via a STALE integ_garrison_cmd from a prior subject
	# would surface the wrong commander's odds on a present-but-leaderless garrison. These ROOT vars are
	# cleared in clear_integ_actors, so they are always this-garrison-fresh. Set in BOTH branches so the
	# single tooltip numeric clause is always truthful. Placed AFTER integ_garrison_edge is computed.
	if = {
		limit = { exists = scope:integ_garrison_cmd }
		# COMMANDER present: his live skill-check odds. Evaluate the char-scope svalue INSIDE his scope
		# (the proven read — the event's random{ chance = svalue } does the same inside this scope), then
		# store the result up onto ROOT via the proven prev-in-ROOT idiom (se_QING_COUNCIL.txt:591).
		# read the char-scope svalue off the SAVED commander scope directly (proven form:
		# se_QING_AMBAN.txt:457 `value = scope:X.combined_stats_council_svalue`). Stays in ROOT scope;
		# the svalue evaluates in the commander's scope via the saved-scope prefix. (Review Finding 2:
		# avoids the novel prev.<svalue> form — scope:X.<svalue> has direct precedent.)
		set_variable = { name = qing_garrison_clean_crush_shown   value = scope:integ_garrison_cmd.qing_garrison_clean_crush_chance_svalue }
		set_variable = { name = qing_garrison_chaotic_crush_shown value = scope:integ_garrison_cmd.qing_garrison_chaotic_crush_chance_svalue }
	}
	else_if = {
		# LEADERLESS, dominant edge (2): the execution crushes cleanly (qing_integ.41.e else_if), so 100/0.
		limit = { var:integ_garrison_size >= 1  has_variable = integ_garrison_edge  var:integ_garrison_edge = 2 }
		set_variable = { name = qing_garrison_clean_crush_shown   value = 100 }
		set_variable = { name = qing_garrison_chaotic_crush_shown value = 0 }
	}
	else = {
		# LEADERLESS, parity or no garrison: the execution wins chaotically, so 0/100.
		set_variable = { name = qing_garrison_clean_crush_shown   value = 0 }
		set_variable = { name = qing_garrison_chaotic_crush_shown value = 100 }
	}
	LOG_line = { sys = SUBJ  msg = "integ actors: garrison size / subject size / edge band computed for subject" }
	LOG_exit = { sys = SUBJ  fn = "SUBJ_QING_resolve_integ_actors"  result = OK }
}

# [#task-27 req-7/R5 2026-08-04] MINISTRY PERFORMANCE COUPLING — the amban's and the garrison's
# performance in these integration events folds into the Lifan Yuan (理藩院, the amban's board) and
# the Ministry of War (兵部, the garrison's board) Grand Council standing respectively.
#
# The two perf meters (qing_min_perf_lifanyuan / qing_min_perf_war, se_QING_MINISTRY.txt) are
# RECOMPUTED WHOLESALE each quarterly pulse from concrete state — a bare change_variable poke on the
# meter would be OVERWRITTEN at the next recompute (the no-restoring-drift ratchet rule + the meters'
# own design). So the coupling is a PERSISTENT, SELF-EXPIRING signed input var that each recompute
# READS and folds (in its office-FILLED branch — inert while the seat is vacant, which is acceptable:
# a vacant board has no head to credit/blame). The var self-expires in 730 days (the coercive-
# absorption-flag idiom, .30.c) so a good/bad showing fades rather than ratcheting forever. These
# helpers SET (not change) the var to a fixed signed magnitude — the most recent outcome is what the
# board is currently judged on; they do not accumulate. Scope: COUNTRY (ROOT = CHI).
# ======================================================================
# Amban outcome → Lifan Yuan. $delta$ = signed magnitude (+ success, − failure, −− death).
SUBJ_QING_lifan_amban_outcome = {
	set_variable = { name = qing_lifan_recent_amban_outcome  value = $delta$  days = 730 }
	LOG_line = { sys = SUBJ  msg = "perf-fold: recent amban outcome recorded for the Lifan Yuan (delta=$delta$)" }
}
# Garrison outcome → Ministry of War. $delta$ = signed magnitude (+ clean crush, − chaotic/failed).
SUBJ_QING_war_garrison_outcome = {
	set_variable = { name = qing_war_recent_garrison_outcome  value = $delta$  days = 730 }
	LOG_line = { sys = SUBJ  msg = "perf-fold: recent garrison outcome recorded for the Ministry of War (delta=$delta$)" }
}

	if = { limit = { has_variable = integ_garrison_edge } remove_variable = integ_garrison_edge }
	if = { limit = { has_variable = integ_edge_delta }    remove_variable = integ_edge_delta }
	if = { limit = { has_variable = integ_edge_delta2 }   remove_variable = integ_edge_delta2 }

	if = { limit = { has_variable = qing_garrison_chaotic_crush_shown } remove_variable = qing_garrison_chaotic_crush_shown }
	# scope:integ_garrison_cmd is a saved scope, not a var — it simply falls out of use, like integ_amban.
}

# ======================================================================
# [#task-7 2026-08-04; REWRITTEN 2026-08-04 flood-fix] SUBJ_QING_crush_revolt_war — the "crush the
# rising" option made a REAL war, via the VANILLA independence-war path.
#
# The chieftain resists = the subject rises in an INDEPENDENCE WAR against its overlord. This is the
# engine-native model: the SUBJECT (still a subject — independence_wargoal is gated on is_subject = yes,
# common/wargoals/00_default.txt:43) declares independence_wargoal on ROOT (CHI); the player then wins
# it militarily = the crush. Scope: COUNTRY (ROOT = CHI); expects scope:target.
#
# PROVEN IDIOM (UPSTREAM): Invictus epigoni.txt:1008 — a subject declares its independence war with the
# BARE engine effect `declare_war_with_wargoal = { war_goal = independence_wargoal  province =
# <own capital>  target = <overlord> }`, from the SUBJECT scope, with NO release_subject. This is the
# proven form and is what this mirrors. (imp19c_setup.2 does the same shape for USA-on-GBR, but that is
# MOD code, not a proof — the proof is epigoni.) independence_wargoal is defined locally
# (common/wargoals/00_default.txt:43, gated is_subject = yes, which scope:target satisfies).
#
# DELIBERATELY NOT via FUNC_declare_war_with_wargoal_province: that wrapper runs an unconditional
# every_subject { add_to_war = scope:current_war_scope } loop, and current_war_scope only saves if its
# random_current_war leader-limit matches — the exact point that FAILED in the prior version and threw
# once per CHI subject (part of the 858k-line boot-18:10 flood). The bare engine verb skips that loop
# entirely, so it cannot reproduce the flood. The prior version ALSO release_subject'd + reparented +
# declared a CONQUER war (overlord-as-declarer) — all wrong for an independence war; all removed.
# ======================================================================
SUBJ_QING_crush_revolt_war = {
	LOG_enter = { sys = SUBJ  fn = "SUBJ_QING_crush_revolt_war" }
	if = {
		limit = {
			exists = scope:target
			scope:target = { is_subject_of = ROOT }
			scope:target = { any_owned_province = { count >= 1 } }
		}
		# The subject rises: it declares its independence war on the overlord (ROOT) with the bare
		# engine verb (upstream epigoni idiom). NO release_subject (independence_wargoal requires the
		# declarer to STILL be a subject) and NO reparent (nothing vanishes — the subject goes to war).
		scope:target = {
			declare_war_with_wargoal = {
				war_goal = independence_wargoal
				province = capital_scope
				target = ROOT
			}
		}
		LOG_line = { sys = SUBJ  msg = "crush-revolt: the subject rises in an INDEPENDENCE WAR against the throne" }
		LOG_exit = { sys = SUBJ  fn = "SUBJ_QING_crush_revolt_war"  result = OK }
	}
	else = {
		LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_crush_revolt_war"  reason = "target not a live landed subject — war skipped" }
	}
}

# ======================================================================
# [#task-7 2026-08-04] SUBJ_QING_marry_into_chieftain — the "marry into his family" option made a
# REAL marriage. Weds an unmarried adult of the imperial house to an unmarried adult child of the
# chieftain (scope:target.current_ruler), binding the two houses. Proven idiom (se_MARRIAGE.txt:305-372,
# Invictus me_bithynia): pick the partner child (opposite sex) via ordered_child, save it, then
# marry_character on our chosen child. Fully guarded: if either side has no eligible child the co-opt
# still lands (gold/prestige, in the calling option) — only the wedding itself is skipped. Scope:
# COUNTRY (ROOT = CHI); expects scope:target with a live ruler.
# ======================================================================
SUBJ_QING_marry_into_chieftain = {
	LOG_enter = { sys = SUBJ  fn = "SUBJ_QING_marry_into_chieftain" }
	if = {
		limit = {
			exists = scope:target
			scope:target = { exists = current_ruler  current_ruler = { is_alive = yes } }
			exists = current_ruler
		}
		# 1. the chieftain's child — an unmarried adult daughter (opposite sex to our groom). order_by
		#    age + max=1 picks one eligible child deterministically (same idiom as se_MARRIAGE.txt:316).
		scope:target.current_ruler = {
			ordered_child = {
				limit = { is_alive = yes  is_adult = yes  is_married = no  prisoner = no  is_female = yes }
				order_by = age
				max = 1
				save_scope_as = integ_bride
			}
		}
		# 2. our house's groom — an unmarried adult male of the imperial family; wed to the bride.
		if = {
			limit = { exists = scope:integ_bride }
			current_ruler = {
				ordered_child = {
					limit = { is_alive = yes  is_adult = yes  is_married = no  prisoner = no  is_male = yes }
					order_by = age
					max = 1
					marry_character = scope:integ_bride
					save_scope_as = integ_groom
				}
			}
			if = {
				limit = { exists = scope:integ_groom }
				LOG_line = { sys = SUBJ  msg = "co-opt: wed an imperial prince to the chieftain's daughter, subject" }
			}
			else = {
				LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_marry_into_chieftain"  reason = "no eligible imperial groom — wedding skipped (co-opt gold/prestige still applied)" }
			}
		}
		else = {
			LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_marry_into_chieftain"  reason = "chieftain has no eligible daughter — wedding skipped (co-opt gold/prestige still applied)" }
		}
		LOG_exit = { sys = SUBJ  fn = "SUBJ_QING_marry_into_chieftain"  result = OK }
	}
	else = {
		LOG_fail = { sys = SUBJ  fn = "SUBJ_QING_marry_into_chieftain"  reason = "no live chieftain ruler / no imperial ruler — wedding skipped" }
	}
}


# ---- [#314 B17] TRIBUTE SIZE selector -----------------------------------
# Repurposes the three formerly-dead "Overseer" policy toggles into a per-subject
# small / medium / large tribute-demand selector. The three tiers are mutually
# exclusive: setting one strips the other two first, so a subject always carries at
# most one qing_tribute_* modifier. The modifier lives on scope:target (the subject) and
# carries only the SUBJECT-FACING loyalty_to_overlord effect (heavier demand => the
# subject resents US more). The income side is NOT a modifier — tribute_income_modifier
# is overlord-global and inert on a leaf subject (the original review-caught bug) — it is
# a CONCRETE quarterly treasury transfer (QING_subject_collect_tribute below), per the
# concrete-over-abstract rule. $tier$ = small|medium|large.
QING_subject_set_tribute = {
	LOG_enter = { sys = SUBJ  fn = "QING_subject_set_tribute -> $tier$" }
	scope:target = {
		# strip any existing tier first (exclusive selection)
		if = { limit = { has_country_modifier = qing_tribute_small }   remove_country_modifier = qing_tribute_small }
		if = { limit = { has_country_modifier = qing_tribute_medium }  remove_country_modifier = qing_tribute_medium }
		if = { limit = { has_country_modifier = qing_tribute_large }   remove_country_modifier = qing_tribute_large }
		add_country_modifier = { name = qing_tribute_$tier$  duration = -1 }
		# remember the current tier so the GUI toggles can show which one is active
		set_variable = { name = qing_tribute_tier  value = flag:$tier$ }
	}
	LOG_exit = { sys = SUBJ  fn = "QING_subject_set_tribute"  result = OK }
}

# ---- [#314 B17] CONCRETE quarterly tribute collection --------------------
# Called on the Qing quarterly pulse (ROOT = c:CHI). For every subject carrying a tribute
# tier, transfer gold from the subject's treasury to the Qing scaled by tier and by the
# subject's income (a fraction of its monthly income, so a richer march pays more). The
# transfer is capped by what the subject actually holds, so a broke subject simply pays
# what it can rather than going negative. This is the real income lever the tooltips
# promise; the qing_tribute_* modifier only carries the loyalty cost. No tier => no
# collection (the historical default is the customary light rendering, modelled as zero
# extra transfer). Tiers: small 2% / medium 5% / large 9% of the subject's monthly income
# per quarter.
QING_subject_collect_tribute = {
	LOG_enter = { sys = SUBJ  fn = "QING_subject_collect_tribute" }
	every_subject = {
		# [#27] never drain a frontier_protectorate MARCH — it is FUNDED by the reverse-tribute subsidy
		# (QING_march_pay_subsidy), not a tribute source. Defense-in-depth: the tribute-tier GUI buttons
		# already exclude marches, so a march should never carry qing_tribute_tier, but guard here too.
		limit = { has_variable = qing_tribute_tier  NOT = { is_subject_type = frontier_protectorate } }
		# fraction of the subject's monthly income, by tier
		save_temporary_scope_as = trib_subject
		if =      { limit = { var:qing_tribute_tier = flag:small }   set_variable = { name = qing_trib_rate  value = 0.02 } }
		else_if = { limit = { var:qing_tribute_tier = flag:medium }  set_variable = { name = qing_trib_rate  value = 0.05 } }
		else_if = { limit = { var:qing_tribute_tier = flag:large }   set_variable = { name = qing_trib_rate  value = 0.09 } }
		else =    { set_variable = { name = qing_trib_rate  value = 0 } }
		# quarterly haul = 3 months' income * rate, clamped to the subject's treasury.
		# [T1-C review-fix] read 3-months' income via the script_value three_months_income_svalue,
		# NOT a bare `value = has_monthly_income` — the latter is parsed as a variable name (none
		# exists), so it silently resolved to 0 AND spammed "Failed to fetch variable" each quarter
		# per subject, killing the whole tribute-gold transfer. A script_value evaluates
		# has_monthly_income as a game read in this (subject) scope. Then scale by the tier rate.
		set_variable = { name = qing_trib_amt  value = three_months_income_svalue }
		change_variable = { name = qing_trib_amt  multiply = var:qing_trib_rate }
		# [BT-11] DEFAULT DETECTION — before the affordability clamp below. A subject whose
		# treasury cannot cover a MEANINGFUL owed sum (>= 5, so a rounding-level shortfall does
		# not count) has defaulted on its tribute this quarter. Stamp a short-lived marker (auto-
		# expires in 100d, so it survives to the next quarterly pulse but not indefinitely) that
		# the Lifan-Yuan default check (QING_tribute_default_check, se_QING_TRIBUTE.txt) reads to
		# route the shortfall through the tributary order and fire the punish/forgive dilemma. At
		# the High Qing zenith subjects are solvent (owed is a small fraction of 3 months' income),
		# so this is inert until real 19th-c. fiscal stress — like the breakaway/interference checks.
		if = {
			limit = {
				# [logfix 2026-07-17] compare treasury to the owed sum via the NAMED svalue —
				# an inline { value = var:qing_trib_amt } block on a comparison RHS is illegal.
				treasury < qing_trib_amt_svalue
				var:qing_trib_amt >= 5
			}
			set_variable = { name = qing_tribute_defaulted  value = 1  days = 100 }
		}
		if = {
			limit = { treasury < qing_trib_amt_svalue }
			set_variable = { name = qing_trib_amt  value = treasury }
		}
		if = {
			limit = { var:qing_trib_amt > 0 }
			add_treasury = { value = var:qing_trib_amt  multiply = -1 }
			ROOT = { add_treasury = scope:trib_subject.var:qing_trib_amt }
			LOG_line = { sys = SUBJ  msg = "tribute collected this quarter from" }
		}
		remove_variable = qing_trib_rate
		remove_variable = qing_trib_amt
	}
	LOG_exit = { sys = SUBJ  fn = "QING_subject_collect_tribute"  result = OK }
}
