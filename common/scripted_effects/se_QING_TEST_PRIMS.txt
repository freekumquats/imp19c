# *** TEMPORARY — REMOVE BEFORE MERGE TO master. ***
# Wired from a temporary on_game_initialized on_action (qing_test_prims_on_actions.txt).
#
# PURPOSE: the march-founding path (se_QING_MARCH.txt QING_found_march) uses the PROVEN
# LAND_release_from_list wrapper, NOT the raw create_country -> change_country_tag combo,
# because the design doc (§5/§8.0) flags the predefined-tag mint + change_country_tag-as-a-
# unit as UNPROVEN (boot-safe on paper, never actually exercised in-game). This probe runs
# BOTH raw primitives once at game start, each preceded by a LOG_line, so a boot either:
#   • logs "[QING] TESTPRIM ... OK" lines through to the end  → both primitives are viable, or
#   • crashes / stops logging at a specific TESTPRIM step      → that primitive is the culprit.
#
# SAFETY: runs ONCE (global flag qing_test_prims_done), CHI-only, and takes only an UNOWNED
# province (Salt Lake City p:59 — verified not in any own_control_core at 1763), so it steals
# no land from any live country. The throwaway country is re-tagged into JPN (a dormant,
# registered tag with NO 00_default block — the proven se_JAPAN_BOSHIN change_country_tag=JPN
# precedent). In a viability boot the JPN side effects don't matter; this is not for real play.
#
# Proven references: create_country + set_as_ruler (common/scripted_guis/release_subject_
# button.txt); change_country_tag = <predefined dormant tag> (se_JAPAN_BOSHIN.txt:201, ROOT
# scope). LOG_line / LOG_fail (se_LOG.txt), sys = QING.
########################################################################

QING_TEST_country_primitives = {
	# run once, ever
	if = {
		limit = { NOT = { has_global_variable = qing_test_prims_done } }
		set_global_variable = qing_test_prims_done
		LOG_line = { sys = QING  msg = "TESTPRIM start: probing create_country + change_country_tag at game init for" }

		# --- guard: the test province must exist and be unowned (else skip cleanly) ---
		if = {
			limit = { exists = p:59  p:59 = { has_owner = no } }
			LOG_line = { sys = QING  msg = "TESTPRIM step 1 OK: test province p:59 exists and is unowned for" }

			# ============================================================
			# TEST A — raw create_country: mint a throwaway country + seat a ruler,
			# then give it the unowned province. Mirrors the proven release_subject_
			# button create_country + set_as_ruler shape.
			# ============================================================
			LOG_line = { sys = QING  msg = "TESTPRIM step 2: about to call create_country for" }
			create_country = {
				save_scope_as = qing_test_country
				create_character = {
					culture = root.culture
					religion = root.religion
					female = no
					save_scope_as = qing_test_ruler
				}
				set_as_ruler = scope:qing_test_ruler
			}
			LOG_line = { sys = QING  msg = "TESTPRIM step 3 OK: create_country returned a country scope for" }

			# give the new country the unowned province (avoids the landless-with-ruler window)
			p:59 = { set_owned_by = scope:qing_test_country }
			LOG_line = { sys = QING  msg = "TESTPRIM step 4 OK: transferred unowned p:59 to the test country for" }

			# validate: the created country now exists and owns >=1 province
			if = {
				limit = { exists = scope:qing_test_country  scope:qing_test_country = { any_owned_province = { count >= 1 } } }
				LOG_line = { sys = QING  msg = "TESTPRIM step 5 OK: test country exists and owns >=1 province (create_country VIABLE) for" }

				# ========================================================
				# TEST B — raw change_country_tag: re-tag the throwaway dynamic country
				# into the dormant predefined tag JPN (Boshin precedent). Run IN the test
				# country's scope. This is the exact primitive the march-mint alternative
				# would rely on.
				# ========================================================
				scope:qing_test_country = {
					LOG_line = { sys = QING  msg = "TESTPRIM step 6: about to change_country_tag = JPN for" }
					change_country_tag = JPN
					LOG_line = { sys = QING  msg = "TESTPRIM step 7 OK: change_country_tag = JPN returned (change_country_tag VIABLE) for" }
				}

				# confirm c:JPN now exists and holds the province
				if = {
					limit = { exists = c:JPN  c:JPN = { owns = 59 } }
					LOG_line = { sys = QING  msg = "TESTPRIM step 8 OK: c:JPN exists and owns p:59 — BOTH primitives fully viable for" }
				}
				else = {
					LOG_fail = { sys = QING  fn = "QING_TEST_country_primitives"  reason = "change_country_tag ran but c:JPN does not own p:59 afterward — change_country_tag may not carry provinces for" }
				}
			}
			else = {
				LOG_fail = { sys = QING  fn = "QING_TEST_country_primitives"  reason = "create_country / set_owned_by did not yield a landed country — create_country path suspect for" }
			}
		}
		else = {
			LOG_fail = { sys = QING  fn = "QING_TEST_country_primitives"  reason = "test province p:59 missing or already owned; primitive probe skipped for" }
		}
		LOG_line = { sys = QING  msg = "TESTPRIM end for" }
	}
}
