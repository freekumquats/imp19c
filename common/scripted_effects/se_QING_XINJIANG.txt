# The 1763 CONSOLIDATION layer for the New Dominion (新疆). After the Qianlong
# conquest of Dzungaria (準噶爾) and Altishahr (回疆) 1755-59, the throne faced the
# decades-long WORK of holding the far west: the 善後 ("settling the aftermath"). This
# feature is the emperor's LEVERS to make the new frontier pacified, integrated and
# self-sustaining — or to bungle it into the revolt that history delivered in 1826
# (Jahangir Khoja, 張格爾之亂).

#     (QING_ili_apply_control_band runs EVERY pulse, unconditionally — so any control

# ======================================================================
# QING_xj_init — seed the consolidation bookkeeping once + the 1763 opening beg corps.
# Scope: COUNTRY (CHI). Idempotent (guards on a dedicated flag, not a count — the pulse
# recompute runs at game start and would set the count first; the harem/#336 lesson).
# The qing_xinjiang_control meter itself is seeded by QING_ili_init (value 40); #367 only
# seeds its OWN state here + the historical begs, then recomputes so the composite score
# and the Lifan fold are live from turn 1.
# ======================================================================
QING_xj_init = {
	LOG_enter = { sys = QING  fn = "QING_xj_init" }
	if = {
		limit = { NOT = { has_variable = qing_xj_initialized } }
		set_variable = { name = qing_xj_initialized  value = 1 }
		set_variable = { name = qing_xj_tuntian        value = 0 }
		set_variable = { name = qing_xj_beg_count       value = 0 }
		set_variable = { name = qing_xj_beg_venal_count value = 0 }
		set_variable = { name = qing_xj_consolidation   value = 50 }
		set_variable = { name = qing_xj_neglect_quarters value = 0 }   # [meter-concretize #10B] seed before the pulse accumulates (read-before-set guard)
		clear_variable_list = qing_xj_begs
		# the 1763 opening beg corps — two real loyal begs of the early consolidation:
		#   額敏和卓 (Emin Khoja, the Turfan leader who aided the Qing conquest and was
		#     made a 郡王 prince — the model loyal beg), and
		#   鄂對 (Odui, hakim beg of Aksu, a pillar of the early Altishahr administration).
		# Only if the Kashgaria subject actually exists at this start (guarded — a game
		# where XNG was released/absorbed simply opens with an empty corps).
		if = {
			limit = { exists = c:XNG }
			QING_xj_seed_beg = { nick = "NICKNAME_QING_XJ_EMINKHOJA"  fin = 6  mar = 6  cha = 7  aged = 69 }
			QING_xj_seed_beg = { nick = "NICKNAME_QING_XJ_ODUI"       fin = 6  mar = 5  cha = 6  aged = 58 }
		}
		# recompute at once so the composite score is tallied and the Lifan fold reads a
		# real value FROM TURN 1 (nothing folds this into a game-start dispatcher, so init
		# must recompute itself — the #336 init-recompute lesson).
		QING_xj_recompute_consolidation = yes
	}
	LOG_exit = { sys = QING  fn = "QING_xj_init"  result = OK }
}

# ======================================================================
# QING_xj_seed_beg — create ONE named historical beg and enrol him in the corps. Scope:
# COUNTRY (CHI). $nick$ = loc nickname key, $fin$/$mar$/$cha$ = stat line, $aged$ = age.
# A Uighur Muslim notable, moved to the Kashgaria subject (his employer), bound to the
# throne. Marked qing_is_xj_beg. Mirrors QING_amban_post's mint (#90 rule — no modifiers
# inside create_character; set_as_minor_character + move_country in the saved scope).
# ======================================================================
QING_xj_seed_beg = {
	create_character = {
		age = $aged$
		culture = uighur
		religion = sunni
		add_finesse = $fin$
		add_martial = $mar$
		add_charisma = $cha$
		add_zeal = 3
		save_scope_as = qing_new_beg
	}
	scope:qing_new_beg = {
		set_as_minor_character = THIS
		move_country = c:XNG
		set_variable = { name = qing_is_xj_beg  value = 1 }
		# a beg confirmed by the throne is bound to it (the loyal-notable model).
		QING_char_bind = yes
		QING_roster_finalize = { nick = $nick$ }
	}
	add_to_variable_list = { name = qing_xj_begs  target = scope:qing_new_beg }
	change_variable = { name = qing_xj_beg_count  add = 1 }
	LOG_line = { sys = QING  msg = "xinjiang: a historical beg is confirmed on the New Dominion" }
}

# ======================================================================
# QING_xj_recompute_consolidation — rebuild the beg corps fresh from the live marked
# characters (alive + employer = c:XNG + NOT a great-office holder — the #362-R2 exclusion),
# tally the count + the venal count, then compute the composite frontier-health score
# qing_xj_consolidation (0..100) from the master grip + institutional depth:
#   base   = qing_xinjiang_control (the ILI-owned grip meter — the spine)
#   + begs  : +4 per loyal beg (a staffed south)         (corps cap 5  -> up to +20)
#   + 屯田  : +3 per tuntian colony, contribution capped   (up to +18)
#   + 協餉  : +8 if the xiexiang subsidy is flowing
#   - venal : -6 per turned beg (a defector on the frontier)
# Clamped 0..100. This score is what folds into the Lifan Yuan office-holder (additively,
# inside that office's own recompute). Self-healing; bounded O(characters). Scope: CHI.
# ======================================================================
QING_xj_recompute_consolidation = {
	# [crash-fix] DO NOT call QING_xj_init here. init calls THIS effect (line ~92), so a
	# QING_xj_init back-call closes a textual recursion init<->recompute. At RUNTIME the
	# qing_xj_initialized guard makes it a harmless no-op, but a scripted_gui button inlines
	# its whole effect chain at COMPILE time — where no guard exists — so the parser recurses
	# forever and stack-overflows (EXCEPTION_ACCESS_VIOLATION, no logged error). This was the
	# 1763-merge boot crash: QING_xinjiang_panel.txt:29 inlines this effect. init runs
	# unconditionally at game start (on_game_initialized) before any standalone recompute, so
	# the seeded vars always exist by the time this runs; the removed call was pure redundancy.
	clear_variable_list = qing_xj_begs
	set_variable = { name = qing_xj_beg_count       value = 0 }
	set_variable = { name = qing_xj_beg_venal_count value = 0 }
	if = {
		limit = { exists = c:XNG }
		every_character = {
			limit = {
				has_variable = qing_is_xj_beg
				is_alive = yes
				employer = c:XNG
				# a beg elevated into a great office (rare, but the office model allows it)
				# LEAVES the corps — he is on the council now, not holding the frontier.
				NOT = { has_variable = qing_office_held }
			}
			ROOT = {
				add_to_variable_list = { name = qing_xj_begs  target = prev }
				change_variable = { name = qing_xj_beg_count  add = 1 }
			}
			if = {
				limit = { has_variable = qing_xj_beg_venal }
				ROOT = { change_variable = { name = qing_xj_beg_venal_count  add = 1 } }
			}
		}
	}
	# --- composite score. [meter-concretize #10B 2026-08-06] Consolidation = control + admin_bias ONLY.
	# control is now a CONCRETE DERIVE (QING_xj_derive_control) of begs/tuntian/xiexiang/venal/secured/etc.,
	# so the old +4·begs / +3·tuntian / +8·xiexiang / −6·venal terms would DOUBLE-COUNT those objects into the
	# Lifan Yuan fold (se_QING_MINISTRY.txt) → the Grand Council. Every concrete input enters exactly once, via
	# the control base. (review CRIT-1/CRIT-4.)
	set_variable = { name = qing_xj_consol_scratch  value = 0 }
	if = { limit = { has_variable = qing_xinjiang_control }  change_variable = { name = qing_xj_consol_scratch  add = var:qing_xinjiang_control } }
	# [law #26 bias 2026-07-24] Xinjiang Administration law biases the consolidation score BEFORE the clamp
	# (provincialize +10 / military-farm +5 / beg-indirect 0). Guarded → default (unset) byte-identical.
	if = { limit = { has_variable = qing_xj_admin_bias }  change_variable = { name = qing_xj_consol_scratch  add = var:qing_xj_admin_bias } }
	# clamp 0..100 into the published score.
	if = { limit = { var:qing_xj_consol_scratch < 0 }   set_variable = { name = qing_xj_consol_scratch  value = 0 } }
	if = { limit = { var:qing_xj_consol_scratch > 100 } set_variable = { name = qing_xj_consol_scratch  value = 100 } }
	set_variable = { name = qing_xj_consolidation  value = var:qing_xj_consol_scratch }
	LOG_line = { sys = QING  msg = "xinjiang: consolidation score recomputed" }
}

# ======================================================================
# [meter-concretize #10B 2026-08-06] QING_xj_derive_control — qing_xinjiang_control is now a PURE ONE-WAY
# CONCRETE DERIVE: a straight set_variable from the real Xinjiang objects each pulse (and after any beat that
# changes those objects). Abstract FOLLOWS concrete — control MODELS grip, never drives it. NO accumulator
# nudges, NO drift. Scope: country (ROOT = CHI). Areas iterated owner-independently (oases are ILI/XNG-held).
# WEIGHTS are PLACEHOLDER-playtest — build-probe LOG the components and pin against the real 1763 value:
#   1763 opens ~40 (ILI-subject 30 + 2 begs ×4 = 38); full-build clamps ~100 (≥85 capstone reachable);
#   a neglected/rotted frontier derives ≤30 (khoja trigger). See DESIGN §11.1/§11.7.
# ======================================================================
QING_xj_derive_control = {
	set_variable = { name = qing_xj_control_tmp  value = 0 }
	# ILI subjecthood — the spine (30).
	if = {
		limit = { exists = c:ILI  c:ILI = { is_subject_of = ROOT } }
		change_variable = { name = qing_xj_control_tmp  add = 30 }
	}
	# loyal beg corps: +4 each (0..5 → 0..20).
	set_variable = { name = qing_xj_ctl_term  value = var:qing_xj_beg_count }
	change_variable = { name = qing_xj_ctl_term  multiply = 4 }
	change_variable = { name = qing_xj_control_tmp  add = var:qing_xj_ctl_term }
	# tuntian military colonies — count the REAL qing_xj_tuntian_colony province modifiers (NOT the int, which
	# the rot path may have decremented in a separate tick), +3 each.
	set_variable = { name = qing_xj_ctl_term  value = 0 }
	area:Dzungaria = { every_area_province = { limit = { has_province_modifier = qing_xj_tuntian_colony }  ROOT = { change_variable = { name = qing_xj_ctl_term  add = 3 } } } }
	area:Tarim     = { every_area_province = { limit = { has_province_modifier = qing_xj_tuntian_colony }  ROOT = { change_variable = { name = qing_xj_ctl_term  add = 3 } } } }
	change_variable = { name = qing_xj_control_tmp  add = var:qing_xj_ctl_term }
	# [#19 2026-08-07] CONCRETE GARRISON term, GATED by xiexiang (funding). Replaces the old flat
	# xiexiang +10 with: count the real historical garrison seats on Xinjiang soil (qing_hist_garrison_prov,
	# the permanent province stamp the OOB garrison raise leaves — imp19c_effects_legion_setup.txt:130/156;
	# #21's guard-widen means Kashgar now carries it), +3 each, capped at +12, BUT ONLY WHEN THE XIEXIANG
	# SUBSIDY FLOWS. Rationale (review): xiexiang IS the garrison-funding proxy ("paid the GARRISON, not the
	# begs", :442) — so make it a GATE on the garrison-existence term, NOT a parallel additive (that would
	# double-score built+paid). Crucially this keeps the term DEGRADABLE: cut the subsidy → garrison term → 0
	# → control can fall into the khoja ≤30 band on a neglected frontier (the intended rot path preserved).
	# The +3/seat matches the tuntian/secured terms; the +12 cap (~4 seats) prevents the many-garrison North
	# from swamping the 0..100 meter. Counted over the same two areas as the tuntian/secured terms.
	if = {
		limit = { has_country_modifier = qing_xj_xiexiang }
		set_variable = { name = qing_xj_ctl_term  value = 0 }
		area:Dzungaria = { every_area_province = { limit = { has_variable = qing_hist_garrison_prov }  ROOT = { change_variable = { name = qing_xj_ctl_term  add = 3 } } } }
		area:Tarim     = { every_area_province = { limit = { has_variable = qing_hist_garrison_prov }  ROOT = { change_variable = { name = qing_xj_ctl_term  add = 3 } } } }
		if = { limit = { var:qing_xj_ctl_term > 12 }  set_variable = { name = qing_xj_ctl_term  value = 12 } }
		change_variable = { name = qing_xj_control_tmp  add = var:qing_xj_ctl_term }
	}
	# secured oasis provinces: +3 each (beat-stamped grip on the map).
	set_variable = { name = qing_xj_ctl_term  value = 0 }
	area:Dzungaria = { every_area_province = { limit = { has_province_modifier = qing_xinjiang_prov_secured }  ROOT = { change_variable = { name = qing_xj_ctl_term  add = 3 } } } }
	area:Tarim     = { every_area_province = { limit = { has_province_modifier = qing_xinjiang_prov_secured }  ROOT = { change_variable = { name = qing_xj_ctl_term  add = 3 } } } }
	change_variable = { name = qing_xj_control_tmp  add = var:qing_xj_ctl_term }
	# frontier secured (Zeng Jize's diplomatic recovery — a durable win referent): +10.
	if = { limit = { has_country_modifier = qing_ili_frontier_secured }  change_variable = { name = qing_xj_control_tmp  add = 10 } }
	# [#13 2026-08-07] CENTRAL-ASIA DOMINION term: subjugating the neighbouring Silk Road khanates (KOK/BUK/KHV)
	# + Kazakh hordes (GKH/ORT/KSH) secures the pacified hinterland behind Xinjiang → firmer grip on the frontier.
	# +5 per subjugated khanate (the Fergana nexus, cap +15 for all three), +2 per subjugated horde (steppe buffer).
	# is_subject_of = ROOT is the concrete on-map test (NOT an abstract "pacified" counter — concrete-over-abstract);
	# added to qing_xj_control_tmp BEFORE the :282 clamp (the #10B pattern — a term in the derive, not a post-publish
	# nudge, so it is re-derived each pulse and never erased). is_subject_of is non-recursive: KSH (an ORT feudatory
	# at 1763) counts only once QING_ca_embrace_kazakh makes it a DIRECT CHI subject (design LOW-8, intended).
	set_variable = { name = qing_xj_ctl_term  value = 0 }
	if = { limit = { exists = c:KOK  c:KOK = { is_subject_of = ROOT } }  change_variable = { name = qing_xj_ctl_term  add = 5 } }
	if = { limit = { exists = c:BUK  c:BUK = { is_subject_of = ROOT } }  change_variable = { name = qing_xj_ctl_term  add = 5 } }
	if = { limit = { exists = c:KHV  c:KHV = { is_subject_of = ROOT } }  change_variable = { name = qing_xj_ctl_term  add = 5 } }
	if = { limit = { exists = c:GKH  c:GKH = { is_subject_of = ROOT } }  change_variable = { name = qing_xj_ctl_term  add = 2 } }
	if = { limit = { exists = c:ORT  c:ORT = { is_subject_of = ROOT } }  change_variable = { name = qing_xj_ctl_term  add = 2 } }
	if = { limit = { exists = c:KSH  c:KSH = { is_subject_of = ROOT } }  change_variable = { name = qing_xj_ctl_term  add = 2 } }
	change_variable = { name = qing_xj_control_tmp  add = var:qing_xj_ctl_term }
	# venal begs drag: −6 each.
	set_variable = { name = qing_xj_ctl_term  value = var:qing_xj_beg_venal_count }
	change_variable = { name = qing_xj_ctl_term  multiply = 6 }
	change_variable = { name = qing_xj_control_tmp  subtract = var:qing_xj_ctl_term }
	# contested oasis provinces: −4 each (beat/rot-stamped loss of grip).
	set_variable = { name = qing_xj_ctl_term  value = 0 }
	area:Dzungaria = { every_area_province = { limit = { has_province_modifier = qing_xinjiang_prov_contested }  ROOT = { change_variable = { name = qing_xj_ctl_term  add = 4 } } } }
	area:Tarim     = { every_area_province = { limit = { has_province_modifier = qing_xinjiang_prov_contested }  ROOT = { change_variable = { name = qing_xj_ctl_term  add = 4 } } } }
	change_variable = { name = qing_xj_control_tmp  subtract = var:qing_xj_ctl_term }
	# clamp 0..100 and publish.
	if = { limit = { var:qing_xj_control_tmp < 0 }   set_variable = { name = qing_xj_control_tmp  value = 0 } }
	if = { limit = { var:qing_xj_control_tmp > 100 } set_variable = { name = qing_xj_control_tmp  value = 100 } }
	set_variable = { name = qing_xinjiang_control  value = var:qing_xj_control_tmp }
	LOG_line = { sys = QING  msg = "xinjiang: control DERIVED from concrete grip objects" }
}

# ======================================================================
# QING_xj_plant_tuntian (屯田) — plant a military colony in Dzungaria. Scope: COUNTRY (CHI).
# Spends treasury; raises the grip; stamps a REAL Dzungaria province with the colony
# modifier; advances the ILI subject's integration a step (settlement knits the north into
# the empire). Guarded on treasury + an unstamped Dzungaria province + the colony cap (8);
# the panel button mirrors this.
# ======================================================================
QING_xj_plant_tuntian = {
	LOG_enter = { sys = QING  fn = "QING_xj_plant_tuntian" }
	QING_xj_init = yes
	if = {
		limit = {
			treasury >= 220
			var:qing_xj_tuntian < 8
			area:Dzungaria = {
				any_area_province = { NOT = { has_province_modifier = qing_xj_tuntian_colony } }
			}
		}
		add_treasury = -220
		TREASURY_LOG_it = { amount = -220 }
		change_variable = { name = qing_xj_tuntian  add = 1 }
		# stamp one unstamped Dzungaria province with the concrete colony modifier.
		area:Dzungaria = {
			random_area_province = {
				limit = { NOT = { has_province_modifier = qing_xj_tuntian_colony } }
				add_province_modifier = { name = qing_xj_tuntian_colony  duration = -1 }
			}
		}
		# [meter-concretize #10B] the +5 control nudge is DROPPED — the planted qing_xj_tuntian_colony
		# modifier is now a control DERIVE term (+3/colony), so control rises via the derive below.
		# settlement tightens the Yili Governorate's integration into the empire.
		if = {
			limit = { exists = c:ILI  c:ILI = { is_subject = yes } }
			c:ILI = { save_scope_as = target }
			SUBJ_QING_advance_integration = { steps = 1 }
		}
		QING_xj_derive_control = yes
		QING_xj_recompute_consolidation = yes
		LOG_line = { sys = QING  msg = "xinjiang: a tuntian colony is planted in Dzungaria" }
	}
	LOG_exit = { sys = QING  fn = "QING_xj_plant_tuntian"  result = OK }
}

# QING_xj_appoint_beg (授伯克) — appoint a loyal beg over a Tarim oasis. Scope: COUNTRY

# Kashgaria subject existing + the corps cap (5); the panel button mirrors this.
# ======================================================================
QING_xj_appoint_beg = {
	LOG_enter = { sys = QING  fn = "QING_xj_appoint_beg" }
	QING_xj_init = yes
	if = {
		limit = {
			exists = c:XNG
			var:qing_xj_beg_count < 5
		}
		QING_xj_mint_beg = yes
		# [meter-concretize #10B] +4 control nudge DROPPED — beg_count is a control DERIVE term (+4/beg).
		QING_xj_derive_control = yes
		QING_xj_recompute_consolidation = yes
		LOG_line = { sys = QING  msg = "xinjiang: a loyal beg is appointed" }
	}
	LOG_exit = { sys = QING  fn = "QING_xj_appoint_beg"  result = OK }
}

# ======================================================================
# QING_xj_discipline_beg (懲辦) — cashier a beg who has turned (a khoja-scare defector,
# qing_xj_beg_venal). Scope: COUNTRY (CHI). Purging the disloyal element restores order —
# he leaves the corps (his marker stripped, so the rebuild drops him) and the grip firms.
# Guarded on a venal beg actually existing; the panel button mirrors this.
# ======================================================================
QING_xj_discipline_beg = {
	LOG_enter = { sys = QING  fn = "QING_xj_discipline_beg" }
	QING_xj_recompute_consolidation = yes
	if = {
		limit = {
			has_variable_list = qing_xj_begs
			any_in_list = {
				variable = qing_xj_begs
				is_alive = yes
				employer = c:XNG
				has_variable = qing_xj_beg_venal
			}
		}
		# pick the ablest defector first (the most dangerous to leave in place).
		ordered_in_list = {
			variable = qing_xj_begs
			limit = {
				is_alive = yes
				employer = c:XNG
				has_variable = qing_xj_beg_venal
			}
			order_by = combined_stats_council_svalue
			check_range_bounds = no
			max = 1
			save_scope_as = qing_xj_purge_pick
		}
		if = {
			limit = { exists = scope:qing_xj_purge_pick }
			# strip his corps + venal marks — the rebuild drops him; he is no longer a beg.
			scope:qing_xj_purge_pick = {
				if = { limit = { has_variable = qing_is_xj_beg }    remove_variable = qing_is_xj_beg }
				if = { limit = { has_variable = qing_xj_beg_venal } remove_variable = qing_xj_beg_venal }
			}
			# [meter-concretize #10B] +3 control nudge DROPPED — cashiering a venal beg drops venal_count
			# (a −6/venal derive drag lifts) and the rebuild drops him from beg_count; control derives up.
			LOG_line = { sys = QING  msg = "xinjiang: a turned beg is cashiered" }
		}
		QING_xj_derive_control = yes
		QING_xj_recompute_consolidation = yes
	}
	LOG_exit = { sys = QING  fn = "QING_xj_discipline_beg"  result = OK }
}

# ======================================================================
# QING_xj_xiexiang_on / _off (協餉) — toggle the interior-province silver subsidy that kept
# Xinjiang solvent. Scope: COUNTRY (CHI). ON applies the drain-for-grip country modifier;
# OFF removes it. The GUI offers whichever is applicable (subsidize when off, cut when on).
# ======================================================================
QING_xj_xiexiang_on = {
	LOG_enter = { sys = QING  fn = "QING_xj_xiexiang_on" }
	QING_xj_init = yes
	if = {
		limit = { NOT = { has_country_modifier = qing_xj_xiexiang } }
		add_country_modifier = { name = qing_xj_xiexiang  duration = -1 }
		# [meter-concretize #10B] +6 control nudge DROPPED — qing_xj_xiexiang now GATES the garrison-seat derive term (#19, up to +12);
		# also resume-of-subsidy stops the neglect rot (the pulse gate reads this modifier).
		QING_xj_derive_control = yes
		QING_xj_recompute_consolidation = yes
		LOG_line = { sys = QING  msg = "xinjiang: the xiexiang silver subsidy begins flowing" }
	}
	LOG_exit = { sys = QING  fn = "QING_xj_xiexiang_on"  result = OK }
}

# ======================================================================
# QING_xj_pulse — quarterly drift. Scope: COUNTRY (CHI). Called from the governance pulse
# (BEFORE the ministry recompute would be irrelevant — the fold reads the published score
# with a one-pulse lag either way). Rebuilds the corps + score, applies the slow
# maintenance/neglect drift on the grip (a paid, staffed frontier holds; a neglected one
# erodes), and on a throttled roll when the grip is low AND the corps is empty/venal, fires
# the khoja revolt-scare (qing_xinjiang.1). Self-guarded; a clean no-op at a firm frontier.
# ======================================================================
QING_xj_pulse = {
	LOG_enter = { sys = QING  fn = "QING_xj_pulse" }
	# [#422 review-fix] ONE-WAY INTEGRATION LOCK. Once the consolidation-tree capstone has
	# declared the New Dominion fully integrated (a normal province — QING_xj_integrate_fully
	# set qing_xj_fully_integrated), the frontier machinery must STOP: otherwise a later cut to
	# the xiexiang subsidy would drift qing_xinjiang_control back down, re-stamp the contested
	# band and re-arm the khoja revolt-scare — silently unwinding the "henceforth a normal
	# province" end-state the capstone promises. Early-return keeps the lock permanent.
	if = {
		limit = { has_variable = qing_xj_fully_integrated }
		LOG_exit = { sys = QING  fn = "QING_xj_pulse"  result = OK }
	}
	else = {
		# [meter-concretize #10B 2026-08-06] NEGLECT ROT — an unpaid frontier garrison hollows out and
		# mutinies (1826 Jahangir pattern). Gated on NO xiexiang ALONE (xiexiang paid the GARRISON, not the
		# begs); begs modulate the RATE via the rot threshold, not whether. This DEGRADES the concrete objects
		# (flip a secured oasis → contested, decay a tuntian colony) so control — a pure derive — falls low
		# because the MAP rotted, reaching the khoja ≤30 trigger even on a built frontier. Replaces the old
		# −1/pulse abstract ratchet. Runs BEFORE the derive so this pulse's control reflects the rot.
		if = {
			limit = { NOT = { has_country_modifier = qing_xj_xiexiang } }
			change_variable = { name = qing_xj_neglect_quarters  add = 1 }
			# rate: loyal corps buffers the bleed; bare/venal collapses fast. [PLACEHOLDER-playtest]
			set_variable = { name = qing_xj_rot_threshold  value = 4 }
			if = { limit = { var:qing_xj_beg_count >= 3  var:qing_xj_beg_venal_count = 0 }  set_variable = { name = qing_xj_rot_threshold  value = 8 } }
			if = { limit = { OR = { var:qing_xj_beg_count = 0  var:qing_xj_beg_venal_count > 0 } }  set_variable = { name = qing_xj_rot_threshold  value = 2 } }
			if = {
				limit = { var:qing_xj_neglect_quarters >= qing_xj_rot_threshold_cmpsvalue }
				# flip ONE secured oasis → contested (garrison mutinies).
				if = {
					limit = { OR = { area:Dzungaria = { any_area_province = { has_province_modifier = qing_xinjiang_prov_secured } }  area:Tarim = { any_area_province = { has_province_modifier = qing_xinjiang_prov_secured } } } }
					if = {
						limit = { area:Dzungaria = { any_area_province = { has_province_modifier = qing_xinjiang_prov_secured } } }
						area:Dzungaria = { random_area_province = { limit = { has_province_modifier = qing_xinjiang_prov_secured }  remove_province_modifier = qing_xinjiang_prov_secured  add_province_modifier = { name = qing_xinjiang_prov_contested  duration = -1 } } }
					}
					else = {
						area:Tarim = { random_area_province = { limit = { has_province_modifier = qing_xinjiang_prov_secured }  remove_province_modifier = qing_xinjiang_prov_secured  add_province_modifier = { name = qing_xinjiang_prov_contested  duration = -1 } } }
					}
				}
				# decay ONE tuntian colony — remove the modifier AND decrement the int in lockstep (else the
				# plant guard var:qing_xj_tuntian<8 soft-locks re-planting — review HIGH fix).
				if = {
					limit = { var:qing_xj_tuntian > 0  area:Dzungaria = { any_area_province = { has_province_modifier = qing_xj_tuntian_colony } } }
					area:Dzungaria = { random_area_province = { limit = { has_province_modifier = qing_xj_tuntian_colony }  remove_province_modifier = qing_xj_tuntian_colony } }
					change_variable = { name = qing_xj_tuntian  subtract = 1 }
				}
				set_variable = { name = qing_xj_neglect_quarters  value = 0 }   # reset after each rot-tick
				LOG_line = { sys = QING  msg = "xinjiang: NEGLECT ROT — an unpaid oasis slips from the throne's grip" }
			}
		}
		else = { set_variable = { name = qing_xj_neglect_quarters  value = 0 } }   # xiexiang resumed → clock resets, rot stops
		# [meter-concretize #10B] DERIVE control from the concrete grip objects (after any rot this pulse),
		# then recompute the consolidation composite off the fresh control base.
		QING_xj_derive_control = yes
		QING_xj_recompute_consolidation = yes
	# --- [#367-R2 review-fix] defensive clear of a STRANDED khoja-scare flag. qing_xj_khoja_pending
	# is set here before queuing qing_xinjiang.1, and is cleared ONLY in that event's immediate. But
	# the event re-checks its own trigger (grip <= 30) at fire time: if the player pushes grip back
	# above 30 during the 10-40 day delay (xiexiang +6, appoint-beg +4), the event silently cancels,
	# its immediate never runs, and the flag leaks forever — permanently disabling every future scare
	# via the NOT={has_variable} guard below. Re-arm on the next pulse once grip recovers out of the
	# danger band, mirroring the qing_amban_here defensive-clear pattern (se_QING_AMBAN.txt).
	if = {
		limit = {
			has_variable = qing_xj_khoja_pending
			OR = {
				NOT = { has_variable = qing_xinjiang_control }
				var:qing_xinjiang_control > 30
			}
		}
		remove_variable = qing_xj_khoja_pending
	}
	# --- khoja revolt-scare risk: a low grip + an empty or disloyal beg corps invites the
	# White-Mountain khojas (backed from Kokand, KOK — the neighbouring Turkic power, per
	# the separatism-backer rule). Throttled roll; the event re-checks its own trigger.
	if = {
		limit = {
			has_variable = qing_xinjiang_control
			var:qing_xinjiang_control <= 30
			OR = {
				var:qing_xj_beg_count = 0
				var:qing_xj_beg_venal_count > 0
			}
			NOT = { has_variable = qing_xj_khoja_pending }
			# [#13 2026-08-07 LINK] the khoja scare is "backed from Kokand" (separatism-backer rule). Suppress it
			# when Kokand is currently beaten (conquered_trigger, a LIVE derive) OR was permanently settled via
			# caravan.3 (yielded_flag, PERMANENT). BOTH triggers, different lifetimes (design §NEW-MEDIUM-2): a
			# caravan.3-settled Kokand stays quiet forever; a merely-driven-off-but-never-settled rump can re-arm.
			NOT = { OR = { QING_kok_conquered_trigger = yes  QING_kok_yielded_flag = yes } }
		}
		random = {
			chance = 15
			set_variable = { name = qing_xj_khoja_pending  value = 1 }
			LOG_line = { sys = QING  msg = "xinjiang: the far west stirs -> queue the khoja revolt-scare" }
			trigger_event = { id = qing_xinjiang.1  days = { 10 40 } }
		}
	}
	}
	LOG_exit = { sys = QING  fn = "QING_xj_pulse"  result = OK }
}

# ======================================================================
# QING_xj_turn_a_beg — a khoja scare TURNS one loyal beg into a defector (marks him
# qing_xj_beg_venal). Scope: COUNTRY (CHI). Called from the fester branch of qing_xinjiang.1.
# Picks a currently-loyal serving beg; a no-op if the corps is empty or already all venal.
# ======================================================================
QING_xj_turn_a_beg = {
	QING_xj_recompute_consolidation = yes
	if = {
		limit = {
			has_variable_list = qing_xj_begs
			any_in_list = {
				variable = qing_xj_begs
				is_alive = yes
				employer = c:XNG
				NOT = { has_variable = qing_xj_beg_venal }
			}
		}
		ordered_in_list = {
			variable = qing_xj_begs
			limit = {
				is_alive = yes
				employer = c:XNG
				NOT = { has_variable = qing_xj_beg_venal }
			}
			order_by = combined_stats_council_svalue
			check_range_bounds = no
			max = 1
			save_scope_as = qing_xj_turn_pick
		}
		if = {
			limit = { exists = scope:qing_xj_turn_pick }
			scope:qing_xj_turn_pick = { set_variable = { name = qing_xj_beg_venal  value = 1 } }
			LOG_line = { sys = QING  msg = "xinjiang: a beg is turned by the khoja" }
		}
		QING_xj_recompute_consolidation = yes
	}
}

# ======================================================================
# QING_xj_remove_on_death — call in a DYING character's scope (hooked into
# on_character_death). If the dead man was a beg, drop him from the corps list on CHI so no
# orphan reference survives + keep the count honest. Self-guards on the marker. Mirrors
# QING_southernstudy_remove_on_death. The corps list lives on CHI (ROOT there is the dying
# char, so we reach CHI via c:CHI — the beg's employer is XNG, not the list-holder).
# ======================================================================
QING_xj_remove_on_death = {
	if = {
		limit = {
			has_variable = qing_is_xj_beg
			exists = c:CHI
		}
		save_scope_as = qing_dead_beg
		c:CHI = {
			if = {
				limit = { has_variable_list = qing_xj_begs }
				remove_list_variable = { name = qing_xj_begs  target = scope:qing_dead_beg }
				if = { limit = { var:qing_xj_beg_count > 0 }  change_variable = { name = qing_xj_beg_count  subtract = 1 } }
				LOG_line = { sys = QING  msg = "xinjiang: a beg died, removed from the corps" }
			}
		}
	}
}

# QING_xj_integrate_fully (新疆一體) — the CONSOLIDATION-TREE CAPSTONE (#422): the New
# Dominion is at last a fully integrated part of the empire — no longer a garrisoned
# frontier under a special regime, but a normal province like any interior one. Scope:
# COUNTRY (CHI). This is the reward for a player who has genuinely built the frontier up
# (deep tuntian, a loyal beg corps, a capable governor, the xiexiang subsidy sustained)
# — the mission gates the ballot on that real state; this effect makes it so:
#   • the control meter is pegged to the top (the grip is total), and the band recompute
#     is run so the country/province bands flip to the SECURED (positive) side for free;
#   • every NEGATIVE Xinjiang/ethnic modifier is stripped off the New Dominion ground
#     (the contested band, the ethnic-tension bands, the restive marks) and the per-
#     province ethnic-tension counter is zeroed — the province is treated as normal;
#   • a lasting integration dividend + the ILI subject is absorbed a step toward full
#     incorporation (settlement has knit the north into the empire for good).
# All verbs proven in-file / in se_QING_ILI: set_variable on the meter, the band
# recompute, area:Dzungaria/Tarim every_area_province, remove_province_modifier,
# SUBJ_QING_advance_integration.
# ======================================================================
QING_xj_integrate_fully = {
	LOG_enter = { sys = QING  fn = "QING_xj_integrate_fully" }
	QING_xj_init = yes
	# [meter-concretize #10B] stamp the whole New Dominion SECURED (concrete), then DERIVE control — it
	# lands ~100 from the full secured set + xiexiang + frontier. The old `set control = 100` is DROPPED
	# (control is a pure derive; setting it would be overwritten next pulse anyway). Then re-band the country.
	QING_ili_apply_prov_band = { INTENT = secured }
	QING_xj_derive_control = yes
	QING_ili_apply_control_band = yes
	# [meter-concretize #10B / review LOW-5] clear any pending khoja scare — a stale qing_xj_khoja_pending set
	# just before integration would otherwise let qing_caravan.2 fire a post-integration route-cut.
	if = { limit = { has_variable = qing_xj_khoja_pending } remove_variable = qing_xj_khoja_pending }
	# strip every NEGATIVE band/tension modifier off the New Dominion provinces, and zero
	# the per-province ethnic-tension counter, so the ground is treated as a normal province.
	area:Dzungaria = {
		every_area_province = {
			if = { limit = { has_province_modifier = qing_xinjiang_prov_contested } remove_province_modifier = qing_xinjiang_prov_contested }
			if = { limit = { has_province_modifier = qing_ethnic_restive }          remove_province_modifier = qing_ethnic_restive }
			if = { limit = { has_variable = qing_prov_ethnic_tension } set_variable = { name = qing_prov_ethnic_tension  value = 0 } }
		}
	}
	area:Tarim = {
		every_area_province = {
			if = { limit = { has_province_modifier = qing_xinjiang_prov_contested } remove_province_modifier = qing_xinjiang_prov_contested }
			if = { limit = { has_province_modifier = qing_ethnic_restive }          remove_province_modifier = qing_ethnic_restive }
			if = { limit = { has_variable = qing_prov_ethnic_tension } set_variable = { name = qing_prov_ethnic_tension  value = 0 } }
		}
	}
	# clear the country-level negative grip band (the recompute above already sets grip_firm).
	if = { limit = { has_country_modifier = qing_xinjiang_grip_lost } remove_country_modifier = qing_xinjiang_grip_lost }
	if = { limit = { has_country_modifier = qing_ethnic_high_tension } remove_country_modifier = qing_ethnic_high_tension }
	# settlement has knit the north in for good — absorb the ILI governorship a step further.
	if = {
		limit = { exists = c:ILI  c:ILI = { is_subject_of = ROOT } }
		c:ILI = { save_scope_as = target }
		SUBJ_QING_advance_integration = { steps = 1 }
	}
	# [P1 2026-07-24] forging the New Dominion into the nation — naturalise the East-Turkic peoples
	# (uighur et al.) and raise the civic-identity ceiling, the same nation-building payoff the
	# settle-frontier arcs grant the Mongol/Manchurian peoples. This "integrate fully" task strips
	# the tension modifiers above but previously never extended CITIZENSHIP (integrate_country_culture)
	# — leaving the Turkic peoples an un-integrated out-group on now-normal ground. forge_nation closes
	# that: permanent engine integration (survives the decline-pulse drift) + a persisting civic ceiling
	# lift. Safe no-op if no un-integrated east_turkic culture is present in the realm.
	QING_settle_forge_nation = { group = east_turkic }
	# a lasting mark: the New Dominion is now provincia like any other.
	set_variable = { name = qing_xj_fully_integrated  value = 1 }
	LOG_line = { sys = QING  msg = "xinjiang: the New Dominion is FULLY INTEGRATED — a normal province" }
	LOG_exit = { sys = QING  fn = "QING_xj_integrate_fully"  result = OK }
}
