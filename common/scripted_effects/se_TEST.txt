		# other culture group matching none of the OR-lists above (mongolic, kartvelian,
		# siberian_turkic, east_turkic, non-CHI chinese_group/jurchen, malagasy, swahili_group,

		# universal. Deliberately NOT a hand-enumerated culture-group list -- that approach is
		# exactly what produced this gap in the first place. Provably complete: any country not
		# already marked TECH_1763_seeded by one of the blocs above lands here by construction.
		every_country = {
			limit = {
				NOT = { has_variable = TECH_1763_seeded }
			}
			TECH_unlock_military_level_0 = yes
			TECH_unlock_civic_level_1 = yes
		}

		# persistent gameplay variable.
		every_country = {
			remove_variable = TECH_1763_seeded
		}
	}
