# se_QING_WAR.txt — the Grand Minister of War (兵部) and military commanders.
#
# Historical anchor: the 兵部 administered the Green Standard Army (綠營),
# military examinations (武舉), the courier relays, and officer appointments.
# Army and navy commanders were its charges. The 19th-century danger: regional
# armies (湘軍/淮軍) whose commanders answered to their patrons, not the throne.
#
# COUPLING to the Grand Minister of War (兵部):
#   - Commanders (characters with military power_base, the loyal-cohorts model)
#     are scored against the War office-holder via QING_pair_friction (the same
#     agree/disagree mechanism governors/residents use).
#   - A capable, congenial War minister keeps his commanders loyal and effective
#     (loyalty up, martial cultivated).
#   - A vacant War office or a clashing pair → the commander drifts toward
#     personal loyalty (the warlord seed); loyalty risk, possible review event.
#   - The effect GUARDS every deref (is_alive / employer / has_variable checks)
#     and does NOT double-count the War accountability metric (war exhaustion /
#     cohort count already measured in se_QING_ACCOUNTABILITY.txt).
#
# State (all on CHI):
#   qing_office_war_holder    the character holding the War office (from se_QING_COUNCIL.txt)
#
# Events (qing_war namespace):
#   qing_war.1   officer review / promotion board under capable War minister
#   qing_war.2   commander too close to his troops (warlord danger)
#   qing_war.3   naval/army commission decision gated on War minister
#   qing_war.4   military-examination (武舉) intake of fresh officers
#
# Prefix = QING_war_
########################################################################

# ======================================================================
# REVIEW COMMANDERS — quarterly pulse (country scope, CHI).
# Iterate the realm's military commanders (characters with military power_base,
# the loyal-cohorts model). Score each against the War office-holder via
# QING_pair_friction and against the throne via QING_char_affinity. A strong,
# loyal War minister + congenial commander = well-managed (loyalty/martial up).
# A vacant War office or a clashing pair = drift toward personal loyalty (the
# warlord seed) — loyalty risk, chance to queue an officer-review event.
# ======================================================================
QING_war_review_commanders = {
	LOG_enter = { sys = QING  fn = "QING_war_review_commanders" }

	# --- guard: War office must be filled and the holder alive
	if = {
		limit = {
			has_variable = qing_office_war_holder
			var:qing_office_war_holder = { is_alive = yes  employer = ROOT }
		}
		var:qing_office_war_holder = { save_scope_as = war_minister }

		# --- iterate commanders: characters with martial focus AND power_base (loyal cohorts model)
		# This is a conservative proxy: commanders who have been granted loyal veterans
		# (via office, regional army sanction, or mission rewards) have power_base >= 1.
		# We filter for martial >= 8 to focus on actual military leaders, not just
		# courtiers with holdings.
		every_character = {
			limit = {
				employer = ROOT
				is_alive = yes
				is_adult = yes
				is_ruler = no
				martial >= 8
				power_base >= 1
			}
			save_scope_as = commander

			# --- [MO-ENH #3] CENTRAL-ARMY COLLAPSE -> WARLORD DRIFT. When BOTH central hosts are
			# rotting (八旗 banner_decay AND 綠營 greenstandard_decay both >= 60), the throne's grip on
			# its provincial commanders slips: a court that cannot field a credible central army invites
			# the 湘軍/淮軍 regionalism. Concrete-over-abstract — the abstract decay counters now bite a
			# real on-map commander's loyalty. An extra estrangement term ON TOP of the friction/affinity
			# scoring below (compounds it; the warlord-event roll there still gates the actual crisis).
			if = {
				limit = {
					ROOT = {
						has_variable = qing_banner_decay
						has_variable = qing_greenstandard_decay
						var:qing_banner_decay >= 60
						var:qing_greenstandard_decay >= 60
					}
				}
				# [loyalty-calibration 2026-08-04] DISTINCT modifier loyalty_qing_army_collapse — NOT the
				# estranged/congenial pair. This army-collapse term is designed to be ADDITIVE ON TOP of
				# the friction/affinity scoring below (comment above); if it used loyalty_qing_estranged,
				# the later well-managed branch's QING_apply_congenial_once would SHED it in the same pass,
				# erasing the compounding. Its own modifier coexists. Idempotent (quarterly pulse).
				QING_apply_army_collapse_once = yes
				LOG_line = { sys = QING  msg = "commander drifting (both central armies rotting)" }
			}
			else = {
				# armies recovered (not both rotting) — lift the collapse pressure so it doesn't linger
				QING_remove_army_collapse = yes
			}

			# --- score the commander against the War minister (pair friction)
			ROOT = {
				QING_pair_friction = { a = scope:war_minister  b = scope:commander }
			}

			# --- also score the commander against the throne (personal affinity)
			QING_char_affinity = yes

			# --- HIGH FRICTION or LOW AFFINITY: the commander drifts from central control
			if = {
				limit = {
					OR = {
						ROOT = { var:qing_pair_friction >= 50 }
						var:qing_char_affinity <= 35
					}
				}
				# loyalty risk
				QING_apply_estranged_once = yes   # [loyalty-calibration 2026-08-04] idempotent (was bare add_loyalty; quarterly re-stamp)
				# chance to queue a warlord-danger event (low prob each pulse to avoid spam)
				# [BT-warlord-spam fix 2026-07-23] do NOT queue another qing_war.2 for a commander already under
				# the warlord-review cooldown (qing_warlord_review_cd, stamped when the event fires). This keeps
				# the queue from stacking copies each pulse; the event's own trigger is the hard backstop.
				if = {
					# [#55 2026-08-10] gate on the shared court slot + the War department cooldown so the warlord-review
					# event joins the one-per-quarter rotation and a fired War event (this or .1/.4) holds it ~3 quarters.
					limit = { NOT = { has_variable = qing_warlord_review_cd }  ROOT = { NOT = { has_variable = qing_gc_event_slot_used }  NOT = { has_variable = qing_dept_cd_war } } }
					ROOT = {
						random = {
							chance = 15
							set_variable = { name = qing_gc_event_slot_used  value = 1 }  # [#55] claim the shared court slot
							set_variable = { name = qing_dept_cd_war  days = 270 }  # [#55] War department stands down ~3 quarters
							trigger_event = { id = qing_war.2  days = 30 }
							LOG_line = { sys = QING  msg = "commander drifting (friction/low affinity); queued warlord review event" }
						}
					}
				}
			}
			# --- LOW FRICTION and HIGH AFFINITY: well-managed commander
			else_if = {
				limit = {
					ROOT = { var:qing_pair_friction < 50 }
					var:qing_char_affinity >= 60
				}
				# loyalty and martial boost (capable War minister cultivates his officers)
				QING_apply_congenial_once = yes   # [loyalty-calibration 2026-08-04] idempotent (was bare add_loyalty; quarterly re-stamp)
				if = {
					limit = { scope:war_minister = { martial >= 10 } }
					# a brilliant War minister sharpens his commanders' martial skill
					QING_char_cultivate = { stat = martial  amount = 1 }
					LOG_line = { sys = QING  msg = "commander well-managed under War minister (low friction, high affinity)" }
				}
			}
		}

		# --- trigger officer review event if the War minister is capable (quarterly chance)
		# [#107 2026-08-09] gate on the shared quarterly GC-event slot + claim it when it fires — was
		# UNGATED, so at 20%/quarter it fired several times a year alongside the other court events (the
		# reported "firing very often" bug). This is the proven throttle idiom the sibling flavour rolls use
		# (00_monthly_country.txt:119+, memory gc-event-throttle-rule): one court event per quarter.
		if = {
			limit = {
				scope:war_minister = { martial >= 10  loyalty >= 50 }
				NOT = { has_variable = qing_gc_event_slot_used }
				# [#55 2026-08-10] War 兵部 department stand-down: after either War roll fires, the whole department
				# stands down ~3 quarters (shared qing_dept_cd_war) so the court slot rotates to other departments.
				NOT = { has_variable = qing_dept_cd_war }
			}
			random = {
				chance = 20
				set_variable = { name = qing_gc_event_slot_used  value = 1 }   # [#107] claim the shared court slot
				set_variable = { name = qing_dept_cd_war  days = 270 }  # [#55] War department stands down ~3 quarters
				trigger_event = { id = qing_war.1  days = 45 }
				LOG_line = { sys = QING  msg = "queued officer review board event (capable War minister)" }
			}
		}

		# --- trigger military-examination event if the War minister is effective
		# [#107 2026-08-09] same throttle as the officer-review roll above (review MEDIUM): gate on the shared
		# court slot + claim it. Was ungated, so it fired several times a year AND could double-queue with
		# qing_war.1 in the same quarter — breaking the one-court-event-per-quarter budget. The qing_war.1
		# roll above claims the slot first, so if it fired this quarter this block is now correctly skipped.
		if = {
			limit = {
				scope:war_minister = { martial >= 12  finesse >= 8 }
				NOT = { has_variable = qing_gc_event_slot_used }
				NOT = { has_variable = qing_dept_cd_war }  # [#55] shared War department cooldown (see the officer-review roll above)
			}
			random = {
				chance = 12
				set_variable = { name = qing_gc_event_slot_used  value = 1 }   # [#107] claim the shared court slot
				set_variable = { name = qing_dept_cd_war  days = 270 }  # [#55] War department stands down ~3 quarters
				trigger_event = { id = qing_war.4  days = 60 }
				LOG_line = { sys = QING  msg = "queued military-examination event (effective War minister)" }
			}
		}
	}
	# --- VACANT War office: commanders drift (no oversight)
	else = {
		every_character = {
			limit = {
				employer = ROOT
				is_alive = yes
				is_adult = yes
				is_ruler = no
				martial >= 8
				power_base >= 1
			}
			save_scope_as = commander
			# no War minister → officers become wayward (loyalty drift)
			# [cross-wave audit] The MO-ENH #3 "both central armies rotting" estrangement term lives
			# ONLY in the filled branch. Here the office is vacant, so the commander is ALREADY estranged
			# unconditionally by the line below; a second identical loyalty_qing_estranged for army decay
			# would be redundant (and floored by the modifier's min anyway), so it is deliberately omitted.
			QING_apply_estranged_once = yes   # [loyalty-calibration 2026-08-04] idempotent (was bare add_loyalty; this vacant-office branch re-stamps every pulse while the War office sits empty)
			# higher chance of warlord-danger event when the office is vacant
			# [BT-warlord-spam fix 2026-07-23] same per-commander cooldown gate as the filled branch — don't
			# stack a fresh qing_war.2 on a man already under review (qing_warlord_review_cd).
			if = {
				# [#55 2026-08-10] same shared-slot + War-department-cooldown gate as the filled branch.
				limit = { NOT = { has_variable = qing_warlord_review_cd }  ROOT = { NOT = { has_variable = qing_gc_event_slot_used }  NOT = { has_variable = qing_dept_cd_war } } }
				ROOT = {
					random = {
						chance = 25
						set_variable = { name = qing_gc_event_slot_used  value = 1 }  # [#55] claim the shared court slot
						set_variable = { name = qing_dept_cd_war  days = 270 }  # [#55] War department stands down ~3 quarters
						trigger_event = { id = qing_war.2  days = 30 }
						LOG_line = { sys = QING  msg = "commander drifting (War office vacant); queued warlord review event" }
					}
				}
			}
		}
		LOG_line = { sys = QING  msg = "War office vacant; commanders drifting without oversight" }
	}

	LOG_exit = { sys = QING  fn = "QING_war_review_commanders"  result = OK }
}

# ======================================================================
# [Phase 4d] QING_war_grant_legion_distinction — the MINISTRY-OF-WAR tie-in for legion
# distinctions. Called from on_legion_raised (00_specific_from_code.txt) in LEGION scope for a
# freshly-raised CHI legion. Distinctions cannot be gated in their definition (oracle-verified),
# so the 兵部's competence is expressed HERE, at the grant: the band of qing_min_perf_war decides
# WHICH distinction the new legion earns.
#   - a bare/failing 兵部 (< 40, or the office vacant): the basic 綠營標營 mark, or nothing.
#   - an adequate 兵部 (40..69): the workhorse marks (綠營/藤牌/索倫/駝城).
#   - a well-run 兵部 (>= 70): the elite corps (健銳營 / 火器營).
# A legion that already carries a Qing distinction is left alone (no re-grant / stack). The
# choice within a band is by the raising province's terrain-ish flavour kept simple: a single
# random_list so repeated musters vary. Guarded on tag + var existence. Scope: LEGION (owner reads
# the country). No iterator over characters here (this is the legion itself) — no compile risk;
# it runs from a real on_action, not a compiled GUI button.
# ======================================================================
QING_war_grant_legion_distinction = {
	# scope: legion. Only for the Qing, and only if this legion has no Qing mark yet.
	owner = { save_scope_as = qing_legion_owner }
	if = {
		limit = {
			scope:qing_legion_owner = { tag = CHI }
			NOR = {
				has_distinction = distinction_qing_green_standard
				has_distinction = distinction_qing_jianrui
				has_distinction = distinction_qing_solon
				has_distinction = distinction_qing_firearms_brigade
				has_distinction = distinction_qing_rattan_shield
				has_distinction = distinction_qing_camel_fort
			}
		}
		if = {
			# ELITE band — a well-run War Ministry (>= 70): the storming/firearms corps.
			limit = {
				scope:qing_legion_owner = {
					has_variable = qing_min_perf_war
					var:qing_min_perf_war >= 70
				}
			}
			random_list = {
				50 = { add_distinction = distinction_qing_jianrui }
				50 = { add_distinction = distinction_qing_firearms_brigade }
			}
			LOG_line = { sys = QING  msg = "war-ministry granted an ELITE legion distinction (兵部 well-run)" }
		}
		else_if = {
			# ADEQUATE band (40..69): the workhorse provincial/frontier marks.
			limit = {
				scope:qing_legion_owner = {
					has_variable = qing_min_perf_war
					var:qing_min_perf_war >= 40
				}
			}
			random_list = {
				25 = { add_distinction = distinction_qing_green_standard }
				25 = { add_distinction = distinction_qing_rattan_shield }
				25 = { add_distinction = distinction_qing_solon }
				25 = { add_distinction = distinction_qing_camel_fort }
			}
			LOG_line = { sys = QING  msg = "war-ministry granted a standard legion distinction (兵部 adequate)" }
		}
		else = {
			# BARE band (< 40 or no var): only the basic Green-Standard mark.
			add_distinction = distinction_qing_green_standard
			LOG_line = { sys = QING  msg = "war-ministry granted only the basic Green-Standard distinction (兵部 weak/vacant)" }
		}
	}
}
