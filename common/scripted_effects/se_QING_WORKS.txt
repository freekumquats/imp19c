# ======================================================================
# QUARTERLY WORKS PULSE — called from QING_GOV_pulse (quarterly cadence).
# [task #209] Mirrors QING_revenue_pulse: gives the Grand Minister of Works
# (工部) domain its own throttled home instead of riding the generic decline
# roll. Because this pulse now OWNS the works events, they are REMOVED from the
# se_QING_DECLINE.txt flavour roll (qing_works.1/.2/.3/.4/.6) — exactly as the
# revenue events were migrated out — so nothing double-fires. The follow-up
# qing_works.5 stays is_triggered_only (chained only from qing_works.1.b).
#
# Two branches, mutually exclusive:
#   VACANT office → the hydraulic works decay. The perennial DECAY crises
#     (dike breach .1, canal silt .2 — each of which already carries a
#     "no Works minister / can't afford it" option path) can still strike,
#     and unsupervised 河工 graft drifts corruption up. The constructive
#     projects (.3 Great Wall, .6 specialty manufactory) and the minister-
#     specific graft scandal (.4) do NOT fire with no holder.
#   FILLED office (holder alive + employer = ROOT + off cooldown) → the full
#     menu on a throttled roll: the two decay crises the minister can address
#     well, the Great Wall prestige project, a new specialty state manufactory,
#     and — in a corrupt court — a river-works embezzlement scandal.
# Cooldown: qing_works_event_cooldown (270 days ≈ 3 quarters), matching
# qing_revenue_event_cooldown. The vacant corruption drift is unconditional
# (mirrors the revenue vacant drift), only the event offer is cooldown-gated.
# ======================================================================
QING_works_pulse = {
	LOG_enter = { sys = QING  fn = "QING_works_pulse" }

	# --- VACANT office: the works decay ----------------------------------
	if = {
		limit = { NOT = { has_variable = qing_office_works_holder } }
		# unsupervised hydraulic-works budget (河工) breeds graft (+corruption)
		QING_DECLINE_nudge = { var = qing_corruption_level  amount = 1 }
		LOG_line = { sys = QING  msg = "works: office vacant -> hydraulic-works oversight lapses (+corruption) for" }
		# the perennial infrastructure crises can still strike (each event's
		# own options carry the "no minister" path). Throttled by the cooldown.
		if = {
			limit = { NOT = { has_variable = qing_works_event_cooldown }  NOT = { has_variable = qing_gc_event_slot_used } }  # [BT-28] share court slot
			set_variable = { name = qing_works_event_cooldown  days = 270 }  # ~3 quarters
			random = {
				chance = 30
				set_variable = { name = qing_gc_event_slot_used  value = 1 }  # [BT-28] claim court slot only after the chance succeeds (mirror frontier roll; avoids majority-case over-suppression)
				random_list = {
					# Yellow River dike breach (黃河河堤)
					20 = {
						trigger = { treasury >= 400 }
						trigger_event = { id = qing_works.1  days = { 5 20 } }
					}
					# the Grand Canal silts up (大運河 漕運)
					20 = {
						trigger = { treasury >= 360 }
						trigger_event = { id = qing_works.2  days = { 5 20 } }
					}
				}
			}
		}
	}
	# --- FILLED office: offer the domain events on a throttled roll -------
	else_if = {
		limit = {
			has_variable = qing_office_works_holder
			var:qing_office_works_holder = { is_alive = yes  employer = ROOT }
			NOT = { has_variable = qing_works_event_cooldown }
			NOT = { has_variable = qing_gc_event_slot_used }  # [BT-28] share court slot
		}
		set_variable = { name = qing_works_event_cooldown  days = 270 }  # ~3 quarters
		random = {
			chance = 40
			set_variable = { name = qing_gc_event_slot_used  value = 1 }  # [BT-28] claim court slot only after the chance succeeds (mirror frontier roll)
			random_list = {
				# Yellow River dike breach (黃河河堤): repair well / cheap / ignore
				16 = {
					trigger = { treasury >= 400 }
					trigger_event = { id = qing_works.1  days = { 5 20 } }
				}
				# the Grand Canal silts up (大運河 漕運): dredge / cheap / abandon
				16 = {
					trigger = { treasury >= 360 }
					trigger_event = { id = qing_works.2  days = { 5 20 } }
				}
				# rebuild a section of the Great Wall (長城): a prestige project
				12 = {
					trigger = { treasury >= 440  stability >= 40 }
					trigger_event = { id = qing_works.3  days = { 5 20 } }
				}
				# [#188] a memorial to raise a new specialty state manufactory
				#   (織造局/御窯廠/茶廠/棉紡織坊/鹽場). Gate on the same finesse >= 7
				#   the build effect requires + a specialty province lacking its works.
				12 = {
					trigger = {
						treasury >= 380
						var:qing_office_works_holder = { finesse >= 7 }
						OR = {
							any_owned_province = { trade_goods = silk           NOT = { has_building = qing_silk_filature_building } }
							any_owned_province = { trade_goods = porcelain      NOT = { has_building = qing_porcelain_kiln_building } }
							any_owned_province = { trade_goods = tea            NOT = { has_building = qing_tea_workshop_building } }
							any_owned_province = { trade_goods = textile_fibres NOT = { has_building = qing_cotton_workshop_building } }
							any_owned_province = { trade_goods = salt           NOT = { has_building = qing_salt_yard_building } }
						}
					}
					trigger_event = { id = qing_works.6  days = { 5 20 } }
				}
				# river-works embezzlement scandal (河工貪腐): only in a corrupt court
				12 = {
					trigger = {
						has_variable = qing_corruption_level
						var:qing_corruption_level >= 50
					}
					trigger_event = { id = qing_works.4  days = { 5 20 } }
				}
			}
		}
	}
	# [#323] The grain-tribute logistics (漕運): recompute canal condition from the
	# depots/dikes + neglect, run the quarterly grain delivery-vs-consumption balance,
	# and apply the concrete consequence (a low capital reserve accelerates banner decay;
	# a critical one triggers the famine-response dilemma). Self-guards; CHI-gated inside.
	# (se_QING_CANAL.txt) Runs every works pulse, independent of the office-vacancy branch
	# above — the grain must flow (or fail to) whether or not the Works event menu fires.
	QING_canal_quarterly_tick = yes
	LOG_exit = { sys = QING  fn = "QING_works_pulse"  result = OK }
}

# ======================================================================
# COMMISSION A RIVER DIKE / LEVEE (河堤) in a high-population province.
# The Yellow River flood-control system. Built in country scope; picks the
# most populous province that doesn't already have one (mirrors the yamen/shuyuan
# idiom in se_QING_GOVERNANCE.txt). Cost + effectiveness scaled by the Works
# holder's finesse: a capable minister (finesse >= 9) builds at standard cost,
# a mediocre one (finesse < 9) incurs overruns (the corruption/graft layer), and
# a VACANT office means the project fails entirely (no holder -> no build).
# ======================================================================
# [#115] Store the STANDARD-tier dike-repair cost into a country var so the dike-breach EVENT
# tooltips (qing_works.1.a/.b, qing_works.5.a) can display the live figure with #114's proven
# GetVariable loc form. Called from each event's immediate (runs before the window renders, so the
# tooltip reads it). Same-tick country set-then-read is proven safe (se_QING_EXAM.txt:84-87). The
# panel button uses the live ScriptValue datafunction instead (no immediate to stash a var).
QING_DIKE_set_display_cost = {
	set_variable = { name = qing_dike_cost_display  value = QING_dike_cost_standard_svalue }
}

QING_works_build_dike = {
	LOG_enter = { sys = QING  fn = "QING_works_build_dike" }
	# $cheap$ = yes/no flag: cheap route skips finesse check, always builds but
	# drives up corruption (the graft-ridden shortcut). If not cheap, gate on holder.
	if = {
		limit = { has_variable = qing_office_works_holder }
		var:qing_office_works_holder = { save_scope_as = works_minister }
	}
	
	if = {
		limit = {
			OR = {
				# cheap route: always builds (gated at call site on no holder / low finesse)
				$cheap$ = yes
				# capable route: Works holder exists + finesse adequate
				AND = {
					has_variable = qing_office_works_holder
					var:qing_office_works_holder = { is_alive = yes  employer = ROOT  finesse >= 7 }
				}
			}
			any_owned_province = { count >= 1  has_minor_river = yes  NOT = { has_building = qing_dike_building } }
			# [#115] corvée manpower self-gate (mirrors the great_wall/grand_canal manpower>=10 self-gate,
			# scaled to the dike's smaller -5 levy). Keeps the manpower levy ATOMIC with the build — the
			# callers also gate manpower>=5, so this only ever backstops a direct verb call.
			manpower >= 5
		}
		# [#115] CHARGE FIRST, then build. add_building_level applies immediately (the province reports
		# has_building right after the call), and QING_dike_count / the cost svalues re-evaluate live on
		# every reference — so charging AFTER the build would read count=N+1 and overcharge by the +40/dike
		# step, breaking both the 4-dike→335/380 baseline and gate==charge. Reading the cost here (before
		# the add_building_level below) evaluates it on the pre-build network of N, matching the caller gates.
		# cost RISES with the dike network (河工 ballooning): expert tier if holder finesse>=9, else the
		# standard/cheap tier. Both scale +40/dike off QING_dike_count; expert is always <= standard.
		if = {
			limit = {
				has_variable = qing_office_works_holder
				var:qing_office_works_holder = { is_alive = yes  finesse >= 9 }
			}
			add_treasury = { value = QING_dike_cost_expert_svalue  multiply = -1 }
			TREASURY_LOG_it = { amount = { value = QING_dike_cost_expert_svalue  multiply = -1 } }
			LOG_line = { sys = QING  msg = "works: build dike (capable minister, expert-tier cost scaling with dike count, +5 corvee manpower) at for" }
		}
		else = {
			add_treasury = { value = QING_dike_cost_standard_svalue  multiply = -1 }
			TREASURY_LOG_it = { amount = { value = QING_dike_cost_standard_svalue  multiply = -1 } }
			# cheap route or mediocre holder -> graft
			if = {
				limit = { $cheap$ = yes }
				QING_DECLINE_nudge = { var = qing_corruption_level  amount = 8 }
				LOG_line = { sys = QING  msg = "works: build dike CHEAPLY (corruption +8, standard-tier cost scaling with dike count, +5 corvee manpower) at for" }
			}
			else = {
				QING_DECLINE_nudge = { var = qing_corruption_level  amount = 4 }
				LOG_line = { sys = QING  msg = "works: build dike (mediocre minister finesse<9, corruption +4, standard-tier cost scaling with dike count, +5 corvee manpower) at for" }
			}
		}
		# [#115] corvée labour: the dike, like the great monuments, costs manpower as well as gold. Levied
		# once inside the atomic build-if; the flat -5 is order-independent of the count-scaled treasury above.
		add_manpower = -5
		# pick the most populous RIVER province without a dike and build one. [#24 sweep] the has_minor_river
		# guard mirrors qing_dike_building's own potential — add_building_level RESPECTS potential, so an
		# unconstrained pop-pick could land on a riverless province and silently drop (paying the cost for nothing).
		ordered_owned_province = {
			limit = { has_minor_river = yes  NOT = { has_building = qing_dike_building } }
			order_by = total_population
			max = 1
			add_building_level = qing_dike_building
			save_scope_as = dike_province
		}
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_works_build_dike"  reason = "no Works holder or no suitable province for" }
	}
	LOG_exit = { sys = QING  fn = "QING_works_build_dike"  result = OK }
}

# COMMISSION A GRAND CANAL DEPOT (漕運倉) in a trade/population centre.
# The grain-tribute canal system. Same finesse-gated cost scaling as dike.
# ======================================================================
QING_works_build_canal_depot = {
	LOG_enter = { sys = QING  fn = "QING_works_build_canal_depot" }
	if = {
		limit = { has_variable = qing_office_works_holder }
		var:qing_office_works_holder = { save_scope_as = works_minister }
	}
	
	if = {
		# [BT-canal-option 2026-07-23] Build whenever there is a province lacking a depot — the throne can
		# ALWAYS choose to dredge, with or without a seated Works Minister. (Previously this required either
		# $cheap$=yes OR a finesse>=7 minister, so the EXPENSIVE proper-dredge option silently built NOTHING
		# when no capable minister sat — the reported "only cheap-fix or abandon" symptom.) The minister still
		# governs COST/QUALITY in the tiers below; his absence no longer blocks the good option outright.
		# [#24 sweep] the region guard mirrors qing_canal_depot_building's own potential (Zhili/Shandong/Jiangsu/
		# Zhejiang — the Grand Canal corridor). add_building_level RESPECTS potential, so an unconstrained pop-pick
		# could land off-corridor and silently drop while still charging the cost.
		limit = {
			any_owned_province = { count >= 1  OR = { is_in_region = Zhili  is_in_region = Shandong  is_in_region = Jiangsu  is_in_region = Zhejiang }  NOT = { has_building = qing_canal_depot_building } }
		}
		ordered_owned_province = {
			limit = { OR = { is_in_region = Zhili  is_in_region = Shandong  is_in_region = Jiangsu  is_in_region = Zhejiang }  NOT = { has_building = qing_canal_depot_building } }
			order_by = total_population
			max = 1
			add_building_level = qing_canal_depot_building
			save_scope_as = canal_province
		}
		if = {
			limit = {
				has_variable = qing_office_works_holder
				var:qing_office_works_holder = { is_alive = yes  finesse >= 9 }
			}
			add_treasury = -320
			TREASURY_LOG_it = { amount = -320 }
			LOG_line = { sys = QING  msg = "works: built canal depot (capable, cost 320) at for" }
		}
		else = {
			add_treasury = -370
			TREASURY_LOG_it = { amount = -370 }
			if = {
				limit = { $cheap$ = yes }
				QING_DECLINE_nudge = { var = qing_corruption_level  amount = 7 }
				LOG_line = { sys = QING  msg = "works: built canal depot CHEAPLY (corruption +7, cost 370) at for" }
			}
			else = {
				QING_DECLINE_nudge = { var = qing_corruption_level  amount = 3 }
				LOG_line = { sys = QING  msg = "works: built canal depot (mediocre minister, corruption +3, cost 370) at for" }
			}
		}
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_works_build_canal_depot"  reason = "no Works holder or no suitable province for" }
	}
	LOG_exit = { sys = QING  fn = "QING_works_build_canal_depot"  result = OK }
}

# ======================================================================
# [#188 task #188] COMMISSION A SPECIALTY-PRODUCTION WORKS — ties the Board of
# Works (工部) to the new 1815 specialty buildings (qing_production_buildings.txt).
# The user's integration example: "the Grand Minister of Works would be directly
# affected by these changes." The Works minister can raise an Imperial Silk
# Manufactory / porcelain kiln / tea works / cotton works / salt yard in a
# province that PRODUCES the matching raw good but lacks the works — extending
# the empire's state industry. Finesse-gated cost, same idiom as dike/canal/wall.
#   $building$ = a qing_production_buildings key
#   $good$     = the trade good the works requires (guards site selection)
#   $cheap$    = yes/no (cheap route skips finesse gate, drives corruption)
# ======================================================================
QING_works_build_specialty = {
	LOG_enter = { sys = QING  fn = "QING_works_build_specialty" }

		limit = { has_variable = qing_office_works_holder }
		var:qing_office_works_holder = { save_scope_as = works_minister }

		limit = {
			OR = {
				$cheap$ = yes

					has_variable = qing_office_works_holder

# ======================================================================
# [#434] THE GREAT MONUMENTS — the Great Wall (長城) and Grand Canal (大運河) proper. Great
# labour projects: each levies BOTH treasury AND manpower (the corvée levies that raised them),
# and is a raise-ONCE standing wonder. Called from the great-project decisions/events. Scope:
# COUNTRY (CHI). add_manpower = -N is the proven manpower-spend idiom (add_manpower widespread,
# negative form proven upstream TI/Invictus). Country-scope prestige reward (add_legitimacy /
# add_stability) lives HERE (valid at country scope), not on the province building.
# ======================================================================
QING_works_build_great_wall = {
	LOG_enter = { sys = QING  fn = "QING_works_build_great_wall" }
	if = {
		limit = {
			NOT = { any_owned_province = { has_building = qing_great_wall_building } }
			# [#24 sweep] require a VALID wall-belt province to exist BEFORE levying the corvée — else the
			# 620 treasury + 10 manpower were charged (and legitimacy/stability granted) while the wall
			# silently dropped off-region. This keeps the cost and the monument atomic.
			any_owned_province = { OR = { is_in_region = Zhili  is_in_region = Shanxi  is_in_region = Shaanxi  is_in_region = Gansu  is_in_region = Liaoning }  NOT = { has_building = qing_great_wall_building } }
			# a seated Minister of Works of finesse 7+ (mirrors the panel is_valid + the tooltip, so the
			# effect is self-guarding if ever called outside the GUI — the great labour needs its director).
			has_variable = qing_office_works_holder
			var:qing_office_works_holder = { is_alive = yes  employer = ROOT  finesse >= 7 }
			treasury >= 620
			manpower >= 10
			exists = capital_scope
		}
		# levy the corvée: gold + manpower both.
		add_treasury = -620
		TREASURY_LOG_it = { amount = -620 }
		add_manpower = -10
		# raise the monument at a most-populous NORTHERN wall-belt province lacking it. [#24 sweep] the
		# region guard mirrors qing_great_wall_building's (now-real) potential — without it the pop-pick
		# could land on a southern city and silently drop the wall (add_building_level respects potential)
		# while the 620 treasury + 10 manpower were still levied and the legitimacy/stability still granted.
		if = {
			limit = { any_owned_province = { OR = { is_in_region = Zhili  is_in_region = Shanxi  is_in_region = Shaanxi  is_in_region = Gansu  is_in_region = Liaoning }  NOT = { has_building = qing_great_wall_building } } }
			ordered_owned_province = {
				limit = { OR = { is_in_region = Zhili  is_in_region = Shanxi  is_in_region = Shaanxi  is_in_region = Gansu  is_in_region = Liaoning }  NOT = { has_building = qing_great_wall_building } }
				order_by = total_population
				max = 1
				add_building_level = qing_great_wall_building
			}
		}
		# the monument's prestige (country-scope, so here not on the building).
		add_legitimacy = 10
		add_stability = 1
		LOG_line = { sys = QING  msg = "works: the Great Wall (長城) is raised — 620 treasury + 10 manpower levied for" }
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_works_build_great_wall"  reason = "already built, no seated Works minister, or insufficient treasury/manpower/capital for" }
	}
	LOG_exit = { sys = QING  fn = "QING_works_build_great_wall"  result = OK }
}
