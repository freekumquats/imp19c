# se_QING_WAGES.txt — [fix, task #2 2026-08-17] FLAT officeholder salary, rendered in the vanilla
# character WEALTH change tooltip.
#
# HISTORY / WHY FLAT:
# #101/#5 attached `monthly_wage_for_character` (a vanilla Jomini field) to every Qing wage-bearing
# character modifier, expecting the engine to pay each holder a % of "nation's income" like vanilla's
# own offices. A live boot confirmed the engine READS the modifiers but pays exactly 0.00 -- this mod
# runs a fully custom income system in parallel with vanilla's native economy (imp19c-two-trade-
# systems), and the vanilla formula scales against a stat the mod never populates.
# Task #9 then paid each post DIRECTLY by a quarterly add_gold of (rate x INCOME_national_total_
# quarterly). That WORKED but could NEVER show in the vanilla character WEALTH tooltip ("Wealth
# changes by X each month due to: ...") -- that tooltip is engine-generated and aggregates ONLY a
# character's `monthly_character_wealth`-bearing modifiers; an add_gold is invisible to it.
#
# TASK #2 REQUIREMENT: the salary MUST appear in that vanilla wealth tooltip. The only lever is a
# static `monthly_character_wealth` on a modifier the holder already carries. A static value cannot
# track live income, so per the user's directive ("just pay all of them a flat rate if that makes it
# easier") the salary is now a FLAT monthly_character_wealth = 5 on each existing per-post marker
# modifier: qing_officeholder (all 13 GC seats incl. chancellor), the 4 commissioner + 3 subpost
# *_office shells, qing_amban_resident, and qing_customs_inspector_general. Those modifiers already
# have a full add-on-appoint / strip-on-depart lifecycle, so the tooltip line appears and vanishes
# automatically, is Qing-only, uses no vanilla offices, and does not override the tooltip GUI.
#
# ====================================================================================
# LOUD REGRESSION NOTE: this REVERSES task #9's income-scaling. Pay is no longer 1% of live quarterly
# income -- it is a flat 5 gold/month/post. Deliberate, user-directed, to satisfy the tooltip
# requirement. Consequence: wages no longer grow with the empire (arguably realistic for fixed Qing
# stipends), and the two #9-review bug classes (deficit-quarter payment inversion; amban employer=ROOT
# filter) are now MOOT -- a flat modifier can't hit either.
# ====================================================================================
#
# WHAT THIS FILE STILL DOES (the modifier IS the payment now): it only TALLIES the quarterly TREASURY
# COST so the topbar income tooltip's wage rows stay honest. WAGE_gc_admin_paid_last /
# WAGE_gc_military_paid_last feed INCOME_cost_{administrator,military}_wages_country
# (INCOME_svalues.txt), which sum into INCOME_national_total_quarterly, which add_treasury reads --
# i.e. these are a REAL treasury outflow, NOT a cosmetic display. Each holder costs the treasury
# (5/month x 3 months) = 15/quarter, matching the 5/month the modifier pays him. NO add_gold, NO
# income-read, NO deficit guard (a flat cost is always incurred, and the modifier pays the character
# from his own wealth pool independent of the treasury balance).
#   Military bucket: War + Guard Commandant GC seats + the imperial guardsman corps.
#   Administrator bucket: every other seat/post (incl. chancellor, emeritus, ambans).
# The flat quarterly cost per holder (15) MUST track the monthly_character_wealth V (5) on the markers.
########################################################################

# Tally ONE Grand Council seat's flat quarterly wage COST. $office$ = office key (matches the council
# enumeration), $bucket$ = admin/military. Scope: COUNTRY (CHI). The holder is PAID via
# monthly_character_wealth on qing_officeholder; this only records the treasury cost. No-op on a
# vacant/dead seat.
QING_pay_gc_seat_wage = {
	if = {
		limit = {
			has_variable = qing_office_$office$_holder
			var:qing_office_$office$_holder = { is_alive = yes  employer = ROOT }
		}
		change_variable = { name = WAGE_gc_$bucket$_paid_tmp  add = 15 }   # 5/mo x 3
	}
}

# Tally ONE single-holder commissioner/customs post's flat quarterly cost. $holder_var$ = the
# country-scope pointer var (e.g. qing_salt_commissioner_holder). All Administrator. Scope: COUNTRY.
QING_pay_single_post_wage = {
	if = {
		limit = {
			has_variable = $holder_var$
			var:$holder_var$ = { is_alive = yes  employer = ROOT }
		}
		change_variable = { name = WAGE_gc_admin_paid_tmp  add = 15 }   # 5/mo x 3
	}
}

# Tally every CHI-EMPLOYED character carrying $marker$ (a subpost corps marker). $bucket$ =
# admin/military. Scope: COUNTRY (CHI). NOT for the amban -- his employer is the subject he's posted
# to, not CHI; see QING_pay_amban_wage below.
QING_pay_marked_corps_wage = {
	every_character = {
		limit = {
			employer = ROOT
			has_variable = $marker$
		}
		ROOT = {
			change_variable = { name = WAGE_gc_$bucket$_paid_tmp  add = 15 }   # 5/mo x 3
		}
	}
}

# Tally every character carrying qing_amban_marker into CHI's Administrator bucket. No employer filter
# -- a posted amban's employer is the subject he resides in, not CHI (se_QING_AMBAN.txt), so an
# employer=ROOT gate would always exclude him. The marker alone is CHI's own, only ever granted by
# CHI's own amban-appointment path, so it is a safe sole filter.
QING_pay_amban_wage = {
	every_character = {
		limit = {
			has_variable = qing_amban_marker
		}
		ROOT = {
			change_variable = { name = WAGE_gc_admin_paid_tmp  add = 15 }   # 5/mo x 3
		}
	}
}

# QING_pay_officeholder_wages — the quarterly entry point. Scope: COUNTRY (CHI, tag-guarded by the
# caller like every other CHI-only quarterly pulse). Tallies the two national accumulators, then
# publishes them as the _last snapshot the income tooltip / cost svalues read (mirrors the
# qing_salt_income_last / qing_caravan_prosperity publish idiom).
QING_pay_officeholder_wages = {
	set_variable = { name = WAGE_gc_admin_paid_tmp     value = 0 }
	set_variable = { name = WAGE_gc_military_paid_tmp  value = 0 }

	# --- Grand Council seats: War + Guard Commandant are Military; the other 10 are Administrator.
	QING_pay_gc_seat_wage = { office = war                bucket = military }
	QING_pay_gc_seat_wage = { office = guard_commandant   bucket = military }
	QING_pay_gc_seat_wage = { office = personnel          bucket = admin }
	QING_pay_gc_seat_wage = { office = revenue            bucket = admin }
	QING_pay_gc_seat_wage = { office = rites              bucket = admin }
	QING_pay_gc_seat_wage = { office = justice            bucket = admin }
	QING_pay_gc_seat_wage = { office = works              bucket = admin }
	QING_pay_gc_seat_wage = { office = censor             bucket = admin }
	QING_pay_gc_seat_wage = { office = lifanyuan          bucket = admin }
	QING_pay_gc_seat_wage = { office = chamberlain        bucket = admin }
	QING_pay_gc_seat_wage = { office = zongli             bucket = admin }
	QING_pay_gc_seat_wage = { office = grand_secretariat  bucket = admin }

	# --- Chancellor: a DISTINCT 13th office (領班軍機大臣; office=chancellor, NOT one of the 12 seats
	# above -- one man holds ONE office), so this is his SOLE tally entry, keyed on his own holder var.
	# He is PAID 5/mo via qing_officeholder like every seat; qing_officeholder_chancellor_bonus stays
	# EMPTY (flat uniform wage -- no 2x tier). This block only records his flat quarterly cost.
	if = {
		limit = {
			has_variable = qing_office_chancellor_holder
			var:qing_office_chancellor_holder = { is_alive = yes  employer = ROOT }
		}
		change_variable = { name = WAGE_gc_admin_paid_tmp  add = 15 }   # 5/mo x 3
	}

	# --- Emperor Emeritus (太上皇): reuses QING_office_appoint (office=emeritus), so he carries
	# qing_officeholder and is PAID 5/mo by monthly_character_wealth like any high officeholder. Fund
	# that cost here so the treasury debit matches (rare abdication path; a retired emperor drew
	# imperial income historically). office=emeritus is NOT among the 12 seats above -- no double-count.
	QING_pay_gc_seat_wage = { office = emeritus  bucket = admin }

	# --- Single-holder commissioner/customs posts (all Administrator).
	QING_pay_single_post_wage = { holder_var = qing_salt_commissioner_holder }
	QING_pay_single_post_wage = { holder_var = qing_caravan_super_holder }
	QING_pay_single_post_wage = { holder_var = qing_hoppo_holder }
	QING_pay_single_post_wage = { holder_var = qing_opium_commissioner_holder }
	QING_pay_single_post_wage = { holder_var = qing_customs_ig_holder }

	# --- Amban (per-subject, multiple possible): Administrator. Dedicated macro -- see
	# QING_pay_amban_wage's own comment for why it can't share QING_pay_marked_corps_wage.
	QING_pay_amban_wage = yes
	# --- Zongli diplomat / censor-inspector subposts: Administrator. Imperial guardsman subpost:
	# Military (the Guard Commandant's own subordinates, per user directive).
	QING_pay_marked_corps_wage = { marker = qing_zongli_diplomat        bucket = admin }
	QING_pay_marked_corps_wage = { marker = qing_is_censor_inspector    bucket = admin }
	QING_pay_marked_corps_wage = { marker = qing_is_imperial_guardsman  bucket = military }

	# --- publish the quarterly snapshot for the income tooltip / cost svalues to read.
	set_variable = { name = WAGE_gc_admin_paid_last     value = var:WAGE_gc_admin_paid_tmp }
	set_variable = { name = WAGE_gc_military_paid_last  value = var:WAGE_gc_military_paid_tmp }
	remove_variable = WAGE_gc_admin_paid_tmp
	remove_variable = WAGE_gc_military_paid_tmp
	LOG_line = { sys = QING  msg = "wages: officeholder salary COST tallied (flat; see WAGE_gc_admin_paid_last / WAGE_gc_military_paid_last)" }
	# [task #2 2026-08-17] emit the two flat quarterly cost totals so the boot confirms the tally fires
	# and the magnitude is sane. debug_log cannot render a variable's value directly in effect context
	# (proven, se_ECON_LOG.txt's own ECON_LOG_curx_exact header) -- reuses that file's proven exact-
	# tick-count idiom instead of inventing a new one.
	debug_log = "IMP19C CURXV LABEL wageadmin"   # x0.5  sum of ~15 admin posts at flat 15/quarter each
	set_variable = { name = ECON_LOG_tickval  value = var:WAGE_gc_admin_paid_last }
	set_variable = { name = ECON_LOG_tickscale  value = 0.5 }
	ECON_LOG_curx_tick_emit = yes
	debug_log = "IMP19C CURXV LABEL wagemilitary"   # x1  fewer/smaller posts (War+Guard Commandant+guardsmen)
	set_variable = { name = ECON_LOG_tickval  value = var:WAGE_gc_military_paid_last }
	set_variable = { name = ECON_LOG_tickscale  value = 1 }
	ECON_LOG_curx_tick_emit = yes
	debug_log = "IMP19C CURXV LABEL end"
}
