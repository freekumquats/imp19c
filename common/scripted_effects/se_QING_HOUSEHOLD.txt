# se_QING_HOUSEHOLD.txt — the Grand Secretary's charge: the Imperial Household
# Department (內務府 / 總管內務府大臣), the privy purse (內帑), the eunuchs
# (太監), and the imperial workshops (造辦處).
#
# The Grand Secretary (內務府) heads the throne's PERSONAL establishment — the
# privy purse (內帑, the emperor's private treasury, distinct from the state
# 戶部 treasury), the palace eunuchs (太監, a perennial danger — the Qing kept
# them tightly checked after past dynasties fell to eunuch power), and the
# imperial workshops (造辦處). The closest man to the sovereign; his charge is
# the emperor's own household and standing.
#
# A capable, loyal, high-charisma Grand Secretary keeps the privy purse full
# (a private reserve the throne can tap), the eunuchs in check (no eunuch
# faction), and the workshops productive (prestige/luxury); a venal or vacant
# office means eunuch overreach, a drained privy purse, and a diminished sovereign.
#
# STATE (CHI country scope):
#   [#424] THE PRIVY PURSE IS NOW current_ruler.wealth — the emperor's REAL vanilla character
#   wealth (內帑 = the sovereign's private silver, exactly what `wealth` models), NOT a 0..100
#   country meter. Every purse flow is add_gold on the emperor; every read is his wealth. Scale:
#   the old meter baseline 50 maps to a comfortable ruler wealth ~200 (old×4); the 內帑 passes to
#   the successor on a change of sovereign (QING_household_carryover_purse, on_ruler_change).
#   qing_eunuch_power      0..100 faction strength (NOT stored — derived from
#                          the presence of a flagged eunuch character; 0 when
#                          no eunuch faction exists)
# Character-scope mark on the faction-leading eunuch (if one exists):
#   qing_eunuch_faction_leader   flag (the dangerous eunuch)
#
# PREFIX: QING_household_
########################################################################

# ======================================================================
# INIT — seed the privy purse. Called once at game start from on_game_initialized.
# ======================================================================
QING_household_init = {
	# [#424] the privy purse is now the emperor's own vanilla wealth (current_ruler.wealth),
	# which the engine already seeds, so there is nothing to initialise. Kept as a no-op stub
	# so the many callers (QING_household_pulse etc.) need no change. The old 0..100
	# qing_privy_purse country meter is RETIRED; the 內帑 is the sovereign's real private silver.
	LOG_enter = { sys = QING  fn = "QING_household_init" }
	# [boot #12] seed the opening palace-eunuch corps (太監) once at game start.
	QING_household_seed_eunuchs = yes
	LOG_exit = { sys = QING  fn = "QING_household_init"  result = OK }
}

# ======================================================================
# [boot #12] PALACE-EUNUCH CORPS (太監). The user saw "The Eunuchs" as a field in the
# Household panel but only a threat-meter stood behind it — no actual eunuch characters.
# The Qing kept a large palace establishment of eunuchs (the 敬事房 administered them);
# they are the concrete objects the eunuch mechanic should manage (concrete-over-abstract
# house rule), and the faction-danger mechanic promotes one of THESE rather than conjuring a
# fresh stranger. Seed a small opening corps + keep a self-healing roster, mirroring the
# proven harem consort roster (se_QING_HAREM.txt QING_harem_mint_consort / _recompute_roster).
#
# STATE:  qing_household_eunuchs  country variable_list of the living palace eunuchs
#         qing_eunuch_count       their count (GUI + perf reads)
# Char:   qing_is_palace_eunuch   marker flag on each eunuch character
# ======================================================================
QING_household_seed_eunuchs = {
	if = {
		limit = { NOT = { has_variable = qing_eunuchs_initialized } }
		set_variable = { name = qing_eunuchs_initialized  value = 1 }
		set_variable = { name = qing_eunuch_count  value = 0 }
		clear_variable_list = qing_household_eunuchs
		# an opening establishment of four palace eunuchs (only if the throne is filled).
		if = {
			limit = { exists = current_ruler }
			QING_household_mint_eunuch = yes
			QING_household_mint_eunuch = yes
			QING_household_mint_eunuch = yes
			QING_household_mint_eunuch = yes
		}
		LOG_line = { sys = QING  msg = "household: seeded the opening palace-eunuch corps for [ROOT.GetTag]" }
	}
}

# QING_household_mint_eunuch — create ONE palace eunuch and add him to the roster.
# Scope: COUNTRY (CHI). Palace eunuchs were overwhelmingly Han commoners bound to the
# throne's household. #90 rule: NO modifiers granted inside create_character; markers set
# in a follow-up scope. Mirrors QING_harem_mint_consort.
QING_household_mint_eunuch = {
	create_character = {
		age = 30
		culture = han            # palace eunuchs were overwhelmingly Han commoners
		religion = confucianism
		# a eunuch's gifts are administrative/social, not martial.
		add_finesse = 4
		add_charisma = 3
		add_zeal = 1
		save_scope_as = qing_new_eunuch
	}
	scope:qing_new_eunuch = {
		set_home_country = ROOT
		set_variable = { name = qing_is_palace_eunuch  value = 1 }
		QING_post_stamp = { post = palace_eunuch }   # [#118]
		# [B12 REVERTED — BOOT-CRASHFIX, then fix #31 2026-08-13] The castrated trait grant was
		# REMOVED here because this function ALSO mints the opening 4-eunuch corps at BOOT
		# (on_game_initialized -> QING_household_init -> QING_household_seed_eunuchs), and
		# `castrated` is type=health — adding a HEALTH trait at gamestate construction (inline or in
		# a same-tick follow-up scope) is the proven boot-crash class. Instead, EVERY eunuch (boot-seeded
		# or later runtime-minted) gets the trait backfilled from QING_household_pulse, every quarter
		# (QING_household_backfill_eunuch_traits, below) — a call site well past construction, so the
		# HEALTH trait grant is boot-safe there. Never grant it here, at construction.
	}
	add_to_variable_list = { name = qing_household_eunuchs  target = scope:qing_new_eunuch }
	change_variable = { name = qing_eunuch_count  add = 1 }
	LOG_line = { sys = QING  msg = "household: a eunuch enters the palace establishment for [ROOT.GetTag]" }
}

# [fix #31 2026-08-13] QING_household_backfill_eunuch_traits — grant the castrated (太監) HEALTH
# trait to every LIVING eunuch who does not yet carry it. Scope: COUNTRY (CHI). Called ONLY from
# QING_household_pulse (every quarter, well past gamestate construction) — NEVER from
# QING_household_init's own boot-time seed (the HEALTH-trait-at-boot crash class). Self-guarded
# (has_trait check) so an already-traited eunuch, or a repeat call, is a clean no-op — never fight
# the trait system, per the established idiom used for #103's `just` grant
# (se_QING_ACCOUNTABILITY.txt:391).
QING_household_backfill_eunuch_traits = {
	every_in_list = {
		variable = qing_household_eunuchs
		limit = { is_alive = yes  NOT = { has_trait = castrated } }
		add_trait = castrated
	}
}

# QING_household_recompute_eunuchs — rebuild the eunuch roster fresh from the live
# characters carrying the marker (alive + still in the household). Self-healing (prunes the
# dead/departed), keeps the count honest for the GUI. Bounded O(characters); run on the
# household pulse + on panel open. Scope: COUNTRY (CHI). Mirrors QING_harem_recompute_roster.
QING_household_recompute_eunuchs = {
	clear_variable_list = qing_household_eunuchs
	set_variable = { name = qing_eunuch_count  value = 0 }
	every_character = {
		limit = {
			has_variable = qing_is_palace_eunuch
			is_alive = yes
			employer = ROOT
		}
		save_scope_as = qing_roster_eunuch
		ROOT = {
			add_to_variable_list = { name = qing_household_eunuchs  target = scope:qing_roster_eunuch }
			change_variable = { name = qing_eunuch_count  add = 1 }
		}
	}
}

# ======================================================================
# [Batch 6 B1 2026-07-25] QING_household_compute_eunuch_power — give the dead qing_eunuch_count a CONSUMER:
# a single 0..100 signal the influence web reads. Scope: COUNTRY (CHI). Called each pulse AFTER the roster
# recompute (so the count is fresh). Formula (design B1, no boolean-in-arithmetic — each term a guarded
# branch): base = count x 8; +30 if a faction leader is entrenched; +10 if corruption >= 50 (a venal court
# lets them run); -15 if the chamberlain is a strong man (charisma >= 8 checks them). The B5 policy law caps
# the ceiling (restrict lowers it, empowered raises it). Clamp 0..100.
# ======================================================================
QING_household_compute_eunuch_power = {
	set_variable = { name = qing_eunuch_power  value = var:qing_eunuch_count }
	change_variable = { name = qing_eunuch_power  multiply = 8 }
	# an entrenched faction leader is the difference between a corps and a cabal.
	if = {
		limit = { any_character = { employer = ROOT  has_variable = qing_eunuch_faction_leader  is_alive = yes } }
		change_variable = { name = qing_eunuch_power  add = 30 }
	}
	# a venal court (high corruption) lets the eunuchs run.
	if = {
		limit = { has_variable = qing_corruption_level  var:qing_corruption_level >= 50 }
		change_variable = { name = qing_eunuch_power  add = 10 }
	}
	# a strong chamberlain (內務府大臣) checks the directorate.
	if = {
		limit = {
			has_variable = qing_office_chamberlain_holder
			var:qing_office_chamberlain_holder = { is_alive = yes  employer = ROOT  charisma >= 8 }
		}
		change_variable = { name = qing_eunuch_power  subtract = 15 }
	}
	# [B5] policy ceiling: restrict (裁抑) caps the corps low; empowered (寵信) lets it climb higher.
	if = {
		limit = { has_variable = qing_eunuch_policy  var:qing_eunuch_policy = 0 }
		if = { limit = { var:qing_eunuch_power > 40 }  set_variable = { name = qing_eunuch_power  value = 40 } }
	}
	else_if = {
		limit = { has_variable = qing_eunuch_policy  var:qing_eunuch_policy = 2 }
		# empowered: no extra cap below the 0..100 ceiling — the directorate may run to the crisis band.
	}
	# clamp 0..100 (standard band).
	if = { limit = { var:qing_eunuch_power < 0 }   set_variable = { name = qing_eunuch_power  value = 0 } }
	if = { limit = { var:qing_eunuch_power > 100 } set_variable = { name = qing_eunuch_power  value = 100 } }
}

# ======================================================================
# [Batch 6 B2 2026-07-25] QING_household_eunuch_consume — the eunuch establishment exerts its influence on
# the court each quarter, gated on qing_eunuch_power bands. Scope: COUNTRY (CHI). All small, all at the
# existing nudge sites (the P7 bias discipline). REVIEW-FIX doom-loop guards baked in: corruption nudge is
# gated < 70 (can't spiral past the crisis band); reform-balance hit fires ONLY at the very-high band (>= 80),
# not a third simultaneous penalty at every level; the restrict policy HALVES the corruption/backlog output.
# ======================================================================
QING_household_eunuch_consume = {
	if = { limit = { NOT = { has_variable = qing_eunuch_power } }  set_variable = { name = qing_eunuch_power  value = 0 } }
	# MID band (>= 50): the directorate skims — graft rises, memorials slow.
	if = {
		limit = { var:qing_eunuch_power >= 50 }
		# corruption creep, capped below the crisis band so eunuch graft can't spiral it out.
		if = {
			limit = { has_variable = qing_corruption_level  var:qing_corruption_level < 70 }
			QING_household_eunuch_corruption_nudge = yes
		}
		# [BT 2026-07-25] the graft is not just an abstract court meter — it shows on the EUNUCHS
		# THEMSELVES: each living palace eunuch's own vanilla character corruption climbs (visible on
		# his character card via [Character.GetCorruption]). Halved under the restrict policy (裁抑),
		# matching the country-level nudge. Bounded iteration over the roster; a self-cap keeps a
		# long-lived eunuch from running to the ceiling.
		QING_household_eunuch_char_corruption = yes
		# 票擬 backlog — the 敬事房 intercepts/slows the palace paperwork.
		if = { limit = { has_variable = qing_secretariat_backlog }  QING_household_eunuch_backlog_nudge = yes }
	}
	# VERY-HIGH band (>= 80): 太監專權 — the eunuchs are a reactionary bloc tilting the whole reform balance.
	if = {
		limit = { var:qing_eunuch_power >= 80  has_variable = qing_reform_faction_balance }
		# raw change_variable with the +-100 clamp (NOT the DECLINE_nudge macro — this meter is signed).
		change_variable = { name = qing_reform_faction_balance  subtract = 2 }
		if = { limit = { var:qing_reform_faction_balance < -100 } set_variable = { name = qing_reform_faction_balance  value = -100 } }
		LOG_line = { sys = QING  msg = "household: 太監專權 — the eunuch bloc drags the reform balance down for [ROOT.GetTag]" }
	}
}

# [BT 2026-07-25] per-character eunuch corruption — the graft shows on each palace eunuch's OWN
# vanilla corruption stat (add_corruption; visible on his character card). Iterates the living-eunuch
# roster list. Self-cap at corruption < 60 so a long-lived eunuch's number rises with the graft but
# does not silently run to the ceiling; halved under the restrict policy (裁抑) to match the country
# nudge. add_corruption is the proven vanilla verb (already used at the faction-leader spawn + the
# check-eunuchs lever below). Country scope (CHI); the list holds only THIS court's eunuchs.
QING_household_eunuch_char_corruption = {
	if = { limit = { NOT = { has_variable = qing_eunuch_policy } }  set_variable = { name = qing_eunuch_policy  value = 1 } }
	every_in_list = {
		variable = qing_household_eunuchs
		limit = { is_alive = yes  corruption < 60 }
		if = {
			limit = { ROOT = { has_variable = qing_eunuch_policy  var:qing_eunuch_policy = 0 } }
			add_corruption = 1   # restrict (裁抑): half output
		}
		else = {
			add_corruption = 2
		}
	}
}

# eunuch corruption creep — halved under the restrict policy. QING_DECLINE_nudge clamps 0..100.
QING_household_eunuch_corruption_nudge = {
	if = {
		limit = { has_variable = qing_eunuch_policy  var:qing_eunuch_policy = 0 }
		QING_DECLINE_nudge = { var = qing_corruption_level  amount = 1 }   # restrict: half output (rounds to +1)
	}
	else = {
		QING_DECLINE_nudge = { var = qing_corruption_level  amount = 2 }
	}
}

# eunuch memorial-interception backlog — halved under the restrict policy. Raw change_variable (backlog is
# a simple accumulator with its own relief elsewhere); mirrors the +6 fresh-rescript nudge, scaled small.
QING_household_eunuch_backlog_nudge = {
	if = {
		limit = { has_variable = qing_eunuch_policy  var:qing_eunuch_policy = 0 }
		change_variable = { name = qing_secretariat_backlog  add = 1 }   # restrict: half output
	}
	else = {
		change_variable = { name = qing_secretariat_backlog  add = 2 }
	}
}

# ======================================================================
# [Batch 6 B2 counterplay 2026-07-25] QING_household_eunuch_purge — the PLAYER lever (from the household
# panel): a strong chamberlain purges the directorate (裁抑內宦). Costs stability + a legitimacy dip (a
# palace purge is disruptive), knocks eunuch power down hard (-30 via a floor), clears the faction-leader
# flag, and relieves the backlog. Cooldown-gated (1825 days) so it is a deliberate act, not a spammable tap.
# Scope: COUNTRY (CHI). Its is_valid (needs a capable chamberlain + off cooldown) lives in the scripted_gui.
# ======================================================================
QING_household_eunuch_purge = {
	LOG_enter = { sys = QING  fn = "QING_household_eunuch_purge" }
	if = {
		limit = { NOT = { has_variable = qing_eunuch_purge_cd } }
		set_variable = { name = qing_eunuch_purge_cd  value = 1  days = 1825 }
		# clear the cabal's leader if one sits.
		every_character = {
			limit = { employer = ROOT  has_variable = qing_eunuch_faction_leader }
			remove_variable = qing_eunuch_faction_leader
		}
		# knock the power down hard (recomputed next pulse, but drop it now for immediate feedback).
		QING_household_compute_eunuch_power = yes
		if = { limit = { has_variable = qing_eunuch_power  var:qing_eunuch_power > 30 }  change_variable = { name = qing_eunuch_power  subtract = 30 } }
		if = { limit = { has_variable = qing_eunuch_power  var:qing_eunuch_power < 0 }   set_variable = { name = qing_eunuch_power  value = 0 } }
		# the purge relieves the intercepted memorials.
		if = { limit = { has_variable = qing_secretariat_backlog  var:qing_secretariat_backlog >= 5 }  change_variable = { name = qing_secretariat_backlog  subtract = 5 } }
		add_stability = -2
		add_legitimacy = -3
		LOG_line = { sys = QING  msg = "household: 裁抑內宦 — the chamberlain purges the eunuch directorate for [ROOT.GetTag]" }
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_household_eunuch_purge"  reason = "purge on cooldown" }
	}
	LOG_exit = { sys = QING  fn = "QING_household_eunuch_purge"  result = OK }
}

# ======================================================================
# [Batch 6 B3 2026-07-25] QING_household_eval_chief_eunuch — the Chief Eunuch seat (掌印太監). Scope:
# COUNTRY (CHI). When the directorate stays powerful (power >= 70) for a sustained spell, its ablest man
# entrenches as the 掌印太監 — the on-map face of 太監專權 (a 李蓮英-type; 1763 starts get an anonymous
# holder, famous eunuchs being post-1763). This is a NON-APPOINTABLE, DISPLAY-ONLY seat: it sets its OWN
# marker (qing_office_chief_eunuch_holder country var) + a standing modifier, and does NOT set
# qing_office_held (which would trip the 1:1 position validator — the design's explicit rule; the holder
# keeps only his qing_is_palace_eunuch marker). Entrenchment needs qing_eunuch_entrench_q >= 4 quarters.
# ======================================================================
QING_household_eval_chief_eunuch = {
	if = { limit = { NOT = { has_variable = qing_eunuch_entrench_q } }  set_variable = { name = qing_eunuch_entrench_q  value = 0 } }
	if = {
		# sustained high power ratchets the entrenchment counter up; anything lower bleeds it back down.
		limit = { has_variable = qing_eunuch_power  var:qing_eunuch_power >= 70 }
		change_variable = { name = qing_eunuch_entrench_q  add = 1 }
		if = { limit = { var:qing_eunuch_entrench_q > 8 }  set_variable = { name = qing_eunuch_entrench_q  value = 8 } }
	}
	else = {
		if = { limit = { var:qing_eunuch_entrench_q > 0 }  change_variable = { name = qing_eunuch_entrench_q  subtract = 1 } }
	}
	# SEAT: entrenched (>=4 quarters sustained) + a living faction leader in the corps -> he takes the seat.
	if = {
		limit = {
			var:qing_eunuch_entrench_q >= 4
			any_character = { employer = ROOT  has_variable = qing_eunuch_faction_leader  is_alive = yes }
		}
		# mirror the ablest serving faction-leader eunuch onto the display-only seat var (NOT qing_office_held).
		ordered_character = {
			limit = { employer = ROOT  has_variable = qing_eunuch_faction_leader  is_alive = yes }
			order_by = finesse
			max = 1
			ROOT = { set_variable = { name = qing_office_chief_eunuch_holder  value = prev } }
		}
		if = { limit = { NOT = { has_country_modifier = qing_chief_eunuch_seated } }  add_country_modifier = { name = qing_chief_eunuch_seated  duration = -1 } }
		LOG_line = { sys = QING  msg = "household: the 掌印太監 (Chief Eunuch) entrenches at court for [ROOT.GetTag]" }
	}
	else = {
		# no longer entrenched (purged, or power fell) -> vacate the seat + drop the modifier.
		if = { limit = { has_variable = qing_office_chief_eunuch_holder }  remove_variable = qing_office_chief_eunuch_holder }
		if = { limit = { has_country_modifier = qing_chief_eunuch_seated }  remove_country_modifier = qing_chief_eunuch_seated }
	}
}

# ======================================================================
# [Batch 6 B4 2026-07-25] QING_household_eunuch_event_roll — fire ONE eunuch-intrigue event per pulse when
# the band warrants, throttled by a cooldown so the court is not spammed. Scope: COUNTRY (CHI). Shares the
# GC event slot via the standard throttle. High power -> the overreach beat (household.8 閹黨); a strong
# chamberlain + elevated power -> the retrenchment opportunity (household.9). All is_triggered_only country
# events (never inlined here — trigger_event is a runtime dispatch).
# ======================================================================
QING_household_eunuch_event_roll = {
	if = {
		limit = {
			NOT = { has_variable = qing_eunuch_event_cd }
			has_variable = qing_eunuch_power
			var:qing_eunuch_power >= 60
		}
		set_variable = { name = qing_eunuch_event_cd  value = 1  days = 1460 }
		if = {
			# a strong chamberlain sees the chance to curb the corps (裁抑內宦).
			limit = {
				has_variable = qing_office_chamberlain_holder
				var:qing_office_chamberlain_holder = { is_alive = yes  employer = ROOT  charisma >= 8 }
			}
			trigger_event = { id = qing_household.9  days = { 5 20 } }
		}
		else = {
			# an unchecked directorate overreaches (閹黨).
			trigger_event = { id = qing_household.8  days = { 5 20 } }
		}
		LOG_line = { sys = QING  msg = "household: a eunuch-intrigue beat arises at court for [ROOT.GetTag]" }
	}
}

# ======================================================================
# [#424] QING_household_carryover_purse — the 內帑 is an institution of the throne: on a
# change of sovereign it passes to the successor, not into the void with the dead emperor's
# personal wealth. Called from on_ruler_change (scope:former_ruler is set there). Reads the
# former ruler's wealth into a country var, then grants it to the new current_ruler — the
# proven inheritance idiom (TI on_character_death_inv: set_variable value=<X.wealth>, then
# add_gold = { value = <var> }). Scope: COUNTRY (CHI).
# ======================================================================
QING_household_carryover_purse = {
	LOG_enter = { sys = QING  fn = "QING_household_carryover_purse" }
	if = {
		limit = { exists = scope:former_ruler  exists = current_ruler }
		set_variable = { name = qing_purse_carryover_tmp  value = scope:former_ruler.wealth }
		# only carry a POSITIVE balance (a bankrupt emperor bequeaths no privy purse; the new
		# sovereign is not saddled with the dead one's personal debts).
		if = {
			limit = { var:qing_purse_carryover_tmp > 0 }
			current_ruler = { add_gold = { value = ROOT.var:qing_purse_carryover_tmp } }
			LOG_line = { sys = QING  msg = "household: the privy purse passes to the new sovereign for [ROOT.GetTag]" }
		}
		remove_variable = qing_purse_carryover_tmp
	}
	LOG_exit = { sys = QING  fn = "QING_household_carryover_purse"  result = OK }
}

# ======================================================================
# PULSE — quarterly drift of the privy purse and eunuch danger, gated on the
# Grand Secretary holder's charisma + loyalty. Called from the quarterly
# governance pulse (after accountability recompute, so the metric is fresh).
# ======================================================================
QING_household_pulse = {
	LOG_enter = { sys = QING  fn = "QING_household_pulse" }
	QING_household_init = yes
	# [boot #12] refresh the palace-eunuch roster (prunes the dead/departed, keeps the count honest).
	QING_household_recompute_eunuchs = yes
	# [fix #31 2026-08-13] backfill the castrated trait on every living eunuch who lacks it, every
	# pulse — this runtime pulse is well past gamestate construction, so the HEALTH-trait grant that
	# QING_household_mint_eunuch's own boot-time seed cannot safely do is safe here. Runs unconditionally
	# (not gated behind the rare eunuch-faction roll below) so a seeded eunuch gets his trait promptly,
	# not only if/when a faction happens to arise.
	QING_household_backfill_eunuch_traits = yes
	# [Batch 6 B1/B2 2026-07-25] compute the eunuch power signal + let it consume into the court.
	QING_household_compute_eunuch_power = yes
	QING_household_eunuch_consume = yes
	# [Batch 6 B3 2026-07-25] evaluate the Chief Eunuch seat (掌印太監) from sustained high power.
	QING_household_eval_chief_eunuch = yes
	# [Batch 6 B4 2026-07-25] the intrigue-event roll (閹黨 / retrenchment / overreach), throttled.
	QING_household_eunuch_event_roll = yes

	# --- PRIVY PURSE DRIFT: capable loyal high-charisma Grand Secretary fills it
	# (+2/quarter), vacant or venal office drains it (-3/quarter).
	if = {
		limit = {
			has_variable = qing_office_chamberlain_holder
			var:qing_office_chamberlain_holder = {
				is_alive = yes
				employer = ROOT
				charisma >= 8
				loyalty >= 60
			}
		}
		if = { limit = { exists = current_ruler }  current_ruler = { add_gold = 8 } }
		LOG_line = { sys = QING  msg = "household pulse: privy purse +8 wealth (capable Grand Secretary) for [ROOT.GetTag]" }
	}
	else_if = {
		limit = {
			OR = {
				NOT = { has_variable = qing_office_chamberlain_holder }
				var:qing_office_chamberlain_holder = {
					OR = {
						NOT = { is_alive = yes }
						NOT = { employer = ROOT }
						loyalty < 40
						has_trait = corrupt
					}
				}
			}
		}
		if = { limit = { exists = current_ruler }  current_ruler = { add_gold = -12 } }
		LOG_line = { sys = QING  msg = "household pulse: privy purse -12 wealth (vacant/venal office) for [ROOT.GetTag]" }
	}

	# --- EUNUCH DANGER CHECK: if the Grand Secretary is absent/weak/disloyal AND
	# corruption is high, a eunuch faction MAY arise (one-time spawn when no faction
	# leader exists yet). CONCRETE: create a real eunuch character, flag him, and
	# grant him a power base (the eunuch danger is a man with clout, not a counter).
	if = {
		limit = {
			var:qing_corruption_level >= 50
			OR = {
				NOT = { has_variable = qing_office_chamberlain_holder }
				var:qing_office_chamberlain_holder = {
					OR = {
						NOT = { is_alive = yes }
						NOT = { employer = ROOT }
						charisma < 6
						loyalty < 50
					}
				}
			}
			# no eunuch faction already exists
			NOT = {
				any_character = {
					employer = ROOT
					has_variable = qing_eunuch_faction_leader
				}
			}
		}
		random = {
			chance = 20
			LOG_line = { sys = QING  msg = "household pulse: eunuch faction arising (corrupt + weak Grand Secretary) for [ROOT.GetTag]" }
			# [boot #12] CONCRETE: the faction leader rises from the EXISTING palace-eunuch corps
			# (the danger is one of the seeded 太監 growing overmighty, not a stranger materialising).
			# Guard against an empty corps (all dead/departed) by minting one first if needed.
			if = {
				limit = {
					NOT = {
						any_character = { employer = ROOT  has_variable = qing_is_palace_eunuch  is_alive = yes }
					}
				}
				QING_household_mint_eunuch = yes
			}
			# promote the most capable serving eunuch to faction leader + give him a power base.
			ordered_character = {
				limit = {
					employer = ROOT
					has_variable = qing_is_palace_eunuch
					is_alive = yes
					NOT = { has_variable = qing_eunuch_faction_leader }
					# [PERF-AUDIT 2026-07-18 C3] skip a eunuch who already holds the trait (or would need an
					# opposite resolved) so the grant is a clean no-op-or-add, matching the opposite-guard
					# idiom used by the accountability 'just' reward. Runtime effect (not construction), so
					# not the boot-AV class — this is consistency hardening.
					NOT = { has_trait = ambitious }
				}
				order_by = finesse
				max = 1
				set_variable = { name = qing_eunuch_faction_leader  value = 1 }
				add_trait = ambitious
				add_loyal_veterans = 3
				add_gold = 50
				add_corruption = 20
				LOG_line = { sys = QING  msg = "household pulse: a palace eunuch leads a faction (power base +3) for [THIS.GetName]" }
			}
		}
	}

	LOG_exit = { sys = QING  fn = "QING_household_pulse"  result = OK }
}

# ======================================================================
# REPLENISH PRIVY PURSE — top up the emperor's private silver. $amount$ (int, 10-30)
# and $cost$ (treasury gold or prestige). Called from events. Country scope.
# ======================================================================
QING_household_replenish_purse = {
	LOG_enter = { sys = QING  fn = "QING_household_replenish_purse" }
	# [#424] grant the replenishment to the emperor's own purse (his wealth). $amount$ is the
	# old-meter delta; callers pass ×4 wealth values now (see qing_household_events).
	if = { limit = { exists = current_ruler }  current_ruler = { add_gold = $amount$ } }
	add_treasury = $cost$
	TREASURY_LOG_it = { amount = $cost$ }
	LOG_line = { sys = QING  msg = "household: privy purse += $amount$ (cost $cost$ treasury) for" }
	LOG_exit = { sys = QING  fn = "QING_household_replenish_purse"  result = OK }
}

# ======================================================================
# TAP PRIVY PURSE — the throne draws on the private reserve to bail out the state
# treasury (a crisis measure). $relief$ (treasury gold gained), $prestige_cost$
# (the personal-prestige price of raiding the privy purse). Drains the purse by
# half. Country scope.
# ======================================================================
QING_household_tap_purse = {
	LOG_enter = { sys = QING  fn = "QING_household_tap_purse" }
	# [#424] the throne raids its OWN privy purse (the emperor's wealth) to bail out the state.
	# Drain HALF the sovereign's wealth into a country var, then subtract it from him. Proven
	# inheritance/transfer idiom (TI on_character_death_inv: read wealth into a var, add_gold={value}).
	if = {
		limit = { exists = current_ruler }
		set_variable = { name = qing_purse_drain_tmp  value = { value = current_ruler.wealth  multiply = 0.5 } }
		if = { limit = { var:qing_purse_drain_tmp < 0 }  set_variable = { name = qing_purse_drain_tmp  value = 0 } }
		current_ruler = { add_gold = { value = 0  subtract = ROOT.var:qing_purse_drain_tmp } }
	}
	add_treasury = $relief$
	TREASURY_LOG_it = { amount = $relief$ }
	if = {
		limit = { exists = current_ruler }
		current_ruler = {
			add_popularity = $prestige_cost$
			LOG_line = { sys = QING  msg = "household: privy purse tapped (ruler popularity $prestige_cost$) for" }
		}
	}
	LOG_exit = { sys = QING  fn = "QING_household_tap_purse"  result = OK }
}

# ======================================================================
# EUNUCH FACTION CHECK — the throne deals with the dangerous eunuch. Character
# scope (this = the eunuch). $outcome$ = check|indulge|instrument.
#   check    — the Qing way: strip his power base hard (remove loyal veterans,
#              confiscate wealth), clear the faction, nudge corruption down.
#   indulge  — buy him off: grant wealth, he stays powerful (danger deferred).
#   instrument — co-opt him as the throne's private agent: he becomes a rival
#              of the Grand Secretary (friction), gains loyal veterans, but
#              loyalty to the throne rises (the dirty-work lever).
# ======================================================================
QING_household_eunuch_check = {
	LOG_enter = { sys = QING  fn = "QING_household_eunuch_check" }
	if = {
		limit = { has_variable = qing_eunuch_faction_leader }
		# strip the flag + power base
		remove_variable = qing_eunuch_faction_leader
		add_loyal_veterans = -3
		add_gold = -50
		add_corruption = -20
		employer = {
			QING_DECLINE_nudge = { var = qing_corruption_level  amount = -3 }
		}
		LOG_line = { sys = QING  msg = "household: eunuch checked (power stripped) for [ROOT.GetTag]" }
	}
	LOG_exit = { sys = QING  fn = "QING_household_eunuch_check"  result = OK }
}

QING_household_eunuch_indulge = {
	LOG_enter = { sys = QING  fn = "QING_household_eunuch_indulge" }
	add_gold = 80
	add_corruption = 10
	employer = {
		QING_DECLINE_nudge = { var = qing_corruption_level  amount = 2 }
	}
	LOG_line = { sys = QING  msg = "household: eunuch indulged (wealth +80, danger deferred) for [ROOT.GetTag]" }
	LOG_exit = { sys = QING  fn = "QING_household_eunuch_indulge"  result = OK }
}

QING_household_eunuch_instrument = {
	LOG_enter = { sys = QING  fn = "QING_household_eunuch_instrument" }
	add_loyal_veterans = 2
	add_loyalty = loyalty_qing_delta_p20
	# [#126 review-fix] set him as rival of the Grand Secretary (if one exists) —
	# friction between the private instrument and the official household head. The
	# prior code used `add_rival = prev` / `is_rival = prev` from inside
	# employer{}->var:...holder{}; since if/limit do not push scope, `prev` there
	# resolved to the COUNTRY (one level up), not the eunuch — so the rivalry was
	# created against the country and the guards tested the wrong scope. Save the
	# eunuch explicitly and reference him by scope so the intended eunuch<->Grand
	# Secretary rivalry is actually formed.
	save_scope_as = qing_eunuch_instrument
	if = {
		limit = {
			employer = {
				has_variable = qing_office_chamberlain_holder
				var:qing_office_chamberlain_holder = {
					is_alive = yes
					employer = ROOT
					NOT = { is_rival = scope:qing_eunuch_instrument }
				}
			}
		}
		employer = {
			var:qing_office_chamberlain_holder = {
				add_rival = scope:qing_eunuch_instrument
			}
		}
	}
	LOG_line = { sys = QING  msg = "household: eunuch co-opted as instrument (loyal veterans +2, loyalty +20) for [ROOT.GetTag]" }
	LOG_exit = { sys = QING  fn = "QING_household_eunuch_instrument"  result = OK }
}
