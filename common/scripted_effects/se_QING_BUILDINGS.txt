# Seed ONE production building at province $P$ (building $B$), guarded by
# ownership + a $G$ trade-good match so a re-mapped province can never mis-seed.
# Skips silently (logged) if the site fails its guard or already has the building.
QING_seed_building = {
	if = {
		limit = {
			exists = p:$P$
			p:$P$ = {
				owner = c:CHI
				trade_goods = $G$
				NOT = { has_building = $B$ }
			}
		}
		p:$P$ = {
			add_building_level = $B$
			LOG_line = { sys = QING  msg = "starting buildings: $B$ seeded at $NAME$ (p:$P$)" }
		}
	}
	else = {
		LOG_line = { sys = QING  msg = "starting buildings: SKIP $B$ at $NAME$ (p:$P$) — guard failed or already present" }
	}
}

# [bookmark-1763 #291] Seed a Board-of-Works infrastructure building (canal depot / river
# dike) at game start. Unlike QING_seed_building these are NOT tied to a trade good, so the
# guard is ownership-only. add_building_level bypasses the building `allow` tech gate (which
# only governs the player build menu). Idempotent (NOT has_building). Scope: any.
QING_seed_works_building = {
	if = {
		limit = {
			exists = p:$P$
			p:$P$ = {
				owner = c:CHI
				NOT = { has_building = $B$ }
			}
		}
		p:$P$ = {
			add_building_level = $B$
			LOG_line = { sys = QING  msg = "starting buildings: $B$ seeded at $NAME$ (p:$P$)" }
		}
	}
	else = {
		LOG_line = { sys = QING  msg = "starting buildings: SKIP $B$ at $NAME$ (p:$P$) — not CHI-owned or already present" }
	}
}

# [#190] Seed a FRONTIER building at province $P$ (building $B$) where the owner is CHI OR a
# CHI subject. Identical to QING_seed_works_building except the ownership guard admits a subject-
# held province: the frontier garrison seats (Heilongjiang / Mukden / Xinjiang / Tibet) are held
# by CHI's autonomous-governorship & protectorate subjects, not by CHI directly, so the CHI-only
# guard would skip them. The OR = { owner = c:CHI  owner = { is_subject_of = c:CHI } } guard is the
# exact idiom the 1815 OOB used to raise the garrison on these same provinces
# (imp19c_effects_legion_setup.txt:83). add_building_level bypasses the building tech/region gate.
QING_seed_frontier_building = {
	if = {
		limit = {
			exists = p:$P$
			p:$P$ = {
				OR = { owner = c:CHI  owner = { is_subject_of = c:CHI } }
				NOT = { has_building = $B$ }
			}
		}
		p:$P$ = {
			add_building_level = $B$
			LOG_line = { sys = QING  msg = "starting buildings: $B$ seeded at $NAME$ (p:$P$)" }
		}
	}
	else = {
		LOG_line = { sys = QING  msg = "starting buildings: SKIP frontier $B$ at $NAME$ (p:$P$) - not CHI/subject-owned or already present" }
	}
}
