########################################################################
# se_QING_PERSONNEL.txt — the Grand Minister of Personnel (吏部尚書) governing
# the provincial officialdom. The 吏部 controlled appointment, evaluation (考課),
# and promotion of the entire civil service; the triennial 大計 reviewed
# provincial officials (governors, treasurers), and the 京察 reviewed
# metropolitan ones. Governors (總督/巡撫) were the Personnel minister's
# most important charges.
#
# COUPLING-FAMILY MODEL: the Personnel holder's skill/loyalty GATES outcomes
# and costs in governor-evaluation events; governors join the character-affinity
# chart via QING_pair_friction with the holder; a VACANT Personnel office = the
# governorates drift without oversight.
#
# State (CHI country):
#   qing_office_personnel_holder         the character holding the office (se_QING_COUNCIL.txt)
#   qing_personnel_daji_cooldown         day-expiry triennial guard for the 大計 review
#
# Prefix = QING_personnel_
########################################################################

# ======================================================================
# EVALUATE GOVERNORS — quarterly pulse called from QING_GOV_pulse.
# Iterate every governor (character with is_governor = yes) in CHI's employ,
# score each against the Personnel office-holder via QING_pair_friction AND
# against the throne via QING_char_affinity. A capable, loyal Personnel
# minister + a congenial governor = the governor is cultivated (loyalty/
# standing up, maybe finesse). A vacant Personnel office OR a clashing pair
# = the governor drifts (loyalty risk, a chance to queue an evaluation event).
# O(governors), quarterly. All derefs guarded.
# ======================================================================
QING_personnel_evaluate_governors = {
	LOG_enter = { sys = QING  fn = "QING_personnel_evaluate_governors" }
	
	# guard: only CHI, only if governors exist
	if = {
		limit = {
			tag = CHI
			any_character = {
				count >= 1
				employer = ROOT
				is_governor = yes
				is_alive = yes
			}
		}
		
		# iterate every governor in the realm
		every_character = {
			limit = {
				employer = ROOT
				is_governor = yes
				is_alive = yes
			}
			save_scope_as = the_governor
			
			# CASE A: Personnel office FILLED
			if = {
				limit = {
					ROOT = {
						has_variable = qing_office_personnel_holder
						var:qing_office_personnel_holder = {
							is_alive = yes
							employer = ROOT
						}
					}
				}
				ROOT.var:qing_office_personnel_holder = { save_scope_as = the_minister }
				
				# score friction between the Personnel minister and this governor
				ROOT = {
					QING_pair_friction = { a = scope:the_minister  b = scope:the_governor }
				}
				
				# LOW FRICTION (<55): they cooperate — minister cultivates the governor
				# [#34-D3 2026-08-09] Threshold raised 40 -> 55. QING_pair_friction (se_QING_AFFINITY.txt:464)
				# is base 20 + 25 for a culture divide, so a Manchu minister vs ANY Han governor scores >=45
				# on the 滿漢 fault line ALONE — which cleared the old >=40 clash gate for EVERY Han governor
				# every quarter (the reported "always clashes"). At 55 a bare culture divide (45) no longer
				# auto-clashes; a clash now needs ADDITIONAL friction (a zeal gulf, a standing rivalry, high
				# corruption, or two proud men) on top — i.e. a genuinely bad pairing, not just Manchu-vs-Han.
				# Raised HERE in the Personnel consumer, NOT in the shared QING_pair_friction (War/Council read
				# the same score and must keep their own thresholds). Cultivation branch keeps the < gate.
				if = {
					limit = { ROOT = { var:qing_pair_friction < 55 } }
					# capable minister (finesse>=9) + congenial governor = effective cultivation
					if = {
						limit = { scope:the_minister = { finesse >= 9 } }
						add_loyalty = loyalty_qing_delta_p3
						# if the governor is also compatible with the throne, promote his standing
						QING_char_affinity = yes
						if = {
							limit = { var:qing_char_affinity >= 60 }
							QING_char_promote_standing = { pop = 2  prom = 1 }
							# a talented governor under a strong minister may gain administrative skill
							random = {
								chance = 15
								QING_char_cultivate = { stat = finesse  amount = 1 }
							}
							# [deferred-fix #3] Queue the 'talented governor flagged for promotion' event
							# HERE, in CASE A. Its trigger requires a FILLED Personnel office with a
							# finesse>=9 minister plus a loyal talented governor — exactly this branch
							# (office filled, able minister, high-affinity governor). It was previously
							# queued ONLY from CASE B (office VACANT), so its own trigger could never pass
							# at fire time and the event was dead. Low per-quarter chance.
							random = {
								chance = 8
								ROOT = {
									trigger_event = {
										id = qing_personnel.3
										days = { 30 90 }
									}
									LOG_line = { sys = QING  msg = "personnel: promotion event queued (talented governor under able minister) for" }
								}
							}
						}
						ROOT = {
							LOG_line = { sys = QING  msg = "personnel: cultivated by able minister (friction=) for" }
						}
					}
					else = {
						# weaker minister (finesse<9) — modest loyalty hold
						add_loyalty = loyalty_qing_delta_p1
						ROOT = {
							LOG_line = { sys = QING  msg = "personnel: maintained by weaker minister for" }
						}
					}
				}
				# HIGH FRICTION (>=55): they clash — risk of recall or drift ([#34-D3] raised from >=40)
				else = {
					# check if the governor is loyal to the throne regardless
					QING_char_affinity = yes
					if = {
						limit = { var:qing_char_affinity >= 65 }
						# loyal governor buffers the clash
						ROOT = {
							LOG_line = { sys = QING  msg = "personnel: clashes with minister but loyal to throne (friction=) for" }
						}
					}
					else = {
						# neither congenial with minister nor loyal to throne = drift risk
						add_loyalty = loyalty_qing_delta_n2
						# chance to queue a recall/clash event (10% per quarter when friction is high).
						# [BT-57] Share the quarterly COURT event slot so the per-GOVERNOR loop cannot fire
						# several "Clash between Ministers" (qing_personnel.2) events in one pulse. Gate on the
						# slot being free (checked EACH governor iteration, so once one governor claims it no
						# further governor fires this pulse) and CLAIM it on the fire path only (after the chance
						# succeeds) — the BT-28 court-slot pattern. Slot cleared at pulse top (00_monthly_country.txt).
						# [#34-D1 2026-08-09] PER-GOVERNOR COOLDOWN. The old dispatch had ONLY the shared court
						# slot + a 10% roll, so the SAME governor could be re-picked to clash quarter after quarter
						# (the reported "same Clash event, always Liu Tongxun 562"). Gate this governor's roll on
						# him NOT having clashed recently, and stamp a ~2-year timed marker ON HIM on the fire path
						# so a fired clash steps that governor aside and the pool rotates to others. (Timed var =
						# self-expiring, no cleanup needed — the proven cooldown idiom.)
						if = {
							limit = {
								ROOT = { NOT = { has_variable = qing_gc_event_slot_used } }
								NOT = { has_variable = qing_personnel_clashed_recently }
							}
							random = {
								chance = 10
								set_variable = { name = qing_personnel_clashed_recently  value = 1  days = 730 }  # [#34-D1] this governor won't be re-rolled for ~2 years
								# [#34-D2 2026-08-09] Save THIS governor as the clash target so qing_personnel.2 targets
								# the man who actually clashed, instead of re-drawing a fresh random_character (which kept
								# landing on the same high-prominence governor, Liu Tongxun 562). The event reads this
								# saved scope first and only random-picks as a fallback. Days-delayed fire, so store on ROOT.
								ROOT = { set_variable = { name = qing_personnel_clash_target  value = scope:the_governor } }
								ROOT = {
									set_variable = { name = qing_gc_event_slot_used  value = 1 }  # [BT-57] claim court slot after the chance succeeds (mirrors frontier/works rolls)
									trigger_event = {
										id = qing_personnel.2
										days = { 30 90 }
									}
									LOG_line = { sys = QING  msg = "personnel: clash event queued (governor vs minister) for" }
								}
							}
						}
						ROOT = {
							LOG_line = { sys = QING  msg = "personnel: drifts (clashes with minister , friction=) for" }
						}
					}
				}
			}
			# CASE B: Personnel office VACANT
			else = {
				# no oversight = governors drift
				QING_char_affinity = yes
				if = {
					limit = { var:qing_char_affinity < 50 }
					# uncongenial governors grow more independent without a Personnel minister
					add_loyalty = loyalty_qing_delta_n1
					# [deferred-fix #3] Removed a queue of qing_personnel.3 here: that event's
					# trigger requires a FILLED Personnel office (finesse>=9 minister), but this
					# is the office-VACANT branch — mutually exclusive, so it could never pass its
					# trigger when queued from here. The promotion path now lives in CASE A.
					ROOT = {
						LOG_line = { sys = QING  msg = "personnel: drifts without Personnel oversight for" }
					}
				}
				else = {
					# congenial governors hold steady even without a minister
					ROOT = {
						LOG_line = { sys = QING  msg = "personnel: holds steady (no Personnel minister, but loyal) for" }
					}
				}
			}
		}
	}
	else = {
		LOG_line = { sys = QING  msg = "personnel: no governors to evaluate for" }
	}
	
	LOG_exit = { sys = QING  fn = "QING_personnel_evaluate_governors"  result = OK }
}

# ======================================================================
# 大計 TRIENNIAL REVIEW — the event-called effect for the Great Reckoning.
# Called from qing_personnel.1 (the 大計 event). Country scope, CHI.
# Rewards able governors, cashiers venal ones, promotes talent based on the
# Personnel holder's finesse. Iterates governors; O(governors).
# ======================================================================
QING_personnel_daji_review = {
	LOG_enter = { sys = QING  fn = "QING_personnel_daji_review" }

	# [#117 review-fix] Removed a vestigial dead write of qing_personnel_daji_last_year
	# via the UNPROVEN current_date.year accessor (demonstrated nowhere in the repo nor
	# in either oracle mod — it spams error.log). That variable was never read anywhere:
	# the real triennial guard is qing_personnel_daji_cooldown, stamped as a day-expiry
	# in qing_personnel.1's immediate. Left over from the #117 design pivot.

	# iterate every governor, scoring and applying outcomes
	every_character = {
		limit = {
			employer = ROOT
			is_governor = yes
			is_alive = yes
		}
		save_scope_as = reviewed_governor
		
		# score the governor on governing skill (finesse + martial, the administrator/pacifier)
		set_variable = { name = qing_daji_score  value = finesse }
		change_variable = { name = qing_daji_score  add = martial }
		
		# ABLE governors (score >= 16) are rewarded
		if = {
			limit = { var:qing_daji_score >= 16 }
			QING_char_promote_standing = { pop = 3  prom = 2 }
			add_loyalty = loyalty_qing_delta_p5
			# if Personnel office filled + minister is capable, the reward is richer
			if = {
				limit = {
					ROOT = {
						has_variable = qing_office_personnel_holder
						var:qing_office_personnel_holder = {
							is_alive = yes
							employer = ROOT
							finesse >= 9
						}
					}
				}
				QING_char_honour_family = { amount = 10 }
				ROOT = {
					LOG_line = { sys = QING  msg = "大計: able governor (score=) rewarded under strong Personnel minister for" }
				}
			}
			else = {
				ROOT = {
					LOG_line = { sys = QING  msg = "大計: able governor (score=) noted for" }
				}
			}
		}
		# VENAL governors (corrupt trait OR corruption>=40) are cashiered
		else_if = {
			limit = {
				OR = {
					has_trait = corrupt
					corruption >= 40
				}
			}
			add_loyalty = loyalty_qing_delta_n10
			add_prominence = -5
			# if Personnel office filled + minister is strong, corruption is purged harder
			if = {
				limit = {
					ROOT = {
						has_variable = qing_office_personnel_holder
						var:qing_office_personnel_holder = {
							is_alive = yes
							employer = ROOT
							finesse >= 9
						}
					}
				}
				QING_char_cleanse = yes
				ROOT = {
					LOG_line = { sys = QING  msg = "大計: venal governor cashiered under strong Personnel minister for" }
				}
			}
			else = {
				ROOT = {
					LOG_line = { sys = QING  msg = "大計: venal governor marked (weak/no Personnel minister) for" }
				}
			}
		}
		# MEDIOCRE governors (score < 16, not venal) are merely noted
		else = {
			ROOT = {
				LOG_line = { sys = QING  msg = "大計: mediocre governor (score=) noted for" }
			}
		}
	}
	
	LOG_exit = { sys = QING  fn = "QING_personnel_daji_review"  result = OK }
}
