# se_QING_WENZHI.txt — IMPERIAL CULTURAL PATRONAGE (文治), task #390.
#
# The 武功 (martial achievement) of the High Qing is modelled by the Ten Great
# Campaigns tree (#63); this is its 文治 (civil / cultural governance) counterpart —
# the emperor as patron of arts, letters, and craft. Qianlong is the archetype and
# the 1763 start sits squarely in his cultural golden age, so the track begins ALIVE
# (patronage seeded to 40, not 0).
#
# WHAT THIS ADDS vs WHAT ALREADY EXISTS. The random cultural-happening roster
# (qing_culture.1-.10, events/imp19c_mod_events/qing_culture_events.txt, #15) and its
# lasting modifiers (qing_culture_modifiers.txt: siku_quanshu, court_painting, jade,
# porcelain, ...) already exist but fire only as UNBIDDEN flavour from QING_culture_roll.
# This module is the missing PLAYER-AGENCY spine: a patronage METER + LAUNCHABLE
# initiatives the throne funds deliberately (spending treasury / the privy purse 內帑),
# each raising patronage AND dispatching the matching existing culture event, so the
# monuments become acts of will, not luck. It also gives patronage an AMBIENT payoff
# (a full patronage court radiates legitimacy) and a slow decay (glory must be renewed).
#
# HISTORY (sources: Chuimei Ho & Bennet Bronson, Splendors of China's Forbidden City:
# The Glorious Reign of Emperor Qianlong; Evelyn Rawski, The Last Emperors):
#   • 造辦處 Imperial Workshops (under the 內務府): jade, Jingdezhen imperial-kiln 御窯廠
#     porcelain, cloisonné, clocks, lacquer — craftsmen drawn to Peking.
#   • 如意館 Ruyiguan painting studio: Jesuit painters at court, above all Giuseppe
#     Castiglione (郎世寧), the Sino-European court style.
#   • 四庫全書 Complete Library of the Four Treasuries (1772-82) — with the 文字獄
#     literary inquisition as its dark side (order and orthodoxy bought with fear).
#   • 南巡 Southern Inspection Tours — cultural/patronage display down the Jiangnan.
#
# COUPLING: the workshops belong to the Imperial Household (內務府, #359) and are funded
# from the privy purse (內帑, qing_privy_purse); the literary projects draw on the Hanlin
# Academy (#358) / Southern Study (#336) scholar corps. Payoffs land in the stats that
# matter in a monarchy — legitimacy (country standing), ruler popularity (personal glory),
# stability — per BT-5/6, NEVER add_prestige (a silent no-op / family stat here).
#
# State (all on CHI):
#   qing_wenzhi_patronage    0..100  the standing of imperial patronage (init 40)
#   qing_wenzhi_*_done       one-shot flags mirroring the culture events' own guards
#
# CHI-only, player-only. Init at game start; drift quarterly from QING_GOV_pulse.
# Prefix = QING_wenzhi_ / sys = QING
########################################################################

# ======================================================================
# INIT — seed the patronage meter. Called once at game start (after QING_DECLINE_init
# so the era flag is available). The 1763 zenith begins mid-high (Qianlong's golden
# age); a later start begins lower. Idempotent. Scope: country (CHI).
# ======================================================================
QING_wenzhi_init = {
	LOG_enter = { sys = QING  fn = "QING_wenzhi_init" }
	if = {
		limit = { NOT = { has_variable = qing_wenzhi_patronage } }
		if = {
			limit = { has_variable = qing_high_qing_era }
			set_variable = { name = qing_wenzhi_patronage  value = 40 }
		}
		else = {
			set_variable = { name = qing_wenzhi_patronage  value = 20 }
		}
	}
	LOG_line = { sys = QING  msg = "wenzhi: patronage meter seeded" }
	LOG_exit = { sys = QING  fn = "QING_wenzhi_init"  result = OK }
}

# ======================================================================
# [#121] SEED CASTIGLIONE (郎世寧) AT GAME START — the named historical court painter, not
# a generic mint. Giuseppe Castiglione (b. 19 July 1688, Milan) had served the Qing court
# since 1715 across three reigns and was ~75 in 1763 (he died in 1766, so a natural-aging
# death within a few years of game start is historically correct, not a bug). No separate
# "Court Jesuit" position is created — he is stamped with the EXISTING qing_court_artist
# marker (se_QING_WENZHI.txt's QING_wenzhi_commission_painting), the same mechanic a
# runtime-minted painter carries, so the Household/Art-Patronage surfaces that read that
# marker see him without any new plumbing. Boot-seed create_character (the proven idiom for
# a historical figure who must already be active at 1763, mirroring QING_missionary_spawn_
# foreign_missionary/_agitator's #90-safe shape: no modifiers granted inside create_character,
# markers set in a follow-up scope). Guarded once by qing_seeded_castiglione. Country scope.
# ======================================================================
QING_wenzhi_seed_castiglione = {
	LOG_enter = { sys = QING  fn = "QING_wenzhi_seed_castiglione" }
	if = {
		limit = { NOT = { has_variable = qing_seeded_castiglione } }
		set_variable = qing_seeded_castiglione
		create_character = {
			age = 75
			culture = north_italian   # [#13 fix] `italian` is a culture GROUP, invalid for culture=; Castiglione (郎世寧) was Milanese -> north_italian
			religion = catholic
			add_charisma = 7
			add_finesse = 9   # the master painter of the Sino-European court style
			add_zeal = 4
			add_martial = 1
			save_scope_as = qing_castiglione
		}
		set_variable = { name = qing_castiglione_char  value = scope:qing_castiglione }   # [#268 idiom] survives reload
		scope:qing_castiglione = {
			set_home_country = ROOT
			# stamped with the SAME marker QING_wenzhi_commission_painting mints/recounts; that
			# effect rebuilds qing_court_artist_count from scratch (every_character scan) every
			# time it runs, so Castiglione is automatically counted against the 5-painter atelier
			# cap the next time a painting is commissioned — no separate counter to keep in sync.
			set_variable = { name = qing_court_artist  value = 1 }
			QING_roster_finalize = { nick = NICKNAME_QING_CASTIGLIONE }
		}
		LOG_line = { sys = QING  msg = "wenzhi: Giuseppe Castiglione (郎世寧) takes his place in the Ruyiguan" }
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_wenzhi_seed_castiglione"  reason = "Castiglione already seeded (qing_seeded_castiglione set)" }
	}
	LOG_exit = { sys = QING  fn = "QING_wenzhi_seed_castiglione"  result = OK }
}

# ======================================================================
# QUARTERLY PULSE — called from QING_GOV_pulse. Patronage decays slowly (the glory of
# the court must be renewed by fresh commissions), and a HIGH standing radiates ambient
# legitimacy (a court famed for its splendour affirms the Mandate). Bounded [0, 100].
# Scope: country (CHI).
# ======================================================================
QING_wenzhi_pulse = {
	if = {
		limit = { has_variable = qing_wenzhi_patronage }
		# slow decay: -1/quarter (a still court's lustre fades)
		QING_DECLINE_nudge = { var = qing_wenzhi_patronage  amount = -1 }
		# [law #16 bias 2026-07-24] Cultural Patronage law (qing_cultural_patronage_law) adds a standing
		# bias to the decay: austere = -1 (lustre fades faster), lavish = +1 (patronage offsets the base
		# decay). Guarded so the default (bias unset) is byte-identical to the plain -1/quarter above.
		if = { limit = { has_variable = qing_wenzhi_patronage_bias }  QING_DECLINE_nudge = { var = qing_wenzhi_patronage  amount = var:qing_wenzhi_patronage_bias } }
		# ambient legitimacy while the court's patronage stands high (>= 60): the splendour
		# of the reign is itself an argument for the Mandate. Small, so it rewards sustained
		# patronage without being a dominant lever.
		if = {
			limit = { var:qing_wenzhi_patronage >= 60 }
			add_legitimacy = 1
		}
		# [MO-ENH #13] ZENITH MILESTONE (文治鼎盛). When imperial patronage first crowns a golden age
		# (>= 80), fire a one-shot capstone beat mirroring the silver-reserve peak (qing_revenue.5): the
		# Hanlin + Southern Study scholars present a memorial praising the reign's 文治. Gated on the
		# shared court event slot (BT-28) + a one-shot flag, so it is a single memorable moment, never
		# spam. Makes the ambient >=60 legitimacy buff feel earned and visible at its apex.
		if = {
			limit = {
				var:qing_wenzhi_patronage >= 80
				NOT = { has_variable = qing_wenzhi_zenith_reached }
				NOT = { has_variable = qing_gc_event_slot_used }
			}
			set_variable = { name = qing_wenzhi_zenith_reached  value = 1 }
			set_variable = { name = qing_gc_event_slot_used  value = 1 }  # [BT-28] claim court slot
			trigger_event = { id = qing_wenzhi.1  days = { 5 20 } }
			LOG_line = { sys = QING  msg = "wenzhi: imperial patronage reached its cultural zenith" }
		}
		# [fix, task #15 2026-08-16, design/DESIGN_ARTPATRONAGE_PAINTER_EVENTS.md] ambient painter
		# flavour: the atelier had no narrative events of its own. Rebuild the count HERE (not just
		# trust the button-maintained qing_court_artist_count) so a stale count from a natural death
		# between button clicks can't wrongly gate this open.
		set_variable = { name = qing_court_artist_count  value = 0 }
		every_character = {
			limit = { has_variable = qing_court_artist  is_alive = yes  employer = ROOT }
			ROOT = { change_variable = { name = qing_court_artist_count  add = 1 } }
		}
		if = {
			limit = {
				var:qing_court_artist_count > 0
				NOT = { has_variable = qing_dept_cd_wenzhi_painter }
				NOT = { has_variable = qing_gc_event_slot_used }
			}
			random = {
				chance = 15
				set_variable = { name = qing_dept_cd_wenzhi_painter  days = 1095 }
				set_variable = { name = qing_gc_event_slot_used  value = 1 }
				trigger_event = { id = qing_wenzhi_painter.1  days = { 5 20 } }
			}
		}
	}
}

# [fix, task #15 2026-08-16, design/DESIGN_ARTPATRONAGE_PAINTER_EVENTS.md] a court painter's death
# had no narrative acknowledgement. Scope: ROOT = the dying character (on_character_death
# convention, qing_mechanics_on_actions.txt). Pure no-op for every non-painter death.
QING_wenzhi_painter_death_check = {
	if = {
		limit = { has_variable = qing_court_artist  exists = employer  employer = { tag = CHI } }
		save_scope_as = qing_dead_painter
		employer = {
			if = {
				limit = { NOT = { has_variable = qing_gc_event_slot_used } }
				set_variable = { name = qing_gc_event_slot_used  value = 1 }
				# [review-fix] no delay -- matches the only in-repo precedent for a permanent
				# save_scope_as read across a scope change on a just-died character (vanilla
				# 00_ambitions.txt:1085/1124, employer={trigger_event=...} with NO days param).
				# A multi-day delay on an already-dead character's scope is unproven here.
				trigger_event = { id = qing_wenzhi_painter.2 }
			}
		}
	}
}

# ======================================================================
# ADVANCE — raise the patronage meter by $amount$, clamped [0, 100]. Scope: country (CHI).
# ======================================================================
QING_wenzhi_advance = {
	QING_wenzhi_init = yes
	QING_DECLINE_nudge = { var = qing_wenzhi_patronage  amount = $amount$ }
	LOG_line = { sys = QING  msg = "wenzhi: patronage advanced" }
}

# ======================================================================
# INITIATIVE: FUND THE IMPERIAL WORKSHOPS (造辦處). The 內務府's ateliers — jade,
# Jingdezhen porcelain, cloisonné — funded from the privy purse (內帑). Grants a lasting
# craft modifier (jade + porcelain, if not already held) and raises patronage. Spends
# treasury + a slice of the privy purse. Scope: country (CHI); called from the panel.
# ======================================================================
QING_wenzhi_fund_workshops = {
	LOG_enter = { sys = QING  fn = "QING_wenzhi_fund_workshops" }
	add_treasury = -255
	TREASURY_LOG_it = { amount = -255 }
	if = { limit = { exists = current_ruler }  current_ruler = { add_gold = -32 } }   # [#424] privy purse (內帑) = ruler wealth
	QING_wenzhi_advance = { amount = 12 }
	if = {
		limit = { NOT = { has_variable = qing_wenzhi_workshops_done } }
		set_variable = qing_wenzhi_workshops_done
		add_country_modifier = { name = qing_culture_imperial_jade  duration = -1 }
		add_country_modifier = { name = qing_culture_porcelain  duration = -1 }
		# [#437] CONCRETE: the 造辦處 ateliers are not only a modifier — raise a real Imperial Kiln
		# (御窯廠, qing_porcelain_kiln_building) at a porcelain-producing province that lacks one (景德鎮
		# and its like), and lift that province's civilisation. add_building_level + add_civilization_value
		# are the proven province-scope idioms (ordered_owned_province / every_owned_province upstream).
		ordered_owned_province = {
			limit = {
				trade_goods = porcelain
				NOT = { has_building = qing_porcelain_kiln_building }
			}
			order_by = total_population
			max = 1
			add_building_level = qing_porcelain_kiln_building
			add_civilization_value = 5
		}
		LOG_line = { sys = QING  msg = "wenzhi: imperial workshops endowed — jade + porcelain renown, an Imperial Kiln raised" }
	}
	else = {
		# already endowed — a fresh commission is a one-off glory, not a new standing modifier.
		current_ruler = { add_popularity = 5 }
		LOG_line = { sys = QING  msg = "wenzhi: a fresh workshop commission (already endowed)" }
	}
	add_legitimacy = 2
	LOG_exit = { sys = QING  fn = "QING_wenzhi_fund_workshops"  result = OK }
}

# ======================================================================
# [#122] SUPPRESS THE JESUITS (禁教) — the Art Patronage panel's suppression lever. Relieves
# every LIVING Jesuit-culture (italian/portuguese) qing_court_artist — expelled from the
# Ruyiguan, exactly the fate of the real 1720s Kangxi/Yongzheng-era rites-controversy
# tension carried forward as a live player choice — then mints ONE fresh Han court artist to
# keep the atelier staffed (reusing the SAME mint shape QING_wenzhi_commission_painting used
# before #121, culture=han/religion=confucianism). Costs: the throne loses the prestige of
# the Sino-European style (a patronage hit) and international standing (a stability hit);
# gains: relieves the SAME anti-Christian social friction meter #91/#309 already model
# (se_QING_MISSIONARY.txt), the concrete payoff for a "concrete-over-abstract" throne choice.
# Scope: country (CHI).
# ======================================================================
QING_wenzhi_suppress_jesuits = {
	LOG_enter = { sys = QING  fn = "QING_wenzhi_suppress_jesuits" }
	every_character = {
		limit = {
			employer = ROOT
			is_alive = yes
			has_variable = qing_court_artist
			OR = { culture.culture_group = culture_group:italian  culture = portuguese }   # [#13 fix] `italian` is a culture GROUP not a culture; match the group. `portuguese` is a valid culture (was cascade-flagged by the bad `italian` on this line)
		}
		remove_variable = qing_court_artist
	}
	# [cr122 fix] the expulsion LOG moved OUT of the every_character loop: inside the loop `this`
	# is the iterated character, not CHI, so the "...for" trailing scope named the wrong actor.
	LOG_line = { sys = QING  msg = "wenzhi: the Jesuit court artists are expelled from the Ruyiguan" }
	# [cr122 fix] rebuild the LIVING atelier count (mirrors QING_wenzhi_commission_painting's own
	# cap rebuild below) so the replacement mint respects the SAME 5-painter cap — otherwise
	# repeated commission/suppress cycling (commission mints italians -> suppress relieves them
	# and mints +1 Han that suppress never relieves) would accumulate unbounded Han court artists.
	set_variable = { name = qing_court_artist_count  value = 0 }
	every_character = {
		limit = { has_variable = qing_court_artist  is_alive = yes  employer = ROOT }
		ROOT = { change_variable = { name = qing_court_artist_count  add = 1 } }
	}
	if = {
		limit = { var:qing_court_artist_count < 5 }
		create_character = {
			age = 30
			culture = han
			religion = confucianism
			add_charisma = 6
			add_finesse = 7
			add_zeal = 4
			add_martial = 1
			save_scope_as = qing_new_court_artist_han
		}
		scope:qing_new_court_artist_han = {
			set_home_country = ROOT
			set_variable = { name = qing_court_artist  value = 1 }
		}
	}
	QING_wenzhi_advance = { amount = -8 }   # the atelier loses its most famous style, a real setback to patronage
	add_stability = -3   # the foreign powers take note; a real, if modest, diplomatic cost
	QING_DECLINE_nudge = { var = qing_mission_social_friction  amount = -8 }   # se_QING_MISSIONARY.txt idiom; the throne's distance from the missions relieves suspicion
	LOG_line = { sys = QING  msg = "wenzhi: the Jesuits are suppressed, a Han artist takes their place" }
	LOG_exit = { sys = QING  fn = "QING_wenzhi_suppress_jesuits"  result = OK }
}

# ======================================================================
# INITIATIVE: COMMISSION THE PAINTING ACADEMY (如意館). The court atelier where even
# the Jesuit Castiglione (郎世寧) serves, fusing Chinese and Western brush. Grants the
# court-painting modifier once + raises patronage + ruler glory. Scope: country (CHI).
# ======================================================================
QING_wenzhi_commission_painting = {
	LOG_enter = { sys = QING  fn = "QING_wenzhi_commission_painting" }
	add_treasury = -220
	TREASURY_LOG_it = { amount = -220 }
	QING_wenzhi_advance = { amount = 10 }
	if = {
		limit = { NOT = { has_country_modifier = qing_culture_court_painting } }
		add_country_modifier = { name = qing_culture_court_painting  duration = -1 }
		LOG_line = { sys = QING  msg = "wenzhi: the Ruyiguan painting academy is commissioned" }
	}
	# [#437] CONCRETE: the 如意館 is staffed by real court painters. Mint one — a learned, charismatic
	# artist of the emperor's household (marked qing_court_artist) — via the proven create_character roster
	# idiom (mirrors QING_exam_mint_scholar; #90 rule: no modifiers granted inside create_character, markers
	# set in a follow-up scope). CAPPED at a resident atelier of 5 living court artists (qing_court_artist_count),
	# self-healing so a court painter who dies frees a place — repeated commissions must not spawn unbounded
	# characters (save-bloat). The rebuild counts the LIVING marked artists first, then mints only if under cap.
	set_variable = { name = qing_court_artist_count  value = 0 }
	every_character = {
		limit = { has_variable = qing_court_artist  is_alive = yes  employer = ROOT }
		ROOT = { change_variable = { name = qing_court_artist_count  add = 1 } }
	}
	if = {
		limit = { var:qing_court_artist_count < 5 }
		# [#121] the 如意館 painters were, historically, predominantly the court's Jesuits
		# (Castiglione/郎世寧 above all, but also Attiret, Sichelbarth after him) — not Han
		# literati. italian + catholic mirrors QING_wenzhi_seed_castiglione below (the SAME
		# pairing that mints him) so a later-commissioned second painter belongs to the same
		# atelier tradition. No separate "Court Jesuit" position — this stays the EXISTING
		# qing_court_artist marker/mechanic, just correctly staffed.
		create_character = {
			age = 30
			culture = north_italian   # [#13 fix] `italian` is a culture GROUP, invalid for culture=; the atelier Jesuits were north-Italian (as Castiglione)
			religion = catholic
			add_charisma = 7
			add_finesse = 6
			add_zeal = 3
			add_martial = 1
			save_scope_as = qing_new_court_artist
		}
		scope:qing_new_court_artist = {
			set_home_country = ROOT
			set_variable = { name = qing_court_artist  value = 1 }
			add_popularity = 10
		}
	}
	current_ruler = { add_popularity = 8 }
	LOG_exit = { sys = QING  fn = "QING_wenzhi_commission_painting"  result = OK }
}

# ======================================================================
# INITIATIVE: LAUNCH THE COMPLETE LIBRARY (四庫全書). The greatest compilation of the
# age — and its shadow, the literary inquisition (文字獄). Raises patronage, then hands
# off to the EXISTING qing_culture.5 event, which presents the compile-with-inquisition
# vs compile-without choice (do not duplicate that dilemma here). Guarded on the same
# one-shot the event uses (qing_culture_siku_done) so the library is founded only once.
# Scope: country (CHI). Draws on the Hanlin (#358) / Southern Study (#336) scholar corps.
# ======================================================================
QING_wenzhi_launch_siku = {
	LOG_enter = { sys = QING  fn = "QING_wenzhi_launch_siku" }
	# [#390 / T2 review-fix] convene only while no convening is already pending: qing_wenzhi_siku_pending
	# is the "dilemma open" guard (the button's is_shown requires it ABSENT). qing_culture.5's options
	# CLEAR pending — incl. the "cannot afford" DECLINE, which does NOT found the library — so the throne
	# may convene again later. That re-arm is the exploit surface: the former code granted the +15
	# patronage on EVERY convening, so decline→re-launch→decline farmed unbounded patronage (and, via the
	# >=60 ambient-legitimacy branch, a permanent legitimacy engine) before the library was ever founded.
	# Gate the +15 on a SEPARATE one-shot flag (qing_wenzhi_siku_patronised) that NO option clears, so the
	# stirring of the scholar corps pays its patronage exactly ONCE across all re-convenings.
	if = {
		limit = { NOT = { has_variable = qing_wenzhi_siku_pending } }
		set_variable = qing_wenzhi_siku_pending
		if = {
			limit = { NOT = { has_variable = qing_wenzhi_siku_patronised } }
			set_variable = qing_wenzhi_siku_patronised
			QING_wenzhi_advance = { amount = 15 }
			# [#437] CONCRETE: the greatest compilation of the age is a real advance in letters and
			# statecraft — grant civic research (the 四庫全書 gathered and systematised the empire's
			# scholarship). add_research { technology value } is the proven country-scope idiom
			# (Invictus 01_armenia.txt:1029). Once, gated by the same one-shot patronage flag.
			add_research = { technology = civic_tech  value = 25 }
		}
		# the scholar corps is stirred; hand off to the existing Siku dilemma event, which
		# self-guards on qing_culture_siku_done and applies the modifiers + the 文字獄 choice.
		trigger_event = { id = qing_culture.5  days = { 3 10 } }
		LOG_line = { sys = QING  msg = "wenzhi: the Complete Library (四庫全書) is convened" }
	}
	LOG_exit = { sys = QING  fn = "QING_wenzhi_launch_siku"  result = OK }
}

# ======================================================================
# INITIATIVE: THE SOUTHERN INSPECTION TOUR (南巡). The emperor progresses down the
# Jiangnan — a display of patronage and majesty that binds the wealthy south to the
# throne, at great cost to the treasury. Raises patronage + ruler glory + stability;
# a one-off grand occasion (no permanent modifier). Scope: country (CHI).
# ======================================================================
QING_wenzhi_southern_tour = {
	LOG_enter = { sys = QING  fn = "QING_wenzhi_southern_tour" }
	add_treasury = -310
	TREASURY_LOG_it = { amount = -310 }
	QING_wenzhi_advance = { amount = 14 }
	current_ruler = { add_popularity = 12 }
	add_legitimacy = 4
	add_stability = 1
	# [#437] CONCRETE: the progress down the Jiangnan is a real civilising splendour on the ground —
	# lift the civilisation of the wealthiest southern cities the tour passes through. The imperial
	# barge, the academies patronised, the works commissioned leave a mark. Cap at the 5 most-developed
	# owned provinces so a vast empire's tour is a Jiangnan progress, not a realm-wide sweep.
	ordered_owned_province = {
		limit = { has_city_status = yes }
		order_by = total_population
		max = 5
		add_civilization_value = 3
	}
	LOG_line = { sys = QING  msg = "wenzhi: the emperor makes a Southern Tour (南巡)" }
