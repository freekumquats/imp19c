# se_QING_GOVERNANCE.txt — the GOVERNANCE engine for the Qing suite: the
# abstracted bureaucratic hierarchy that the Grand Council (軍機處) sits atop,
# plus the Grand Council hub and the character-held great offices.
#
# Design mirrors se_QING_DECLINE.txt exactly: a small set of 0..100 COUNTER
# variables on the Qing country, nudged EVENT-DRIVEN, read O(1) by a throttled
# pulse (QING_GOV_pulse) that swaps the modifier band matching each counter.
# It reuses QING_DECLINE_nudge (generic over var name) for all clamped nudges.
#
# The hierarchy itself (tens of thousands of clerks) is NOT modelled per-clerk.
# [meter-concretize #1 2026-08-06] The two old stored 0..100 drift meters
# qing_bureau_capacity and qing_exam_ladder are RETIRED — their update/drift effects and
# band-track spawn loops are deleted. Administrative reach and examination reach are now
# DERIVED live from the real 衙門 (yamen) and 書院 (academy) networks on the map:
#   QING_bureau_reach  0..100  = covered county seats ÷ coverable seats  (yamen coverage)
#   QING_exam_reach    0..100  = academy-covered provinces × 3           (書院 coverage)
# (both in QING_governance_svalues.txt; cached once per pulse into qing_bureau_reach_cache /
# qing_exam_reach_cache by QING_GOV_cache_reach so hot readers stay O(1)). The player owns the
# count — seed at 1763, build from the province menu, lose to war — and reach just reads it.
# Still stored/nudged here:
#   qing_bureau_integrity  0..100  merit (科舉) vs patronage (捐納) staffing
#   qing_council_effectiveness  0..100  derived from seated councillors
# The mission gates that once read qing_bureau_capacity now read the concrete ADMIN_available_
# country (the real administrative surplus/deficit). See design/DESIGN_BUREAU_CAPACITY_CONCRETIZE.md.
#
#   ROOT / this = the Qing overlord (CHI, human)
#
# Prefix = QING_GOV  (council seat/office helpers: QING_council_ / QING_office_)
########################################################################

# ======================================================================
# INIT — seed the governance counters once for a fresh CHI game.
# ======================================================================
QING_GOV_init = {
	LOG_enter = { sys = QING  fn = "QING_GOV_init" }
	if = { limit = { NOT = { has_variable = qing_bureau_integrity } }      set_variable = { name = qing_bureau_integrity  value = 60 } }
	if = { limit = { NOT = { has_variable = qing_council_effectiveness } } set_variable = { name = qing_council_effectiveness  value = 50 } }
	# [meter-concretize #1 2026-08-06] qing_bureau_capacity + qing_exam_ladder RETIRED. Both are now
	# DERIVED live from the real 衙門/書院 networks (QING_bureau_reach / QING_exam_reach svalues,
	# QING_governance_svalues.txt), cached once per pulse below. Drop the stale saved-game meters and
	# their target/band-track bookkeeping so nothing lingers (design §5/§9d). The derived svalues read
	# live buildings already in the save — no migration needed.
	remove_variable = qing_bureau_capacity
	remove_variable = qing_bureau_capacity_target
	remove_variable = qing_gov_corruption_drag
	remove_variable = qing_yamen_band_cur
	remove_variable = qing_yamen_band_prev
	remove_variable = qing_exam_ladder
	remove_variable = qing_exam_ladder_target
	remove_variable = qing_exam_corruption_drag
	remove_variable = qing_shuyuan_band_cur
	remove_variable = qing_shuyuan_band_prev
	# [meter-concretize #1] guarantee the derived-reach caches exist from the first frame, so any
	# event-fired reader (e.g. QING_censorate_find_corrupt) that runs BEFORE the first governance
	# pulse still reads a real coverage value, not a missing var. Refreshed every pulse by
	# QING_GOV_cache_reach.
	if = { limit = { NOT = { has_variable = qing_bureau_reach_cache } } set_variable = { name = qing_bureau_reach_cache  value = QING_bureau_reach } }
	if = { limit = { NOT = { has_variable = qing_exam_reach_cache } }   set_variable = { name = qing_exam_reach_cache    value = QING_exam_reach } }
	LOG_exit = { sys = QING  fn = "QING_GOV_init"  result = OK }
}

# ======================================================================
# [meter-concretize #1 2026-08-06] CACHE the two DERIVED reach meters once per pulse from the
# concrete building networks. A from-scratch recompute (never nudged) — NOT an accumulator; it
# just spares the many hot readers (censorate per-candidate gate, GUI, pass-rate base, bands,
# selfstr backing) an every_owned_province sweep each read. Scope: CHI.
# ======================================================================
QING_GOV_cache_reach = {
	LOG_enter = { sys = QING  fn = "QING_GOV_cache_reach" }
	set_variable = { name = qing_bureau_reach_cache  value = QING_bureau_reach }
	set_variable = { name = qing_exam_reach_cache    value = QING_exam_reach }
	LOG_line = { sys = QING  msg = "reach cached: bureau=(reach=) exam= derived from live 衙門/書院 networks for [ROOT.GetTag]" }
	LOG_exit = { sys = QING  fn = "QING_GOV_cache_reach"  result = OK }
}

# ======================================================================
# DERIVED UPDATES — recompute the two abstracted meters from live state.
# All reads are O(1) counter/variable reads; no pop or province sweeps.
# ======================================================================

# Integrity drifts toward (100 - corruption), penalised while purchased ranks
# are stuffing the bureaucracy (the 捐納 modifier from se_QING_MECHANICS.txt).
QING_GOV_update_integrity = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = QING  fn = "QING_GOV_update_integrity" }
	set_variable = { name = qing_bureau_integrity_target  value = 100 }
	change_variable = { name = qing_bureau_integrity_target  subtract = var:qing_corruption_level }
	if = {
		limit = { has_country_modifier = qing_office_purchased_ranks }
		change_variable = { name = qing_bureau_integrity_target  subtract = 20 }
	}
	if = { limit = { var:qing_bureau_integrity_target < 0 }   set_variable = { name = qing_bureau_integrity_target  value = 0 } }
	if = { limit = { var:qing_bureau_integrity_target > 100 } set_variable = { name = qing_bureau_integrity_target  value = 100 } }
	# smooth drift toward the target (+/-2 per pulse)
	if = {
		limit = { var:qing_bureau_integrity < qing_bureau_integrity_target_cmpsvalue }
		QING_DECLINE_nudge = { var = qing_bureau_integrity  amount = 2 }
	}
	else_if = {
		limit = { var:qing_bureau_integrity > qing_bureau_integrity_target_cmpsvalue }
		QING_DECLINE_nudge = { var = qing_bureau_integrity  amount = -2 }
	}
	LOG_exit = { sys = QING  fn = "QING_GOV_update_integrity"  result = OK }
}

# ======================================================================
# BAND APPLICATION — one active modifier per family, matching the DERIVED reach.
# [meter-concretize #1] Bands now key off qing_bureau_reach_cache (the yamen coverage fraction,
# 0..100). The decline SEMANTICS shift is intended (design §4.5): the 1763 seed covers all city
# provinces so reach starts ~100; the empire reaches "paralysed" by out-expanding its administration
# (annexing uncovered land / losing covered seats), not by governance rotting under corruption — the
# corruption-driven rot path now lives on the corruption band (DESIGN_CORRUPTION_CONCRETIZE).
# ======================================================================
QING_GOV_apply_capacity_band = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = QING  fn = "QING_GOV_apply_capacity_band" }
	remove_country_modifier = qing_bureau_capable
	remove_country_modifier = qing_bureau_strained
	remove_country_modifier = qing_bureau_paralysed
	if = {
		limit = { var:qing_bureau_reach_cache >= 70 }
		add_country_modifier = { name = qing_bureau_capable  duration = -1 }
		LOG_line = { sys = QING  msg = "band capacity = capable for (reach=)" }
	}
	else_if = {
		limit = { var:qing_bureau_reach_cache < 30 }
		add_country_modifier = { name = qing_bureau_paralysed  duration = -1 }
		LOG_line = { sys = QING  msg = "band capacity = paralysed for (reach=)" }
	}
	else_if = {
		limit = { var:qing_bureau_reach_cache < 50 }
		add_country_modifier = { name = qing_bureau_strained  duration = -1 }
		LOG_line = { sys = QING  msg = "band capacity = strained for (reach=)" }
	}
	LOG_exit = { sys = QING  fn = "QING_GOV_apply_capacity_band"  result = OK }
}

# THE GOVERNANCE PULSE ENTRY POINT
# Called from the throttled monthly pulse (00_monthly_country.txt) right after
# the decline pulse, so the council recompute below sees fresh decline counters.

	QING_ministry_recompute_all_perf = yes

	# 0b. [1:1 fix 2026-07-21 / one-sweep 2026-08-06] The 1:1 character:position VALIDATION SWEEP is now folded

	#     recompute runs on every court change (not just this quarterly pulse), so violations are caught fresher,
	#     and the audit shares the same every_character sweep as the officialdom-corruption sum (one full-pool
	#     walk instead of two). (se_QING_COUNCIL.txt)

	# 1. recompute the Grand Council's effectiveness from its seated members (+ 1:1 audit + corruption sum)

	#      stands in for it). O(1) scope reads; se_QING_SEATS.txt.

	#     top-N ordered_character court sort that was warming a GUI cache the player may never open that

	#    exam-ladder are no longer stored drift meters — QING_GOV_cache_reach recomputes

	# 2a. [#41 2026-08-13] re-apply the ADMIN_state_loyalty_from_province_drain/_gain state modifiers
