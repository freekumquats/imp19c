# se_QING_CARAVAN.txt — [#370] Silk Road / Central Asian Caravan Trade-Node System (絲路商道 / 回疆貿易).
#
# The ECONOMIC twin of the Xinjiang Consolidation arc (#367). Where se_QING_XINJIANG.txt models the
# GRIP on the New Dominion (garrisons, begs, tuntian, the 協餉 subsidy), this models the COMMERCE that
# ran alongside it: the caravan trade between Qing Altishahr (回疆, the Tarim Basin) and the Central
# Asian khanates — tea / rhubarb / silk / silver west, horses / jade / cotton / dried fruit east —
# funnelled through the oasis entrepôts of Kashgar (喀什噶爾), Yarkand (葉爾羌), Aksu and Khotan (和闐).
#
# HISTORY (Millward, Beyond the Pass / Eurasian Crossroads; Newby, The Empire and the Khanate;
# Fletcher): after the 1759 conquest the Qing taxed the caravans lightly (~5-10% ad valorem, levied
# chiefly at Yarkand), ran the frontier at a fiscal LOSS covered by the 協餉 inter-provincial subsidy,
# and used commerce as a PACIFICATION instrument — a prosperous oasis rebels less. The great crisis
# was COMMERCIAL: the Khanate of Kokand's escalating demands — the aqsaqal (阿奇木伯克) consul, the
# right to levy its OWN customs on third-party traders, extraterritoriality — culminated in the 1832
# settlement, often called the first "unequal treaty" the Qing granted, a decade before Nanking. Kokand
# weaponised the White-Mountain khoja pretenders (Jahangir 1826, the Seven Khojas 1847) whenever its
# demands were rebuffed.
#
# LAYER, DON'T DUPLICATE:
#   • OWNS one new meter: qing_caravan_prosperity (0..100, 商道繁榮). Joins the EXISTING shared counter
#     family (se_QING_DECLINE.txt) — same clamped-nudge idiom (QING_DECLINE_nudge), same init, same
#     band-swap (QING_DECLINE_apply_caravan_band). No new counter machinery.
#   • COUPLES to #367 (does not duplicate): the target reads the #367 grip meter qing_xinjiang_control
#     and the pending khoja-scare flag qing_xj_khoja_pending; prosperity nudges the grip back (a
#     thriving trade eases the hold, a cut route frays it). The Kokand-concession lever couples to the
#     SAME qing_xj_kokand_emboldened opinion #367 already owns.
#   • GRAND-COUNCIL FOLD is the SAME office, a SECOND additive term: caravan prosperity folds into the
#     LIFAN YUAN (理藩院) Director — the Court that historically governed Xinjiang trade relations — as
#     term (d) inside QING_ministry_recompute_perf_lifanyuan, alongside the #367 consolidation term (c).
#     Additive-inside (that recompute SETS the meter fresh each pulse), scaled small (/5), so the two
#     frontier terms are complementary, not double-counted.
#   • REVENUE: the caravan customs are an ad-valorem levy on the REAL Silk Road goods the empire produces
#     (GOODS_national_production_ tea/silk/silver west, wool/livestock/gems east — the same real-goods basis
#     the Canton customs use, #423 family), scaled by route throughput (prosperity) x the customs rate and
#     capped to the oasis road's carrying capacity. [Task #32 2026-07-25: reworked off an abstract
#     prosperity x rate-factor onto the real trade base.] The historically-small-but-real fiscal
#     contribution that now MOVES with the economy the player builds (the 協餉 cost side lives in #367).
#
# STATE (CHI country scope):
#   qing_caravan_prosperity        0..100  the trade-node health/volume (商道繁榮). Eases toward target.
#   qing_caravan_prosperity_target scratch: the recomputed target the meter eases toward (hysteresis)
#   qing_caravan_customs_rate      0/1/2   輕稅 light / 中稅 moderate / 重稅 heavy
#   qing_caravan_aqsaqal_granted   flag    the Kokand aqsaqal concession is in force (1832-style)
#   [#10A 2026-08-06] qing_caravan_yarkand_market RETIRED — replaced by the real qing_oasis_bazaar_building
#     (counted into qing_caravan_bazaar_count each recompute; the building IS the source of truth).
#   qing_caravan_bazaar_count      int     cached count of qing_oasis_bazaar_building in the oases (GUI read)
#   qing_caravan_initialized       init guard
# Reads/drives EXISTING state: qing_xinjiang_control (#367 grip), qing_xj_consolidation (#367),
#   qing_xj_khoja_pending (#367 scare), the KOK opinion qing_xj_kokand_emboldened.
#
# PREFIX: QING_caravan_ / sys = QING. CHI-only. Init from on_game_initialized (after QING_xj_init, whose
# grip meter the target reads). Drifts quarterly via QING_caravan_pulse (se_QING_GOVERNANCE.txt, AFTER
# QING_xj_pulse so it reads that pulse's fresh grip/scare state). All logged sys = QING, STATIC msgs only.
# Proven idioms only: QING_DECLINE_nudge clamped meter (amount=var:), add_treasury=var:
# (se_QING_MECHANICS:675), SUBJ_QING_advance_integration (se_QING_XINJIANG tuntian lever), the KOK
# opinion + khoja-scare vars (#367), banded country modifiers. NO create_character (#90 rule N/A — this
# is a pure counter/lever model). Kokand is the NEIGHBOURING Turkic power (separatism-backer rule
# satisfied — reuses #367's KOK coupling, not a far-flung power).
#
#   ROOT / this = the Qing country (CHI, player); root.current_ruler = the emperor.

# ======================================================================
# QING_caravan_init — seed the caravan bookkeeping once. Scope: COUNTRY (CHI). Idempotent (guards on a
# dedicated flag, not a count — the #336 init-recompute lesson). Seeds prosperity 45 (a going concern in
# the 1763 zenith, the routes open and the oases prosperous), customs_rate 1 (moderate, the historical
# default), market 0. Then recomputes so the target + the Lifan fold read a real value from turn 1.
# ======================================================================
QING_caravan_init = {
	LOG_enter = { sys = QING  fn = "QING_caravan_init" }
	if = {
		limit = { NOT = { has_variable = qing_caravan_initialized } }
		set_variable = { name = qing_caravan_initialized      value = 1 }
		set_variable = { name = qing_caravan_prosperity        value = 45 }
		set_variable = { name = qing_caravan_customs_rate      value = 1 }
		set_variable = { name = qing_caravan_bazaar_count      value = 0 }   # [#10A] cached bazaar-building count (recompute rebuilds it); qing_caravan_yarkand_market RETIRED
		# [#10A 2026-08-06] NO 1763 bazaar seed — opens at 0 bazaars (like the old counter), so prosperity
		# stays ~45 and does NOT trip the ≥55 Kokand-ultimatum gate. The player builds bazaars via the lever.
		# recompute at once so the target is tallied and the Lifan fold reads a real value FROM TURN 1
		# (nothing folds this into a game-start dispatcher, so init must recompute itself — #336 lesson).
		QING_caravan_recompute_target = yes
	}
	# [#112a] Seed the superintendent (參贊大臣) state with PER-VAR guards OUTSIDE the initialized block above,
	# so an already-initialized save still backfills these newly-added vars + seats a superintendent (matching
	# QING_canton_init's per-var pattern; the single qing_caravan_initialized guard would skip them otherwise).
	if = { limit = { NOT = { has_variable = qing_caravan_super_squeeze } }  set_variable = { name = qing_caravan_super_squeeze  value = 30 } }
	if = { limit = { NOT = { has_variable = qing_caravan_super_finesse } }  set_variable = { name = qing_caravan_super_finesse  value = 7 } }
	# the 參贊大臣 was a real, always-filled post — seat one now if none is seated (a PICK, not a create: safe
	# at init, no #90 boot-crash risk), so the panel shows him + his stats grade the yield from turn one.
	if = { limit = { NOT = { has_variable = qing_caravan_super_holder } }  QING_caravan_super_appoint = yes }
	LOG_exit = { sys = QING  fn = "QING_caravan_init"  result = OK }
}

# ======================================================================
# QING_caravan_recompute_target — write qing_caravan_prosperity_target on CHI from the frontier grip,
# the oasis market infrastructure, the Kokand concession, the customs burden, and any active route
# disruption. The meter (in the pulse) EASES toward this (hysteresis), matching the #367 / currency /
# population designs. All O(1) cached reads. Scope: COUNTRY (CHI).
# ======================================================================
QING_caravan_recompute_target = {
	# --- BASE: a going trade in a held frontier.
	set_variable = { name = qing_caravan_prosperity_target  value = 40 }

	# --- [meter-concretize #10 2026-08-06] + REAL OASIS TRADE (replaces the old qing_xinjiang_control/4
	# meter-of-meter term, which also broke the caravan↔control circular loop). Sum num_goods_produced across
	# the Dzungaria + Tarim oasis provinces — the actual goods moving through the Western Regions. These are
	# SUBJECT-held (ILI/XNG), so we MUST iterate by AREA, not every_owned_province (which returns 0 for CHI —
	# the #91 blocker, se_QING_ILI.txt:301-314). Proven area:+every_area_province effect idiom (se_QING_ILI:320);
	# num_goods_produced bare province read (GOODS_svalues.txt:1352). Scaled by S so the 1763 target lands ~45
	# (base 40 − 5 customs drag + S×oasis ≈ 45 → S×oasis ≈ 10; S PLACEHOLDER-playtest against the logged raw).
	# NOT the #32 income rework (country GOODS_national_production_*) — this is the narrower oasis subset.
	# read the oasis-trade sum from the proven script_value (area:+every_area_province+bare num_goods_produced
	# — see QING_governance_svalues.txt; NOT prev.num_goods_produced in an effect, which is unproven).
	set_variable = { name = qing_caravan_oasis_trade  value = QING_caravan_oasis_trade_svalue }
	# [PLACEHOLDER-playtest] scale factor S = 0.5 (raw oasis goods → prosperity points). Build-probe the raw
	# qing_caravan_oasis_trade at 1763 via the LOG below and re-pick S so the target opens ~45.
	set_variable = { name = qing_caravan_oasis_tmp  value = var:qing_caravan_oasis_trade }
	change_variable = { name = qing_caravan_oasis_tmp  multiply = 0.5 }
	change_variable = { name = qing_caravan_prosperity_target  add = var:qing_caravan_oasis_tmp }
	remove_variable = qing_caravan_oasis_tmp
	LOG_line = { sys = QING  msg = "caravan: oasis-trade term (raw num_goods_produced summed over Dzungaria+Tarim) for" }

	# --- [meter-concretize #10A 2026-08-06] + MARKET INFRASTRUCTURE, now CONCRETE: count the real
	# qing_oasis_bazaar_building on the map (replaces the abstract qing_caravan_yarkand_market counter). Each
	# bazaar +5. Area-iterated (owner-independent — the oases are subject-held; every_owned_province returns 0).
	# Cache the count into qing_caravan_bazaar_count for the GUI (a GUI GetVariable cannot read a script_value).
	set_variable = { name = qing_caravan_bazaar_count  value = 0 }
	area:Dzungaria = { every_area_province = { limit = { has_building = qing_oasis_bazaar_building }  ROOT = { change_variable = { name = qing_caravan_bazaar_count  add = 1 } } } }
	area:Tarim     = { every_area_province = { limit = { has_building = qing_oasis_bazaar_building }  ROOT = { change_variable = { name = qing_caravan_bazaar_count  add = 1 } } } }
	set_variable = { name = qing_caravan_mkt_tmp  value = var:qing_caravan_bazaar_count }
	change_variable = { name = qing_caravan_mkt_tmp  multiply = 5 }
	# --- AQSAQAL CONCESSION as a THROUGHPUT MULTIPLIER (USER 2026-08-06: aqsaqal = a diplomatic concession,
	# not a building): the Kokand consul draws the Andijani merchants THROUGH the bazaars, so its value SCALES
	# with the bazaar network (×1.5) instead of a flat +8. A concession with no markets draws little; a full
	# network swells the volume. (multiply=1.5 is proven in-file, se_QING_CARAVAN.txt.)
	if = {
		limit = { has_variable = qing_caravan_aqsaqal_granted }
		change_variable = { name = qing_caravan_mkt_tmp  multiply = 1.5 }
	}
	change_variable = { name = qing_caravan_prosperity_target  add = var:qing_caravan_mkt_tmp }
	remove_variable = qing_caravan_mkt_tmp

	# --- - CUSTOMS DRAG: heavy tariffs depress volume and breed smuggling. light 0 / moderate -5 /
	# heavy -15 (the historical trade-off — more revenue per unit, less trade).
	if = {
		limit = { has_variable = qing_caravan_customs_rate  var:qing_caravan_customs_rate = 2 }
		change_variable = { name = qing_caravan_prosperity_target  subtract = 15 }
	}
	else_if = {
		limit = { has_variable = qing_caravan_customs_rate  var:qing_caravan_customs_rate = 1 }
		change_variable = { name = qing_caravan_prosperity_target  subtract = 5 }
	}

	# --- - ROUTE DISRUPTION: a pending khoja scare (#367) cuts the caravan road (-15).
	if = {
		limit = { has_variable = qing_xj_khoja_pending }
		change_variable = { name = qing_caravan_prosperity_target  subtract = 15 }
	}

	# clamp 0..100.
	if = { limit = { var:qing_caravan_prosperity_target > 100 } set_variable = { name = qing_caravan_prosperity_target  value = 100 } }
	if = { limit = { var:qing_caravan_prosperity_target < 0 }   set_variable = { name = qing_caravan_prosperity_target  value = 0 } }
}

# ======================================================================
# QING_caravan_pulse — quarterly, from QING_GOV_pulse AFTER QING_xj_pulse (so it reads that pulse's
# fresh grip + scare state). Recomputes the target, eases the meter toward it, pays the quarterly
# customs, couples prosperity back into the grip, and offers the crisis events. Scope: COUNTRY (CHI).
# Cheap: O(1). Self-guarded; a clean no-op at a quiet frontier.
# ======================================================================
QING_caravan_pulse = {
	LOG_enter = { sys = QING  fn = "QING_caravan_pulse" }
	QING_caravan_init = yes   # idempotent — ensures the meter exists

	# --- recompute the target and EASE the meter 1/4 of the way toward it (hysteresis).
	QING_caravan_recompute_target = yes
	set_variable = { name = qing_caravan_ease_tmp  value = var:qing_caravan_prosperity_target }
	change_variable = { name = qing_caravan_ease_tmp  subtract = var:qing_caravan_prosperity }
	change_variable = { name = qing_caravan_ease_tmp  divide = 4 }
	QING_DECLINE_nudge = { var = qing_caravan_prosperity  amount = var:qing_caravan_ease_tmp }
	remove_variable = qing_caravan_ease_tmp

	# --- PAY THE CUSTOMS: an ad-valorem levy on the REAL goods the caravans carried, scaled by the route's
	# throughput (prosperity) and the customs rate. [Task #32 real-goods rework 2026-07-25] The take used to
	# be a pure abstract prosperity x rate-factor / 40. Now it is levied on the actual Silk Road trade base
	# the empire produces — the same GOODS_national_production_* reads the Canton customs use (se_QING_CANTON
	# :94-97) and #423's opium model: tea / silk / silver went WEST down the road, wool / livestock / gems
	# (the steppe & oasis goods — horses, jade, dried fruit map to these engine keys) came EAST. A larger
	# real trade economy = a fatter caravan customs, so the frontier income now MOVES with the economy the
	# player builds, not an abstract meter alone.
	#   base = (tea + silk + silver + wool + livestock + gems national production) / 20   [the tradeable volume]
	#   income = base x (prosperity/100)  [route throughput]  x rate-factor  [light .5 / moderate 1 / heavy 1.5]
	# A going concern lands near the old ~4/quarter at moderate rate, but now tracks the real export base.
	set_variable = { name = qing_caravan_goods_tmp  value = 0 }
	change_variable = { name = qing_caravan_goods_tmp  add = GOODS_national_production_tea }
	change_variable = { name = qing_caravan_goods_tmp  add = GOODS_national_production_silk }
	change_variable = { name = qing_caravan_goods_tmp  add = GOODS_national_production_silver }
	change_variable = { name = qing_caravan_goods_tmp  add = GOODS_national_production_wool }
	change_variable = { name = qing_caravan_goods_tmp  add = GOODS_national_production_livestock }
	change_variable = { name = qing_caravan_goods_tmp  add = GOODS_national_production_gems }
	change_variable = { name = qing_caravan_goods_tmp  divide = 20 }
	# cap the tradeable base (the Yarkand road's throughput ceiling — a caravan route is not a seaport):
	# an export boom must not balloon the frontier customs beyond what the oasis entrepôts can move. The
	# cap sits above a healthy going concern so the base still MOVES with the economy up to that ceiling
	# (mirrors the Canton yield cap, se_QING_CANTON:102, scaled to the smaller caravan volume).
	if = { limit = { var:qing_caravan_goods_tmp > 18 }  set_variable = { name = qing_caravan_goods_tmp  value = 18 } }
	# route throughput: the oasis trade only carries what the road can bear (prosperity 0..100 as a fraction).
	set_variable = { name = qing_caravan_income_tmp  value = var:qing_caravan_goods_tmp }
	change_variable = { name = qing_caravan_income_tmp  multiply = var:qing_caravan_prosperity }
	change_variable = { name = qing_caravan_income_tmp  divide = 100 }
	remove_variable = qing_caravan_goods_tmp
	# customs rate-factor: heavy skims more per unit (x1.5) even as it shrinks the base (the target penalty
	# above already cut prosperity for heavy/moderate); light takes half (x0.5); moderate is the x1 baseline.
	if = {
		limit = { var:qing_caravan_customs_rate = 2 }
		change_variable = { name = qing_caravan_income_tmp  multiply = 1.5 }
	}
	else_if = {
		limit = { var:qing_caravan_customs_rate = 0 }
		change_variable = { name = qing_caravan_income_tmp  multiply = 0.5 }
	}
	# moderate rate (1): factor 1 — leave income_tmp as the throughput-scaled base.
	# [#112a] CONTINUOUS SUPERINTENDENT-EFFECTIVENESS FACTOR — the 喀什噶爾參贊大臣 grades the take the way the
	# Hoppo grades Canton (#111a): a capable, honest superintendent delivers more of the road's worth to the
	# throne; a venal one lets his squeeze swallow it. Reads the LAGGED mirrors (qing_caravan_super_finesse /
	# _squeeze), set later this pulse in the reconcile block — the same one-quarter lag as the Hoppo (the yield
	# is computed BEFORE the holder is reconciled, so a live read here would risk a dead/absent holder). Both
	# guarded on has_variable (clean x1 only if ABSENT; init seeds them 7/30, so the squeeze shave applies
	# from pulse one — the intended Hoppo behavior, se_QING_CANTON:64-66).
	#   factor = 1 + (finesse - 7) x 0.03 - squeeze x 0.004 , clamped [0.5, 1.3].
	# Applied to the throughput-scaled base BEFORE the Kokand haircut below. NOTE: caravan income is add_treasury
	# ONLY (no silver, unlike Canton), so this cannot perturb the #23 currency model; the clamp is treasury sanity.
	if = {
		limit = { has_variable = qing_caravan_super_finesse }
		set_variable = { name = qing_caravan_super_factor  value = 1 }
		set_variable = { name = qing_caravan_super_fin_tmp  value = var:qing_caravan_super_finesse }
		change_variable = { name = qing_caravan_super_fin_tmp  subtract = 7 }
		change_variable = { name = qing_caravan_super_fin_tmp  multiply = 0.03 }
		change_variable = { name = qing_caravan_super_factor  add = var:qing_caravan_super_fin_tmp }
		remove_variable = qing_caravan_super_fin_tmp
		if = {
			limit = { has_variable = qing_caravan_super_squeeze }
			set_variable = { name = qing_caravan_super_sq_tmp  value = var:qing_caravan_super_squeeze }
			change_variable = { name = qing_caravan_super_sq_tmp  multiply = 0.004 }
			change_variable = { name = qing_caravan_super_factor  subtract = var:qing_caravan_super_sq_tmp }
			remove_variable = qing_caravan_super_sq_tmp
		}
		if = { limit = { var:qing_caravan_super_factor > 1.3 }  set_variable = { name = qing_caravan_super_factor  value = 1.3 } }
		if = { limit = { var:qing_caravan_super_factor < 0.5 }  set_variable = { name = qing_caravan_super_factor  value = 0.5 } }
		change_variable = { name = qing_caravan_income_tmp  multiply = var:qing_caravan_super_factor }
		remove_variable = qing_caravan_super_factor
	}
	if = {
		# [#13 2026-08-07] once a beaten Kokand is SETTLED (terms dictated OR route absorbed), conquest supersedes
		# the treaty — the throne collects FULL customs even if an aqsaqal was historically granted (a treasury
		# gain, NEW-MEDIUM-1). [#13-R impl-review MED-2] Guard on QING_kok_yielded_flag (ORs both end-states) not
		# just the dictate flag, else the ABSORB monopoly leaves a foreign consul skimming a route the throne now
		# supposedly owns outright — an incoherent end-state on the grant→conquer→absorb path.
		limit = { has_variable = qing_caravan_aqsaqal_granted  NOT = { QING_kok_yielded_flag = yes } }
		change_variable = { name = qing_caravan_income_tmp  divide = 2 }   # Kokand skims the third-party customs
	}
	if = {
		limit = { var:qing_caravan_income_tmp > 0 }
		add_treasury = var:qing_caravan_income_tmp
		LOG_line = { sys = QING  msg = "caravan: the Yarkand customs pay their quarterly take for" }
	}
	# [#430] publish the quarterly caravan take for the Revenue panel 回疆貿易 ledger line AND the Board of
	# Revenue inflow term (#431 reads qing_caravan_income_last). A real on-books treasury stream, so the 戶部
	# gets credit — a Yarkand trade that pays well lifts the minister even if the land tax lags. Published
	# even at 0 so a dead route reads 0, not a stale prior value.
	set_variable = { name = qing_caravan_income_last  value = var:qing_caravan_income_tmp }
	remove_variable = qing_caravan_income_tmp

	# [#112a] RECONCILE THE SUPERINTENDENT (參贊大臣). Mirrors the Hoppo (#69, se_QING_CANTON:255-347). Runs
	# AFTER the income is banked + published (income_last persists; income_tmp is already removed above). (a)
	# relieve a double-booked holder; (b) backfill a dead/missing one; (c) graft creep; (d) SIPHON from the
	# road's take as REAL char wealth; (e) mirror squeeze(=corruption) + finesse for next pulse's yield grade.
	# (a) double-book relief — a superintendent who later took a command/governorship, OR who was
	# already drawn into another courtier post (e.g. the Customs Inspector-General), loses this post.
	# [fix #28 2026-08-13] added qing_customs_ig_marker/qing_court_artist: QING_char_holds_court_
	# position (qing_dynasty_triggers.txt) already excludes seated holders of BOTH markers from
	# FUTURE draws (#27), but this reconcile's own double-book gate never checked either — so a man
	# who was drawn as caravan superintendent BEFORE #27 shipped (or via any other stale-save path)
	# would never be relieved here, unlike the general/admiral/governor/qing_officer_marker cases
	# already covered. Mirrors the Hoppo's identical reconcile at se_QING_CANTON.txt:255-270.
	if = {
		limit = {
			has_variable = qing_caravan_super_holder
			var:qing_caravan_super_holder = { is_alive = yes }
			var:qing_caravan_super_holder = {
				OR = {
					is_general = yes
					is_admiral = yes
					is_governor = yes
					has_variable = qing_officer_marker
					has_variable = qing_customs_ig_marker
					has_variable = qing_court_artist
				}
			}
		}
		var:qing_caravan_super_holder = { if = { limit = { has_variable = qing_caravan_super_marker }  remove_variable = qing_caravan_super_marker } }
		remove_variable = qing_caravan_super_holder
		LOG_line = { sys = QING  msg = "caravan: relieved a double-booked superintendent (also a serving commander/officer/other courtier post) for" }
	}
	# (b) backfill a dead/missing superintendent — appoint a fresh man (a new broom, low corruption).
	if = {
		limit = { OR = { NOT = { has_variable = qing_caravan_super_holder }  var:qing_caravan_super_holder = { is_alive = no } } }
		if = { limit = { has_variable = qing_caravan_super_holder }  remove_variable = qing_caravan_super_holder }
		QING_caravan_super_appoint = yes
		LOG_line = { sys = QING  msg = "caravan: the superintendent's post fell vacant and was backfilled for" }
	}
	if = {
		limit = { has_variable = qing_caravan_super_holder  var:qing_caravan_super_holder = { is_alive = yes } }
		# (c) graft creep — his corruption climbs quarterly (the source of the squeeze), capped < 90.

# ======================================================================
# LEVER: GRANT THE AQSAQAL CONCESSION (許阿奇木) — recognise the Kokand consul and its right to levy its

# commercial upside: prosperity swells, Kokand relations mend (clear the emboldened grievance + goodwill),

QING_caravan_grant_aqsaqal = {

	QING_caravan_init = yes
	if = {
		limit = { NOT = { has_variable = qing_caravan_aqsaqal_granted } }

		QING_DECLINE_nudge = { var = qing_caravan_prosperity  amount = 10 }
		# Kokand is mollified — clear the grievance and lay in real goodwill.
		if = {
			limit = { exists = c:KOK }
			c:KOK = {
				# remove_opinion no-ops if the grievance is absent (proven unguarded in se_QING_DIPLO).
				remove_opinion = { modifier = qing_xj_kokand_emboldened  target = ROOT }
				add_opinion = { modifier = qing_gp_relation_opinion  target = ROOT }
			}
		}

		if = {
			limit = { has_variable = qing_xj_khoja_pending }
			remove_variable = qing_xj_khoja_pending
		}

		add_legitimacy = -6

		current_ruler = { add_popularity = -15 }

	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_caravan_grant_aqsaqal"  reason = "aqsaqal concession already in force" }
	}
	# [#112b] Seat the real Kokandi aqsaqal (阿奇木伯克) at the SINGLE choke point (rev-112bc C1): EVERY grant
	# path (qing_caravan.1.grant, .2.negotiate, the panel lever, the trade-concession law on_enact) routes

	QING_caravan_aqsaqal_seat = yes
	LOG_exit = { sys = QING  fn = "QING_caravan_grant_aqsaqal"  result = OK }

			}
			remove_variable = qing_caravan_aqsaqal_holder
			LOG_line = { sys = QING  msg = "caravan: the aqsaqal concession is revoked, the Kokandi consul reverts to Kokand for" }
		}
		# the throne recovers its dignity and its revenue.
		add_legitimacy = 3
		# but Kokand is affronted — the grievance returns and the frontier grip frays.
		if = {
			limit = { exists = c:KOK }
			# add_opinion refreshes if already present (proven unguarded in se_QING_DIPLO).
			c:KOK = { add_opinion = { modifier = qing_xj_kokand_emboldened  target = ROOT } }
		}
		# [meter-concretize #10B] −6 control nudge DROPPED — the frayed grip is stamped CONTESTED on an oasis
		# (gated on not-fully-integrated, review H1); control DERIVES down. Gated so a normalised province
		# can't be re-destabilised post-integration.
		if = { limit = { NOT = { has_variable = qing_xj_fully_integrated } }  QING_ili_apply_prov_band = { INTENT = contested }  QING_xj_derive_control = yes }
		QING_caravan_recompute_target = yes
		LOG_line = { sys = QING  msg = "caravan: the Kokand aqsaqal concession is revoked for" }
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_caravan_revoke_aqsaqal"  reason = "no aqsaqal concession in force" }
	}
	LOG_exit = { sys = QING  fn = "QING_caravan_revoke_aqsaqal"  result = OK }
}
