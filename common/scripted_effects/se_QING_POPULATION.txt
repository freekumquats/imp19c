# se_QING_POPULATION.txt — [#369] Population Pressure / Famine Deep Model (人口壓力 / 人地矛盾 / 饑荒).
#
# A STANDING Malthusian population-pressure driver over the mod's EXISTING one-shot demographic
# pieces. The suite already models the story in fragments — the New World crop boom fires ONCE
# and forks ONCE (qing_migration.20-.23), qing_granary_stock drifts as a famine reserve, and
# se_MIGRATION.txt is a full bottom-up migration engine with migr_gov_push/pull government levers
# — but NOTHING continuously drives them off the realm's crowding. #369 supplies that: a live
# qing_pop_pressure meter (0..100, 人口壓力) that MOUNTS with the population boom and involution
# (內卷), RELAXES with New World crops / frontier resettlement / granary relief, and continuously
# COUPLES the existing pieces — driving the migration heartland-push, feeding qing_sect_pressure
# (the famine->rebellion linkage: White Lotus 1794, Taiping 1850), and dragging the Board of
# Revenue's performance.
#
# This is Hong Liangji's 洪亮吉 治平篇 (c.1793) thesis — "China's Malthus", predating Malthus by
# five years — made into a live meter: 戶口 (households) outrun 田廬 (fields and dwellings).
#
# LAYER, DON'T DUPLICATE. qing_pop_pressure joins the EXISTING shared counter family
# (se_QING_DECLINE.txt: corruption/sect/currency/granary) — same clamped-nudge idiom
# (QING_DECLINE_nudge), same band-swap (QING_DECLINE_apply_bands), same init. It owns no new
# meter machinery. The migration coupling reuses QING_COLON_heartland_push/clear (se_QING_COLON.txt)
# and QING_COLON_frontier_pull; the famine reserve is the existing qing_granary_stock.
#
# THE GRAND-COUNCIL FOLD IS A REAL PER-OFFICE DRAG (not the transitive harmony fold): population
# pressure is a matter of STATE ADMINISTRATION, so it folds into the Board of Revenue (戶部),
# whose QING_ministry_recompute_perf_revenue already scores granary coverage / salt / currency /
# corruption. The 戶部 historically held the 黃冊 (Yellow Registers, the census) and the 常平倉
# (ever-normal granaries) — the two instruments of demographic management — so a realm buckling
# under 人地矛盾 IS a Board of Revenue failure. #369 adds ONE drag term (f) there (see
# se_QING_MINISTRY.txt), the same shape as the currency-stress (d) and corruption (e) drags.
#
# ALL PRIMITIVES PROVEN (no oracle needed): the clamped counter idiom (QING_DECLINE_nudge),
# total_population at country scope (slave_revolt_svalue 00_event_values.txt:113; DIPLOMACY_svalues
# .txt:527), banded country modifiers, the migration levers (migr_gov_push/pull, se_QING_COLON.txt),
# the frontier regions (Turkestan/Yunnan/Guizhou/Sichuan_Kham/Gansu/Qinghai — the historically-OPEN
# valves; NOT Manchuria, which was legally closed by the 柳條邊 Willow Palisade until ~1860).
#
#   ROOT / this = CHI (guarded). Prefix = QING_pop · sys tag = QING.
#   State (on CHI):
#     qing_pop_pressure          0..100  the standing Malthusian meter (人口壓力). Eases toward target.
#     qing_pop_pressure_target   scratch: the recomputed target the meter eases toward (hysteresis)
#     qing_frontier_resettlement flag: a standing 移民實邊 policy is open (a relief valve the target reads)
#   Existing state it drives/reads: qing_sect_pressure, qing_granary_stock, qing_newworld_crops,
#     qing_migr_crop_boom_golden (modifier), migr_gov_push/pull (se_MIGRATION coupling).
########################################################################

# ======================================================================
# TARGET RECOMPUTE — write qing_pop_pressure_target on CHI from the realm's crowding and the
# relief valves. The meter (below) EASES toward this target (hysteresis), so a shock does not
# snap the band — it matches the currency-stress design. All O(1) cached reads. Scope: COUNTRY.
# ======================================================================
QING_pop_recompute_target = {
	# [BT 2026-07-25 logfix] Seed the target to 0 BEFORE the total_population read. On a rare early
	# pulse total_population evaluated to `none` (the country's pop index not yet warm), so the
	# set_variable below did not land and every downstream `var:qing_pop_pressure_target < 0` compare
	# then read an UNSET var — the "compare_value returned an unset scope" / "value of type none" flood
	# (3x at 18:23:42). A cheap unconditional seed guarantees the var is always numeric for the compares
	# even if the total_population read is momentarily none; the real value overwrites it on the next line.
	set_variable = { name = qing_pop_pressure_target  value = 0 }
	# --- CROWDING (the core driver): total_population vs a carrying baseline. High Qing China
	# is already crowded; the target rises as population grows past the baseline. total_population
	# reads at country scope as the COUNT of pop objects (proven: slave_revolt_svalue subtracts
	# num_of_slaves; housing_svalues total_population_tenth = total_population*0.1). CHI ships ~40k
	# pop objects at the 1763 start, so /1200 lands the base crowding term at ~34 — a sane High-Qing
	# baseline that the involution (+18) and thin-granary (+15) offsets push into the strain (>=45),
	# crisis (>=75) and event bands. [#371-R: was /12000, which pinned the target to ~4 so the meter
	# could NEVER cross any band and the whole Malthusian dynamic was inert — a 10x scale error.]
	# [logfix] MUST wrap as value={ value=total_population }: a BARE `value = total_population` in a
	# set_variable is parsed as a SCOPE link (empty), not a number, so the var lands non-numeric and the
	# divide below + every later read flood ("Variable not of the 'value' scope type" / "not set"). The
	# proven numeric idiom is the wrapped form (se_LAND.txt:160 EDU_*_governorship).
	# [BT 2026-07-25 logfix R2] GUARD the read on total_population > 0. On a rare early pulse the pop
	# index is not warm and total_population itself evaluates to `none` — the wrapped set_variable then
	# logged "Value of wrong type ... Got value of type 'none'" at THIS line (the R1 seed above stopped
	# the downstream compare flood, but not this read itself). total_population is a valid literal-RHS
	# trigger (naval_raiding.txt:57, decentralize_realm_button.txt:25). When it's 0/none the seeded 0
	# stands (a realm with no counted pops has no man-land crisis), so the crowding term is simply skipped.
	if = {
		# [logfix 2026-08-07] the guard trigger was `total_population > 0` — but total_population is a
		# PROVINCE-only TRIGGER; in this country-scope effect it errored "Wrong scope for trigger: country,
		# expected province" (8x/boot). (It still works as a VALUE on the next line — the engine coerces the
		# country-scope sum there.) Use country_population > 0, the PROVEN country-scope pop trigger
		# (EE_lists.txt:38), which co-warms with total_population off the same pop index — same "pops are
		# warm" intent, no scope error. The /1200 calibration on the total_population VALUE is unchanged.
		limit = { country_population > 0 }
		set_variable = { name = qing_pop_pressure_target  value = { value = total_population } }
		change_variable = { name = qing_pop_pressure_target  divide = 1200 }
	}
	# floor at 0 (a tiny realm has no man-land crisis).
	if = { limit = { var:qing_pop_pressure_target < 0 } set_variable = { name = qing_pop_pressure_target  value = 0 } }

	# --- + INVOLUTION (內卷): the New World crop boom has fired but NOT resolved to the golden
	# path — more mouths on the same marginal hill land (maize/sweet-potato involution). +18.
	if = {
		limit = {
			has_variable = qing_newworld_crops
			NOT = { has_country_modifier = qing_migr_crop_boom_golden }
		}
		change_variable = { name = qing_pop_pressure_target  add = 18 }
	}

	# --- + FAMINE STRESS: a thin granary reserve is no buffer against the next bad harvest.
	# qing_granary_stock is the existing 0..100 famine reserve (se_QING_DECLINE.txt).
	if = {
		limit = { has_variable = qing_granary_stock  var:qing_granary_stock < 30 }
		change_variable = { name = qing_pop_pressure_target  add = 15 }
	}
	# a well-stocked reserve lets the state feed the crowd — ease the target a little.
	if = {
		limit = { has_variable = qing_granary_stock  var:qing_granary_stock >= 60 }
		change_variable = { name = qing_pop_pressure_target  subtract = 8 }
	}

	# --- - RELIEF VALVES: a standing frontier-resettlement policy (移民實邊) and the golden crop
	# path (the safety valves working) both bleed off pressure.
	if = {
		limit = { has_variable = qing_frontier_resettlement }
		change_variable = { name = qing_pop_pressure_target  subtract = 12 }
	}
	# [#40 Batch 4f 2026-07-25] FRONTIER SETTLEMENT law bias (移民實邊). The qing_frontier_settlement_law
	# selector writes qing_frontier_settle_policy (0 managed default / 1 open / 2 flood); it scales the
	# pressure relief ON TOP of the one-shot panel-lever flag above (which is unchanged — no regression:
	# a game that only used the lever behaves exactly as before). Open bleeds an extra -8, flood -16 — the
	# player CHOOSING how hard to open the marches. Guarded on has_variable (unset until enacted -> the
	# default managed stance is a clean no-op). LHS var, literal RHS (RHS-comparison rule). Clamp below bounds it.
	if = {
		limit = { has_variable = qing_frontier_settle_policy  var:qing_frontier_settle_policy = 1 }
		change_variable = { name = qing_pop_pressure_target  subtract = 8 }
	}
	else_if = {
		limit = { has_variable = qing_frontier_settle_policy  var:qing_frontier_settle_policy = 2 }
		change_variable = { name = qing_pop_pressure_target  subtract = 16 }
	}
	if = {
		limit = { has_country_modifier = qing_migr_crop_boom_golden }
		change_variable = { name = qing_pop_pressure_target  subtract = 10 }
	}
	# [#88 2026-08-13] NEW WORLD AGRICULTURE (settle-frontier mission #65 reward): the realm has
	# organized New World crop cultivation across its farmsteads. NOTE: this REPLACES the mission's
	# old standalone global_population_growth/global_population_capacity_modifier effect — this
	# custom pressure meter has no capacity term of its own (its crowding driver is a fixed
	# total_population/1200 ratio, above), so the old vanilla-engine capacity lift is not "re-homed"
	# here, it is RETIRED; this pressure-meter credit is the mission's entire replacement reward, not
	# an addition to a capacity effect that still exists elsewhere. Magnitude -6, below the golden-crop
	# relief's -10: that path is gated behind a risky, RNG-contingent chain that can fail and instead
	# grant a malus (qing_migr_overpopulation), while this mission's path (treasury>=60 + >=3
	# farmsteads) is deterministic and risk-free — equal weighting would over-reward the strictly-
	# easier path. OVERNIGHT ASSUMPTION; boot-tune against pop logs like every other term here.
	if = {
		limit = { has_variable = qing_newworld_agriculture }
		change_variable = { name = qing_pop_pressure_target  subtract = 6 }
	}

	# clamp target 0..100.
	if = { limit = { var:qing_pop_pressure_target > 100 } set_variable = { name = qing_pop_pressure_target  value = 100 } }
	if = { limit = { var:qing_pop_pressure_target < 0 }   set_variable = { name = qing_pop_pressure_target  value = 0 } }
}

# ======================================================================
# THE PULSE — quarterly, from QING_GOV_pulse after QING_princes_pulse. Recomputes the target,
# eases the meter toward it, couples pressure into migration + sects, and offers the crisis events.
# Scope: COUNTRY (CHI). Cheap: O(1) reads + the O(owned) heartland-push (already used by the boom).
# ======================================================================
QING_pop_pulse = {
	LOG_enter = { sys = QING  fn = "QING_pop_pulse" }
	QING_DECLINE_init = yes   # idempotent — ensures qing_pop_pressure exists (see se_QING_DECLINE.txt init)

	# --- recompute the target and EASE the meter 1/4 of the way toward it (hysteresis) ---
	QING_pop_recompute_target = yes
	# delta = (target - current) / 4, applied via the clamped nudge.
	set_variable = { name = qing_pop_ease_tmp  value = var:qing_pop_pressure_target }
	change_variable = { name = qing_pop_ease_tmp  subtract = var:qing_pop_pressure }
	change_variable = { name = qing_pop_ease_tmp  divide = 4 }
	QING_DECLINE_nudge = { var = qing_pop_pressure  amount = var:qing_pop_ease_tmp }
	remove_variable = qing_pop_ease_tmp

	# --- COUPLE INTO MIGRATION: a crowded core sheds people to the frontier; a relieved one
	# settles back. Reuses the existing COLON levers (migr_gov_push, read by MIGRATION_push_province).
	if = {
		limit = { var:qing_pop_pressure >= 60 }
		QING_COLON_heartland_push = { count = 3 }
		LOG_line = { sys = QING  msg = "population: the crowded heartland sheds its surplus to the frontier" }
	}
	else_if = {
		# [#371-R] clear the heartland push ONLY when no crop-boom-driven migration is meant to be
		# live. QING_COLON_clear_heartland_push strips migr_gov_push from EVERY owned province, and
		# the crop-boom events (qing_migration.20/.23.b) set that SAME variable on the top provinces
		# to drive the boom's out-migration over the decade migration pulse. Without this guard the
		# quarterly clear wiped the boom's push within one quarter (in the common low-pressure state),
		# silently defeating the boom's frontier settlement. The golden path (.22) clears its own push.
		limit = {
			var:qing_pop_pressure < 35
			NOR = {
				has_country_modifier = qing_migr_crop_boom
				has_country_modifier = qing_migr_overpopulation
			}
		}
		QING_COLON_clear_heartland_push = yes
	}

	# --- COUPLE INTO SECTS: severe pressure feeds the landless into the millenarian sects (the
	# historical White Lotus / Taiping recruiting ground — 棚民 shack-people on the hill margins).
	if = {
		limit = { var:qing_pop_pressure >= 70 }
		QING_DECLINE_nudge = { var = qing_sect_pressure  amount = 2 }
		LOG_line = { sys = QING  msg = "population: the landless swell the millenarian sects" }
	}

	# --- OFFER THE MEMORIAL once pressure first crosses the danger band (player-only; the event
	# re-checks its own state). The once-only "seen" flag is set in qing_population.1's own
	# immediate, NOT here — the #366/#368 flag-leak fix, so a re-check failure during the delay
	# never strands it.
	if = {
		limit = {
			var:qing_pop_pressure >= 65
			NOT = { has_variable = qing_population_memorial_seen }
			is_ai = no
		}
		trigger_event = { id = qing_population.1  days = { 5 20 } }
	}
	# --- OFFER THE FAMINE when pressure is high AND the granary reserve is exhausted.
	else_if = {
		limit = {
			var:qing_pop_pressure >= 60
			has_variable = qing_granary_stock
			var:qing_granary_stock <= 15
			NOT = { has_variable = qing_population_famine_seen }
			is_ai = no
		}
		trigger_event = { id = qing_population.2  days = { 5 20 } }
	}
	# [#429] NOTE: real famine deaths are NOT handled here — the vanilla shortage.1 event fires monthly
	# for CHI (player AND AI) off the engine's real food shortage and does the killing (weighted pops +
	# recent_famine). The Qing side reacts via QING_pop_famine_feedback (called from shortage.1). The
	# qing_population.2 event above is the player-facing Malthusian memorial/choice layered on top.
	LOG_exit = { sys = QING  fn = "QING_pop_pulse"  result = OK }
}

# ======================================================================
# [#57, design/DESIGN_57_NEWWORLD_CROPS_CONCRETE.md] NEW WORLD CROP POP-BOOM PULSE — makes the
# "New World crops -> boom -> pressure -> migration" loop production-linked instead of a one-shot
# die roll (`qing_migration.20-23`, unchanged, stays the narrative wrapper). Piggybacks QING_GOV_
# pulse's cadence (called alongside QING_pop_pulse, se_QING_GOVERNANCE.txt). CHI-only (design's
# open question 1, v1 scope — the farmstead building itself stays generic/worldwide).
#
# TWO DISTINCT quantities, deliberately separate vars so a tier crossing spawns a small FIXED
# batch, never scaling with the realm's total farmstead count (design's own round-1 warning against
# reusing one variable for both):
#   qing_newworld_farmstead_count — the GATE input (farmstead count), recomputed every pulse.
#   pops_per_tier (literal 2 below) — a separate, small, fixed constant [ASSUMPTION, boot-tune
#     per Rule 1a] — pops actually created on a tier crossing. NEVER derived from farmstead_count.
#   qing_newworld_pop_tier — the ratchet: SET (never incremented) to floor(farmstead_count /
#     TIER_THRESHOLD) on a crossing, so a multi-tier jump in one pulse still only grants ONE
#     pops_per_tier batch (deliberate, accepted simplification per the design doc).
# ======================================================================
QING_pop_newworld_growth_pulse = {
	LOG_enter = { sys = QING  fn = "QING_pop_newworld_growth_pulse" }
	if = { limit = { NOT = { has_variable = qing_newworld_pop_tier } } set_variable = { name = qing_newworld_pop_tier  value = 0 } }

	# --- GATE INPUT: count owned provinces with a New World farmstead.
	set_variable = { name = qing_newworld_farmstead_count  value = 0 }
	every_owned_province = {
		limit = { has_building = new_world_farmstead_building }
		change_variable = { name = qing_newworld_farmstead_count  add = 1 }
	}

	# floor(farmstead_count / TIER_THRESHOLD) — TIER_THRESHOLD=10 [ASSUMPTION, boot-tune per Rule
	# 1a; see design doc's 16-province 1763 baseline sanity bound]. Scratch var, removed below.
	set_variable = {
		name = qing_newworld_pop_tier_current
		value = {
			value = var:qing_newworld_farmstead_count
			divide = 10
			round = floor
		}
	}

	if = {
		# UPWARD tier crossing only (the ratchet only ever grows here; the removal branch below is
		# the separate, unconditional downward-move check).
		limit = { var:qing_newworld_pop_tier_current > qing_newworld_pop_tier_cmpsvalue }
		if = {
			limit = {
				any_owned_province = {
					OR = {
						trade_goods = maize
						trade_goods = sweet_potato
						trade_goods = potato
						trade_goods = peanut
						trade_goods = chili
					}
				}
			}
			ordered_owned_province = {
				order_by = total_population
				limit = {
					OR = {
						trade_goods = maize
						trade_goods = sweet_potato
						trade_goods = potato
						trade_goods = peanut
						trade_goods = chili
					}
				}
				max = 2   # pops_per_tier [ASSUMPTION, boot-tune per Rule 1a]
				create_state_pop = freemen
				set_variable = { name = migr_gov_push  value = 6 }
			}
			LOG_line = { sys = QING  msg = "newworld crop boom: pop batch created + heartland push set" }
		}
		else = {
			LOG_fail = { sys = QING  fn = "QING_pop_newworld_growth_pulse"  reason = "tier crossed but no owned New World crop province" }
		}
		# [fix, review 2026-08-14] the event chain (qing_migration.21-23) SWAPS qing_migr_crop_boom
		# away for one of two resolved states (_golden or qing_migr_overpopulation) once the boom
		# reckoning fires -- an unconditional re-grant here would reintroduce the neutral boom's own
		# growth/tax terms ALONGSIDE whichever resolved state is now active on a later tier crossing
		# (they are distinct modifier keys, so both would stack, undoing the crisis path's intended
		# malus / double-crediting the golden path's growth). Guard against both resolved states.
		if = {
			limit = {
				NOR = {
					has_country_modifier = qing_migr_crop_boom_golden
					has_country_modifier = qing_migr_overpopulation
				}
			}
			add_country_modifier = { name = qing_migr_crop_boom  duration = -1 }
		}
		set_variable = { name = qing_newworld_pop_tier  value = var:qing_newworld_pop_tier_current }
		# diagnostic: literal per-tier band string (this engine can't render raw numeric values in
		# logs, per design/LOGGING_TEMPLATE.md:37-38 — proven band-ladder shape only, matching
		# QING_DECLINE_apply_pop_pressure_band's idiom).
		if = { limit = { var:qing_newworld_pop_tier_current = 1 } LOG_line = { sys = QING  msg = "newworld crop boom: pop tier 1 reached" } }
		else_if = { limit = { var:qing_newworld_pop_tier_current = 2 } LOG_line = { sys = QING  msg = "newworld crop boom: pop tier 2 reached" } }
		else_if = { limit = { var:qing_newworld_pop_tier_current = 3 } LOG_line = { sys = QING  msg = "newworld crop boom: pop tier 3 reached" } }
		else = { LOG_line = { sys = QING  msg = "newworld crop boom: pop tier 4+ reached" } }
		# population-band snapshot at the moment of the crossing — bands sized around the 1763
		# baseline (~40k pop objects, se_QING_POPULATION.txt:63-64) so a human reader can compare the
		# band across quarters to see whether growth accelerated after a crossing.
		if = { limit = { country_population >= 60000 } LOG_line = { sys = QING  msg = "newworld crop boom: total_population band >= 60000" } }
		else_if = { limit = { country_population >= 50000 } LOG_line = { sys = QING  msg = "newworld crop boom: total_population band >= 50000" } }
		else_if = { limit = { country_population >= 40000 } LOG_line = { sys = QING  msg = "newworld crop boom: total_population band >= 40000" } }
		else = { LOG_line = { sys = QING  msg = "newworld crop boom: total_population band < 40000" } }
	}

	# --- REMOVAL: unconditional, SEPARATE branch, evaluated every pulse independent of the
	# tier-crossing gate above — nesting removal under the upward gate (which by construction never
	# fires on a downward move) would make removal structurally unable to ever fire, reproducing the
	# stuck-modifier bug the design doc's round-3 fix exists to prevent. qing_newworld_pop_tier
	# itself is NOT reset — the pop/tier ratchet is a permanent, one-way counter by design; only the
	# flavour modifier (a live read of CURRENT conditions) is reversible.
	if = {
		limit = {
			has_country_modifier = qing_migr_crop_boom
			var:qing_newworld_farmstead_count < 10   # TIER_THRESHOLD, matching the divide above
		}
		remove_country_modifier = qing_migr_crop_boom
	}

	remove_variable = qing_newworld_pop_tier_current
	LOG_exit = { sys = QING  fn = "QING_pop_newworld_growth_pulse"  result = OK }
}

# ======================================================================
# LEVER: RELIEF & RESETTLEMENT (賑濟移墾) — relieve a famine and resettle the surplus. Tops up the
# granary reserve, cuts pressure + sect pressure, and opens a frontier pull so the crowd resettles
# (以工代賑 work-relief + 移民實邊). Treasury cost. Scope: COUNTRY (CHI). Gated by the caller/panel.
# ======================================================================
QING_pop_relief_resettle = {
	LOG_enter = { sys = QING  fn = "QING_pop_relief_resettle" }
	# [fix, task #72 2026-08-14, design/DESIGN_72_POPULATION_LEVERS_CONCRETE.md] the tooltip narrates
	# "open the granaries to feed the stricken" (開倉放糧, distributing EXISTING stored grain) but the
	# old code ADDED 150 to qing_granary_food (a market purchase funded by treasury) -- backwards from
	# the narrated action, and disconnected from the granary system entirely. Now SUBTRACTS real stock,
	# gated on having enough to distribute (mirrors the treasury affordability gate's own idiom: an
	# action that can't be funded/stocked simply can't be taken). QING_DECLINE_granary_rederive already
	# floors the pool at 0, so this cannot go negative even if a preceding famine drained it further.
	if = {
		limit = {
			treasury >= 310
			has_variable = qing_granary_food
			var:qing_granary_food >= 150
		}
		add_treasury = -310
		TREASURY_LOG_it = { amount = -310 }
		change_variable = { name = qing_granary_food  subtract = 150 }
		QING_DECLINE_granary_rederive = yes
		QING_DECLINE_nudge = { var = qing_pop_pressure    amount = -15 }
		QING_DECLINE_nudge = { var = qing_sect_pressure   amount = -6 }
		# open the frontier valve so the relieved surplus actually resettles.
		QING_pop_open_frontier_valve = yes
		add_stability = 1
		LOG_line = { sys = QING  msg = "population: 賑濟移墾 relief distributed real granary stock, resettlement mounted" }
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_pop_relief_resettle"  reason = "treasury below 310 or granary stock below 150 -- cannot fund relief" }
	}
	LOG_exit = { sys = QING  fn = "QING_pop_relief_resettle"  result = OK }
}

# ======================================================================
# LEVER: TAX REMISSION (蠲免) — the classic imperial mercy: forgive the land tax in the stricken
# provinces. Eases pressure + a stability lift, at a treasury cost (forgone revenue). Immediate
# relief, nothing structural. Scope: COUNTRY (CHI).
# ======================================================================
QING_pop_tax_remission = {
	LOG_enter = { sys = QING  fn = "QING_pop_tax_remission" }
	if = {
		limit = { treasury >= 220 }
		add_treasury = -220
		TREASURY_LOG_it = { amount = -220 }
		QING_DECLINE_nudge = { var = qing_pop_pressure  amount = -8 }
		add_stability = 1
		# [fix, task #72 2026-08-14, design/DESIGN_72_POPULATION_LEVERS_CONCRETE.md] the tooltip
		# narrates "remit the land tax of the stricken provinces" but the old code touched no tax
		# mechanism at all -- only the abstract pressure meter. Now suspends REAL land tax for a year
		# via a proven modifier key (local_tax_modifier = -1.0, common/modifiers/imp19c_province_
		# modifiers.txt). Scoped to every owned city-status province -- qing_pop_pressure has no
		# per-province breakdown anywhere in this codebase to target "the stricken provinces" more
		# narrowly than that; a known, logged simplification (design doc §"known limitations"), not
		# an oversight.
		every_owned_province = {
			limit = { has_city_status = yes }
			add_province_modifier = { name = qing_tax_remission_active  duration = 365 }
		}
		LOG_line = { sys = QING  msg = "population: 蠲免 tax remission proclaimed, real land tax suspended for a year" }
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_pop_tax_remission"  reason = "treasury below 220" }
	}
	LOG_exit = { sys = QING  fn = "QING_pop_tax_remission"  result = OK }
}

# ======================================================================
# LEVER: PROMOTE FRONTIER SETTLEMENT (移民實邊) — a STANDING policy (vs the one-shot relief): open
# and hold the frontier resettlement valve, and set qing_frontier_resettlement (a relief valve the
# target recompute reads, so it durably lowers the pressure ceiling). Prestige cost (the throne
# commits to peopling the marches). Scope: COUNTRY (CHI).
# ======================================================================
QING_pop_promote_resettlement = {
	LOG_enter = { sys = QING  fn = "QING_pop_promote_resettlement" }
	if = {
		limit = { NOT = { has_variable = qing_frontier_resettlement } }
		set_variable = { name = qing_frontier_resettlement  value = 1 }
	}
	QING_pop_open_frontier_valve = yes
	# [T3 BT-5/6] was current_ruler add_prestige (FAMILY prestige — a near-inert no-op on a character);
	# the throne's political cost of the resettlement policy lands as ruler popularity instead.
	current_ruler = { add_popularity = -8 }
	QING_DECLINE_nudge = { var = qing_pop_pressure  amount = -5 }
	LOG_line = { sys = QING  msg = "population: 移民實邊 frontier resettlement promoted" }
	LOG_exit = { sys = QING  fn = "QING_pop_promote_resettlement"  result = OK }
}

# ======================================================================
# OPEN THE FRONTIER VALVE — mark the Qing's HELD frontier provinces as settlement draws
# (migr_gov_pull, read by MIGRATION_pull_province). Uses the historically-OPEN valves: Xinjiang
# 屯田 (Turkestan), the southwest 改土歸流 (Yunnan / Guizhou / Sichuan_Kham), and Gansu / Qinghai.
# NOT Manchuria — the 柳條邊 Willow Palisade kept it legally closed to Han settlement until ~1860.
# Bounded O(owned frontier provinces). Scope: COUNTRY (CHI).
# ======================================================================
QING_pop_open_frontier_valve = {
	every_owned_province = {
		limit = {
			OR = {
				is_in_region = Turkestan
				is_in_region = Yunnan
				is_in_region = Guizhou
				is_in_region = Sichuan_Kham
				is_in_region = Gansu
				is_in_region = Qinghai
			}
		}
		set_variable = { name = migr_gov_pull  value = 12 }
	}
	LOG_line = { sys = QING  msg = "population: frontier resettlement valve opened (Xinjiang/southwest/Gansu)" }
}
