# [#201 2026-07-30] Ideology-specific apotheosis FLOURISHES for the 48 thinker-deities.
#
# PROBLEM: the 48 ideology thinker-deities (common/deities/04_ideology_pantheons.txt) all reuse the
# same EIGHT generic category base effects (military_apotheosis_capital_freemen_effect, economy_income_effect,
# ...). So a Liberalism war-deity (Montesquieu) and a Communism war-deity (Marx) fire an IDENTICAL
# apotheosis. That is the defect this file fixes: each of the six ideologies gets a distinct doctrine
# flourish that is APPENDED to the category base in every deity's on_activate. Result: every one of the
# 48 is a unique (ideology-flourish x category-base) pair — distinct across ideologies (flourish differs)
# AND within an ideology (base differs).
#
# IDIOM: this mirrors the upstream wrapper pattern in common/scripted_effects/00_apotheosis.txt
# (tur_apotheosis_effect = military_apotheosis_capital_freemen_effect + add_trait; midas_apotheosis_effect
# = economy_apotheosis_province_improvement_effect + add_treasury; pylaemenes_apotheosis_effect =
# military_apotheosis_military_experience_effect + add_popularity + add_legitimacy). Every primitive below
# (add_political_influence / add_stability / add_legitimacy / current_ruler add_popularity / add_manpower /
# add_military_experience / every_country_state add_state_food / add_research) is used by that vanilla file,
# so nothing here is unproven engine capability.
#
# Each flourish is guarded on `exists = scope:deified_ruler` (matching the base effects) and carries a
# custom_tooltip so the deity-activation preview shows the extra line. Levers are chosen so all six differ:
#   Liberalism  -> political influence  (the open forum, free press, rule of law)
#   Conservatism-> stability            (prudent reform, the settled order)
#   Monarchism  -> legitimacy + ruler popularity (the crown exalted, divine right)
#   Nationalism -> manpower + military experience (la nation en armes)
#   Socialism   -> state food to every state (bread and the social question)
#   Communism   -> civic research (the scientific vanguard, planned progress)

# --- Liberalism (自由主義): the open forum ---
ideology_apotheosis_liberalism_effect = {
	if = {
		limit = { exists = scope:deified_ruler }
		custom_tooltip = ideology_apotheosis_liberalism_tt
		add_political_influence = 10
	}
}

# --- Conservatism (保守主義): the settled order ---
ideology_apotheosis_conservatism_effect = {
	if = {
		limit = { exists = scope:deified_ruler }
		custom_tooltip = ideology_apotheosis_conservatism_tt
		add_stability = 5
	}
}

# --- Monarchism (尊王): the crown exalted ---
ideology_apotheosis_monarchism_effect = {
	if = {
		limit = { exists = scope:deified_ruler }
		custom_tooltip = ideology_apotheosis_monarchism_tt
		if = {
			limit = { is_monarchy = yes }
			add_legitimacy = 10
		}
		current_ruler = {
			add_popularity = 5
		}
	}
}

# --- Nationalism (民族主義): la nation en armes ---
ideology_apotheosis_nationalism_effect = {
	if = {
		limit = { exists = scope:deified_ruler }
		custom_tooltip = ideology_apotheosis_nationalism_tt
		if = {
			limit = { manpower_percentage < 0.8 }
			add_manpower = {
				value = scope:deified_ruler.martial
				multiply = 0.5
				min = 1
				max = 5
			}
		}
		else = {
			add_military_experience = 3
		}
	}
}

# --- Socialism (社會主義): bread and the social question ---
ideology_apotheosis_socialism_effect = {
	if = {
		limit = { exists = scope:deified_ruler }
		custom_tooltip = ideology_apotheosis_socialism_tt
		hidden_effect = {
			every_country_state = {
				add_state_food = {
					value = 25
					multiply = scope:deified_ruler.charisma
					min = 25
					max = 250
				}
			}
		}
	}
}

# --- Communism (共產主義): the scientific vanguard ---
ideology_apotheosis_communism_effect = {
	if = {
		limit = { exists = scope:deified_ruler }
		custom_tooltip = ideology_apotheosis_communism_tt
		add_research = {
			technology = civic_tech
			value = 15
		}
	}
}
