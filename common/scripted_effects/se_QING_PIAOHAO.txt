# Seed one qing_draft_bank_building at province $P$, guarded (exists + CHI-owned +
# not already present). add_building_level bypasses the allow tech gate.
# [fix, boot-test 2026-08-16] $NAME$ used to be referenced NOWHERE in this macro's body (the
# comment's own claim that it was "only a readable label on the call site" was the bug): an
# argument a macro body never references is a load-time compile failure ("Compiling source for
# QING_piaohao_found_at failed for unknown arguments: NAME"), silently voiding the WHOLE call --
# all 5 of qing_piaohao.1's draft-bank seeds (Taiyuan/Beijing/Suzhou/Jiangning/Guangzhou) failed
# to compile every boot. Per the corrected log-string-macro-rule (only a literal '#' inside a
# LOG string is dangerous; $param$ substitution itself is cosmetic-only, proven safe at scale by
# QING_seed_works_building's 90+ working call sites), $NAME$ now appears in both LOG_line
# strings below, matching that exact proven idiom.
QING_piaohao_found_at = {
	if = {
		limit = {
			exists = p:$P$
			p:$P$ = {
				owner = c:CHI
				has_city_status = yes   # re-guard the building's own potential (code-review: don't rely on add_building_level bypassing potential; honest SKIP if a target ever lacks city status)
				NOT = { has_building = qing_draft_bank_building }
			}
		}
		p:$P$ = {
			add_building_level = qing_draft_bank_building
			LOG_line = { sys = QING  msg = "piaohao: qing_draft_bank_building founded — $NAME$" }
		}
	}
	else = {
		LOG_line = { sys = QING  msg = "piaohao: SKIP draft bank ($NAME$) — not CHI-owned or already present" }
	}
}
