########################################################################
# se_LOG.txt — Diagnostic logging module
#
# Purpose: emit richly-tagged lines to logs/debug.log so a play session run
#          with the -debug_mode launch option produces a diagnostic trace that
#          can be handed back for analysis. The debug_log effect writes to
#          <userdir>/Paradox Interactive/Imperator/logs/debug.log (NOT game.log).
#          Engine errors land in logs/error.log automatically; logical failures
#          are surfaced here in debug.log with a loud tag.
#
# IMPORTANT: debug_log ONLY outputs when the game is launched in -debug_mode.
#            All lines below are silently dropped in a normal session, so this
#            module is zero-cost in ordinary play.
#
# [#90-fix cosmetic] The tag WAS "[IMP19C][$sys$]". SQUARE BRACKETS inside a
# debug_log / loc string are parsed by the engine as data-function ([GetX] )
# syntax, so "[IMP19C][$sys$]" was mangled on output.
#
# [#54 2026-08-07 CORRECTION — the old claim here was FALSE] The old comment said
# "$arg$ substitution into debug_log DOES work (proven: se_PURCHASE)". It does NOT.
# Proven at byte level from the Aug-7 boot (od -c on debug.log): a $param$ inside a
# debug_log string mangles — the engine EATS the single char immediately before the
# param, inserts the value, and APPENDS a trailing literal $. So "IMP19C $sys$: $msg$"
# rendered "IMP19CSUBJ$: <msg>$" (the space before $sys$ eaten, sys tag fused to
# IMP19C, stray $). se_PURCHASE's "$tradegood$" mangles the same way; it was never
# checked against output. This is unavoidable — the Invictus/TI oracles only ever
# debug_log FULLY-LITERAL strings (they never interpolate). See design/LOGGING_TEMPLATE.md.
#
# MITIGATION (best achievable without a clean-interpolation primitive): put TWO spaces
# before $sys$ so ONE survives the eat — the grep KEY renders clean ("IMP19C QING")
# even though a stray $ trails the tag. The trailing $ is structural and harmless to
# grep on the tag. Numeric VALUES still cannot be rendered at all (use band ladders,
# per LOGGING_TEMPLATE.md). Grep the trace with the tag form, e.g.:  grep "IMP19C QING"
#
# All lines share the "IMP19C " prefix so the whole trace can be extracted:
#     grep "IMP19C " debug.log > imp19c_trace.log
# Per-system lines add a second tag, e.g. "IMP19C SUBJ:", so a single
# subsystem can be isolated:
#     grep "IMP19C SUBJ:" debug.log > subj_trace.log
# Failures carry a FAIL tag:
#     grep "IMP19C FAIL" debug.log > fail_trace.log
#
# Prefix = LOG
########################################################################

# ---- Core line writer -------------------------------------------------
# One tagged message. $sys$ = subsystem tag (SUBJ, TRADE, CURR, ...),
# $msg$ = free text. Pass numbers/vars pre-formatted into $msg$ by the caller.
# [task #83, 2026-08-14] added debug_log_scopes=yes -- this is the SAME idiom LOG_state/
# LOG_fail already use (proven working: a real boot's debug.log shows a genuine
# "Country Derge [420] / Root: Country Great Qing [419]" scope-stack dump following one of
# those calls). The old ""/"" bracket-expression idiom used
# at ~1433 LOG_line call sites across the codebase to inline a scope identifier into the
# MESSAGE STRING never worked (renders literal "ERROR:[ROOT.GetTag]" text -- bracket
# data-function syntax is not evaluated inside a debug_log string). Rather than editing
# every call site, fixing it HERE means every existing LOG_line call automatically gets a
# real, working scope-stack dump for free, with no site-by-site changes needed. The dead
# ""/"ERROR:[...]" text left in messages is now just cosmetic noise -- harmless, but
# safe to strip in a follow-up mechanical pass.
# NOT applied to LOG_enter/LOG_exit below: the dangling-"for" defect this fixes never
# occurred in either (checked directly -- all 1433 sites are LOG_line), and both are the
# highest-frequency call-tracing macros (thousands of calls/tick); adding a multi-line scope
# dump to them would multiply an already-1.7GB-per-boot debug.log for zero benefit.
LOG_line = {
	# Function: write a single tagged diagnostic line to logs/debug.log
	# Scope:    any
	# [#54] TWO spaces before $sys$: the engine eats the one char before a $param$ (proven), so
	# one space survives and the grep KEY renders "IMP19C QING" (not the fused "IMP19CQING"). A
	# stray $ still trails each param — structural, harmless to a tag grep. See LOGGING_TEMPLATE.md.
	debug_log = "IMP19C  $sys$: $msg$"
	debug_log_scopes = yes
}

# ---- Call tracing -----------------------------------------------------
# Mark entry into a code path. $fn$ = effect / event name being entered.
LOG_enter = {
	# Function: trace that a code path began executing
	# Scope:    any
	debug_log = "IMP19C  $sys$: ENTER $fn$"
}

# Mark exit, recording the outcome. $result$ = OK / FAIL / SKIP / <detail>.
LOG_exit = {
	# Function: trace that a code path finished, with its result
	# Scope:    any
	debug_log = "IMP19C  $sys$: EXIT $fn$ result=$result$"
}

# ---- Game-state capture ----------------------------------------------
# Dump the full scope stack (ROOT, this, prev, and all saved/named scopes
# with their values) plus a header, so the log records exactly what state
# the code was operating on. Use this at any point a failure needs context.
LOG_state = {
	# Function: header line + full scope-stack dump for game-state capture
	# Scope:    any
	debug_log = "IMP19C  $sys$: STATE $note$ (scope dump follows)"
	debug_log_scopes = yes
}

# ---- Failure surfacing -----------------------------------------------
# Loud, greppable failure line. $fn$ = where it failed, $reason$ = why.
# Pairs a FAIL tag with a scope dump so a silent no-op becomes a diagnosable
# event. Extract just failures with: grep "IMP19C FAIL" debug.log
LOG_fail = {
	# Function: surface a logical failure that would otherwise be a silent no-op
	# Scope:    any
	debug_log = "IMP19C FAIL $sys$: $fn$ : $reason$"
	debug_log_scopes = yes
}

# ---- Guarded assertion -----------------------------------------------
# Evaluate $check$-style logic at the call site instead (script has no
# generic assert), then call LOG_ok or LOG_fail. This effect just logs a
# passed check for symmetry when tracing a validated precondition.
LOG_ok = {
	# Function: record that a precondition/branch passed as expected
	# Scope:    any
	debug_log = "IMP19C  $sys$: OK $fn$ : $detail$"
}

# ---- GUI-click observability probe -----------------------------------
# [R2-7/R2-10 diagnosis] debug_log emitted from INSIDE a scripted_gui invoked by a
# button's onclick Execute() did NOT appear in the boot debug.log (all 35k IMP19C
# lines came from on_action/event paths, none from GUI clicks). That left it
# unprovable whether the Appoint/Dynastic-Match buttons' Execute actually ran, or
# whether debug_log simply cannot fire from the GUI-Execute context. This probe
# sidesteps the question: the GUI effect STAMPS a country variable (no logging), and
# the CHI quarterly pulse — an on_action path we KNOW logs reliably — READS the stamp
# and logs it. A stamp that advances between boots proves the button's Execute fired.
#
# LOG_gui_probe_stamp: call at the top of a GUI-invoked scripted_gui effect. $probe$ =
# a short kebab id (e.g. council_appoint, marriage_open). Increments gui_probe_$probe$
# on the country scope so the count survives to the next pulse. Scope: country.
LOG_gui_probe_stamp = {
	if = { limit = { NOT = { has_variable = gui_probe_$probe$ } }  set_variable = { name = gui_probe_$probe$  value = 0 } }
	change_variable = { name = gui_probe_$probe$  add = 1 }
}

# LOG_gui_probe_report: call from an on_action (the CHI pulse). Logs the current stamp
# count for $probe$ (0 = the GUI effect has never run since game start → its Execute
# never fired; >0 = it fired that many times). Runs on the reliable on_action log path.
LOG_gui_probe_report = {
	if = {
		limit = { has_variable = gui_probe_$probe$ }
		LOG_line = { sys = PROBE  msg = "GUI probe $probe$ has fired (count in scope var gui_probe_$probe$)" }
	}
	else = {
		LOG_line = { sys = PROBE  msg = "GUI probe $probe$ has NEVER fired since game start (button Execute not reaching the effect)" }
	}
}
