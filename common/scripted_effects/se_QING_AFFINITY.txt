# se_QING_AFFINITY.txt — CHARACTER COMPATIBILITY with the throne.
#
# The Grand Council / great-office / examination events don't treat courtiers as
# interchangeable counters: how a man gets on with the reigning emperor is scored
# from the WHOLE Imperator character model, exactly as the engine itself weighs
# characters (see the Characters wiki page):
#
#   • CHARISMA  — "the ability to charm and persuade others." A charismatic
#     official is easier to deal with and rises smoothly; a graceless one grates.
#   • ZEAL GAP  — zeal is the gift of inspiring faith. A wide gap in fervour
#     between sovereign and servant (a zealot under a worldly throne, or a
#     sceptic under a devout one) breeds friction; matched zeal binds.
#   • CULTURE   — a Manchu-culture courtier shares the conquest elite's world with
#     a Manchu throne; a Han official (or an alien one) stands at a remove. Match
#     draws them together, mismatch apart (the 滿漢 fault line, given teeth).
#   • RELIGION  — likewise a shared faith binds and a divergent one distances.
#   • FRIEND / RIVAL — the engine's own secondary relations: a friend of the
#     emperor is bound to him (+Loyalty in the base game), a rival set against him.
#   • KINSHIP   — a close relative of the imperial house is bound by blood (and is
#     the very stuff of the overmighty-kinsman danger).
#   • AGE & HEALTH — an aged or ailing official is a fading asset; the old grow
#     "increasingly susceptible to terminal illnesses," so their troubles — and
#     their threat — may resolve themselves.
#
# QING_char_affinity  — call inside a courtier's character scope (this = courtier).
#   Scores compatibility with root.current_ruler into qing_char_affinity (0..100,
#   50 = neutral) and logs it. A pure read — no side effects on the character.
# QING_char_bind      — call inside a courtier's scope AFTER QING_char_affinity.
#   Turns the score into consequences: a highly compatible man is bound to the
#   throne (loyalty + a friendship of the sovereign), an incompatible one is
#   alienated (loyalty loss + a rivalry). This is what "makes characters more or
#   less friendly" per the factors above.
#
# CHI-only, player-only suite. All logged to sys = QING.
#   ROOT = the Qing country (CHI, player); root.current_ruler = the emperor.
#
# Prefix = QING_char_
########################################################################

# ======================================================================
# SCORE — read every factor, store 0..100 in qing_char_affinity, log it.
# ======================================================================
QING_char_affinity = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = QING  fn = "QING_char_affinity" }
	set_variable = { name = qing_char_affinity  value = 50 }

	# [fix, boot-test 2026-08-16] guards the SOURCE of the cascade, not just one symptom.
	# A boot's error.log showed BOTH var:qing_aff_zeal_gap (below) AND the OUTPUT var
	# var:qing_char_affinity itself (set at the very first statement above, then read again
	# at the clamp near the end) reported "not being set" in perfect lockstep (5076 hits
	# each, same invocations) -- meaning the whole scope goes invalid partway through this
	# macro's execution, not that one set_variable call failed in isolation (an independent
	# review caught this: guarding only the two qing_aff_zeal_gap reads, as an earlier version
	# of this fix did, cannot prevent the clamp's OWN later unset-var error). The one thing
	# every affected statement in this macro has in common is reading root.current_ruler
	# (zeal/culture/religion/friend/rival/kinship) -- an interregnum (no seated ruler) is a
	# real, if rare, game state where that link fails. Guarding root.current_ruler's EXISTENCE
	# once, up front, and skipping the whole ruler-relative scoring when absent, matches this
	# doc's own stated design ("50 = neutral" is already the pre-set default) and prevents the
	# cascade at its source instead of patching each downstream symptom.
	if = {
			# [fix #11, 2026-08-18] ADD is_alive = yes. The block below reads `this`-attributes
		# (charisma/zeal/culture/religion/age/health/power_base). On a dead-but-EXISTING character
		# scope those stat reads throw ("unset scope" / "invalid left side" / a failed var op) -- the
		# Script-system-error class this task chases. is_alive is a native char TRIGGER: on a dead/
		# stale scope it returns false and the whole stat block is skipped cleanly (no error). The
		# neutral default (qing_char_affinity = 50, set unconditionally above) then stands, and the
		# clamp below reads that var (never a stat), so it stays safe OUTSIDE this guard. Two prior
		# diagnoses (zeal-gap arithmetic; a whole-scope exists guard) were refuted in review; this
		# guards exactly the statements that can throw on a bad current scope. The exists=root.
		# current_ruler check stays (the ruler scope is a different scope from `this`).
		limit = { is_alive = yes  exists = root.current_ruler }
		# --- CHARISMA: +1 per point of the courtier's charm above the average of 5.
		change_variable = { name = qing_char_affinity  add = charisma }
		change_variable = { name = qing_char_affinity  subtract = 5 }

		# --- ZEAL MISMATCH: subtract the ABSOLUTE gap in zeal from throne to servant.
		# [fix, boot-test 2026-08-16] the 3 var:qing_aff_zeal_gap reads below were unguarded --
		# 15228 "Script system error" hits in one boot's error.log (the single largest error class,
		# ~46% of all "Script system error" instances) traced to this exact site via QING_char_
		# promote_standing -> QING_char_affinity. If root.current_ruler.zeal ever fails to resolve
		# (e.g. an interregnum with no seated ruler), change_variable's subtract silently leaves the
		# var in a bad state, and the next read throws "unset scope"/"invalid left side" even though
		# the set two lines above looks unconditional. has_variable-guarded, matching the proven
		# read-before-set idiom used throughout this codebase; degrades safely to "no zeal-gap
		# contribution this call" instead of erroring.
		set_variable = { name = qing_aff_zeal_gap  value = zeal }
		change_variable = { name = qing_aff_zeal_gap  subtract = root.current_ruler.zeal }
		if = {
			limit = { has_variable = qing_aff_zeal_gap  var:qing_aff_zeal_gap < 0 }
			change_variable = { name = qing_aff_zeal_gap  multiply = -1 }
		}
		if = {
			limit = { has_variable = qing_aff_zeal_gap }
			change_variable = { name = qing_char_affinity  subtract = var:qing_aff_zeal_gap }
		}

		# --- CULTURE match / mismatch (the 滿漢 dyarchy at the personal level)
		if = {
			limit = { culture = root.current_ruler.culture }
			change_variable = { name = qing_char_affinity  add = 10 }
		}
		else = {
			change_variable = { name = qing_char_affinity  subtract = 8 }
		}

		# --- RELIGION match / mismatch
		# [logfix] bare `religion = X` is a COUNTRY trigger — invalid on this CHARACTER scope
		# (error.log: "religion trigger [ Wrong scope: character, expected country ]", 210x).
		# Unlike `culture` (a valid char trigger, used above), faith must be read as a SCOPE
		# LINK on both sides: `this.religion = X.religion` (proven, party_types 00_default_republic.txt:154).
		if = {
			limit = { this.religion = root.current_ruler.religion }
			change_variable = { name = qing_char_affinity  add = 8 }
		}
		else = {
			change_variable = { name = qing_char_affinity  subtract = 8 }
		}

		# --- FRIENDSHIP / RIVALRY with the sovereign (engine secondary relations)
		if = {
			limit = { is_friend = root.current_ruler }
			change_variable = { name = qing_char_affinity  add = 15 }
		}
		if = {
			limit = { is_rival = root.current_ruler }
			change_variable = { name = qing_char_affinity  subtract = 20 }
		}

		# --- KINSHIP & FAMILY PRESTIGE: a close relative of the imperial house is
		# bound by blood, and a member of the reigning family more so — the great
		# families' standing (family prestige) is itself a courtier's stake in the
		# dynasty. A scion of a prestigious house is invested in the throne.
		if = {
			limit = { is_close_relative = root.current_ruler }
			change_variable = { name = qing_char_affinity  add = 8 }
		}
		if = {
			limit = { from_ruler_family = yes }
			change_variable = { name = qing_char_affinity  add = 8 }
		}
		# family prestige, banded (family.prestige is unbounded, so we don't add it raw)
		if = {
			limit = { family = { prestige >= 500 } }
			change_variable = { name = qing_char_affinity  add = 6 }
		}
		else_if = {
			limit = { family = { prestige >= 100 } }
			change_variable = { name = qing_char_affinity  add = 3 }
		}

		# --- STATESMANSHIP: the governing skills (finesse the administrator's gift,
		# martial the general's). A capable man serves the throne well and knows it —
		# a modest confidence bonus above the average pair (10 = 5+5).
		set_variable = { name = qing_aff_statecraft  value = finesse }
		change_variable = { name = qing_aff_statecraft  add = martial }
		change_variable = { name = qing_aff_statecraft  subtract = 10 }
		change_variable = { name = qing_aff_statecraft  divide = 2 }
		change_variable = { name = qing_char_affinity  add = var:qing_aff_statecraft }

		# --- STANDING: popularity (how the people see him) and prominence (his fame)
		# both raise a man's sense of his own worth to the state. Banded, not raw.
		if = { limit = { popularity >= 40 }   change_variable = { name = qing_char_affinity  add = 5 } }
		else_if = { limit = { popularity <= -20 } change_variable = { name = qing_char_affinity  subtract = 5 } }
		if = { limit = { prominence >= 60 }   change_variable = { name = qing_char_affinity  add = 4 } }

		# --- EXISTING LOYALTY feeds back: a man already loyal is congenial, a
		# disloyal one estranged (loyalty is a 0..100 state-loyalty read).
		if = { limit = { loyalty >= 65 }  change_variable = { name = qing_char_affinity  add = 6 } }
		else_if = { limit = { loyalty < 40 }  change_variable = { name = qing_char_affinity  subtract = 8 } }

		# --- VENALITY: a corrupt official is a self-server; an honest / just one a
		# pillar. Corruption also drags in step with the realm's corruption level.
		if = { limit = { has_trait = corrupt } change_variable = { name = qing_char_affinity  subtract = 8 } }
		if = { limit = { OR = { has_trait = honest  has_trait = just } } change_variable = { name = qing_char_affinity  add = 5 } }
		# [continuation fix 2026-08-19, user-report] DISGRACE: a publicly disgraced man's standing
		# with the throne is itself part of his affinity, not just whatever residual corrupt trait
		# he happens to still carry (QING_char_cleanse strips that trait as part of the SAME
		# punishment, so without this the disgrace itself contributed nothing here). Matches the
		# two disgrace traits' own severity ordering (QING_char_apply_hard_disgrace escalates PAST
		# the lesser taint, se_QING_AFFINITY.txt:438-471) -- completely_disgraced is the heavier hit.
		if = { limit = { has_trait = completely_disgraced }  change_variable = { name = qing_char_affinity  subtract = 15 } }
		else_if = { limit = { has_trait = disgraced }  change_variable = { name = qing_char_affinity  subtract = 8 } }
		# corruption is ALSO a numeric character stat (0..100), distinct from the
		# corrupt trait: the higher a man's own venality, the more he serves himself.
		if = { limit = { corruption >= 40 } change_variable = { name = qing_char_affinity  subtract = 8 } }
		else_if = { limit = { corruption >= 15 } change_variable = { name = qing_char_affinity  subtract = 4 } }
		# a corrupt court makes even honest men warier of the throne — the realm's
		# corruption level (the decline counter) drags every courtier's affinity.
		if = {
			limit = { ROOT = { has_variable = qing_corruption_level  var:qing_corruption_level >= 60 } }
			change_variable = { name = qing_char_affinity  subtract = 5 }
		}

		# --- PERSONAL WEALTH & POWER BASE: the Heshen danger. A courtier rich in his
		# own right and thick with loyal retainers (power_base — holdings + veterans,
		# which the engine already docks 0.55 loyalty per unit) is powerful but LESS
		# dependent on the throne's favour, and so a warier, more dangerous servant.
		if = { limit = { wealth >= 300 }   change_variable = { name = qing_char_affinity  subtract = 5 } }
		if = { limit = { power_base >= 15 } change_variable = { name = qing_char_affinity  subtract = 6 } }
		else_if = { limit = { power_base >= 8 } change_variable = { name = qing_char_affinity  subtract = 3 } }

		# --- AGE & HEALTH: an ailing or aged official is a fading (less reliable) asset
		if = {
			limit = { health < 0.35 }
			change_variable = { name = qing_char_affinity  subtract = 10 }
		}
		if = {
			limit = { age >= 70 }
			change_variable = { name = qing_char_affinity  subtract = 6 }
		}
	}

	# --- clamp 0..100
	# [error.log fix 2026-08-19] guard the clamp reads with has_variable, matching this same file's
	# qing_aff_zeal_gap idiom above. Despite the unconditional set_variable at the top of this macro
	# and the existing is_alive/exists=root.current_ruler guard (fix #11), a boot's error.log still
	# showed 1160x "Failed to fetch variable for 'qing_char_affinity' due to not being set" at exactly
	# these two clamp lines, via the newly-restored Censorate impeach path (QING_censorate_impeach_
	# uphold -> QING_char_cleanse -> QING_char_affinity). Root cause (why the scope/var still goes
	# unset here) is unresolved after two prior fix passes on this macro; this stops the flood safely
	# with the same degrade-cleanly idiom used throughout this codebase, without masking a real stat.
	# [read-before-set fix, task #28 2026-08-26] nested instead of a flat AND-limit per pair: a flat
	# `has_variable = X  var:X > N` still evaluates the second leaf even when has_variable already
	# fails — the same class fixed for QING_office_appoint in task #16 (ccb2dca00). One shared
	# has_variable guard, two inner compares; same net effect, spurious log removed.
	if = {
		limit = { has_variable = qing_char_affinity }
		if = { limit = { var:qing_char_affinity > 100 } set_variable = { name = qing_char_affinity  value = 100 } }
		if = { limit = { var:qing_char_affinity < 0 }   set_variable = { name = qing_char_affinity  value = 0 } }
	}

	LOG_line = { sys = QING  msg = "char affinity: w/ throne = (zeal gap)" }
	LOG_exit = { sys = QING  fn = "QING_char_affinity"  result = OK }
}

# ======================================================================
# APPLY — turn the score into loyalty + a friendship or rivalry.
# High affinity (>=70): the man is bound to the throne — loyalty and, at the top,
# a personal friendship of the sovereign. Low affinity (<=30): he is alienated —
# loyalty falls and, at the bottom, he becomes the emperor's rival. The mid band
# leaves the relations untouched. Auto-scores first if not yet scored.
# ======================================================================
QING_char_bind = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = QING  fn = "QING_char_bind" }
	if = {
		limit = { NOT = { has_variable = qing_char_affinity } }
		QING_char_affinity = yes
	}
	if = {
		limit = { var:qing_char_affinity >= 70 }
		# [loyalty-calibration 2026-08-04] APPLY ONCE via the shared helper (idempotent + sheds the
		# opposite band). QING_char_bind is called from many re-bind pulses; a bare add_loyalty
		# re-stamped every call.
		QING_apply_congenial_once = yes
		# a genuinely congenial servant becomes a friend of the throne
		if = {
			limit = {
				var:qing_char_affinity >= 85
				NOT = { is_friend = root.current_ruler }
				NOT = { is_rival = root.current_ruler }
			}
			add_friend = root.current_ruler
			LOG_line = { sys = QING  msg = "char bind: befriended the throne (affinity high)" }
		}
	}
	else_if = {
		limit = { var:qing_char_affinity <= 30 }
		# [loyalty-calibration 2026-08-04] APPLY ONCE via the shared helper. The old bare add_loyalty
		# here, re-stamped every pulse, is what pinned low-affinity chars (Guanting via this path;
		# Songchun/Fulu also via QING_war_review_commanders, now likewise guarded) toward the floor.
		QING_apply_estranged_once = yes
		# a deeply incompatible servant turns rival
		if = {
			limit = {
				var:qing_char_affinity <= 15
				NOT = { is_rival = root.current_ruler }
				NOT = { is_friend = root.current_ruler }
			}
			add_rival = root.current_ruler
			LOG_line = { sys = QING  msg = "char bind: set against the throne (affinity low)" }
		}
	}
	LOG_line = { sys = QING  msg = "char bind applied: affinity=" }
	LOG_exit = { sys = QING  fn = "QING_char_bind"  result = OK }
}

# ======================================================================
# APPLY-ONCE loyalty helpers (this = a character). [loyalty-calibration 2026-08-04]
# loyalty_qing_estranged / _congenial are applied from SEVERAL quarterly pulses
# (QING_char_bind here + QING_war_review_commanders in se_QING_WAR). A bare
# add_loyalty re-stamps every pulse, piling toward the floor/ceiling. These helpers
# make the application idempotent (add only if not already carried) and shed the
# OPPOSITE band on a cross, so a character carries at most one of the pair and it
# lands exactly once. has_loyalty / remove_loyalty are proven (on_office_gained.txt,
# 00_ambitions.txt). Call these instead of a bare add_loyalty at any recurring site.
# ======================================================================
QING_apply_estranged_once = {
	# Scope: character
	if = {
		limit = { has_loyalty = loyalty_qing_congenial }
		remove_loyalty = loyalty_qing_congenial
	}
	if = {
		limit = { NOT = { has_loyalty = loyalty_qing_estranged } }
		add_loyalty = loyalty_qing_estranged
	}
}
QING_apply_congenial_once = {
	# Scope: character
	if = {
		limit = { has_loyalty = loyalty_qing_estranged }
		remove_loyalty = loyalty_qing_estranged
	}
	if = {
		limit = { NOT = { has_loyalty = loyalty_qing_congenial } }
		add_loyalty = loyalty_qing_congenial
	}
}
# [loyalty-calibration 2026-08-04] Army-collapse pressure (MO-ENH #3) is a DISTINCT modifier that
# coexists WITH the congenial/estranged pair (it must not be shed by the well-managed branch — see
# loyalty def note). Apply-once; the paired remove restores it when the central armies recover.
QING_apply_army_collapse_once = {
	# Scope: character
	if = {
		limit = { NOT = { has_loyalty = loyalty_qing_army_collapse } }
		add_loyalty = loyalty_qing_army_collapse
	}
}
QING_remove_army_collapse = {
	# Scope: character
	if = {
		limit = { has_loyalty = loyalty_qing_army_collapse }
		remove_loyalty = loyalty_qing_army_collapse
	}
}

# ======================================================================
# MUTATORS — events don't only READ these factors, they can CHANGE them. A man
# can be sinicized, converted, cultivated into a more charismatic statesman, or
# have his fervour kindled or cooled. Each is call-site-agnostic (this = the
# character), logs the change, and re-scores affinity so the new state takes
# effect immediately. This is the loop the user asked for: the events are shaped
# by these factors AND reshape them.
# ======================================================================

# --- Assimilate a courtier to the imperial culture (漢化 / 滿化, per the throne's
# own culture): the sinicization of a Manchu, or a Han official taken into the
# banner establishment. Binds him closer to the conquest elite.
QING_char_assimilate = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = QING  fn = "QING_char_assimilate" }
	set_culture = root.current_ruler.culture
	LOG_line = { sys = QING  msg = "char mutate: assimilated to culture" }
	QING_char_affinity = yes
	LOG_exit = { sys = QING  fn = "QING_char_assimilate"  result = OK }
}

# --- Convert a courtier to the imperial faith.
QING_char_convert = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = QING  fn = "QING_char_convert" }
	set_character_religion = root.current_ruler.religion
	LOG_line = { sys = QING  msg = "char mutate: converted to" }
	QING_char_affinity = yes
	LOG_exit = { sys = QING  fn = "QING_char_convert"  result = OK }
}

# --- Cultivate a courtier's ability. $stat$ = charisma|zeal|martial|finesse,
# $amount$ = signed points (accepts negatives — a disgrace or a wasting illness
# can DIMINISH a man). Used by the mentorship / office-seasoning / scandal events.
QING_char_cultivate = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = QING  fn = "QING_char_cultivate" }
	add_$stat$ = $amount$
	LOG_line = { sys = QING  msg = "char mutate: $stat$ += $amount$" }
	QING_char_affinity = yes
	LOG_exit = { sys = QING  fn = "QING_char_cultivate"  result = OK }
}

# --- Enrich or fine a courtier (personal wealth). $amount$ signed. A grant of
# silver buys goodwill; a confiscation (negative) strips his private fortune —
# the fiscal lever on the power-base danger.
QING_char_endow = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = QING  fn = "QING_char_endow" }
	add_gold = $amount$
	LOG_line = { sys = QING  msg = "char mutate: wealth += $amount$" }
	QING_char_affinity = yes
	LOG_exit = { sys = QING  fn = "QING_char_endow"  result = OK }
}

# --- Advance or diminish a courtier's public standing. $pop$ / $prom$ signed
# (popularity / prominence). Raising a man to the public eye or casting him down.
QING_char_promote_standing = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = QING  fn = "QING_char_promote_standing" }
	add_popularity = $pop$
	add_prominence = $prom$
	LOG_line = { sys = QING  msg = "char mutate: popularity += $pop$, prominence += $prom$" }
	QING_char_affinity = yes
	LOG_exit = { sys = QING  fn = "QING_char_promote_standing"  result = OK }
}

# --- Elevate the whole house — a grant of family prestige (爵位/世職 honours).
# $amount$ signed; applied to the character's family scope.
QING_char_honour_family = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = QING  fn = "QING_char_honour_family" }
	if = {
		limit = { is_minor_character = no }
		family = { add_prestige = $amount$ }
		LOG_line = { sys = QING  msg = "char mutate: family prestige += $amount$" }
	}
	QING_char_affinity = yes
	LOG_exit = { sys = QING  fn = "QING_char_honour_family"  result = OK }
}

# --- Brand a courtier corrupt: adds the corrupt TRAIT (guarded) and drives up
# the numeric corruption STAT — the sale-of-offices rot at the personal level.
QING_char_taint = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = QING  fn = "QING_char_taint" }
	if = {
		limit = { NOT = { has_trait = corrupt } }
		add_trait = corrupt
	}
	add_corruption = 20
	LOG_line = { sys = QING  msg = "char mutate: tainted (corrupt +trait, corruption +20)" }
	QING_char_affinity = yes
	LOG_exit = { sys = QING  fn = "QING_char_taint"  result = OK }
}

# --- Rehabilitate a reformed official: clears the corrupt trait (guarded) and
# cuts the numeric corruption stat back.
QING_char_cleanse = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = QING  fn = "QING_char_cleanse" }
	if = {
		limit = { has_trait = corrupt }
		remove_trait = corrupt
	}
	add_corruption = -20
	LOG_line = { sys = QING  msg = "char mutate: cleansed (corrupt -trait, corruption -20)" }
	QING_char_affinity = yes
	LOG_exit = { sys = QING  fn = "QING_char_cleanse"  result = OK }
}

# ======================================================================
# [#84 2026-08-08] DISGRACE-AS-TRAIT wrappers. The disgrace status is now a card TRAIT (disgraced /
# completely_disgraced) instead of a character modifier. add_trait does NOT auto-strip its opposites
# (proven by the exam-degree ladder, which strips manually), so these wrappers keep the two disgrace
# traits mutually exclusive and centralise the (formerly per-site) add/lift logic. Scope: CHARACTER.
# ----------------------------------------------------------------------
# LESSER TAINT (輕貶) — a botched-but-survivable failure. Deprioritizes (svalue -100) but does not bar.
# Auto-expires after 5 years. The engine has no trait timer, so the authoritative clock is a TIMED
# VARIABLE (qing_taint_expiry, 1825d, auto-lapsing) that RE-TAINTING refreshes; a hidden qing_censorate.10
# removes the trait only once that var has actually lapsed (re-arming itself while it is still set). This
# makes the LAST taint's clock govern — a bare per-add scheduled removal would let an earlier copy strip
# the trait prematurely on re-taint.
QING_char_apply_taint = {
	if = { limit = { has_trait = completely_disgraced }  remove_trait = completely_disgraced }
	if = { limit = { NOT = { has_trait = disgraced } }  add_trait = disgraced }
	# 5-year auto-expiry (1825 days). The engine has no trait timer, so a hidden qing_censorate.10 is
	# scheduled to remove the trait — BUT a bare +1825d event is wrong under RE-TAINTING: a man tainted
	# again before the first clock runs out would have TWO .10s pending, and the earlier one would strip
	# the trait ~prematurely. So the authoritative clock is a TIMED VARIABLE (qing_taint_expiry, 1825d,
	# auto-lapsing): re-tainting REFRESHES it, and .10 only removes the trait once the var has actually
	# lapsed (NOT has_variable). A stale early .10 then finds the var still set and no-ops. We still fire
	# the .10 at +1825 as the CHECK trigger; on refresh the newer .10 (or a later pulse) does the removal.
	set_variable = { name = qing_taint_expiry  value = 1  days = 1825 }
	trigger_event = { id = qing_censorate.10  days = 1826 }
	LOG_line = { sys = QING  msg = "char mutate: lesser taint applied (disgraced trait, 5y expiry)" }
}
# HARD DISGRACE (身敗名裂) — corruption sacking / upheld impeachment. The universal eligibility bar.
# PERMANENT until the Censorate's Rehabilitate lifts it (no auto-expiry — deliberate design change
# from the old 10y-timed modifier; clearing it is an act of imperial grace, not mere lapse).
QING_char_apply_hard_disgrace = {
	if = { limit = { has_trait = disgraced }  remove_trait = disgraced }
	# escalating from taint to hard disgrace: drop the lesser-taint clock (its trait is gone now).
	if = { limit = { has_variable = qing_taint_expiry }  remove_variable = qing_taint_expiry }
	if = { limit = { NOT = { has_trait = completely_disgraced } }  add_trait = completely_disgraced }
	LOG_line = { sys = QING  msg = "char mutate: HARD disgrace applied (completely_disgraced trait)" }
}
# LIFT — clear whichever disgrace trait the man carries (the Rehabilitate action + the taint expiry).
QING_char_lift_disgrace = {
	if = { limit = { has_trait = completely_disgraced }  remove_trait = completely_disgraced }
	if = { limit = { has_trait = disgraced }  remove_trait = disgraced }
	# clear the lesser-taint clock too, so a lifted man leaves no dangling expiry var (a pending
	# qing_censorate.10 then simply finds no trait to remove — a clean no-op).
	if = { limit = { has_variable = qing_taint_expiry }  remove_variable = qing_taint_expiry }
	LOG_line = { sys = QING  msg = "char mutate: disgrace lifted (eligibility restored)" }
}

# --- Move the numeric corruption stat alone (no trait change). $amount$ signed.
# For gradual venality creep or an audit's partial clean-up.
QING_char_corruption = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = QING  fn = "QING_char_corruption" }
	add_corruption = $amount$
	LOG_line = { sys = QING  msg = "char mutate: corruption += $amount$" }
	QING_char_affinity = yes
	LOG_exit = { sys = QING  fn = "QING_char_corruption"  result = OK }
}

# ======================================================================
# PAIRWISE FRICTION — TRANSIENT, never stored per-pair.
# Two councillors' compatibility with EACH OTHER, computed on the fly at event
# time from the same character factors and written to ONE country-scope scratch
# variable (qing_pair_friction) that the calling event reads immediately and lets
# fall out of use. We deliberately do NOT persist an affinity between every pair
# of characters (that would be O(n^2) saved variables — the memory blow-up the
# user warned against); the council-infighting events resolve the clash in the
# moment and store nothing.
#
# Call at COUNTRY scope with two saved character scopes:
#   QING_pair_friction = { a = scope:feud_a  b = scope:feud_b }
# Result: var:qing_pair_friction (0..100; HIGH = they clash badly).
# ======================================================================
QING_pair_friction = {
	# [#98-fix task #98] se_LOG enter/exit wiring (deep-audit)
	LOG_enter = { sys = QING  fn = "QING_pair_friction" }
	set_variable = { name = qing_pair_friction  value = 20 }

	# culture divide (the 滿漢 fault line between two ministers)
	if = {
		limit = { NOT = { $a$ = { culture = $b$.culture } } }
		change_variable = { name = qing_pair_friction  add = 25 }
	}
	# faith divide
	# [logfix] inside $a$={} the scope is a CHARACTER; bare `religion =` is a COUNTRY trigger
	# (Wrong scope: character). Navigate the link on the left too: this.religion = $b$.religion
	# (proven, party_types 00_default_republic.txt:154). culture above is a valid char trigger.
	if = {
		limit = { NOT = { $a$ = { this.religion = $b$.religion } } }
		change_variable = { name = qing_pair_friction  add = 15 }
	}
	# a zeal gulf between them (fervent vs worldly)
	set_variable = { name = qing_pair_zeal_gap  value = $a$.zeal }
	change_variable = { name = qing_pair_zeal_gap  subtract = $b$.zeal }
	if = { limit = { var:qing_pair_zeal_gap < 0 }  change_variable = { name = qing_pair_zeal_gap  multiply = -1 } }
	change_variable = { name = qing_pair_friction  add = var:qing_pair_zeal_gap }
	# two proud, prominent men rub worse (rivalry of the eminent)
	if = {
		limit = { $a$ = { prominence >= 50 }  $b$ = { prominence >= 50 } }
		change_variable = { name = qing_pair_friction  add = 10 }
	}
	# venality breeds mutual distrust — the corrupt TRAIT...
	if = {
		limit = { OR = { $a$ = { has_trait = corrupt }  $b$ = { has_trait = corrupt } } }
		change_variable = { name = qing_pair_friction  add = 8 }
	}
	# ...and the numeric CORRUPTION stat too (a grasping minister sows distrust
	# among colleagues even before the trait crystallises; user 2026-07-07).
	if = {
		limit = { OR = { $a$ = { corruption >= 40 }  $b$ = { corruption >= 40 } } }
		change_variable = { name = qing_pair_friction  add = 8 }
	}
	else_if = {
		limit = { OR = { $a$ = { corruption >= 20 }  $b$ = { corruption >= 20 } } }
		change_variable = { name = qing_pair_friction  add = 4 }
	}
	# FRIENDS / RIVALS of each other (engine secondary relations): a friendship
	# smooths the council table; a standing rivalry poisons it (user 2026-07-07).
	if = {
		limit = { $a$ = { is_friend = $b$ } }
		change_variable = { name = qing_pair_friction  subtract = 15 }
	}
	if = {
		limit = { $a$ = { is_rival = $b$ } }
		change_variable = { name = qing_pair_friction  add = 20 }
	}
	# HIGHBORN vs LOWBORN: a great-house grandee and a low-born colleague grate
	# across the gulf of birth; two proud houses rival one another for precedence
	# (family prestige, banded — mirrors QING_char_affinity's prestige bands).
	if = {
		limit = {
			OR = {
				AND = { $a$ = { family = { prestige >= 300 } }  $b$ = { NOT = { family = { prestige >= 100 } } } }
				AND = { $b$ = { family = { prestige >= 300 } }  $a$ = { NOT = { family = { prestige >= 100 } } } }
			}
		}
		change_variable = { name = qing_pair_friction  add = 10 }
	}
	else_if = {
		limit = { $a$ = { family = { prestige >= 500 } }  $b$ = { family = { prestige >= 500 } } }
		change_variable = { name = qing_pair_friction  add = 8 }
	}
	# GENERATIONAL GULF: an aged grandee and a young upstart clash over precedence
	# and pace of change (age, banded-contrast — age>=N is the proven trigger form).
	if = {
		limit = {
			OR = {
				AND = { $a$ = { age >= 60 }  $b$ = { age <= 35 } }
				AND = { $b$ = { age >= 60 }  $a$ = { age <= 35 } }
			}
		}
		change_variable = { name = qing_pair_friction  add = 8 }
	}
	# WEALTH DISPARITY: a rich minister and a poor one distrust each other's means
	# and motives (wealth, banded-contrast).
	if = {
		limit = {
			OR = {
				AND = { $a$ = { wealth >= 200 }  $b$ = { wealth < 50 } }
				AND = { $b$ = { wealth >= 200 }  $a$ = { wealth < 50 } }
			}
		}
		change_variable = { name = qing_pair_friction  add = 6 }
	}
	# RELATIVE POPULARITY: when one minister markedly outshines the other in the
	# public eye, the eclipsed colleague resents it (popularity, banded-contrast).
	if = {
		limit = {
			OR = {
				AND = { $a$ = { popularity >= 40 }  $b$ = { popularity <= 0 } }
				AND = { $b$ = { popularity >= 40 }  $a$ = { popularity <= 0 } }
			}
		}
		change_variable = { name = qing_pair_friction  add = 6 }
	}
	# HEALTH: one hale and one failing — the ailing man clings to his post, the
	# vigorous one presses his advantage (health, banded-contrast).
	if = {
		limit = {
			OR = {
				AND = { $a$ = { health < 0.35 }  $b$ = { health >= 0.5 } }
				AND = { $b$ = { health < 0.35 }  $a$ = { health >= 0.5 } }
			}
		}
		change_variable = { name = qing_pair_friction  add = 6 }
	}
	# but kinship softens it
	if = {
		limit = { $a$ = { is_close_relative = $b$ } }
		change_variable = { name = qing_pair_friction  subtract = 15 }
	}

	if = { limit = { var:qing_pair_friction > 100 } set_variable = { name = qing_pair_friction  value = 100 } }
	if = { limit = { var:qing_pair_friction < 0 }   set_variable = { name = qing_pair_friction  value = 0 } }

	LOG_line = { sys = QING  msg = "pair friction: vs =" }
	LOG_exit = { sys = QING  fn = "QING_pair_friction"  result = OK }
}
