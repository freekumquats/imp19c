# se_QING_CANTON.txt — THE CANTON SYSTEM (粵海關 / 一口通商), task #389.
#
# From 1757 to 1842 all Western maritime trade was funnelled through Guangzhou
# (廣州) under the 一口通商 single-port decree — in full force at the 1763 start.
# This replaces the display-only open/close regime stub (#386) with the real
# mechanic, whose HISTORICAL HOOK is the money's routing:
#
#   THE CANTON CUSTOMS WERE THE EMPEROR'S MONEY, NOT THE STATE'S. The 粵海關監督
#   (Hoppo, Superintendent of Maritime Customs) was NOT a civil-service post — he
#   was an Imperial Household Department (內務府) appointee, a Manchu bondservant
#   (包衣 booi) personally chosen by the throne for a 3-year term. The Canton customs
#   SURPLUS flowed as "tribute" straight to the emperor's privy purse (內帑) via the
#   Neiwufu, BYPASSING the regular Board of Revenue (戶部) state treasury. Only a
#   fixed quota (~1 million taels/yr) went to Beijing as ordinary revenue; the
#   surplus — and the Hoppo's vast personal squeeze — was off the fiscal state's books.
#
# So this module splits the Canton yield each quarter:
#   • the bulk accrues to the EMPEROR (qing_privy_purse 內帑 + qing_canton_customs, the
#     running privy haul) — the money the 戶部 fisc cannot touch;
#   • a smaller quota reaches the STATE treasury (add_treasury);
#   • the Hoppo's squeeze (qing_hoppo_squeeze) climbs as he "milks the trade" (Morse),
#     feeding corruption — the post was the most lucrative in the empire and its holder
#     paid bribes to win it, keep it, and leave it.
# The Cohong (公行) merchant monopoly carries the merchant-debt / bankruptcy risk that a
# high squeeze eventually triggers (qing_canton.1).
#
# Yield scales with the regime being OPEN + Guangzhou's port development (the concrete
# on-map trade capacity, port_building on province 9298) + the Hoppo's competence, so a
# richer, better-run port pays the throne more. All Canton figures here are in 萬兩 (10k-tael units).
# [2026-08-06 CORRECTION] This once read "same scale as the #372 戶部銀庫 reserve" — that is STALE and
# WRONG since #425 retired the old qing_silver_reserve (萬兩) counter and moved the 戶部 mechanic onto the
# engine reserve silver_reserve_size, which is counted in 千兩 (1000-tael units). So Canton 萬兩 and the
# reserve 千兩 differ by 10× (1 萬兩 = 10 千兩). Any code crediting a Canton 萬兩 figure into
# silver_reserve_size MUST multiply by 10 first (see design/DESIGN_CANTON_SILVER_RESERVE.md §5).
#
# Sources: Van Dyke, The Canton Trade (2005); Chang Te-Ch'ang, "The Economic Role of the
# Imperial Household in the Ch'ing Dynasty" (JAS 1972); Torbert, The Ch'ing Imperial
# Household Department; H.B. Morse on the Hoppo.
#
# State (all on CHI):
#   qing_canton_regime      flag:open|closed  (seeded open at 1763, se_MECHANICS)
#   qing_canton_customs     running privy-purse haul from Canton, 萬兩 (display)
#   qing_canton_last_yield  the last quarter's total customs yield, 萬兩 (display)
#   qing_hoppo_squeeze      0..100 the Hoppo's corruption / over-exaction
#   qing_hoppo_holder       the character serving as Hoppo (a Neiwufu appointee), if any
#   qing_canton_cohong_cooldown  throttle on the Cohong-debt crisis event
#
# CHI-only, player-only. Init at game start; pulse quarterly from QING_GOV_pulse.
# Prefix = QING_canton_ / sys = QING
########################################################################

# ======================================================================
# INIT — seed the Canton bookkeeping. Called once at game start. Scope: country (CHI).
# ======================================================================
QING_canton_init = {
	LOG_enter = { sys = QING  fn = "QING_canton_init" }
	if = { limit = { NOT = { has_variable = qing_canton_customs } }     set_variable = { name = qing_canton_customs  value = 0 } }
	if = { limit = { NOT = { has_variable = qing_canton_last_yield } }  set_variable = { name = qing_canton_last_yield  value = 0 } }
	if = { limit = { NOT = { has_variable = qing_canton_last_state } }  set_variable = { name = qing_canton_last_state  value = 0 } }   # [#426] state share of the yield (Revenue ledger)
	if = { limit = { NOT = { has_variable = qing_canton_last_silver_in } }  set_variable = { name = qing_canton_last_silver_in  value = 0 } }   # [2026-08-06] Canton→reserve specie inflow last qtr, 千兩 (panel-safe first read)
	# the Hoppo's post ran on squeeze even at the zenith — seed a moderate baseline.
	if = { limit = { NOT = { has_variable = qing_hoppo_squeeze } }      set_variable = { name = qing_hoppo_squeeze  value = 30 } }
	# [#111a] seed the lagged FINESSE mirror to a NEUTRAL 7 (= no finesse lift) so the continuous
	# Hoppo-effectiveness factor still applies its squeeze shave on the very first pulse — before the
	# reconcile/graft block first mirrors the seated Hoppo's real finesse. Re-derived every pulse thereafter.
	if = { limit = { NOT = { has_variable = qing_hoppo_finesse } }      set_variable = { name = qing_hoppo_finesse  value = 7 } }
	# [#426] the Canton revenue split (percent of the yield to the emperor's privy purse 內帑 vs the
	# state treasury 戶部). Player-set via the Canton window toggle: 0 / 50 / 100. Default 50.
	if = { limit = { NOT = { has_variable = qing_canton_purse_share } } set_variable = { name = qing_canton_purse_share  value = 50 } }
	# [#66 2026-08-08] SEAT A HOPPO AT GAME START. The 粵海關監督 was a real, always-filled post; the
	# prior model only created him when the player clicked Rotate, so a fresh game had a squeeze meter
	# with no man behind it (the reported "I cannot find the Hoppo"). Appoint one now if none is seated,
	# so the Canton window can show him, his corruption can drive the siphon (#69), and he can be
	# impeached (#68) from turn one. Guarded so a re-run of init never double-seats.
	if = {
		limit = { NOT = { has_variable = qing_hoppo_holder } }
		QING_canton_appoint_hoppo = yes
	}
	LOG_line = { sys = QING  msg = "canton: bookkeeping seeded for" }
	LOG_exit = { sys = QING  fn = "QING_canton_init"  result = OK }
}

# ======================================================================
# QUARTERLY PULSE — the Canton customs collection. Called from QING_GOV_pulse. When the
# regime is OPEN, compute the quarter's yield and split it emperor / state; drift the
# Hoppo's squeeze; on a high squeeze, roll the Cohong-debt crisis. A CLOSED regime yields
# nothing (the whole point of 閉關 is that the silver stops). Scope: country (CHI).
# ======================================================================
QING_canton_pulse = {
	if = {
		limit = {
			has_variable = qing_canton_regime
			var:qing_canton_regime = flag:open
			# [#389 review-fix] the customs only flow while the Qing actually HOLDS Guangzhou (9298);
			# a regime flag left open after the port is lost yields nothing.
			exists = p:9298
			p:9298 = { owner = ROOT }
		}
		LOG_enter = { sys = QING  fn = "QING_canton_pulse" }
		# --- [#436] base quarterly yield DERIVED FROM REAL EXPORT GOODS, not a flat 30萬兩 constant.
		# The 粵海關 customs were an ad-valorem levy on the tea / silk / porcelain (茶/絲/瓷) the empire
		# shipped out through Canton — the very goods the Western traders came for. Read the empire's
		# real production of those three via the proven country-scope aggregates (GOODS_national_production_*,
		# the same family #423's opium model reads: GOODS_svalues.txt:195). A larger export economy = a
		# richer customs house; a shrunken one (goods provinces lost or converted away) = a thinner take.
		# Scaled /8 to the model's 萬兩 figure (production numbers are large) and floored at 8 so the
		# customs house always yields a baseline while any export trade remains — the scale constant is a
		# conversion factor, NOT a magic tael figure. At the 1763 zenith (China the world's workshop) this
		# lands near the historical ~30萬兩/quarter, but now MOVES with the real export base.
		set_variable = { name = qing_canton_yield_tmp  value = 0 }
		change_variable = { name = qing_canton_yield_tmp  add = GOODS_national_production_tea }
		change_variable = { name = qing_canton_yield_tmp  add = GOODS_national_production_silk }
		change_variable = { name = qing_canton_yield_tmp  add = GOODS_national_production_porcelain }
		change_variable = { name = qing_canton_yield_tmp  divide = 8 }
		if = { limit = { var:qing_canton_yield_tmp < 8 }  set_variable = { name = qing_canton_yield_tmp  value = 8 } }
		# a runaway export boom must not balloon the customs beyond the port's throughput — cap the raw
		# goods-basis at 45萬兩/quarter (before the port/squeeze multipliers below), the ceiling of what
		# the single Canton entrepôt could clear under the 一口通商 single-port system.
		if = { limit = { var:qing_canton_yield_tmp > 45 }  set_variable = { name = qing_canton_yield_tmp  value = 45 } }
		# scale by Guangzhou's (province 9298) port development — a bigger port, more trade.
		# port_building levels run 0..3 at 1763; each level above 1 adds a fifth again.
		if = {
			limit = { exists = p:9298  p:9298 = { num_of_port_building >= 3 } }
			change_variable = { name = qing_canton_yield_tmp  multiply = 1.4 }
		}
		else_if = {
			limit = { exists = p:9298  p:9298 = { num_of_port_building >= 2 } }
			change_variable = { name = qing_canton_yield_tmp  multiply = 1.2 }
		}
		# a capable, honest Hoppo delivers more of the trade's worth to the throne; a venal one
		# lets his squeeze swallow it (high squeeze shaves the yield that actually reaches Peking).
		# [#69 note] this reads LAST quarter's qing_hoppo_squeeze — the squeeze is re-derived from the
		# Hoppo's corruption LATER in this pulse (deliberate one-quarter lag; corruption creeps only
		# ±1/qtr so the steady-state error is negligible; a freshly-seated/impeached Hoppo simply
		# over/under-shaves for one quarter, self-correcting next pulse — not worth reordering).
		# [#111a] CONTINUOUS HOPPO-EFFECTIVENESS FACTOR (replaces the old squeeze>=60 -> x0.7 CLIFF).
		# The factor grades the yield on the seated Hoppo's fiscal competence AND his graft, both read from
		# LAGGED meters (mirrored later this pulse from his real char stats): qing_hoppo_finesse and
		# qing_hoppo_squeeze (=corruption). The deliberate one-quarter lag is unchanged from the old cliff
		# (stats creep only ~1/qtr so the steady-state error is negligible; the yield chain runs BEFORE the
		# holder is reconciled, so a live read here would risk a dead/double-booked holder). Both meters are
		# guarded on has_variable: before the first reconcile (or while transiently vacant) the factor is a
		# clean x1 no-op.
		#   factor = 1 + (finesse - 7) x 0.03  -  squeeze x 0.004 , clamped [0.5, 1.3].
		# At the ~30-wan-tael zenith baseline: a master Hoppo (finesse 12, corruption ~10) ~= 1 +0.15 -0.04
		# = 1.11; a venal dolt (finesse 4, corruption 80) ~= 1 -0.09 -0.32 = 0.59 (near the 0.5 floor). The
		# old cliff's x0.7 at squeeze 60 now lands ~= 1 -0.24 = 0.76 at neutral finesse -- a smoother, graded
		# version of the same shave. The [0.5,1.3] band bounds the max uplift so it cannot perturb the
		# currency model (the silver inflow below inherits this factor via qing_canton_yield_tmp, bounded too).
		if = {
			limit = { has_variable = qing_hoppo_finesse }
			set_variable = { name = qing_canton_hoppo_factor  value = 1 }
			# finesse lift: (finesse - 7) x 0.03
			set_variable = { name = qing_canton_hoppo_fin_tmp  value = var:qing_hoppo_finesse }
			change_variable = { name = qing_canton_hoppo_fin_tmp  subtract = 7 }
			change_variable = { name = qing_canton_hoppo_fin_tmp  multiply = 0.03 }
			change_variable = { name = qing_canton_hoppo_factor  add = var:qing_canton_hoppo_fin_tmp }
			remove_variable = qing_canton_hoppo_fin_tmp
			# corruption/squeeze shave: squeeze x 0.004 (graded; replaces the x0.7 cliff). Guarded -- squeeze
			# is seeded at game start, but belt-and-suspenders for a pre-first-reconcile pulse.
			if = {
				limit = { has_variable = qing_hoppo_squeeze }
				set_variable = { name = qing_canton_hoppo_sq_tmp  value = var:qing_hoppo_squeeze }
				change_variable = { name = qing_canton_hoppo_sq_tmp  multiply = 0.004 }
				change_variable = { name = qing_canton_hoppo_factor  subtract = var:qing_canton_hoppo_sq_tmp }
				remove_variable = qing_canton_hoppo_sq_tmp
			}
			# clamp the factor to the sane band [0.5, 1.3] (var-vs-literal, bare-legal).
			if = { limit = { var:qing_canton_hoppo_factor > 1.3 }  set_variable = { name = qing_canton_hoppo_factor  value = 1.3 } }
			if = { limit = { var:qing_canton_hoppo_factor < 0.5 }  set_variable = { name = qing_canton_hoppo_factor  value = 0.5 } }
			change_variable = { name = qing_canton_yield_tmp  multiply = var:qing_canton_hoppo_factor }
			remove_variable = qing_canton_hoppo_factor
		}
		# [#30/#31 Batch 4g 2026-07-25] CANTON TARIFF bias (關稅則例). The qing_canton_tariff_law selector
		# sets the ad-valorem rate the customs house levies on the real Canton trade (the yield above IS the
		# real-goods base — tea/silk/porcelain). qing_canton_tariff_rate: 1 = 協定關稅 low fixed treaty tariff
		# (takes LESS per unit, x0.8 — the historical low rate the treaty ports were pinned to), 2 = 關稅自主
		# autonomous high tariff (takes MORE, x1.3). Default (0/unset) = the customary rate, x1 (unchanged).
		# Bias-B: guarded on has_variable, LHS var / literal RHS, applied to the yield BEFORE the purse split.
		if = {
			limit = { has_variable = qing_canton_tariff_rate  var:qing_canton_tariff_rate = 1 }
			change_variable = { name = qing_canton_yield_tmp  multiply = 0.8 }
		}
		else_if = {
			limit = { has_variable = qing_canton_tariff_rate  var:qing_canton_tariff_rate = 2 }
			change_variable = { name = qing_canton_yield_tmp  multiply = 1.3 }
		}
		# --- SPLIT [#426]: the player sets (via the Canton window) what SHARE of the yield goes to
		# the emperor's privy purse (內帑) vs the state treasury (戶部): qing_canton_purse_share = 0 / 50
		# / 100 (percent to the purse). The state gets the remainder — 0% routes the whole Canton haul
		# onto the 戶部 books (shown as the Revenue panel's Canton Contribution); 100% keeps it private.
		set_variable = { name = qing_canton_emperor_tmp  value = var:qing_canton_yield_tmp }
		change_variable = { name = qing_canton_emperor_tmp  multiply = var:qing_canton_purse_share }
		change_variable = { name = qing_canton_emperor_tmp  divide = 100 }
		# accrue the emperor's Canton silver (the running privy haul the panel shows)...
		change_variable = { name = qing_canton_customs  add = var:qing_canton_emperor_tmp }
		# ...and lift the emperor's own privy purse (內帑 = current_ruler.wealth, #424), scaled by share.
		if = {
			limit = { exists = current_ruler  var:qing_canton_purse_share > 0 }
			set_variable = { name = qing_canton_purse_gain_tmp  value = 12 }
			change_variable = { name = qing_canton_purse_gain_tmp  multiply = var:qing_canton_purse_share }
			change_variable = { name = qing_canton_purse_gain_tmp  divide = 100 }
			current_ruler = { add_gold = { value = ROOT.var:qing_canton_purse_gain_tmp } }
			remove_variable = qing_canton_purse_gain_tmp
		}
		# the state's share reaches the 戶部 treasury (the on-books Canton Contribution).
		set_variable = { name = qing_canton_state_tmp  value = var:qing_canton_yield_tmp }
		change_variable = { name = qing_canton_state_tmp  subtract = var:qing_canton_emperor_tmp }
		add_treasury = var:qing_canton_state_tmp
		# record the total yield + the state share for the panel read-outs (the Revenue panel's
		# Canton Contribution line reads qing_canton_last_state).
		set_variable = { name = qing_canton_last_yield  value = var:qing_canton_yield_tmp }
		set_variable = { name = qing_canton_last_state  value = var:qing_canton_state_tmp }

		# --- [Canton silver inflow] the specie the Western traders paid IN for the tea/silk/porcelain
		# they shipped OUT. The customs YIELD above is only the ad-valorem LEVY (→ treasury/purse); the
		# silver that actually entered the empire was the PAYMENT for the goods, far larger than the levy,
		# and it is credited here to the engine's currency-backing reserve (silver_reserve_size), the
		# historical pre-Opium-War specie inflow the money model previously never modelled. Derived from
		# qing_canton_yield_tmp so it inherits every gating already applied above (open regime, port level,
		# Hoppo squeeze, tariff law) for free — close the port → yield 0 → inflow 0.
		# UNITS: qing_canton_yield_tmp is 萬兩; silver_reserve_size is 千兩 → ×10 conversion (fixed), then
		# ×1.5 trade-specie multiple tuned to the sourced ~100 tons/yr Canton inflow (Deng 2008;
		# research/1763_CANTON_SILVER_INFLOW.md): 45萬兩 → ×10=450 → ×1.5=675 千兩/qtr ≈ ~100 tons/yr.
		# NO upper cap (the 1777 peak is a milestone EVENT, not a ceiling — the peak clamp was removed
		# from QING_revenue_reserve_drift 2026-08-06); this is a data-driven MARKET writer, so real trade
		# specie may carry the reserve above the historic peak. Moves silver_reserve_size ONLY — does NOT
		# write silver_reserve_actual_change (a LIVE minting input; mirrors the revenue drift's abstention).
		# See design/DESIGN_CANTON_SILVER_RESERVE.md.
		if = {
			limit = { has_variable = silver_reserve_size }
			set_variable = { name = qing_canton_silver_in_tmp  value = var:qing_canton_yield_tmp }
			change_variable = { name = qing_canton_silver_in_tmp  multiply = 10 }   # 萬兩 → 千兩 unit conversion (fixed)
			change_variable = { name = qing_canton_silver_in_tmp  multiply = 1.5 }  # trade-specie multiple (tuned knob)
			change_variable = { name = silver_reserve_size          add = var:qing_canton_silver_in_tmp }
			set_variable    = { name = qing_canton_last_silver_in    value = var:qing_canton_silver_in_tmp }  # panel readout
			remove_variable = qing_canton_silver_in_tmp
			LOG_line = { sys = QING  msg = "canton: silver inflow credited to the reserve for" }
		}
		else = {
			# reserve unseeded (should never happen for CHI — seeded in CURRENCY_setup_all_reserves): honest 0.
			set_variable = { name = qing_canton_last_silver_in  value = 0 }
		}

		# [#76 2026-08-08 reopened] RECONCILE a double-booked Hoppo BEFORE anything reads him. The
		# reported save already had Aligun/585 both commanding the Jingzhou garrison AND seated as Hoppo
		# (the pre-fix draw slipped the is_general gate — see the appoint limit below). A living Hoppo who
		# is ALSO a serving commander/officer/governor is relieved of the customs post here: strip his
		# qing_hoppo_marker (which clears the second title — the offices custom-loc renders it off that
		# marker, 00_offices.txt:178/238) and vacate qing_hoppo_holder. Clearing the holder makes block (a)
		# below see an empty post THIS pulse and backfill a free courtier, so the customs house is never
		# left headless. Mirrors the study-chief / sub-post double-book strip (#79). The command outranks
		# the customs sinecure — a garrison general is not pulled off the wall to run a customs house.
		if = {
			limit = {
				has_variable = qing_hoppo_holder
				var:qing_hoppo_holder = { is_alive = yes }
				var:qing_hoppo_holder = {
					OR = {
						is_general = yes
						is_admiral = yes
						is_governor = yes
						has_variable = qing_officer_marker
					}
				}
			}
			var:qing_hoppo_holder = { if = { limit = { has_variable = qing_hoppo_marker }  remove_variable = qing_hoppo_marker } }
			remove_variable = qing_hoppo_holder
			LOG_line = { sys = QING  msg = "canton: relieved a double-booked Hoppo (also a serving commander/officer) for" }
		}

		# [#69 2026-08-08] THE HOPPO'S GRAFT MADE CONCRETE. The squeeze is no longer a free-floating
		# meter; it is DERIVED from the seated Hoppo's own vanilla CORRUPTION, and his exactions are
		# REAL character wealth he pockets (concrete-over-abstract rule). Backfill first so the post is
		# never empty (the window + impeachment always have a subject).
		# (a) BACKFILL a dead/missing Hoppo — appoint a fresh man (a new broom, low corruption).
		if = {
			limit = {
				OR = {
					NOT = { has_variable = qing_hoppo_holder }
					var:qing_hoppo_holder = { is_alive = no }
				}
			}
			# [#76] This backfill fires ONLY for a dead/missing Hoppo (the outer limit: no holder, or holder
			# is_alive=no). An ALIVE ex-Hoppo never reaches here — the only paths that vacate a living Hoppo
			# (censorate impeach, rotate) both strip qing_hoppo_marker off the man before removing the holder
			# var, so QING_char_holds_court_position never reads a stale post on a live former Hoppo. A DEAD
			# ex-Hoppo keeps the marker on his corpse, but a dead char renders no card and is never re-picked,
			# so it is inert (same as the study-chief death case). Nothing to strip here.
			if = { limit = { has_variable = qing_hoppo_holder }  remove_variable = qing_hoppo_holder }
			QING_canton_appoint_hoppo = yes
			LOG_line = { sys = QING  msg = "canton: the Hoppo's post fell vacant and was backfilled for" }
		}
		# (b) GRAFT CREEP — the seated Hoppo's own corruption climbs quarterly (the source of the
		# squeeze), capped < 90 so a long-tenured man's number rises but never pegs. add_corruption is
		# the proven vanilla char verb (se_QING_HOUSEHOLD.txt:223). The 粵海關章程 regulation bias now
		# adjusts THIS creep (the source), not a separate meter: -1 board-audited slows it, +1 tax-farmed
		# speeds it. Guarded on has_variable (unset until the law is enacted).
		if = {
			limit = { has_variable = qing_hoppo_holder  var:qing_hoppo_holder = { is_alive = yes } }
			var:qing_hoppo_holder = {
				if = {
					limit = { corruption < 90 }
					add_corruption = 1
					# the 粵海關章程 regulation bias adjusts the creep. It is only ever 0 / -1 / +1 (the three
					# law options), so apply it as explicit LITERAL branches — add_corruption takes a literal,
					# NOT a variable amount (no add_<stat> = var: form is attested; proven negatives at
					# se_QING_MECHANICS.txt:129). +1 tax-farmed speeds the graft; -1 board-audited slows it.
					if = {
						limit = { ROOT = { has_variable = qing_hoppo_regulation_bias  var:qing_hoppo_regulation_bias = 1 } }
						add_corruption = 1
					}
					else_if = {
						limit = { ROOT = { has_variable = qing_hoppo_regulation_bias  var:qing_hoppo_regulation_bias = -1 } }
						add_corruption = -1
					}
				}
			}
			# (c) SIPHON — the Hoppo's "squeeze" was extracted FROM THE HONG MERCHANTS (over-charging on
			# fees and measurement), NOT skimmed from the throne's official customs — so this does NOT
			# reduce the yield or the emperor/state split (the throne's reduced take at high squeeze is
			# ALREADY modelled by the ×0.7 yield-shave above). He pockets a slice scaled by his corruption
			# as REAL personal wealth: siphon = yield × corruption / 200 (corruption 30 ≈ 15% of a quarter's
			# customs-equivalent; 90 ≈ 45%). add_gold to him = his hoard, confiscated on impeachment (#68/#70).
			# This touches only his CHARACTER wealth stat — NOT add_treasury, silver_reserve_size, or the
			# money supply — so it cannot perturb the currency sim (#23-caution); the only treasury effect is
			# the one-shot 抄家 confiscation when he is impeached.
			var:qing_hoppo_holder = { save_scope_as = qing_hoppo_seated }
			set_variable = { name = qing_hoppo_siphon_tmp  value = var:qing_canton_yield_tmp }
			change_variable = { name = qing_hoppo_siphon_tmp  multiply = scope:qing_hoppo_seated.corruption }
			change_variable = { name = qing_hoppo_siphon_tmp  divide = 200 }
			if = { limit = { var:qing_hoppo_siphon_tmp < 0 }  set_variable = { name = qing_hoppo_siphon_tmp  value = 0 } }
			scope:qing_hoppo_seated = { add_gold = { value = ROOT.var:qing_hoppo_siphon_tmp } }
			# (d) the squeeze meter is now a DERIVED reflection of his corruption (0..100), for the
			# existing consumers (the yield-shave, the corruption-level leak, the Cohong-crisis gate).
			set_variable = { name = qing_hoppo_squeeze  value = scope:qing_hoppo_seated.corruption }
			# [#111a] LAGGED FINESSE mirror — the seated Hoppo's fiscal competence, cached alongside the
			# squeeze(=corruption) mirror so the CONTINUOUS yield-grading factor (QING_canton_pulse yield
			# chain) can read it WITHOUT a live char read at the yield site. The yield is computed + banked
			# EARLIER in this pulse (before this reconcile/graft block), so reading var:qing_hoppo_holder.finesse
			# there would risk a dead/double-booked/absent holder — the same one-quarter-lag design the squeeze
			# already uses (see the yield-chain note). save_scope_as → scope:X.finesse is the proven read
			# (se_QING_CANAL.txt:118-120 warns the bare var:X.finesse form reads 0 silently).
			set_variable = { name = qing_hoppo_finesse  value = scope:qing_hoppo_seated.finesse }
			remove_variable = qing_hoppo_siphon_tmp
		}
		else = {
			# transiently vacant (before backfill, or the appoint pool was empty): the exactions run
			# unchecked, the squeeze drifts up as under the old model. The lagged finesse holds its last
			# value (no seated man to re-read); the grading factor guards on has_variable.
			QING_DECLINE_nudge = { var = qing_hoppo_squeeze  amount = 3 }
		}
		# a high squeeze leaks into the empire-wide corruption counter (guarded — it may be unset
		# very early, though QING_DECLINE_init seeds it at game start).
		if = {
			limit = {
				var:qing_hoppo_squeeze >= 55
				has_variable = qing_corruption_level
			}
			QING_DECLINE_nudge = { var = qing_corruption_level  amount = 1 }
		}

		# --- the Cohong (公行) debt crisis: at a punishing squeeze the Hong merchants buckle
		# under levies and bankruptcy looms. Throttled + gated on the court event slot (BT-28),
		# self-guards on high squeeze, so it is a rare crisis, inert at the well-run zenith.
		if = {
			limit = {
				var:qing_hoppo_squeeze >= 65
				NOT = { has_variable = qing_canton_cohong_cooldown }
				NOT = { has_variable = qing_gc_event_slot_used }
			}
			set_variable = { name = qing_canton_cohong_cooldown  days = 1095 }   # ~3y throttle
			set_variable = { name = qing_gc_event_slot_used  value = 1 }  # [BT-28] claim court slot
			trigger_event = { id = qing_canton.1  days = { 5 20 } }
			LOG_line = { sys = QING  msg = "canton: Cohong debt crisis brews (high Hoppo squeeze) for" }
		}

		# clear the per-pulse scratch vars
		remove_variable = qing_canton_yield_tmp
		remove_variable = qing_canton_emperor_tmp
		remove_variable = qing_canton_state_tmp
		LOG_line = { sys = QING  msg = "canton: customs collected (emperor 內帑 + state quota) for" }
		LOG_exit = { sys = QING  fn = "QING_canton_pulse"  result = OK }
	}
}

# ======================================================================
# ROTATE THE HOPPO (粵海關監督). The player's proactive lever on the squeeze: send a fresh
# Neiwufu man to Canton on the historical 3-year cycle. Not a Grand Council office (the
# Hoppo is a bondservant appointee, outside the great-office roster) — a lightweight
# character marker. Self-contained (picks a capable courtier itself, so it needs no runtime
# GUI picker): a fresh, accountable appointee steadies the squeeze; a higher-finesse man
# steadies it more. Country-scope (ROOT = CHI); called from the Canton window button.
# [#389 review-fix] REPLACES the button-less QING_canton_appoint_hoppo, whose absence made
# the squeeze a one-way ratchet with no player agency (the Cohong crisis was inevitable).
# ======================================================================
# ---- [#66 2026-08-08] APPOINT A HOPPO (extracted). Picks the ablest-finesse free courtier and
# seats him as the 粵海關監督, stamping qing_hoppo_holder (the country link) + qing_hoppo_marker
# (on the man). No squeeze nudge here — the caller decides the appointment's effect on the graft
# (rotate eases it; init/backfill just fills the seat). A fresh appointee carries his OWN current
# corruption, so an honest new man runs the customs straight and a venal one does not — the #69
# concrete model (squeeze is derived from his corruption in the pulse). Country scope (ROOT = CHI).
QING_canton_appoint_hoppo = {
	ordered_character = {
		limit = {
			employer = ROOT
			is_adult = yes
			is_alive = yes
			is_ruler = no
			# [#76 2026-08-08] one-man-one-post: exclude anyone already holding ANY court/sub-post
			# (QING_char_holds_court_position now includes the Hoppo marker) AND serving commanders/
			# governors/heir — the SAME exclusions the GC pickers use. The reported bug was the Hoppo
			# (Aligun) being drawn while he was the Jingzhou garrison commander; is_general/is_admiral
			# close that. QING_char_hard_disgraced keeps a disgraced man out too.
			NOT = { QING_char_holds_court_position = yes }
			is_general = no
			is_admiral = no
			is_governor = no
			# [#76 2026-08-08 reopened] ALSO exclude a serving garrison/army FIELD OFFICER. The reported
			# Aligun/585 case: he commands the Jingzhou banner garrison, attached via set_as_commander at
			# setup — which stamps qing_officer_marker (imp19c_effects_legion_setup.txt:189) but does NOT
			# reliably set is_general (the same BT-7 lesson QING_char_holds_court_position learned at
			# qing_dynasty_triggers.txt:213). Without this line he passed is_general=no and was drawn as
			# Hoppo while still commanding, showing BOTH titles. qing_officer_marker is the authoritative
			# officer-corps flag; gating it here closes the draw for good.
			NOT = { has_variable = qing_officer_marker }
			NOT = { QING_char_hard_disgraced = yes }
			NOT = { AND = { exists = ROOT.primary_heir  this = ROOT.primary_heir } }
		}
		order_by = finesse
		max = 1
		save_scope_as = qing_hoppo_pick
		ROOT = {
			set_variable = { name = qing_hoppo_holder  value = scope:qing_hoppo_pick }
		}
		# qing_hoppo_marker doubles as the per-char post marker: it feeds QING_char_holds_court_position
		# (mutual exclusion both ways) and the character-card title (offices custom-loc), like the other
		# sub-post markers (#76).
		set_variable = { name = qing_hoppo_marker  value = 1 }
	}
}

QING_canton_rotate_hoppo = {
	LOG_enter = { sys = QING  fn = "QING_canton_rotate_hoppo" }
	# relieve any prior Hoppo first (the post turns over on a 3-year cycle).
	if = {
		limit = { has_variable = qing_hoppo_holder }
		if = {
			limit = { var:qing_hoppo_holder = { is_alive = yes } }
			var:qing_hoppo_holder = { if = { limit = { has_variable = qing_hoppo_marker }  remove_variable = qing_hoppo_marker } }
		}
		remove_variable = qing_hoppo_holder
	}
	# pick a capable adult courtier (highest finesse — the fiscal administrator's gift) as the new Hoppo.
	QING_canton_appoint_hoppo = yes
	# a fresh, accountable appointee steadies the squeeze (a new broom, and the bribes to win
	# the post are paid — he has room to run it straight for a while).
	QING_DECLINE_nudge = { var = qing_hoppo_squeeze  amount = -12 }
	# a high-finesse man polices the exactions harder still.
	if = {
		limit = {
			has_variable = qing_hoppo_holder
			var:qing_hoppo_holder = { finesse >= 7 }
		}
		QING_DECLINE_nudge = { var = qing_hoppo_squeeze  amount = -6 }
	}
	LOG_line = { sys = QING  msg = "canton: a new Hoppo (粵海關監督) is sent to milk the trade for" }
	LOG_exit = { sys = QING  fn = "QING_canton_rotate_hoppo"  result = OK }
}

# ======================================================================
# [#426] QING_canton_set_purse_share — set what SHARE of the Canton yield accrues to the
# emperor's privy purse (內帑) vs the state treasury (戶部). $share$ = 0 | 50 | 100 (percent
# to the purse). The quarterly Canton pulse reads qing_canton_purse_share and splits the
# yield accordingly; a lower share routes more onto the 戶部 books (the Revenue panel's
# Canton Contribution line). Scope: COUNTRY (CHI). Called from the three Canton-window toggles.
# ======================================================================
QING_canton_set_purse_share = {
	QING_canton_init = yes
	set_variable = { name = qing_canton_purse_share  value = $share$ }
	LOG_line = { sys = QING  msg = "canton: privy-purse share set (percent to 內帑) for" }
}
