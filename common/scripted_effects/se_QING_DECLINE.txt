# se_QING_DECLINE.txt — the shared "dynastic decline" state engine for the
# Qing player-experience mechanics suite.
#
# Design: the mechanics A (silver stress), C (banner decay), D (corruption),
# G (sects), #1 (ethnic tension) and the reform pressure that unlocks #9b all
# share a small set of COUNTER variables on the Qing country. Each counter is
# nudged EVENT-DRIVEN (never recomputed by sweeping pops/provinces), and a
# single throttled pulse (QING_DECLINE_apply_bands) reads the counters and
# swaps the country modifier that matches each counter's current band.
#
# This is why the whole suite stays cheap: the counters are plain numbers,
# reads are O(1), and the only per-tick work is a handful of threshold checks
# gated to the single human Qing player (see 00_monthly_country.txt).
#
#   ROOT / this = the Qing overlord (CHI, human)
#
# State variables (all on CHI):
#   qing_corruption_level     0..100   (D)   raised by war/office-sales, cut by audits/reform
#   qing_sect_pressure        0..100   (G)   raised by famine/unrest/tax, cut by suppression
#   qing_banner_decay         0..100   (C)   climbs slowly with time, cut by reform drills
#   qing_ethnic_tension       0..100   (#1)  raised by integration/strife, cut by dyarchy
#   qing_currency_stress      0..100   (A)   tracks reserve-ratio stress + opium drain
#   qing_reform_pressure      0..100   (#9)  fed by ALL the above; unlocks the reform track
#   qing_granary_stock        0..100   (F)   built in good years, spent in famine
#
# Prefix = QING_DECLINE
########################################################################

# ---- Generic clamped-counter helpers ---------------------------------
# Nudge one 0..100 counter by $amount$ (may be negative) and clamp.
# Scope: country (CHI).  $var$ = counter name, $amount$ = signed integer.
QING_DECLINE_nudge = {
	if = {
		limit = { NOT = { has_variable = $var$ } }
		set_variable = { name = $var$  value = 0 }
	}
	change_variable = { name = $var$  add = $amount$ }
	if = { limit = { var:$var$ > 100 }  set_variable = { name = $var$  value = 100 } }
	if = { limit = { var:$var$ < 0 }    set_variable = { name = $var$  value = 0 } }
	LOG_line = { sys = QING  msg = "counter $var$ += $amount$ ->" }
}

# ---- SIGNED nudge (NO 0-floor) — for two-store meters whose LEVEL is derived --
# [meter-concretize #3 2026-08-06] Like QING_DECLINE_nudge but WITHOUT the 0/100 clamp, so the
# target var may go NEGATIVE and be eased from BOTH sides. Used for qing_currency_stress_residual:
# the currency-stress LEVEL is clamp(reserve-derived base + residual), so the residual itself must
# stay signed (a surplus/relief event pushes it negative; a crisis pushes it positive) — the plain
# floored nudge would make every negative event inert once the residual hit 0, discarding the
# High-Qing calm that comes from trade-surplus relief. The clamp happens only on the LEVEL, in
# QING_DECLINE_recompute_currency_stress. Scope: country. $var$ = var name, $amount$ = signed int.
QING_DECLINE_nudge_signed = {
	if = {
		limit = { NOT = { has_variable = $var$ } }
		set_variable = { name = $var$  value = 0 }
	}
	change_variable = { name = $var$  add = $amount$ }
	LOG_line = { sys = QING  msg = "signed counter $var$ += $amount$ ->" }
}

# Initialise every counter to 0 if unset. Called once on game start for CHI.
QING_DECLINE_init = {
	LOG_enter = { sys = QING  fn = "QING_DECLINE_init" }
	# [bookmark-1763 #304] HIGH QING ERA detection. This suite is tuned for the 1815
	# decline; at a pre-Heshen start (game begins before 1772.1.1, i.e. the 1763 bookmark)
	# the dynasty is at its zenith, not sliding. Set a one-shot era flag + grant the
	# prosperity modifier so the pulse suppresses the passive decline creep until the
	# Heshen inflection (qing_decline.40/.41) ends it. current_date at this init call
	# = the game-start date (QING_DECLINE_init runs at on_game_initialized). The 1772.1.1
	# threshold MATCHES the qing_decline.40 checker-arming gate in oa_economy_setup, so the
	# era flag is never granted without the checker also being armed (no permanent golden
	# age from a start in a threshold gap). Guarded so an 1815 start (current_date past 1772)
	# never enters the High Qing branch, and so a mid-game re-init after the era has ended
	# cannot re-grant it.
	if = {
		limit = {
			current_date < 1772.1.1
			NOT = { has_variable = qing_high_qing_era }
			NOT = { has_variable = qing_high_qing_era_ended }
		}
		set_variable = { name = qing_high_qing_era  value = 1 }
		add_country_modifier = { name = qing_high_qing_prosperity  duration = -1 }
		LOG_line = { sys = QING  msg = "HIGH QING ERA active (pre-1775 zenith): prosperity granted, decline creep suppressed for" }
	}
	# [bookmark-1763 #304] Corruption starts LOW in the High Qing (baseline administrative
	# friction, not the systemic 陋規/Heshen graft of the post-1775 court); the 1815 start
	# keeps the historical value 0 baseline (the pulse ramps it). Seed 12 for a High Qing
	# start so the corruption band opens in its lowest tier, 0 otherwise.
	if = {
		limit = { NOT = { has_variable = qing_corruption_level } }
		if = {
			limit = { has_variable = qing_high_qing_era }
			set_variable = { name = qing_corruption_level  value = 12 }
		}
		else = { set_variable = { name = qing_corruption_level  value = 0 } }
	}
	if = { limit = { NOT = { has_variable = qing_sect_pressure } }     set_variable = { name = qing_sect_pressure  value = 0 } }
	if = { limit = { NOT = { has_variable = qing_banner_decay } }      set_variable = { name = qing_banner_decay  value = 10 } }
	# task #88 — the Green Standard (綠營) rots IN PARALLEL to the banners: a second
	# army-decay vector on the Han line infantry. Starts slightly higher than the
	# banners: the Green Standard was already the more neglected, garrison-bound force.
	if = { limit = { NOT = { has_variable = qing_greenstandard_decay } } set_variable = { name = qing_greenstandard_decay  value = 15 } }
	# task #88 — Han provincial military power (勇營 → 湘軍/淮軍 → 軍閥). Starts at 0:
	# in 1815 the throne still commands its own armies. Rises only when BOTH central
	# armies fail and the throne must lean on regional magnates (derived in the pulse).
	if = { limit = { NOT = { has_variable = qing_han_provincial_power } } set_variable = { name = qing_han_provincial_power  value = 0 } }
	# task #88 — the modern national-army share (新軍/levee). Starts at 0: no modern
	# centrally-commanded force exists at the 1815 start. Built by Self-Strengthening
	# (Ever-Victorious) and the alt-history Napoleon levee; drifts toward a
	# self-strengthening-gated ceiling, with founding floors front-loading it.
	if = { limit = { NOT = { has_variable = qing_modernarmy_share } }     set_variable = { name = qing_modernarmy_share  value = 0 } }
	if = { limit = { NOT = { has_variable = qing_ethnic_tension } }    set_variable = { name = qing_ethnic_tension  value = 20 } }
	if = { limit = { NOT = { has_variable = qing_currency_stress } }   set_variable = { name = qing_currency_stress  value = 0 } }
	# [meter-concretize #3 2026-08-06] the SIGNED event-shock residual for the two-store currency-stress
	# meter (the base is derived from the reserve ratio in QING_DECLINE_recompute_currency_stress; this
	# holds the discrete event nudges and decays toward 0). Seeded 0; may go negative.
	if = { limit = { NOT = { has_variable = qing_currency_stress_residual } } set_variable = { name = qing_currency_stress_residual  value = 0 } }
		# [#67] paper-money depreciation meter (0..100). Seeded 0; rises with paper over-issue while the regime is
		# active, decays slowly otherwise (trust rebuilds; scarring on re-entry). Drives the mint-allowance shrink
		# (CURRENCY_qing_paper_mint_allowance) + a bounded stress term. Only meaningful post-1853 crisis unlock.
		if = { limit = { NOT = { has_variable = qing_paper_depreciation } }  set_variable = { name = qing_paper_depreciation  value = 0 } }
	if = { limit = { NOT = { has_variable = qing_reform_pressure } }   set_variable = { name = qing_reform_pressure  value = 0 } }
	if = { limit = { NOT = { has_variable = qing_granary_stock } }     set_variable = { name = qing_granary_stock  value = 30 } }
	# [#369] Population pressure (人口壓力 / 人地矛盾) — the standing Malthusian meter. Starts LOW
	# in the High Qing zenith (the land can still feed the boom); seeded 20 so the strain band
	# (>=45) is a few pulses away, and driven up by the target recompute (se_QING_POPULATION.txt).
	if = { limit = { NOT = { has_variable = qing_pop_pressure } }      set_variable = { name = qing_pop_pressure  value = 20 } }
	# [#370] Caravan prosperity (商道繁榮) — the Central Asian trade-node health. Starts at a going 45
	# in the 1763 zenith (routes open, oases prosperous); driven by the target recompute (se_QING_CARAVAN.txt)
	# off the Xinjiang grip + market investment + the Kokand concession.
	if = { limit = { NOT = { has_variable = qing_caravan_prosperity } } set_variable = { name = qing_caravan_prosperity  value = 45 } }
	# Civic (national) identity — the POSITIVE nation-building meter (§11 L4 ext).
	# A shared "Chinese"/中華 civic identity that can transcend the empire's ethnic
	# fault lines. Starts at 0: in 1815 the realm is a dynastic-ethnic order, not a
	# nation. Rises only as the Self-Strengthening movement forges the connective
	# institutions of a nation (schools, telegraph, press) — see the drift below.
	if = { limit = { NOT = { has_variable = qing_civic_identity } }    set_variable = { name = qing_civic_identity  value = 0 } }
	# [CROSSWIRE 2026-07-21 rites->mandate] 天命 — the Mandate of Heaven as a standing 0..100 meter.
	# Previously the mandate lived only as vanilla legitimacy (se_QING_MANDATE.txt disaster hook); this
	# gives it a persistent counter so ritual efficacy (祭天, se_QING_RITES) can RAISE it and dynastic
	# crisis can erode it, and so it can GATE heterodox rebellion (a lost mandate invites rising — the
	# consumer is in QING_rebellion_roll below). Seeded HIGH (80) at the High-Qing zenith: the dynasty's
	# fitness to rule is unquestioned in 1763/1815. Erodes in the pulse when decline runs deep.
	if = { limit = { NOT = { has_variable = qing_mandate_strength } }  set_variable = { name = qing_mandate_strength  value = 80 } }
	LOG_exit = { sys = QING  fn = "QING_DECLINE_init"  result = OK }
}

# ======================================================================
# BAND APPLICATION
# Read each counter and (re)apply the matching modifier band. Called from the
# throttled monthly pulse. Each block removes the other bands for its family
# first so exactly one is active, then adds the correct one. This is the only
# recurring work in the suite and it is all O(1) counter reads.
# ======================================================================

# --- A: currency stress feeds off the live reserve ratio, then bands --------
# The reserve-ratio -> stress transfer function now lives in the GENERIC engine
# se_CURRENCY_STRESS.txt (CURR_STRESS_update): the Qing meter EXTENDS it rather
# than owning a private copy. We pass our own counter var (qing_currency_stress)
# so the qing decline suite keeps its own meter and its feed into reform pressure
# and the opium/treaty events — only the underlying transfer function is shared.
# (Historically the "silver drain": a low reserve ratio = silver drained away.)
# [meter-concretize #3 2026-08-06] TWO-STORE recompute — the LEVEL is now DERIVED from the concrete
# monetary state, not a 1-bit drift accumulator (design/DESIGN_CURRENCY_STRESS_CONCRETIZE.md).
#   qing_currency_stress = clamp( base + opium_flow + residual , 0, 100 )
# where
#   base    = the CALIBRATED reserve-backing transfer (silver famine 銀荒 = a drained 戶部銀庫), NOT the
#             naive (1−impact)×100 (CHI is silver_standard w/o public_debt → impact = reserve_ratio_total÷3,
#             a high bar; the naive map would fire silver_drain at the High-Qing zenith). Knee at 0.6
#             (lowered from 0.7 per review MEDIUM — headroom for the deflation term below): impact 0.6→0,
#             0.5→~17, 0.4→~33 (strain), 0.3→~50, 0→100. Logs bound CHI's 1763 impact in [0.5,1.0] → base
#             opens ≤~17, and even the MAX deflation term (+10) keeps the opening ≤~27 (< the 30 strain
#             threshold) — so no false silver_drain at the zenith regardless of where in [0.5,1.0] the true
#             impact sits OR whether 1763 carries any deflation. PLUS that deflation term (銀荒 is a
#             DEFLATIONARY silver scarcity — private_cash_ratio<1, review M2), 0..+10.
#   opium   = the CONCRETE trade-balance flow (tea/silk/porcelain export inflow − opium-import drain),
#             computed in QING_opium_assess_trade_balance and stored SIGNED in qing_opium_stress_flow
#             (surplus negative = eases, drain positive = worsens). A FLOW term, not the reserve STOCK —
#             complementary, not double-counting (design §3).
#   residual = the SIGNED, PROPORTIONALLY-DECAYING store of discrete event shocks (rebellion/canton/
#             reform resets/etc.), retargeted off the old direct-to-level nudges. Decays ×0.85/quarter
#             toward 0 from both sides so a shock fades over ~a decade and no persistent flow integrates
#             to the rails.
# The generic (non-CHI) CURR_STRESS_update is UNTOUCHED — this is CHI's own recompute (design §4b).
QING_DECLINE_recompute_currency_stress = {
	LOG_enter = { sys = QING  fn = "QING_DECLINE_recompute_currency_stress" }
	if = {
		limit = { has_variable = official_currency }
		# --- base: reserve-backing transfer, knee 0.6 (review MEDIUM: headroom for deflation) ---------
		# base = clamp( (0.6 − impact) / 0.6 × 100 , 0, 100 ). Stage impact, then transform.
		set_variable = { name = qing_curr_base_tmp  value = CURRENCY_reserve_ratio_impact }
		change_variable = { name = qing_curr_base_tmp  multiply = -1 }
		change_variable = { name = qing_curr_base_tmp  add = 0.6 }
		change_variable = { name = qing_curr_base_tmp  divide = 0.6 }
		change_variable = { name = qing_curr_base_tmp  multiply = 100 }
		if = { limit = { var:qing_curr_base_tmp < 0 }   set_variable = { name = qing_curr_base_tmp  value = 0 } }
		if = { limit = { var:qing_curr_base_tmp > 100 } set_variable = { name = qing_curr_base_tmp  value = 100 } }
		# --- deflation term: a silver-scarcity money shortage adds stress (0..~10) ------
		# CURRENCY_amt_circulated_deflation = (1 − private_cash_ratio)/10 when ratio<1, else 0.
		# ×100 puts a full deflation (ratio→0) at ~10 stress points; 0 during inflation/glut.
		set_variable = { name = qing_curr_defl_tmp  value = CURRENCY_amt_circulated_deflation }
		change_variable = { name = qing_curr_defl_tmp  multiply = 100 }
		change_variable = { name = qing_curr_base_tmp  add = var:qing_curr_defl_tmp }
		# --- level = base + opium flow + residual + monetary-policy bias, clamped 0..100 -----------------------
		set_variable = { name = qing_currency_stress  value = var:qing_curr_base_tmp }
		if = {
			limit = { has_variable = qing_opium_stress_flow }
			change_variable = { name = qing_currency_stress  add = var:qing_opium_stress_flow }
		}
		# [#63] MONETARY-POLICY LAW BIAS — the player's monetary_policy_setting stance threads in here as one
		# more DIRECT additive term into the LEVEL (rebuilt from qing_curr_base_tmp each pulse → no ratchet;
		# NOT the decaying residual). A debasement stance (more_minting/issue_bonds, bias +) raises monetary
		# stress; sound money (currency_recall/limited_minting, bias −) lowers it; the neutral standard_minting
		# sets it 0. Guarded on has_variable so a realm that never enacted a stance is byte-identical to today.
		# qing_monetary_bias is set ONLY by the law option on_enact (00_monetary_policy_setting.txt).
		if = {
			limit = { has_variable = qing_monetary_bias }
			change_variable = { name = qing_currency_stress  add = var:qing_monetary_bias }
		}
		# [#67] PAPER-MONEY DEPRECIATION → stress. A BOUNDED contribution (≤ +15, NOT the raw 0..100 meter — the
		# review MED: a big term would peg the shared meter + swamp the opium/reserve signals). = depreciation×0.15,
		# so a fully-collapsed paper regime (dep 100) adds +15 on top of the +8 bias. The PRIMARY depreciation
		# feedback is the allowance-shrink (CURRENCY_qing_paper_mint_allowance), not this — this only colours stress.
		if = {
			limit = { has_variable = qing_paper_depreciation }
			set_variable = { name = qing_paper_stress_tmp  value = var:qing_paper_depreciation }
			change_variable = { name = qing_paper_stress_tmp  multiply = 0.15 }
			change_variable = { name = qing_currency_stress  add = var:qing_paper_stress_tmp }
			remove_variable = qing_paper_stress_tmp
		}
		if = {
			limit = { has_variable = qing_currency_stress_residual }
			change_variable = { name = qing_currency_stress  add = var:qing_currency_stress_residual }
		}
		if = { limit = { var:qing_currency_stress < 0 }   set_variable = { name = qing_currency_stress  value = 0 } }
		if = { limit = { var:qing_currency_stress > 100 } set_variable = { name = qing_currency_stress  value = 100 } }
		remove_variable = qing_curr_base_tmp
		remove_variable = qing_curr_defl_tmp
		# --- decay the event residual toward 0 (proportional, both sides) ---------------
		if = {
			limit = { has_variable = qing_currency_stress_residual }
			change_variable = { name = qing_currency_stress_residual  multiply = 0.85 }
		}
		LOG_line = { sys = QING  msg = "currency stress recomputed: base+opium+residual -> level for" }
	}
	LOG_exit = { sys = QING  fn = "QING_DECLINE_recompute_currency_stress"  result = OK }
}

# Band application maps the SHARED band level (CURR_STRESS_classify: 0 calm /
# 1 strain / 2 crisis, thresholds 30/60) onto the Qing-flavoured modifiers. The
# modifier names are the consumer's own — the generic classifier deliberately
# names no modifiers (see se_CURRENCY_STRESS.txt design seam).
QING_DECLINE_apply_currency_band = {
	CURR_STRESS_classify = { counter = qing_currency_stress  out = qing_curr_band_tmp }
	remove_country_modifier = qing_curr_silver_drain
	remove_country_modifier = qing_curr_monetary_crisis
	if = {
		limit = { var:qing_curr_band_tmp = 2 }
		add_country_modifier = { name = qing_curr_monetary_crisis  duration = -1 }
		LOG_line = { sys = QING  msg = "band currency = monetary_crisis for (stress=)" }
	}
	else_if = {
		limit = { var:qing_curr_band_tmp = 1 }
		add_country_modifier = { name = qing_curr_silver_drain  duration = -1 }
		LOG_line = { sys = QING  msg = "band currency = silver_drain for (stress=)" }
	}
	remove_variable = qing_curr_band_tmp
}

QING_DECLINE_apply_corruption_band = {
	remove_country_modifier = qing_corrupt_endemic
	remove_country_modifier = qing_corrupt_rampant
	remove_country_modifier = qing_corrupt_clean_government
	if = {
		limit = { var:qing_corruption_level >= 60 }
		add_country_modifier = { name = qing_corrupt_rampant  duration = -1 }
		LOG_line = { sys = QING  msg = "band corruption = rampant for (level=)" }
	}
	else_if = {
		limit = { var:qing_corruption_level >= 30 }
		add_country_modifier = { name = qing_corrupt_endemic  duration = -1 }
		LOG_line = { sys = QING  msg = "band corruption = endemic for (level=)" }
	}
	else_if = {
		limit = { var:qing_corruption_level <= 10 }
		add_country_modifier = { name = qing_corrupt_clean_government  duration = -1 }
		LOG_line = { sys = QING  msg = "band corruption = clean_government for (level=)" }
	}
}

# ======================================================================
# [meter-concretize #2 2026-08-06] CORRUPTION — drift the live meter toward the character-derived target.
# Single-store mirror of qing_council_effectiveness (design DESIGN_CORRUPTION_CONCRETIZE.md §2): the target
# qing_officialdom_corruption_target = avg seated-holder vanilla `corruption` + baseline-K (computed in
# QING_council_recompute, se_QING_COUNCIL.txt). The ~108 event nudges STAY on the live var — they are
# impulses that decay as the drift re-converges toward the character truth (the damped behavior effectiveness
# already ships). So a scandal spikes corruption then it eases back toward what the actual officials are.
# ±2/pulse (slower than effectiveness's ±3 — graft is stickier). RHS-compare needs the _cmpsvalue wrapper.
# Gated on the target existing (recompute has run at least once); self-seeds the level if absent.
# ======================================================================
QING_DECLINE_drift_corruption = {
	if = { limit = { NOT = { has_variable = qing_corruption_level } }  set_variable = { name = qing_corruption_level  value = 12 } }
	if = {
		limit = { has_variable = qing_officialdom_corruption_target }
		if = {
			limit = { var:qing_corruption_level < qing_officialdom_corruption_target_cmpsvalue }
			QING_DECLINE_nudge = { var = qing_corruption_level  amount = 2 }
		}
		else_if = {
			limit = { var:qing_corruption_level > qing_officialdom_corruption_target_cmpsvalue }
			QING_DECLINE_nudge = { var = qing_corruption_level  amount = -2 }
		}
	}
}

# MODERN NATIONAL-ARMY SHARE — DERIVED drift (task #88).
# The share of the realm's army that is a modern, centrally-commanded force
# drifts toward a ceiling set by Self-Strengthening progress: you cannot field a
# modern national army without the arsenals, drill and schools the movement
# builds. Founding effects (Ever-Victorious, Napoleon's levee) front-load it with
# floors; between those events it drifts toward the selfstr ceiling. Hollow
# modernization (the refraction flag) halves the ceiling — a paper army is not a
# real modern force. Cheap O(1). Mirrors QING_DECLINE_drift_civic_identity.
# ======================================================================
QING_DECLINE_drift_modernarmy_share = {
	# [meter-concretize #11 2026-08-06] The share is now DERIVED from the REAL modern-legion count on the map,
	# not the abstract qing_selfstr_progress ceiling (design/DESIGN_MODERNARMY_SHARE_CONCRETIZE.md). The
	# founding events (Ever-Victorious / Beiyang / Nanyang / Napoleon levée / spawn_newarmy / capstone) each
	# raise a REAL legion of qing_ever_victorious sub-units; the derive counts those legions × K. The WIN: a
	# modern army annihilated (the Yalu 1894 collapse) now DROPS the share — which re-arms the warlord vector
	# via #6 (QING_DECLINE_drift_han_provincial_power subtracts this share). No permanent stored boost / floor.
	if = { limit = { NOT = { has_variable = qing_modernarmy_share } }  set_variable = { name = qing_modernarmy_share  value = 0 } }
	# --- count the country's modern legions. BARE any_sub_unit filter — NO commander/owner guard: the
	# founding raises attach no commander (copying #9's has_commander guard would filter them out → count 0),
	# and country-scope every_unit only iterates ROOT's own legions (proven se_QING_COUNCIL.txt), so
	# foreign/rebel EVA is already excluded (review C1). Count LEGIONS, not sub-units.
	set_variable = { name = qing_modernarmy_legion_ct  value = 0 }

		limit = { any_sub_unit = { sub_unit_type = qing_ever_victorious } }

	# the current founding band-crossings; spawn_newarmy adds a 4th at share>=60 → settles ~100).
	set_variable = { name = qing_modernarmy_target_tmp  value = var:qing_modernarmy_legion_ct }
	change_variable = { name = qing_modernarmy_target_tmp  multiply = 30 }
	# a hollow, for-show modernization fields a paper force — halve the count-target (the flag survives from
	# the refraction test; a real legion is real, but a hollow movement fields fewer effective ones).
	if = {
		limit = { has_variable = qing_selfstr_hollow_flag  var:qing_selfstr_hollow_flag = 1 }
		change_variable = { name = qing_modernarmy_target_tmp  divide = 2 }
	}
	# [#15 Batch 4i] DRILL POSTURE bias (操練之制). qing_drill_posture_law writes qing_drill_posture_bias;
	# intensive 新式操練 lifts the target the share chases, a lax posture lowers it. Relocated from the old
	# selfstr ceiling onto the count-target (review L4). Guarded → default byte-identical.
	if = {
		limit = { has_variable = qing_drill_posture_bias }
		change_variable = { name = qing_modernarmy_target_tmp  add = var:qing_drill_posture_bias }
	}
	if = { limit = { var:qing_modernarmy_target_tmp < 0 }    set_variable = { name = qing_modernarmy_target_tmp  value = 0 } }
	if = { limit = { var:qing_modernarmy_target_tmp > 100 }  set_variable = { name = qing_modernarmy_target_tmp  value = 100 } }
	if = {
		limit = { var:qing_modernarmy_share < qing_modernarmy_target_tmp_cmpsvalue }

		QING_DECLINE_nudge = { var = qing_modernarmy_share  amount = -2 }

	LOG_line = { sys = QING  msg = "modernarmy share: real modern-legion count -> target (legions=) for" }

# ======================================================================
# HAN PROVINCIAL MILITARY POWER — DERIVED drift (task #88).
# The genuinely-new decline vector. Its target is a function of the OTHER army
# meters, capturing the historical causal chain: when BOTH central armies have
# rotted (banner + green-standard decay high) the throne is FORCED to let Han
# governors-general raise personal regional armies (勇營 → 湘軍/淮軍) — which then
# accrete autonomy. A strong modern NATIONAL army (新軍/levee) is the throne's way
# out: it SUPPRESSES the target, because the centre no longer depends on the
# warlords. So:
#     target = (banner_decay + greenstandard_decay)/2  - modernarmy_share
#   clamped 0..100, drifted +/-2 (slow: regional power bases form over years).
# Governance levers (a green-standard drill, banner reform) cut the central-army
# decay counters and thus indirectly starve this vector; committing to the modern
# national army cuts it directly. Cheap O(1).
# ======================================================================
QING_DECLINE_drift_han_provincial_power = {
	if = { limit = { NOT = { has_variable = qing_han_provincial_power } }  set_variable = { name = qing_han_provincial_power  value = 0 } }
	# [meter-concretize #6 2026-08-06] BLEND a CONCRETE magnate-governor COUNT into the target. The decay
	# proxy below models the LATENT pressure (rotting central armies FORCE devolution); this term adds the
	# REALIZED devolution — how many Han governors-general actually hold personal power bases. COUNT
	# (not a cohort-sum): num_loyal_cohorts is a TRIGGER only (proven char_events.txt:175), NOT a readable
	# value, so we count magnate-governors. A magnate = an alive Han governor employed by the throne who
	# EITHER carries the #105 sanction modifier (qing_regional_magnate) OR commands >= 10 loyal cohorts.
	# every_character{employer=ROOT is_governor is_alive} is the proven effect-iterator (se_QING_PERSONNEL.txt:47);
	# has_culture=han is the proven selection (QING_mechanics_actions.txt:153). Opens at 0 in 1763 (no
	# sanctioned magnates, no Han governors with private armies at the zenith) — matches the seed.
	set_variable = { name = qing_han_magnate_tally  value = 0 }
	every_character = {
		limit = {
			employer = ROOT
			is_governor = yes
			is_alive = yes
			has_culture = han
			OR = {
				has_character_modifier = qing_regional_magnate
				num_loyal_cohorts >= 10
			}
		}
		ROOT = { change_variable = { name = qing_han_magnate_tally  add = 1 } }
	}
	# target = mean of the two central-army decay meters (LATENT pressure)...
	set_variable = { name = qing_han_prov_target_tmp  value = var:qing_banner_decay }
	change_variable = { name = qing_han_prov_target_tmp  add = var:qing_greenstandard_decay }
	change_variable = { name = qing_han_prov_target_tmp  divide = 2 }
	# ...minus the modern national-army share (the centralizing counterweight).
	if = { limit = { NOT = { has_variable = qing_modernarmy_share } }  set_variable = { name = qing_modernarmy_share  value = 0 } }
	change_variable = { name = qing_han_prov_target_tmp  subtract = var:qing_modernarmy_share }
	# ...PLUS the concrete magnate-count (REALIZED devolution). 12/magnate (user-decided 2026-08-06):
	# 6 magnates = "heavily-devolved" reference → 12×6 = 72 from the count, + the decay-pressure term once
	# central armies rot → crosses the ~80 separatism-warlord arm floor (se_SEPARATISM). A mildly-devolved
	# realm (1-2 magnates → 12-24) stays well below. WEIGHTED SUM (not max) so a modern national army still
	# suppresses the vector. PLACEHOLDER-playtest the 12.
	set_variable = { name = qing_han_magnate_term_tmp  value = var:qing_han_magnate_tally }
	change_variable = { name = qing_han_magnate_term_tmp  multiply = 12 }
	change_variable = { name = qing_han_prov_target_tmp  add = var:qing_han_magnate_term_tmp }
	# [law #14 bias 2026-07-24] Provincial Militarization law biases the han-provincial-power TARGET BEFORE
	# the clamp (delegated +10 / centralized -10 / sanctioned 0). Biasing the TARGET (not the +/-2 chase
	# nudge) is what gives a durable effect — the meter eases toward this. Guarded → default byte-identical.
	if = { limit = { has_variable = qing_provmil_bias }  change_variable = { name = qing_han_prov_target_tmp  add = var:qing_provmil_bias } }
	if = { limit = { var:qing_han_prov_target_tmp < 0 }    set_variable = { name = qing_han_prov_target_tmp  value = 0 } }
	if = { limit = { var:qing_han_prov_target_tmp > 100 }  set_variable = { name = qing_han_prov_target_tmp  value = 100 } }
	# [meter-concretize #6 / error-logging rule 2026-08-06] report the concrete magnate tally + derived
	# target so the han-provincial-power drift is traceable in debug.log (values decorative per LOG idiom).
	LOG_line = { sys = QING  msg = "han provincial power: magnate tally + decay-pressure -> target (tally=) for" }
	if = {
		limit = { var:qing_han_provincial_power < qing_han_prov_target_tmp_cmpsvalue }
		QING_DECLINE_nudge = { var = qing_han_provincial_power  amount = 2 }
	}
	else_if = {
		limit = { var:qing_han_provincial_power > qing_han_prov_target_tmp_cmpsvalue }
		QING_DECLINE_nudge = { var = qing_han_provincial_power  amount = -2 }
	}
}

# ETHNIC TENSION — DEEP RE-DERIVATION FROM PROVINCE DEMOGRAPHICS (§11 L3, Qing).
#
# The abstract ethnic-tension counter is now largely COMPUTED from the real
# ethnic geography of the realm rather than hand-tuned nudges. It reads the
# GENERIC de jure baseline (se_DEJURE.txt froze each province's rightful 1815
# culture into var:dejure_culture at game start) and measures how much of the
# realm is "restive": land whose rightful people the Qing has NOT integrated.
#
# The manchu-primary / Han-realm subtlety is handled for free: a country's OWN
# primary culture is always integrated, so manchu de jure land is never restive;
# and because we CALIBRATE against the 1815 baseline (the huge static mass of
# un-integrated Han de jure land is the baseline and nets to zero deviation), a
# stable historical Qing sits at the ~20 start value. The counter then MOVES as
# the realm changes: conquering more un-integrated foreign de jure land raises
# it, integrating cultures or losing that land lowers it, and Han settlers
# demographically displacing a frontier people (live plurality != frozen de
# jure) adds a displacement premium — the historical driver of frontier strife.
#
# PERFORMANCE (firm, user): the heavy every_owned_province scan runs ANNUALLY
# (yearly_country_pulse, CHI + human only) and writes a cached target; the
# cheap O(1) quarterly pulse just drifts the live counter toward that target.
# Ethnic tension only moves through slow processes (conquest, integration,
# migration) that do not shift within a year, so annual granularity loses almost
# nothing while cutting the scan cost ~4x versus running it every pulse.
# ======================================================================
QING_DECLINE_scan_ethnic_target = {
	LOG_enter = { sys = QING  fn = "QING_DECLINE_scan_ethnic_target" }

	# Weighted tally of restive land. base 1 per restive province, +1 if actively
	# unhappy (unrest high), +1 if its rightful people have been demographically
	# displaced (live plurality no longer the frozen de jure culture).
	set_variable = { name = qing_ethnic_restive_weighted  value = 0 }
	# [ethnic-gov] Separately track the net governor–subject affinity contribution
	# so the trace can show how much of the tally came from alien/native governors.
	set_variable = { name = qing_ethnic_gov_signal  value = 0 }
	every_owned_province = {
		limit = {
			is_sea = no
			total_population > 0
			has_variable = dejure_culture
		}
		# Pull the frozen rightful culture into a named scope so it can be tested
		# unambiguously across scope changes (a saved scope is globally visible
		# within the effect; a prev-chain into any_country_culture would resolve
		# to the country, where the variable does not live).
		var:dejure_culture = { save_scope_as = dj_rightful_c }
		# restive = the Qing has NOT integrated this land's rightful culture.
		# (manchu, the primary culture, is always integrated -> never restive.)
		if = {
			limit = {
				NOT = {
					ROOT = {
						any_country_culture = {
							is_integrated = yes
							is_culture = scope:dj_rightful_c
						}
					}
				}
			}
			ROOT = { change_variable = { name = qing_ethnic_restive_weighted  add = 1 } }
			# actively restless land weighs more
			if = {
				limit = { province_unrest >= 4 }
				ROOT = { change_variable = { name = qing_ethnic_restive_weighted  add = 1 } }
			}
			# displacement premium: the rightful people are no longer the plurality
			# (e.g. Han settlers swamped a frontier province) — a strife driver.
			if = {
				limit = { NOT = { dominant_province_culture = scope:dj_rightful_c } }
				ROOT = { change_variable = { name = qing_ethnic_restive_weighted  add = 1 } }
			}
			# [ethnic-gov] Governor–subject affinity. Whether the man SET OVER this
			# restive land is one of its people materially shapes the strife: a
			# native governor-general soothes it, an alien overlord inflames it.
			# Concrete, not abstract — this reads the REAL governing character on the
			# map (governorship.governor_or_ruler, which falls back to the ruler for
			# the capital region), not a bookkeeping counter. Three-tier ladder,
			# both directions (culture is the strong signal, culture_group the weak):
			#   exact culture match          -> -2 (a local son; strong soothe)
			#   same group, different culture -> -1 (a kinsman people; mild soothe)
			#   different culture group       -> +1 (an alien overlord; aggravate)
			# Verified idiom (TI/Invictus): province -> governorship -> governor_or_
			# ruler; culture = <culture scope> and culture.culture_group = <..group>.
			governorship = {
				if = {
					limit = { exists = governor_or_ruler }
					governor_or_ruler = {
						if = {
							# strong soothe: the governor IS of the rightful people
							limit = { culture = scope:dj_rightful_c }
							ROOT = {
								change_variable = { name = qing_ethnic_restive_weighted  subtract = 2 }
								change_variable = { name = qing_ethnic_gov_signal       subtract = 2 }
							}
						}
						else_if = {
							# mild soothe: same broad group, different exact culture
							limit = { culture.culture_group = scope:dj_rightful_c.culture_group }
							ROOT = {
								change_variable = { name = qing_ethnic_restive_weighted  subtract = 1 }
								change_variable = { name = qing_ethnic_gov_signal       subtract = 1 }
							}
						}
						else = {
							# alien overlord: a different culture group entirely
							ROOT = {
								change_variable = { name = qing_ethnic_restive_weighted  add = 1 }
								change_variable = { name = qing_ethnic_gov_signal       add = 1 }
							}
						}
					}
				}
			}
		}
	}

	# On the FIRST scan, freeze the 1815 weighted tally as the baseline so the
	# historical realm sits at the ~20 start value and only DEVIATIONS move it.
	if = {
		limit = { NOT = { has_variable = qing_ethnic_restive_base } }
		set_variable = { name = qing_ethnic_restive_base  value = var:qing_ethnic_restive_weighted }
		LOG_line = { sys = QING  msg = "ethnic baseline frozen: restive_base=" }
	}

	# target = 20 + (current restive - baseline restive) / 2, clamped 0..100.
	# Each net-new restive province adds ~0.5 to the tension target.
	set_variable = { name = qing_ethnic_tension_target  value = var:qing_ethnic_restive_weighted }
	change_variable = { name = qing_ethnic_tension_target  subtract = var:qing_ethnic_restive_base }
	change_variable = { name = qing_ethnic_tension_target  divide = 2 }
	change_variable = { name = qing_ethnic_tension_target  add = 20 }
	if = { limit = { var:qing_ethnic_tension_target > 100 }  set_variable = { name = qing_ethnic_tension_target  value = 100 } }
	if = { limit = { var:qing_ethnic_tension_target < 0 }    set_variable = { name = qing_ethnic_tension_target  value = 0 } }

	# [ethnic-gov] Surface the net governor-affinity contribution to the tally so a
	# trace shows how much of this year's tension came from who governs whom.
	LOG_line = { sys = QING  msg = "ethnic scan: restive_weighted= gov_signal=" }

	LOG_exit = { sys = QING  fn = "QING_DECLINE_scan_ethnic_target"  result = OK }
}

# CIVIC ASSIMILATION — the annual nation-building pass (§11 L4 ext).
# Runs once a year, right after the demographic scan (CHI + human), so it reuses
# the same annual cadence and adds NO new polling hook. What a coalescing civic
# nation actually DOES to its ethnic map, in two complementary registers:
#
#   (A) INTEGRATION BRIDGE (civic citizenship — inclusive). Once civic identity is
#       genuinely established (>= 50), the realm extends citizenship to its
#       peoples: it integrates ONE not-yet-integrated country-culture per year

#       year's scanned target. No culture is erased — minorities become citizens.

#
#   (B) SINICIZATION TRICKLE (assimilative). A DISTINCT, popular-level process from
#       the elite Manchu-identity meter (qing_manchu_identity, se_QING_MECHANICS)
#       and the customs-service sinicization (QING_customs_sinicize): here a small,
#       slow fraction of pops in provinces WHERE HAN ARE ALREADY THE PLURALITY but
#       the frozen rightful culture is non-Han (i.e. displacement has already
#       happened, L2) drift to the province's dominant Sinitic culture. Gated to
#       follow demographic reality, never force it — so it complements the bottom-
#       up migration philosophy rather than fighting it. Only fires at high civic
#       identity (>= 70), and touches at most a couple of provinces a year.
#   ROOT / this = the Qing overlord (CHI, human).
# ======================================================================
QING_DECLINE_civic_assimilation = {
	LOG_enter = { sys = QING  fn = "QING_DECLINE_civic_assimilation" }
	if = { limit = { NOT = { has_variable = qing_civic_identity } }  set_variable = { name = qing_civic_identity  value = 0 } }

	# (A) INTEGRATION BRIDGE — one culture per year, once civic identity is established.
	# SCOPE DIRECTION (see the warning at QING_DECLINE_scan_ethnic_target): the frozen
	# rightful culture lives as a variable on the PROVINCE, so we must iterate provinces
	# and pull it into a named scope — a culture scope has no owned-province list, and a
	# prev-chain the other way would resolve to the country. We therefore walk our own
	# restive de jure provinces, and integrate the first un-integrated rightful culture
	# we find. A body-side "done" guard caps it at ONE per year (an every_ limit cannot
	# cap mid-iteration — the candidate list is built once, up front).
	if = {
		limit = { var:qing_civic_identity >= 50 }
		set_variable = { name = qing_civic_bridge_done  value = 0 }
		every_owned_province = {
			limit = {
				has_variable = dejure_culture
				total_population > 0
			}
			if = {
				limit = { ROOT.var:qing_civic_bridge_done = 0 }
				var:dejure_culture = { save_scope_as = civic_candidate_culture }
				# integrate only if this rightful culture is ACTUALLY PRESENT among our
				# pops AND not yet a citizen culture. The positive any_country_culture
				# test does both at once: it only enumerates cultures that exist in the
				# realm, so a fully-displaced rightful people (no pops left — e.g. a
				# frontier swamped by Han settlers) is correctly skipped rather than
				# wasting the year's single integration on an absent culture.
				# (is_integrated = no auto-excludes our own always-integrated primary.)
				if = {
					limit = {
						ROOT = {
							any_country_culture = {
								is_integrated = no
								is_culture = scope:civic_candidate_culture
							}
						}
					}
					ROOT = {
						integrate_country_culture = scope:civic_candidate_culture
						set_variable = { name = qing_civic_bridge_done  value = 1 }
					}
					LOG_line = { sys = QING  msg = "civic: extended citizenship (integrated a restive culture) for (civic=)" }
				}
			}
		}
		if = {
			limit = { var:qing_civic_bridge_done = 0 }
			LOG_fail = { sys = QING  fn = "QING_DECLINE_civic_assimilation"  reason = "integration bridge: no un-integrated culture anchors restive de jure land (all such peoples already citizens)" }
		}
		remove_variable = qing_civic_bridge_done
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_DECLINE_civic_assimilation"  reason = "integration bridge: civic identity below 50, citizenship not yet extended" }
	}

	# (B) SINICIZATION — a demographic MELTING POT driven by MAJORITY WEIGHT, not by
	# strife (user, firm): even a HAPPILY-INTEGRATED minority pop assimilates under
	# enough surrounding majority pressure — the more dominant the local Sinitic
	# plurality, the faster minorities merge into it. This is the popular, bottom-up
	# counterpart to the elite Manchu-identity meter (qing_manchu_identity) and the
	# customs-service sinicization; it deliberately does NOT filter on unrest or pop
	# happiness. The national project (high civic identity) is what mobilises common
	# schooling/print/administration into assimilation pressure, so it is gated on
	# civic identity >= 70; WITHIN that, INTENSITY scales with the majority's weight.
	if = {
		limit = { var:qing_civic_identity >= 70 }
		# sentinel so we can LOG_fail the "gate passed but no province matched" no-op
		# path (standing rule), mirroring part (A)'s qing_civic_bridge_done.
		set_variable = { name = qing_civic_sinic_done  value = 0 }
		# Touch the several most-populous provinces where a Sinitic culture already
		# holds a majority. Assimilation follows demography — it never manufactures a
		# majority that isn't there (so it complements L2 bottom-up migration).
		ordered_owned_province = {
			order_by = total_population
			max = 4
			limit = {
				is_sea = no
				total_population > 0
				# [F6 boot-test fix] was `dominant_province_culture.culture_group = chinese_group` — that
				# province-level dot-chain fails to parse ("Illegal use of operator ="); the proven province
				# idiom is the dedicated keyword dominant_province_culture_group (oracle: Invictus
				# 00_steppe_inv.txt:247). This is the ordered_owned_province scope (a province), so it applies directly.
				dominant_province_culture_group = chinese_group
				# a genuine majority of the province is already Sinitic (>= 50%).
				# (any_pops_in_province takes its match conditions FLAT beside
				# percent/count — NO limit = {} wrapper; proven decentralize_realm_button.)
				any_pops_in_province = {
					percent >= 0.5
					pop_culture_group = chinese_group
				}
				# and at least one ELIGIBLE non-Sinitic pop remains to assimilate. This
				# must match QING_civic_assimilate_one_pop's own eligibility (non-slave,
				# non-noble) so the outer gate never admits a province the helper will
				# no-op on (which would fire a misleading success LOG_line).
				any_pops_in_province = {
					count >= 1
					NOT = { pop_type = slaves }
					NOT = { pop_type = nobles }
					NOT = { pop_culture_group = chinese_group }
				}
			}
			save_scope_as = civic_sinic_province
			# set the sentinel on ROOT (country) — the loop body scope is the PROVINCE.
			ROOT = { set_variable = { name = qing_civic_sinic_done  value = 1 } }
			# Base assimilation: one non-Sinitic pop merges into the local Sinitic
			# plurality culture. set_pop_culture is the proven engine effect. Happy or
			# not — majority weight, not grievance, drives this.
			QING_civic_assimilate_one_pop = yes
			# Majority-weight INTENSITY: an OVERWHELMING Sinitic majority (>= 75%)
			# exerts stronger pull — a second pop assimilates the same year.
			if = {
				limit = {
					scope:civic_sinic_province = {
						any_pops_in_province = {
							percent >= 0.75
							pop_culture_group = chinese_group
						}
					}
				}
				QING_civic_assimilate_one_pop = yes
			}
			LOG_line = { sys = QING  msg = "civic: sinicization melting-pot in for" }
		}
		if = {
			limit = { var:qing_civic_sinic_done = 0 }
			LOG_fail = { sys = QING  fn = "QING_DECLINE_civic_assimilation"  reason = "sinicization: no majority-Sinitic province with an assimilable (non-slave/non-noble) minority left" }
		}
		remove_variable = qing_civic_sinic_done
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_DECLINE_civic_assimilation"  reason = "sinicization: civic identity below 70, no assimilation pressure mobilised" }
	}
	LOG_exit = { sys = QING  fn = "QING_DECLINE_civic_assimilation"  result = OK }
}

# Assimilate ONE non-Sinitic, non-slave pop in scope:civic_sinic_province into the
# province's dominant (Sinitic) culture. Split out so the majority-weight intensity
# logic can call it once or twice. Guarded: no-op + LOG_fail if no eligible pop.
QING_civic_assimilate_one_pop = {
	if = {
		limit = {
			scope:civic_sinic_province = {
				any_pops_in_province = {
					count >= 1
					NOT = { pop_type = slaves }
					NOT = { pop_type = nobles }
					NOT = { pop_culture = scope:civic_sinic_province.dominant_province_culture }
				}
			}
		}
		scope:civic_sinic_province = {
			random_pops_in_province = {
				limit = {
					NOT = { pop_type = slaves }
					NOT = { pop_type = nobles }
					NOT = { pop_culture = scope:civic_sinic_province.dominant_province_culture }
				}
				set_pop_culture = scope:civic_sinic_province.dominant_province_culture
			}
		}
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_civic_assimilate_one_pop"  reason = "no eligible non-Sinitic freemen/townsfolk pop left to assimilate in the province" }
	}
}

# Drift the live ethnic-tension counter toward the annually-scanned demographic
# target, by +/-3 per quarterly pulse. Cheap O(1). The stance drift
# (QING_ethnic_stance_drift) still nudges on top as a player lever, so good
# governance (dyarchy) can hold tension below the raw demographic baseline.
#
# CIVIC RELIEF (§11 L4 ext, "strong relief" per design): a shared civic national
# identity lets the SAME ethnic geography produce LESS separatist strife — Bretons
# became Frenchmen. We subtract a civic-relief term from the EFFECTIVE tension
# target the counter drifts toward, so a high-civic realm can be pulled well below
# the raw demographic baseline and largely escape L4 separatism (SEPARATISM_qing_
# pulse gates on tension >= 45). Relief = civic_identity / 2, so a maxed civic
# identity (100) can shave up to 50 points off the demographic target. Nation-
# building is thus a real, powerful answer to the separatist threat the L1-4
# program creates. The relief is computed into a scratch var each pulse (never
# mutating the scanned target itself, which the annual scan owns).
QING_DECLINE_drift_ethnic_tension = {
	if = {
		limit = { NOT = { has_variable = qing_ethnic_tension_target } }
		# scan has not run yet (start of game) — nothing to drift toward.
		LOG_fail = { sys = QING  fn = "QING_DECLINE_drift_ethnic_tension"  reason = "no demographic target yet; annual scan not run" }
	}
	else = {
		# effective target = scanned demographic target - civic relief, clamped 0..100.
		set_variable = { name = qing_ethnic_eff_target  value = var:qing_ethnic_tension_target }
		if = { limit = { NOT = { has_variable = qing_civic_identity } }  set_variable = { name = qing_civic_identity  value = 0 } }
		set_variable = { name = qing_civic_relief_tmp  value = var:qing_civic_identity }
		change_variable = { name = qing_civic_relief_tmp  divide = 2 }
		change_variable = { name = qing_ethnic_eff_target  subtract = var:qing_civic_relief_tmp }
		if = { limit = { var:qing_ethnic_eff_target < 0 }    set_variable = { name = qing_ethnic_eff_target  value = 0 } }
		if = { limit = { var:qing_ethnic_eff_target > 100 }  set_variable = { name = qing_ethnic_eff_target  value = 100 } }
		if = {
			limit = { var:qing_ethnic_tension < qing_ethnic_eff_target_cmpsvalue }
			QING_DECLINE_nudge = { var = qing_ethnic_tension  amount = 3 }
		}
		else_if = {
			limit = { var:qing_ethnic_tension > qing_ethnic_eff_target_cmpsvalue }
			QING_DECLINE_nudge = { var = qing_ethnic_tension  amount = -3 }
		}
	}
}

# ======================================================================
# CIVIC (NATIONAL) IDENTITY — the positive nation-building meter (§11 L4 ext).
#
# Drifts toward a target derived from Self-Strengthening progress: the modern
# schools, telegraph and national press that the movement builds ARE the
# connective tissue that forges a civic nation out of a multiethnic empire, so
# civic identity CANNOT outrun self-strengthening. The inclusive dyarchy stance
# accelerates it (shared citizenship); ethnic favouritism (banner/han) caps it
# low. Hollow modernization (refraction flag) dampens the target — a nation built
# on brittle for-show institutions does not cohere.
#   target = selfstr_progress, then:
#     - if the movement is hollow, halve it (brittle institutions bind no nation);
#     - +15 ceiling lift while the dyarchy stance is held;
#     - -20 cap while a favouritist stance (banner/han) is held.
#   clamped 0..100. Drifts +/-2 per quarterly pulse (slower than tension: a
#   national identity is built over generations).
# ======================================================================
QING_DECLINE_drift_civic_identity = {
	if = { limit = { NOT = { has_variable = qing_civic_identity } }  set_variable = { name = qing_civic_identity  value = 0 } }
	QING_selfstr_init = yes
	# base ceiling = self-strengthening progress.
	set_variable = { name = qing_civic_target_tmp  value = var:qing_selfstr_progress }
	# hollow modernization forges no nation — halve the ceiling.
	if = {
		limit = { has_variable = qing_selfstr_hollow_flag  var:qing_selfstr_hollow_flag = 1 }
		change_variable = { name = qing_civic_target_tmp  divide = 2 }
	}
	# inclusive rule teaches shared citizenship; favouritism forecloses it.
	if = {
		limit = { var:qing_ethnic_stance_active = flag:dyarchy }
		change_variable = { name = qing_civic_target_tmp  add = 15 }
	}
	else_if = {
		limit = {
			OR = {
				var:qing_ethnic_stance_active = flag:banner
				var:qing_ethnic_stance_active = flag:han
			}
		}
		change_variable = { name = qing_civic_target_tmp  subtract = 20 }
	}
	# [P1 2026-07-24] the frontier-settlement arcs raise the civic-identity CEILING durably:
	# forging the nation out of the settled frontier peoples lifts the target the meter drifts to,
	# so the gain persists (a one-shot nudge on the meter would decay back — the no-restoring-drift
	# ratchet rule). qing_civic_identity_settle_bonus is accrued by QING_settle_forge_nation
	# (+8/arc, capped 40 there). Guarded read (may be unset before any arc completes).
	if = {
		limit = { has_variable = qing_civic_identity_settle_bonus }
		change_variable = { name = qing_civic_target_tmp  add = var:qing_civic_identity_settle_bonus }
	}
	# [law #42 bias 2026-07-24] National Integration law lifts the civic-identity TARGET durably
	# (civic-nation +10 / multi-ethnic +5 / dynastic 0), alongside the settle-bonus above — a persistent
	# ceiling lift (ratchet-safe; a meter nudge would decay). Guarded → default (unset) byte-identical.
	if = { limit = { has_variable = qing_civic_law_bias }  change_variable = { name = qing_civic_target_tmp  add = var:qing_civic_law_bias } }
	if = { limit = { var:qing_civic_target_tmp < 0 }    set_variable = { name = qing_civic_target_tmp  value = 0 } }
	if = { limit = { var:qing_civic_target_tmp > 100 }  set_variable = { name = qing_civic_target_tmp  value = 100 } }
	# drift toward it, slowly.
	if = {
		limit = { var:qing_civic_identity < qing_civic_target_tmp_cmpsvalue }
		QING_DECLINE_nudge = { var = qing_civic_identity  amount = 2 }
	}
	else_if = {
		limit = { var:qing_civic_identity > qing_civic_target_tmp_cmpsvalue }
		QING_DECLINE_nudge = { var = qing_civic_identity  amount = -2 }
	}
}

# ======================================================================
# THE PULSE ENTRY POINT
# One call, gated CHI/human/throttled by 00_monthly_country.txt. Refreshes the
# derived counters and re-applies every band. All O(1).
# ======================================================================
QING_DECLINE_pulse = {
	LOG_enter = { sys = QING  fn = "QING_DECLINE_pulse" }

	QING_DECLINE_init = yes

	# Banner decay creeps up with time (the martial caste rots); reform drills
	# and events push back. Small monthly creep.
	# task #88 (user): a FULLY MODERNIZED realm (Self-Strengthening capstone) has
	# escaped the dynastic cycle — the passive martial-caste rot no longer creeps.
	# Player choices can still move the meters, but they no longer regenerate on
	# their own. Guarded on the qing_fully_modernized flag set by the capstone.
	# [bookmark-1763 #304] the passive martial-caste rot is ALSO suppressed during the High
	# Qing era (pre-1775 zenith): the Banners had just destroyed the Zunghars (1759) and the
	# Green Standard held a realm at peace, so neither creeps until the Heshen inflection
	# (qing_decline.40) ends the era. The 1815 start has no qing_high_qing_era flag, so it
	# rots as before.
	# [meter-concretize #9 2026-08-06] BANNER/GS DECAY is now DERIVED from garrison-commander rot (廢弛): the
	# targets qing_banner_decay_target / qing_greenstandard_decay_target = base + K×(avg commander corruption
	# + (10 − avg commander martial)) are computed in QING_council_recompute (se_QING_COUNCIL.txt). The old
	# passive +1/+1 creep + the GS-accel are DELETED — the rot IS the decay (USER), not a synthetic creep on
	# top. Drift the live counters ±2/pulse toward the targets. Gated NOT fully_modernized AND NOT
	# high_qing_era (mirrors the old creep gate — protects the self-str capstone hard-set-to-5 and the calm
	# zenith). Target existence guarded (recompute has run). NOTE (USER ruling b): nothing artificially drives
	# commander corruption up — a clean corps keeps decay low; a dedicated rot-driver mechanic comes later.
	if = {
		limit = {
			NOT = { has_variable = qing_fully_modernized }
			NOT = { has_variable = qing_high_qing_era }
		}
		if = {
			limit = { has_variable = qing_banner_decay_target }
			if = {
				limit = { var:qing_banner_decay < qing_banner_decay_target_cmpsvalue }
				QING_DECLINE_nudge = { var = qing_banner_decay  amount = 2 }
			}
			else_if = {
				limit = { var:qing_banner_decay > qing_banner_decay_target_cmpsvalue }
				QING_DECLINE_nudge = { var = qing_banner_decay  amount = -2 }
			}
		}
		if = {
			limit = { has_variable = qing_greenstandard_decay_target }
			if = {
				limit = { var:qing_greenstandard_decay < qing_greenstandard_decay_target_cmpsvalue }
				QING_DECLINE_nudge = { var = qing_greenstandard_decay  amount = 2 }
			}
			else_if = {
				limit = { var:qing_greenstandard_decay > qing_greenstandard_decay_target_cmpsvalue }
				QING_DECLINE_nudge = { var = qing_greenstandard_decay  amount = -2 }
			}
		}
	}

	# [meter-concretize #9] NOTE: the military-upkeep law-bias (qing_banner_upkeep_bias / _greenstandard_) is
	# now folded into the decay TARGET inside QING_council_recompute (se_QING_COUNCIL.txt), NOT nudged here —
	# a per-pulse counter nudge would be washed out by the ±2 drift (the Task #32 dead-law flaw), and adding
	# it to the target in THIS pulse would be clobbered by the recompute's set_variable next quarter. Biasing
	# the target at its compute site gives the law a durable, drift-respected effect.

	# task #88 — the modern national-army share drifts toward its self-strengthening
	# ceiling FIRST, so the Han-provincial derivation below reads a fresh value (a
	# modern national army is what suppresses the warlord vector).
	QING_DECLINE_drift_modernarmy_share = yes
	# task #88 — Han provincial military power is DERIVED from the two central-army
	# decay meters minus the modern-army share: it rises only when the centre's own
	# armies have failed AND no modern national army has replaced them. Runs AFTER
	# both inputs are fresh this pulse.
	QING_DECLINE_drift_han_provincial_power = yes

	# [#366] The Silver/Opium trade-balance assessment runs BEFORE update_currency_stress, so
	# the net-silver-flow nudge (tea/silk/porcelain export inflow minus the opium-import drain)
	# and the reserve-ratio drift below COMPOUND into qing_currency_stress in the same pulse.
	# A High-Qing trade surplus bleeds stress down; the 19thC opium drain drives it up — and via
	# the existing term (d) of QING_ministry_recompute_perf_revenue that stress folds straight
	# into the Board of Revenue minister's Grand Council standing. (se_QING_OPIUM.txt)
	QING_opium_assess_trade_balance = yes
	# [#67] PAPER-MONEY DEPRECIATION doom-loop — update the meter BEFORE the stress recompute (so this pulse's
	# stress + next month's mint allowance read the fresh value). While the paper regime is active, depreciation
	# rises with real OVER-ISSUE (how much of the paper headroom the player actually minted); when the regime is
	# off, it decays slowly (trust rebuilds; scarring on re-entry). All arithmetic clamped 0..100.
	if = {
		limit = { has_variable = qing_paper_regime_active }
		# over_issue = minted rate ABOVE the non-paper baseline cap = the portion of the paper allowance used.
		# = CURRENCY_minting_rate − (CURRENCY_minting_rate_cap − paper_allowance), clamped >= 0. RHS-safe (temps).
		set_variable = { name = qing_paper_over_tmp  value = CURRENCY_minting_rate_cap }
		change_variable = { name = qing_paper_over_tmp  subtract = CURRENCY_qing_paper_mint_allowance }   # = baseline (non-paper) cap
		set_variable = { name = qing_paper_used_tmp  value = var:CURRENCY_minting_rate }
		change_variable = { name = qing_paper_used_tmp  subtract = var:qing_paper_over_tmp }              # minted above baseline = paper used
		if = {
			limit = { var:qing_paper_used_tmp > 0 }
			# rise scaled by how hard the allowance is leaned on (relative to the reserve → a ratio); accelerates
			# as depreciation is already high (non-linear confidence cliff): rise = used/reserve × (1 + dep/50) × k.
			set_variable = { name = qing_paper_rise_tmp  value = var:qing_paper_used_tmp }
			if = { limit = { has_variable = silver_reserve_size  var:silver_reserve_size > 0 }  change_variable = { name = qing_paper_rise_tmp  divide = var:silver_reserve_size } }
			change_variable = { name = qing_paper_rise_tmp  multiply = 40 }   # k — base rise per full-reserve over-issue (best-guess; tune on boot)
			# non-linear: multiply by (1 + dep/50) so a deep depreciation rises faster (the cliff).
			set_variable = { name = qing_paper_accel_tmp  value = var:qing_paper_depreciation }
			change_variable = { name = qing_paper_accel_tmp  divide = 50 }
			change_variable = { name = qing_paper_accel_tmp  add = 1 }
			change_variable = { name = qing_paper_rise_tmp  multiply = var:qing_paper_accel_tmp }
			change_variable = { name = qing_paper_depreciation  add = var:qing_paper_rise_tmp }
			remove_variable = qing_paper_rise_tmp
			remove_variable = qing_paper_accel_tmp
		}
		else = {
			# regime active but restrained (minting at/below baseline) — depreciation eases modestly.
			if = { limit = { var:qing_paper_depreciation > 0 }  change_variable = { name = qing_paper_depreciation  subtract = 2 } }
		}
		remove_variable = qing_paper_over_tmp
		remove_variable = qing_paper_used_tmp
	}
	else = {
		# regime OFF — trust rebuilds slowly toward 0 (scarring: re-entry starts pre-depreciated).
		if = { limit = { has_variable = qing_paper_depreciation  var:qing_paper_depreciation > 0 }  change_variable = { name = qing_paper_depreciation  subtract = 3 } }
	}
	if = { limit = { has_variable = qing_paper_depreciation  var:qing_paper_depreciation < 0 }   set_variable = { name = qing_paper_depreciation  value = 0 } }
	if = { limit = { has_variable = qing_paper_depreciation  var:qing_paper_depreciation > 100 } set_variable = { name = qing_paper_depreciation  value = 100 } }
	# [#67] LOG the paper-money meter each pulse the regime is active OR the meter is still scarring down — so the
	# boot surfaces the doom-loop (STATIC label, no macro/# in string; -debug_mode via LOG_line). Guarded so a
	# non-paper realm (the 1763 baseline + every non-crisis game) stays silent — no flood.
	if = {
		limit = { OR = { has_variable = qing_paper_regime_active  AND = { has_variable = qing_paper_depreciation  var:qing_paper_depreciation > 0 } } }
		LOG_line = { sys = QING  msg = "paper-money regime: depreciation meter + mint allowance recomputed for" }
	}
	# [meter-concretize #3] REPLACES QING_DECLINE_update_currency_stress (the deleted 1-bit drift wrapper).
	# Runs AFTER the opium assessment so it reads this pulse's fresh qing_opium_stress_flow. Two-store

						# ignition path: when the dynasty's mandate has visibly failed (< 30), the millenarian
						# panic needs no other spark. This is the consumer that gives the mandate meter teeth —
						# a court that neglects the 祭天 rites (which raise it) courts the archetypal 邪教 rising.

#   has_state_food          — a state's CURRENT stored food (governor_policies:154; TI svalues)
#   has_state_food_capacity — a state's MAX food storage        (00_governor_policies:140; TI)
#   add_state_food = <v>    — move food into (+) / out of (-) a state (TI christianity/disasters_inv)
#   every_country_state     — iterate the country's states       (TI, 128 uses)
#
# State (on CHI):
#   qing_granary_food  — the pooled grain in REAL food units. Capped at qing_granary_count*200 (the
#                        summed local_food_capacity of the actual ever-normal granary BUILDINGS —
#                        qing_granary_count is kept fresh by the Works sweep, se_QING_MINISTRY:497).
#   qing_granary_stock — KEPT: now DERIVED = qing_granary_food / cap * 100 (a 0..100 fill-ratio), so
#                        every existing consumer (pop-pressure target, famine gate, ministry perf,
#                        canal) keeps working — the number is now a real fill %, not a free counter.
#
# FILL: skim 1/20 of capacity from each NEAR-MAXED state (has_state_food >= 90% of capacity — a
#   near-max test, NOT exact ==, since vanilla food may plateau just below the ceiling and never
#   hit it exactly). The skimmed food leaves the province (add_state_food = -skim) and enters the
#   pool (up to the cap — grain the granaries cannot hold is left in the province, not conjured).
# SHIP: to each state in real shortage (has_state_food < 1/5 capacity) while the pool holds grain,
#   send a relief shipment (add_state_food = +ship), drain the pool, and stamp qing_grain_relief.
#   The pool drains as it ships, so a widespread (national) famine empties it before covering all
#   the stricken states — only a localized dearth is fully relieved (the historical 常平倉 limit).
# Scope: COUNTRY (CHI). One every_country_state sweep (fill+ship in the same pass). ROOT = CHI.
# ======================================================================
QING_DECLINE_granary_pool = {
	LOG_enter = { sys = QING  fn = "QING_DECLINE_granary_pool" }
	# [PERF-AUDIT 2026-07-18 M2] GATE this whole-CHI every_country_state grain sweep to a ~180-day cadence.
	# It formerly ran every DECLINE pulse (~90d); the ever-normal reserve moves grain slowly and the derived
	# qing_granary_stock / _capacity vars PERSIST in the save between runs, so every consumer (famine gate,
	# pop-pressure, ministry perf, canal, GUI) keeps reading the last-computed values — a half-year cadence
	# halves the sweep cost with only a modest lag in relief timing. Self-throttle cooldown idiom.
	if = {
		limit = { NOT = { has_variable = qing_granary_pool_cd } }
		set_variable = { name = qing_granary_pool_cd  days = 180 }
	# the reserve CAP = the real ever-normal granary capacity: count * the building's 200 food_capacity.
	# qing_granary_count is maintained by the Works sweep (se_QING_MINISTRY:497). [review-fix] that sweep
	# runs in QING_GOV_pulse, which fires AFTER QING_DECLINE_pulse in 00_monthly_country.txt — so this
	# reads LAST quarter's count (a harmless one-quarter lag; the count changes at most by the ~1 granary
	# the Revenue path builds per quarter). Default 0 until the count first exists / no granaries built.
	set_variable = { name = qing_granary_cap_tmp  value = 0 }
	if = {
		limit = { has_variable = qing_granary_count }
		set_variable = { name = qing_granary_cap_tmp  value = var:qing_granary_count }
		change_variable = { name = qing_granary_cap_tmp  multiply = 200 }
	}
	# init the pooled grain ONCE to EMPTY. [review-fix] no game-start granaries exist in any setup file,
	# so the ever-normal reserve genuinely begins empty and fills only as granaries are built and skim
	# real surplus (the old "half-full bootstrap" was dead anyway — cap is 0 on the first pulse before
	# any granary). While cap==0 the derive-block below LEAVES qing_granary_stock at its DECLINE_init
	# baseline (30), so an as-yet-granaryless realm is NOT false-flagged into famine.
	if = { limit = { NOT = { has_variable = qing_granary_food } }  set_variable = { name = qing_granary_food  value = 0 } }

	# --- FILL from near-maxed surplus states + SHIP to shortage states, one real-food sweep. ------
	every_country_state = {
		# save the state so the pool write below reads the skim/ship amount back via the PROVEN
		# scope:X.var:Y idiom (upstream TI/Invictus use scope:X.var: heavily; prev.var: is NOT attested).
		save_scope_as = qing_gran_state
		# [logfix 2026-07-16] Precompute EVERY comparison operand into a plain STATE var first. A trigger
		# (has_state_food_capacity) or a cross-scope ROOT.var: inside a comparison RHS { value = } block,
		# evaluated in this every_country_state scope, floods the live 1763 log with "Unsupported scope
		# operator '<'" / "Unknown trigger type: value" — so the FILL/SHIP limits NEVER passed and the
		# ever-normal pool never moved grain (the granary relief was dead). Comparing a plain state var to
		# { value = <state var> } in the SAME scope is the proven-silent form (se_QING_COUNCIL/GOVERNANCE,
		# 0 log hits). Reading has_state_food(_capacity)/ROOT.var: into set_variable is fine — only the
		# comparison RHS is the problem. These temps are re-set each state and removed at the loop end.
		set_variable = { name = qing_gran_cap_tmp      value = has_state_food_capacity }
		set_variable = { name = qing_gran_food_tmp     value = has_state_food }
		set_variable = { name = qing_gran_fill_thr     value = has_state_food_capacity }
		change_variable = { name = qing_gran_fill_thr  multiply = 0.9 }
		set_variable = { name = qing_gran_ship_thr     value = has_state_food_capacity }
		change_variable = { name = qing_gran_ship_thr  multiply = 0.2 }
		set_variable = { name = qing_gran_pool_tmp     value = ROOT.var:qing_granary_food }
		set_variable = { name = qing_gran_poolcap_tmp  value = ROOT.var:qing_granary_cap_tmp }
		# FILL: this state's stores are near-max (>= 90% capacity), so its surplus would be wasted —
		# skim 1/20 of capacity into the national pool, but only while the pool has room under the cap.
		if = {
			limit = {
				var:qing_gran_cap_tmp > 0
				var:qing_gran_food_tmp >= qing_gran_fill_thr_cmpsvalue
				var:qing_gran_pool_tmp < qing_gran_poolcap_tmp_cmpsvalue
			}
			set_variable = { name = qing_gran_skim_tmp  value = has_state_food_capacity }
			change_variable = { name = qing_gran_skim_tmp  divide = 20 }
			# do not overfill the pool past its cap: clamp the skim to the remaining headroom.
			set_variable = { name = qing_gran_room_tmp  value = ROOT.var:qing_granary_cap_tmp }
			change_variable = { name = qing_gran_room_tmp  subtract = ROOT.var:qing_granary_food }
			if = { limit = { var:qing_gran_skim_tmp > qing_gran_room_tmp_cmpsvalue }  set_variable = { name = qing_gran_skim_tmp  value = var:qing_gran_room_tmp } }
			if = {
				limit = { var:qing_gran_skim_tmp > 0 }
				add_state_food = { value = var:qing_gran_skim_tmp  multiply = -1 }   # leaves the province
				ROOT = { change_variable = { name = qing_granary_food  add = scope:qing_gran_state.var:qing_gran_skim_tmp } }   # into the pool
			}
			remove_variable = qing_gran_skim_tmp
			remove_variable = qing_gran_room_tmp
		}
		# SHIP: this state is in real shortage (< 1/5 capacity) — if the pool holds grain, send relief.
		# (same precomputed-operand form as the FILL leg above — see the logfix note.)
		else_if = {
			limit = {
				var:qing_gran_cap_tmp > 0
				var:qing_gran_food_tmp < qing_gran_ship_thr_cmpsvalue
				var:qing_gran_pool_tmp > 0
			}
			# ship up to 1/10 of the state's capacity, bounded by what the pool actually holds.
			set_variable = { name = qing_gran_ship_tmp  value = has_state_food_capacity }
			change_variable = { name = qing_gran_ship_tmp  divide = 10 }
			if = { limit = { var:qing_gran_ship_tmp > qing_gran_pool_tmp_cmpsvalue }  set_variable = { name = qing_gran_ship_tmp  value = var:qing_gran_pool_tmp } }   # [logfix] was ROOT.var: on the RHS (cross-scope in state ctx — illegal); use the precomputed pool temp
			if = {
				limit = { var:qing_gran_ship_tmp > 0 }
				add_state_food = { value = var:qing_gran_ship_tmp }                            # grain arrives (block form for parity with the FILL leg; bare = var:X is unattested)
				ROOT = { change_variable = { name = qing_granary_food  subtract = scope:qing_gran_state.var:qing_gran_ship_tmp } }   # drawn from the pool
				# stamp the relief on the state's provinces (real food + province loyalty + happiness +
				# unrest reduction for the duration — see the qing_grain_relief modifier). Only on a FRESH
				# grant (the NOT guard) so it does not re-stack while the shipment stands. Count fresh grants
				# into a state var for the one-time governor-loyalty bump below.
				set_variable = { name = qing_gran_fresh_tmp  value = 0 }
				every_state_province = {
					limit = { NOT = { has_province_modifier = qing_grain_relief } }
					add_province_modifier = { name = qing_grain_relief  duration = 365 }
					scope:qing_gran_state = { change_variable = { name = qing_gran_fresh_tmp  add = 1 } }
				}
				# GOVERNOR LOYALTY: governor_loyalty is a COUNTRY-scope modifier key (not valid on a
				# province modifier), so the fed region's governor is thanked with a one-time loyalty
				# bump instead — the throne's grain kept his province quiet. Fires ONLY on a fresh grant
				# (qing_gran_fresh_tmp > 0). [review-fix] the governor is reached THROUGH A PROVINCE
				# (province.governor is the PROVEN link — imp19c se_ARMY:60 scope:arm_province.governor,
				# TI syracuse_province.governor; `governor` from STATE scope is NOT attested). Take the
				# first owned province and bump its governor once (all provinces in a state share one).
				if = {
					limit = { var:qing_gran_fresh_tmp > 0 }
					random_state_province = {
						limit = { exists = governor }
						governor = { add_loyalty = loyalty_qing_commended }
					}
				}
				remove_variable = qing_gran_fresh_tmp
			}
			remove_variable = qing_gran_ship_tmp
		}
		# [logfix 2026-07-16] clear the per-state comparison temps (set at the top of every iteration).
		remove_variable = qing_gran_cap_tmp
		remove_variable = qing_gran_food_tmp
		remove_variable = qing_gran_fill_thr
		remove_variable = qing_gran_ship_thr
		remove_variable = qing_gran_pool_tmp
		remove_variable = qing_gran_poolcap_tmp
	}

	# --- clamp the pool [0, cap] and DERIVE the 0..100 stock ratio every consumer reads. ----------
	# [#95 slice 5] extracted to the shared QING_DECLINE_granary_rederive helper (below) so the player
	# levers (invest/release/relief) that now change the real qing_granary_food refresh the cache the same
	# way. It recomputes cap_tmp = count×200 (identical to the top-of-effect compute), clamps food to
	# [0,cap], derives qing_granary_stock (cap>0 only; else leaves the baseline 30 — the granaryless
	# invariant), and persists qing_granary_capacity for the GUI. Behaviour-identical to the old inline tail.
	QING_DECLINE_granary_rederive = yes
	LOG_line = { sys = QING  msg = "granary: ever-normal pool moved real food; stock derived from pooled grain / capacity for" }
	}   # [M2] close the ~180-day cadence gate opened at the top of the effect.
	LOG_exit = { sys = QING  fn = "QING_DECLINE_granary_pool"  result = OK }
}

# ======================================================================
# [#95 slice 5] QING_DECLINE_granary_rederive — Scope: COUNTRY (CHI). Recompute the derived
# qing_granary_stock (0..100) + qing_granary_capacity from the REAL pooled food + granary count,
# WITHOUT running the whole every_country_state fill/ship sweep. Behaviour-identical to the pool
# effect's clamp+derive tail (:2490-2506) — extracted so the player levers (invest / release /
# relief), which now change the real qing_granary_food, can refresh the cache IMMEDIATELY instead
# of waiting on (and being reverted by) the 180-day pool sweep. cap = qing_granary_count × 200
# (the building's local_food_capacity); while cap==0 the derive is skipped and qing_granary_stock
# is LEFT at its DECLINE_init baseline (30) — an as-yet-granaryless realm is not false-flagged into
# famine (the same invariant the pool guards). qing_granary_cap_tmp_cmpsvalue already exists
# (00_event_values.txt:1881). ROOT = CHI.
# ======================================================================
QING_DECLINE_granary_rederive = {
	set_variable = { name = qing_granary_cap_tmp  value = 0 }
	if = {
		limit = { has_variable = qing_granary_count }
		set_variable = { name = qing_granary_cap_tmp  value = var:qing_granary_count }
		change_variable = { name = qing_granary_cap_tmp  multiply = 200 }
	}
	if = { limit = { NOT = { has_variable = qing_granary_food } }  set_variable = { name = qing_granary_food  value = 0 } }
	# clamp the pooled food to [0, cap] then derive the 0..100 ratio (cap>0 only; else leave baseline 30).
	if = { limit = { var:qing_granary_food > qing_granary_cap_tmp_cmpsvalue }  set_variable = { name = qing_granary_food  value = var:qing_granary_cap_tmp } }
	if = { limit = { var:qing_granary_food < 0 }  set_variable = { name = qing_granary_food  value = 0 } }
	if = {
		limit = { var:qing_granary_cap_tmp > 0 }
		set_variable = { name = qing_granary_stock  value = var:qing_granary_food }
		change_variable = { name = qing_granary_stock  multiply = 100 }
		change_variable = { name = qing_granary_stock  divide = var:qing_granary_cap_tmp }
	}
	set_variable = { name = qing_granary_capacity  value = var:qing_granary_cap_tmp }
	remove_variable = qing_granary_cap_tmp
}

# ======================================================================
# [#91 item K]  GRANARY STOCK -> REAL 糧倉 BUILDINGS in the Yellow River basin.
# The abstract qing_granary_stock counter (se_QING_DECLINE.txt:25,66) is KEPT as
# the summary/AI layer; this hangs the CONCRETE object off it. In a good year with
# reserves high (stock >= 60) the throne raises a real qing_granary_building
# (defined in common/buildings/qing_granary_buildings.txt) in a Yellow River (黃河)
# region province that lacks one; in a famine with stock exhausted (stock <= 15)
# the granaries are eaten — remove one building level. The Yellow River basin is
# the regions Zhili / Shandong / Henan (the historical 常平倉 heartland; see the
# flood event .12). add_building_level / remove_building_level are proven engine
# effects (QING_selfstr_build; TI 05_joseon_mission remove_building_level).
# Cheap: ordered_owned_province capped at 1, gated on the stock bands.
#   ROOT / this = the Qing overlord (CHI, human).
# ======================================================================
QING_DECLINE_granary_concrete = {
	LOG_enter = { sys = QING  fn = "QING_DECLINE_granary_concrete" }
	if = {
		# GOOD YEAR, reserves high: build ONE new granary where the basin lacks one.
		# [fix 2026-08-14] BUILD specifically throttled to ~3 years (1095 days, best-guess/boot-
		# tunable). QING_DECLINE_granary_concrete itself only runs on the ~90-day governance pulse
		# already (qing_mechanics_pulse_cooldown), but with MANY Yellow River basin cities lacking a
		# granary at a 1763 start, this branch matched and built a new one on EVERY one of those
		# quarterly firings until the backlog was exhausted -- reported as "auto-built very quickly,
		# every quarter". A real institution-building programme, not a market reaction, so a slower
		# multi-year cadence fits better than the governance pulse's own rate.
		# [review-fix, same date] the cooldown must gate ONLY this build sub-block. An earlier draft
		# put it in this outer limit, which also silently gated the RESTOCK block below (a good-year
		# stock recovery would then fail to lift qing_granary_empty for up to 3 years, leaving healthy
		# granaries wrongly flagged empty) -- restock is now its own sibling `if` below, ungated by
		# this cooldown, exactly as the original code intended.
		limit = {
			NOT = { has_variable = qing_granary_build_cd }
			var:qing_granary_stock >= 60
			treasury >= 255   # [#92] the state-funded expansion must be affordable (proven idiom se_QING_CARAVAN:482); else it waits
			any_owned_province = {
				has_city_status = yes
				OR = { is_in_region = Zhili  is_in_region = Shandong  is_in_region = Henan }
				NOT = { has_building = qing_granary_building }
			}
		}
		set_variable = { name = qing_granary_build_cd  days = 1095 }
		ordered_owned_province = {
			order_by = total_population
			max = 1
			limit = {
				has_city_status = yes
				OR = { is_in_region = Zhili  is_in_region = Shandong  is_in_region = Henan }
				NOT = { has_building = qing_granary_building }
			}
			add_building_level = qing_granary_building
			save_scope_as = qing_gran_built_province   # [#92] the funded province, for the notification event loc
			LOG_line = { sys = QING  msg = "granary: real granary built in (Yellow River basin), state-funded 255 for (stock=)" }
		}
		# [#92] the state-funded expansion is NEITHER free NOR silent (was both before #92): charge the treasury
		# 255 (same order as the Caravan oasis-bazaar endowment; add_treasury=-255 is the proven charge idiom,
		# se_QING_CARAVAN:494) and fire qing_revenue.6 so the player is told the ever-normal reserve has expanded.
		# Guarded on the save-scope so it only charges/notifies when a granary WAS actually built this pulse
		# (the ordered_owned_province may match nothing if every basin city already has one).
		if = {
			limit = { exists = scope:qing_gran_built_province }
			add_treasury = -255
			TREASURY_LOG_it = { amount = -255 }
			trigger_event = { id = qing_revenue.6  days = 1 }
		}
	}
	# [#429] reserves are healthy again — RESTOCK the drained granaries: lift the empty flag from
	# every basin granary that carries it (the stores are refilled; the building was never lost).
	# [review-fix 2026-08-14] pulled out of the good-year build `if` above -- this must fire on the
	# stock band ALONE, every pulse, with no build cooldown/treasury/backlog dependency, or a granary
	# emptied by famine stays wrongly flagged empty for up to 3 years after reserves actually recover.
	if = {
		limit = { var:qing_granary_stock >= 60 }
		every_owned_province = {
			limit = { has_province_modifier = qing_granary_empty }
			remove_province_modifier = qing_granary_empty
			LOG_line = { sys = QING  msg = "granary: 糧倉 restocked, stores refilled in for (stock=)" }
		}
	}
	else_if = {
		# FAMINE, reserves exhausted: the granaries are EMPTIED — but NOT razed. [#429 fix] an empty
		# warehouse does not collapse; the physical 常平倉 building stands, its stores just run dry. Flag
		# every stocked-basin granary province with qing_granary_empty (negates the food + stabilisation
		# bonus while the reserve is drained) instead of remove_building_level (the old #435-era model
		# wrongly destroyed the institution). The modifier is cleared on the good-year restock branch above.
		limit = {
			var:qing_granary_stock <= 15
			any_owned_province = {
				has_building = qing_granary_building
				NOT = { has_province_modifier = qing_granary_empty }
				OR = { is_in_region = Zhili  is_in_region = Shandong  is_in_region = Henan }
			}
		}
		every_owned_province = {
			limit = {
				has_building = qing_granary_building
				NOT = { has_province_modifier = qing_granary_empty }
				OR = { is_in_region = Zhili  is_in_region = Shandong  is_in_region = Henan }
			}
			add_province_modifier = { name = qing_granary_empty  duration = -1 }
			LOG_line = { sys = QING  msg = "granary: 糧倉 stores run dry in famine (building stands, empty) in for (stock=)" }
		}
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_DECLINE_granary_concrete"  reason = "no granary object change: stock in the middle band, or no eligible Yellow River province to build/empty/restock" }
	}
	LOG_exit = { sys = QING  fn = "QING_DECLINE_granary_concrete"  result = OK }
}

			add_province_modifier = { name = qing_smuggler_den  duration = -1 }
			LOG_line = { sys = QING  msg = "currency: foreign smuggler pod rooted in Canton (p:9298) for (currency_stress=)" }
		}
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_DECLINE_spawn_smuggler"  reason = "no smuggler pod: currency stress below 60, on spawn cooldown, or Canton (p:9298) not held / already a smuggler den" }
	}
	LOG_exit = { sys = QING  fn = "QING_DECLINE_spawn_smuggler"  result = OK }
}

# [#80/#81/#84 2026-08-11] QING_frontier_office_ease_squeeze — shared one-quarter squeeze-easing bridge for the
# three frontier fiscal offices (salt 兩淮鹽政 / caravan 參贊大臣 / hoppo 粵海關監督). Called by BOTH the
# auto-pick rotate effects (event/AI path, e.g. qing_canton.2.b/.3.b) AND the new player picker handler, so
# every rotate path eases the squeeze by the SAME curve tied to the NEW holder's corruption (#81), replacing
# the old flat -12 / -6-if-finesse.
#   easing = corruption/5 - 20  (in [-20, 0]): corruption 0 -> -20, 50 -> -10, 100 -> 0 (a venal man cleans nothing).
# NOTE (design v2 finding F4): the quarterly office pulse OVERWRITES $squeeze$ = seated.corruption, so this
# nudge is a ONE-QUARTER BRIDGE, not a durable mover — the durable lever is seating a low-corruption man (the
# candidate list surfaces each man's corruption for exactly this choice). Country scope (ROOT = CHI).
#   $squeeze$ = the squeeze var name (qing_salt_squeeze | qing_hoppo_squeeze | qing_caravan_super_squeeze)
#   $holder$  = the holder var name  (qing_salt_commissioner_holder | qing_hoppo_holder | qing_caravan_super_holder)
# ======================================================================
QING_frontier_office_ease_squeeze = {
	if = {
		limit = { has_variable = $holder$  var:$holder$ = { is_alive = yes } }
		var:$holder$ = { save_scope_as = qing_frontier_ease_char }
		if = { limit = { NOT = { has_variable = $squeeze$ } }  set_variable = { name = $squeeze$  value = 30 } }
		set_variable = { name = qing_frontier_ease_tmp  value = scope:qing_frontier_ease_char.corruption }
		change_variable = { name = qing_frontier_ease_tmp  divide = 5 }
		change_variable = { name = qing_frontier_ease_tmp  subtract = 20 }
		change_variable = { name = $squeeze$  add = var:qing_frontier_ease_tmp }
		if = { limit = { var:$squeeze$ < 0 }    set_variable = { name = $squeeze$  value = 0 } }
		if = { limit = { var:$squeeze$ > 100 }  set_variable = { name = $squeeze$  value = 100 } }
		remove_variable = qing_frontier_ease_tmp
		LOG_line = { sys = QING  msg = "frontier office rotate: squeeze eased by new holder corruption (one-quarter bridge) for" }
	}
}
