COTTAGEIND_scale_production = {
	# Scope: Governorship
	# Function: Scale the production from cottage industries to the appropriate level based on pops, industrialisation etc.
	# [#MG-flood-diag — READ-ONLY, REMOVABLE] This is the exact effect whose change_variable
	# below throws "Variable not of the 'value' scope type. Type: empty" (~30 goods, a fixed
	# ~8 scopes/boot). Log the governorship scope AT THE FAILURE SITE when the suspected
	# empty-source precondition holds: (a) owner link unresolved — GOODS_governorship_<raw>_
	# produced does `multiply = owner.MODIFIER_*`, so an unresolved owner makes that svalue
	# empty, which propagates through COTTAGEIND_raw_* -> COTTAGEIND_produced_$output$ ->
	# the multiply here; or (b) either multiply operand var was never set on this scope.
	# (has_variable does NOT catch a var SET to an empty value, so the owner check is the
	# primary detector.) debug_log + scope dump ONLY — no logic/variable change; silent
	# no-op outside -debug_mode. Remove this if block to revert.
	if = {
		limit = {
			OR = {
				NOT = { exists = owner }
				NOT = { has_variable = COTTAGEIND_pops_output }
				NOT = { has_variable = COTTAGEIND_produced_$output$ }
			}
		}
		LOG_fail = {
			sys = ECON
			fn = "COTTAGEIND_scale_production"
			reason = "empty-var precondition at scale entry (ownerless governorship and/or pops_output/produced-var unset) -> the change_variable multiply below errors empty; scope dump identifies this governorship + its owner link"
		}
	}
	change_variable = {
		name = COTTAGEIND_produced_$output$
		multiply = var:COTTAGEIND_pops_output
	}
	change_variable = {
		name = COTTAGEIND_produced_$output$
		multiply = {
			value = $efficiency$
			multiply = 0.5
			multiply = TECH_cottage_industry_overall_bonus
		}
	}
	change_variable = {
		name = $output$_stockpile
		add = var:COTTAGEIND_produced_$output$
	}
}
