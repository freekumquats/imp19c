########################################################################
# se_ROW_BUILDINGS.txt — [bookmark-1763 #291] REST-OF-WORLD starting specialty
# production buildings.
#
# Companion to SE_qing_starting_buildings (se_QING_BUILDINGS.txt). The Qing get five
# SPECIFIC, historically-named specialty works seeded into hand-picked provinces. The
# rest of the world instead gets the two GENERIC, abstract production buildings
# (row_manufactory_building / row_plantation_building, common/buildings/
# row_production_buildings.txt) — deliberately lower-granularity per the granular-China
# / broad-ROW ruling. Rather than hand-map provinces worldwide, this seeds them
# DATA-DRIVEN: every substantial non-Chinese country gets one manufactory in its
# most-developed finished-good province and one plantation in its most-developed
# cash-crop province, so the 1763/1815 world opens with a plausible spread of existing
# industry (the Lancashire mills, the Caribbean sugar estates, the Bengal muslin
# houses) instead of a blank map the player/AI must build from zero.
#
# add_building_level bypasses the building `allow` tech gate (only the player build
# menu is gated) — same as the Qing seed. Guarded + idempotent via a global sentinel.
# Gate: NOT chinese_group (they use their own named works) + a development floor so
# tiny/tribal tags don't get free industry. Placement mirrors QING_works_build_specialty
# (ordered_owned_province, order_by = total_population, max = 1).
#
# Concrete-over-abstract per standing rule. All steps logged (sys = SETUP).
# Called once from oa_economy_setup.txt at on_game_initialized, right after the Qing seed.
#
# NO BOM (se_*.txt convention).
########################################################################

SE_row_starting_buildings = {
	# [bookmark-1763 #291] seed the two generic ROW specialty buildings across the world's
	# substantial non-Chinese economies. Guarded, idempotent.
	LOG_enter = { sys = SETUP  fn = "SE_row_starting_buildings" }
	if = {
		limit = { NOT = { has_global_variable = row_starting_buildings_done } }

		every_country = {
			# Skip the Qing culture group (their own named works cover them) and require a
			# development floor — a real state with meaningful urban industry, not a tribe or
			# a one-province minor. num_of_cities >= 3 keeps this to organized realms.
			limit = {
				exists = capital_scope
				NOT = { OR = {
					country_culture_group = jurchen        # exclude the Qing (Manchu/jurchen) — they seed their own named works
					country_culture_group = chinese_group   # and any Han-Chinese successor
				} }
				num_of_cities >= 3
			}
			ROW_seed_country_buildings = yes
		}

		# [#230/#231 2026-07-31] The genuine large "works" of 1763 were the NAVAL DOCKYARDS and
		# ROYAL ARSENALS (the factory system is post-1763 — see the worldwide manufacturing brief;
		# memory imp19c-1763-row-seeding lists the ranked sites). These were the biggest industrial
		# enterprises in the Western world. Seed them as arsenal_building (the generic large state
		# military/naval works — carries local munitions production + defence; its `allow` tech gate
		# is bypassed by add_building_level). Province-keyed, owner-agnostic guard
		# (ROW_seed_row_arsenal: exists + city-status + not already present) — at 1763 the owner IS
		# the historical power, and broad-ROW abstraction doesn't warrant per-tag checks. HIGH-
		# confidence sites only; Saint-Étienne (state manufactory 1764) omitted as anachronistic.
		ROW_seed_row_arsenal = { P = 5294  NAME = "Portsmouth Royal Dockyard" }
		ROW_seed_row_arsenal = { P = 9936  NAME = "Chatham Royal Dockyard (Medway)" }
		ROW_seed_row_arsenal = { P = 3388  NAME = "Woolwich Royal Arsenal (London)" }
		ROW_seed_row_arsenal = { P = 5582  NAME = "Brest naval arsenal" }
		ROW_seed_row_arsenal = { P = 8110  NAME = "Toulon naval arsenal" }
		ROW_seed_row_arsenal = { P = 3174  NAME = "Kronstadt / St Petersburg Baltic yard" }
		ROW_seed_row_arsenal = { P = 6142  NAME = "Ferrol royal dockyard" }
		ROW_seed_row_arsenal = { P = 1135  NAME = "Venetian Arsenal" }
		ROW_seed_row_arsenal = { P = 7709  NAME = "Istanbul Imperial Arsenal (Tersane-i Amire)" }
		ROW_seed_row_arsenal = { P = 7891  NAME = "Tula arms works" }

		set_global_variable = row_starting_buildings_done
		LOG_line = { sys = SETUP  msg = "ROW starting buildings: generic specialty-production seed complete" }
	}
	LOG_exit = { sys = SETUP  fn = "SE_row_starting_buildings"  result = OK }
}

# Seed at most one manufactory + one plantation into THIS country (called in country scope
# from the every_country loop). Each goes to the most-populous owned province carrying a
# matching good that doesn't already have the building. Skips silently (logged) when the
# country has no eligible province. Mirrors QING_works_build_specialty placement.
ROW_seed_country_buildings = {
	# Scope: country (the iterated country from every_country). NOTE: inside a
	# scripted_effect called from every_country, ROOT is the on_action root, NOT the
	# iterated country (se_ECON_LOG.txt:57-58), so we stage a self-scope handle for the
	# log line rather than reading [ROOT.GetTag].
	save_scope_as = row_seed_country

	# --- manufactory: finished / craft goods ---
	if = {
		limit = {
			any_owned_province = {
				NOT = { has_building = row_manufactory_building }
				OR = {
					trade_goods = cloth
					trade_goods = wool
					trade_goods = silk
					trade_goods = dye
					trade_goods = textile_fibres
					trade_goods = industrial_fibres
				}
			}
		}
		ordered_owned_province = {
			limit = {
				NOT = { has_building = row_manufactory_building }
				OR = {
					trade_goods = cloth
					trade_goods = wool
					trade_goods = silk
					trade_goods = dye
					trade_goods = textile_fibres
					trade_goods = industrial_fibres
				}
			}
			order_by = total_population
			max = 1
			add_building_level = row_manufactory_building
			# [R2 error.log fix] the [scope:row_seed_country.GetTag] promote fails to resolve inside this
			# LOG string (data factory: "Could not find promote for scope:row_seed_country"), 256 errors.
			# Keep the log line static — the tag is not needed for the diagnostic.
			LOG_line = { sys = SETUP  msg = "ROW starting buildings: row_manufactory_building seeded" }
		}
	}

	# --- plantation: cash crops ---
	if = {
		limit = {
			any_owned_province = {
				NOT = { has_building = row_plantation_building }
				OR = {
					trade_goods = sugar
					trade_goods = tobacco
					trade_goods = coffee
					trade_goods = spices
					trade_goods = opium
				}
			}
		}
		ordered_owned_province = {
			limit = {
				NOT = { has_building = row_plantation_building }
				OR = {
					trade_goods = sugar
					trade_goods = tobacco
					trade_goods = coffee
					trade_goods = spices
					trade_goods = opium
				}
			}
			order_by = total_population
			max = 1
			add_building_level = row_plantation_building
			LOG_line = { sys = SETUP  msg = "ROW starting buildings: row_plantation_building seeded" }
		}
	}

	# =========================================================================================
	# [#230/#231 2026-07-31] GENERIC CIVIC / MILITARY / ECONOMIC layer. The granularity rule means
	# ROW REUSES the shared generic building palette (no bespoke per-country types — those are
	# China's), but it must still seed ALL the period-appropriate generic types, scaled to each
	# realm's development — not just a manufactory+plantation. add_building_level bypasses each
	# building's allow tech-gate (build-menu only); the two port types have a province `potential`
	# (can_have_port / has_minor_river) that IS respected, so they are guarded accordingly.
	# EXCLUDED as 1763-anachronistic: INF_railway_upgrade (steam loco), IND_electric_plant,
	# IND_gasworks/blast_furnace (manufactories = industrial revolution), IND_coal_mine (mining
	# rails), INF_hospital (clinical science), INF_sewer — all post-1763 tech eras.
	# -----------------------------------------------------------------------------------------

	# CAPITAL civic core — every organized realm's seat gets administration + a fortress, and a
	# port if the capital is a harbour. This is the universal baseline.
	capital_scope = {
		if = {
			limit = { NOT = { has_building = URB_administration_district } }
			add_building_level = URB_administration_district
		}
		if = {
			limit = { NOT = { has_building = fortress_building } }
			add_building_level = fortress_building
		}
		if = {
			limit = { can_have_port = yes  NOT = { has_building = port_building } }
			add_building_level = port_building
		}
		LOG_line = { sys = SETUP  msg = "ROW starting buildings: capital civic core (admin/fortress/port) seeded" }
	}

	# COMMERCE district — the realm's single most-populous province (the primary market/entrepot).
	if = {
		limit = { any_owned_province = { NOT = { has_building = URB_commerce_district } } }
		ordered_owned_province = {
			limit = { NOT = { has_building = URB_commerce_district } }
			order_by = total_population
			max = 1
			add_building_level = URB_commerce_district
		}
	}

	# RESIDENTIAL districts — the top TWO population centres get housing (urbanised cores).
	if = {
		limit = { any_owned_province = { NOT = { has_building = URB_residential_district } } }
		ordered_owned_province = {
			limit = { NOT = { has_building = URB_residential_district } }
			order_by = total_population
			max = 2
			add_building_level = URB_residential_district
		}
	}

	# FORTRESSES — the top THREE population centres are walled (beyond the capital), so defence is
	# not a lone-capital affair. Skips any that already got one (e.g. the capital above).
	if = {
		limit = { any_owned_province = { NOT = { has_building = fortress_building } } }
		ordered_owned_province = {
			limit = { NOT = { has_building = fortress_building } }
			order_by = total_population
			max = 3
			add_building_level = fortress_building
		}
	}

	# SCHOOLS — basic education in the top TWO centres (literate administrative/mercantile cores).
	if = {
		limit = { any_owned_province = { NOT = { has_building = EDU_school } } }
		ordered_owned_province = {
			limit = { NOT = { has_building = EDU_school } }
			order_by = total_population
			max = 2
			add_building_level = EDU_school
		}
	}

	# ADVANCED tier — realms with a developed, civilised capital (civilization_value >= 20: the
	# European/Ottoman/Japanese/urban-Indian states, not tribal/frontier tags) additionally get a
	# UNIVERSITY and a CULTURAL district at the capital (the seats of learning & high culture:
	# Paris, London, Madrid, Istanbul, Edo, etc.). This is the development-scaling that keeps a
	# steppe khanate from getting a university while Bourbon France does.
	if = {
		limit = { capital_scope = { civilization_value >= 20 } }
		capital_scope = {
			if = {
				limit = { NOT = { has_building = EDU_university } }
				add_building_level = EDU_university
			}
			if = {
				limit = { NOT = { has_building = URB_cultural_district } }
				add_building_level = URB_cultural_district
			}
			LOG_line = { sys = SETUP  msg = "ROW starting buildings: advanced-realm university + cultural district seeded" }
		}
	}

	# INDUSTRIAL ESTATE — a realm with a real finished-goods province (the manufactory site) gets a
	# generic industrial estate there too (proto-industrial workshop concentration). Craft goods only.
	if = {
		limit = {
			any_owned_province = {
				NOT = { has_building = IND_industrial_estate }
				OR = { trade_goods = cloth  trade_goods = wool  trade_goods = silk  trade_goods = dye  trade_goods = textile_fibres  trade_goods = industrial_fibres  trade_goods = iron  trade_goods = copper }
			}
		}
		ordered_owned_province = {
			limit = {
				NOT = { has_building = IND_industrial_estate }
				OR = { trade_goods = cloth  trade_goods = wool  trade_goods = silk  trade_goods = dye  trade_goods = textile_fibres  trade_goods = industrial_fibres  trade_goods = iron  trade_goods = copper }
			}
			order_by = total_population
			max = 1
			add_building_level = IND_industrial_estate
		}
	}
}

# [#230/#231] Seed a naval dockyard / royal arsenal (arsenal_building) at province $P$. Owner-
# agnostic (the site's 1763 owner IS the historical power); guarded on existence + city-status +
# not-already-present so it is safe and idempotent. add_building_level bypasses the building's
# firearms/civilization allow gate (build-menu only). Scope: any. Skips (logged) if guard fails.
ROW_seed_row_arsenal = {
	if = {
		limit = {
			exists = p:$P$
			p:$P$ = {
				has_city_status = yes
				NOT = { has_building = arsenal_building }
			}
		}
		p:$P$ = {
			add_building_level = arsenal_building
			LOG_line = { sys = SETUP  msg = "ROW starting buildings: arsenal_building seeded at $NAME$ (p:$P$)" }
		}
	}
	else = {
		LOG_line = { sys = SETUP  msg = "ROW starting buildings: SKIP arsenal at $NAME$ (p:$P$) — no city / already present" }
	}
}
