# se_QING_EXAM.txt — [#321 / overnight-build-#1] THE SCHOLAR HIRE-POOL (科舉).
#
# The keju (qing_keju_events.txt) already stages the examination CYCLE as a rich
# flavour beat, and the abstract qing_exam_ladder (0..100, se_QING_GOVERNANCE.txt)
# already summarises the ladder's health. What was MISSING — and what this module
# adds, per the concrete-over-abstract rule — is a PERSISTENT product of the exam:
# real graduate CHARACTERS who enter a HIRING POOL and are drawn on FIRST when a
# great office falls vacant, before the engine conjures a stranger. The exam thus
# stops being pure flavour and becomes the empire's actual talent pipeline.
#
# THE POOL (concrete objects, not a counter):
#   • qing_scholar_pool — a variable_list of jinshi/juren graduate characters the
#     realm has produced but not yet placed in office. Proven variable_list idiom
#     (add_to_variable_list / ordered_in_list / remove_list_variable / variable_list_size —
#     exactly as se_QING_COUNCIL tracks qing_council_members and se_QING_FACTION its
#     lobby blocs). NOT a dynamic name@scope var (no precedent in this codebase).
#   • Each graduate is a create_character with the degree status trait + a scholar
#     marker var (qing_is_pool_scholar) so death-cleanup and the office-fill picker
#     can recognise him. Bound to the throne via the affinity model (天子門生).
#
# WIRED:
#   • QING_exam_graduate_cohort — called from the palace-exam event (qing_keju.2)
#     to mint the cohort of graduates for the cycle (scaled by exam_ladder health).
#   • QING_exam_fill_office_from_pool — the pool-first office-fill: pick the ablest
#     matching scholar, remove him from the pool, seat him via QING_office_appoint.
#   • QING_exam_pool_tick — quarterly maintenance (qing_mechanics_pulse_on_action):
#     age out stale scholars (a graduate un-placed for ~15 years retires), keep the
#     count var fresh for the GUI, and log.
#   • on_character_death — a dying pool scholar is removed from the list (no orphan
#     refs) — hooked in 00_specific_from_code.txt.
#
# CHI-only, player-facing (the keju cycle that feeds it is CHI + is_ai=no gated).
# All logged sys = QING, STATIC msgs only (#253). no-BOM/LF. Prefix = QING_exam_.
#
#   ROOT / this = the Qing country (CHI, player); root.current_ruler = the emperor.
#
# Sources (feasibility only): Benjamin Elman, A Cultural History of Civil
#   Examinations (2000); Ichisada Miyazaki, China's Examination Hell (tr. 1976);
#   Chung-li Chang, The Chinese Gentry (1955). Engine idioms cited inline.

# ======================================================================
# QING_exam_init — seed the pool bookkeeping once. Scope: COUNTRY. Idempotent.
# ======================================================================
QING_exam_init = {
	LOG_enter = { sys = QING  fn = "QING_exam_init" }
	if = {
		limit = { NOT = { has_variable = qing_exam_initialized } }
		set_variable = { name = qing_exam_initialized  value = 1 }
		set_variable = { name = qing_scholar_pool_count  value = 0 }
		# [#321 deferred] the metropolitan pass-rate metric (0..100) the cohort-mint reads.
		# Seeded mid-band; recomputed each triennium from ladder health vs corruption drag.
		set_variable = { name = qing_exam_pass_rate  value = 50 }
	}
	LOG_exit = { sys = QING  fn = "QING_exam_init"  result = OK }
}

# ======================================================================
# [#321 deferred] QING_exam_compute_pass_rate — Scope: COUNTRY (CHI). Derive the
# metropolitan examination's PASS-RATE metric (0..100) from the ladder's health and
# the corruption/patronage drag, so the cohort that graduates each triennium scales
# with how well the ladder actually works (a vigorous, clean ladder passes more able
# men; a clogged, bought one passes few). This is the abstracted stand-in for the
# ~1% historical metropolitan rate — NOT a literal probability, but a 0..100 band the
# cohort-mint and the GUI read. Kept prestige/counter-neutral (pure derive-and-store).
#
# FORMULA (all proven counter reads, mirroring QING_GOV_update_exam_ladder's blend):
#   base   = exam_ladder health (the ladder is the pipeline's throughput)
#   minus  = corruption/2 (examiner bribery + bought shengyuan places choke the intake)
#   minus  = 20 if purchased-ranks modifier is live (捐納 crowds out merit intake)
#   clamp 0..100.
# ======================================================================
QING_exam_compute_pass_rate = {
	QING_exam_init = yes
	# [meter-concretize #1 2026-08-06] base on the DERIVED academy-coverage reach (QING_exam_reach, the
	# 書院 network read live) instead of the retired qing_exam_ladder meter. Semantic shift: the pass-rate
	# base now tracks "how many academies prepare candidates," not a rotting ladder counter. The
	# subsequent corruption/purchased-rank/civic/curriculum terms below are UNCHANGED (design §9 row 6).
	set_variable = { name = qing_exam_pass_rate  value = QING_exam_reach }
	# corruption drag (examiner bribery): halve the corruption level and subtract it.
	if = {
		limit = { has_variable = qing_corruption_level }
		set_variable = { name = qing_exam_pass_drag  value = var:qing_corruption_level }
		change_variable = { name = qing_exam_pass_drag  divide = 2 }
		change_variable = { name = qing_exam_pass_rate  subtract = var:qing_exam_pass_drag }
		remove_variable = qing_exam_pass_drag
	}
	# an active 捐納 (sale-of-offices) regime crowds merit out of the intake.
	if = {
		limit = { has_country_modifier = qing_office_purchased_ranks }
		change_variable = { name = qing_exam_pass_rate  subtract = 20 }
	}
	# [MO-ENH #6] CIVIC IDENTITY WIDENS THE TALENT POOL. A modernizing realm (national schools, a
	# broader literate press, the Self-Strengthening education apparatus — the qing_civic_identity
	# meter, se_QING_DECLINE.txt §11) feeds a wider, better-prepared candidate base up the exam
	# ladder, lifting the metropolitan pass-rate. Only a HIGH civic identity (>= 50) helps, and the
	# boost scales gently ((civic-50)/4, so a maxed civic of 100 adds +12.5, roughly offsetting a
	# mid corruption drag). Purely additive to the derived rate; the clamp below bounds it.
	if = {
		limit = { has_variable = qing_civic_identity  var:qing_civic_identity >= 50 }
		set_variable = { name = qing_exam_civic_boost  value = var:qing_civic_identity }
		change_variable = { name = qing_exam_civic_boost  subtract = 50 }
		change_variable = { name = qing_exam_civic_boost  divide = 4 }
		change_variable = { name = qing_exam_pass_rate  add = var:qing_exam_civic_boost }
		remove_variable = qing_exam_civic_boost
	}
	# [#45 Batch 4a 2026-07-24] EXAM CURRICULUM bias (科舉科目). The qing_exam_curriculum_law selector
	# writes qing_exam_curriculum_bias on enact (0 = 八股 classical default, +10 = 經世 practical). A
	# practical curriculum lifts the metropolitan pass-rate; additive, applied BEFORE the clamp below so
	# it stays bounded. Guarded on has_variable so a game that never enacted the law reads nothing (bias-B).
	if = {
		limit = { has_variable = qing_exam_curriculum_bias }
		change_variable = { name = qing_exam_pass_rate  add = var:qing_exam_curriculum_bias }
	}
	# clamp 0..100.
	if = { limit = { var:qing_exam_pass_rate < 0 }   set_variable = { name = qing_exam_pass_rate  value = 0 } }
	if = { limit = { var:qing_exam_pass_rate > 100 } set_variable = { name = qing_exam_pass_rate  value = 100 } }
}

# ======================================================================
# [#321 deferred] QING_exam_schedule_triennial — Scope: COUNTRY (CHI). The 鄉試/會試
# TRIENNIAL CLOCK. The examination cycle was held on a fixed three-year cadence (the
# 子/卯/午/酉 years), not at random — so this replaces the old stochastic firing of
# qing_keju.1 (a weight in QING_frontier_flavour_roll) with a real ~3-year timer. Run
# on the quarterly mechanics pulse; a 1095-day cooldown var gives the triennial rhythm
# (the proven days= cooldown idiom, cf. QING_tribute/QING_canal/QING_works). Recomputes
# the pass-rate before convening so qing_keju.1/.2 read a fresh value. Player-only (the
# pulse is CHI + is_ai=no gated); a clean no-op while the cooldown is live.
# ======================================================================
QING_exam_schedule_triennial = {
	QING_exam_init = yes
	if = {
		limit = { NOT = { has_variable = qing_exam_triennial_cooldown } }
		LOG_enter = { sys = QING  fn = "QING_exam_schedule_triennial" }
		# [BT-qing-statutes chunk 2.4] cooldown length is now set by the Examination Cadence law
		# (qing_law_exam_cadence: 0 triennial / 1 frequent / 2 sparse). Proven if/else_if literal-days
		# pattern (cf. se_QING_TRIBUTE.txt:257-266) — NOT a var-driven `days = var:X` (unproven). Absent
		# the var (law never enacted) it falls through to the historical 1095-day triennial default.
		if =      { limit = { has_variable = qing_law_exam_cadence  var:qing_law_exam_cadence = 1 }  set_variable = { name = qing_exam_triennial_cooldown  days = 730 } }   # ~biennial (frequent)
		else_if = { limit = { has_variable = qing_law_exam_cadence  var:qing_law_exam_cadence = 2 }  set_variable = { name = qing_exam_triennial_cooldown  days = 2190 } }  # ~sexennial (sparse)
		else =    { set_variable = { name = qing_exam_triennial_cooldown  days = 1095 } }             # ~3 years (the triennial default; selector 0 or unset)
		QING_exam_compute_pass_rate = yes
		# convene the examination cycle (the host prompt); a short lead so it does not
		# land on the same day as the pulse that scheduled it.
		trigger_event = { id = qing_keju.1  days = { 5 20 } }
		LOG_line = { sys = QING  msg = "exam: the triennial examination is proclaimed for" }
		LOG_exit = { sys = QING  fn = "QING_exam_schedule_triennial"  result = OK }
	}
}

# ======================================================================
# QING_exam_mint_scholar — create ONE graduate character and add him to the pool.
# Scope: COUNTRY (CHI). $degree$ = the status trait (jinshi | juren). The graduate
# is a Han commoner of real ability (the gentry-scholar archetype), bound to the
# throne. Marked qing_is_pool_scholar + stamped with his intake date so the tick
# can retire the long-unplaced. Uses the proven create_character idiom (roster).
# ======================================================================
QING_exam_mint_scholar = {
	create_character = {
		age = 28
		culture = han
		religion = confucianism   # [#61 2026-07-23] was root.religion — INVALID create_character religion FIELD (flood; cf. SOUTHERNSTUDY:97); literal + Design-3-safe.
		# a fresh graduate is learned and eloquent, not martial — finesse/charisma
		# lead, with a spread so the pool is not identical clones.
		add_finesse = 7
		add_charisma = 5
		add_martial = 1
		add_zeal = 4
		add_trait = $degree$
		save_scope_as = qing_new_scholar
	}
	scope:qing_new_scholar = {
		set_home_country = ROOT
		# mark him a pool scholar (death-cleanup + fill-picker + on-seating drop recognise
		# this). The retirement tick keys on AGE (created 28, retired ~55 = decades un-placed),
		# so no separate intake-date var is needed.
		set_variable = { name = qing_is_pool_scholar  value = 1 }
		# [playtest logfix] the degree is conferred under the throne's eye (天子門生) — but the
		# affinity SCORE + bind is DEFERRED to the next quarterly pool tick (QING_exam_pool_tick).
		# A character just made by create_character THIS tick cannot have its own script variables
		# read back the same tick (set_variable writes silently, but the immediate change_variable
		# read-back fails), so QING_char_bind's inline affinity read here flooded the log (~6k lines:
		# qing_char_affinity / qing_aff_zeal_gap "not being set"). By the next tick the scholar is
		# committed and the affinity model reads clean — mirrors the Southern/Upper Study seed→bind
		# split (se_QING_SOUTHERNSTUDY.txt: mint, then bind in a later block).
		# Stamp a dedicated one-shot BIND marker (a write-only var — safe this tick, cf. the
		# qing_is_pool_scholar write above; only same-tick READ-back fails). The pool tick binds
		# exactly the scholars carrying it. We do NOT gate the deferred bind on the affinity var
		# itself, because QING_office_appoint scores a man (QING_char_affinity, deliberately WITHOUT
		# bind — se_QING_COUNCIL.txt) so a scholar autofilled before the next tick would have his
		# affinity var already set and would wrongly skip binding.
		set_variable = { name = qing_needs_bind  value = 1 }
	}
	# add to the pool list + keep the count var in step (GUI reads the count).
	add_to_variable_list = { name = qing_scholar_pool  target = scope:qing_new_scholar }
	change_variable = { name = qing_scholar_pool_count  add = 1 }
	# [#34 review-fix] LOG string must never carry a macro $param$ (it can derail the
	# parser — standing rule); the $degree$ interpolation is dropped for a static line.
	LOG_line = { sys = QING  msg = "exam: minted a degreed graduate into the hiring pool for" }
}

# ======================================================================
# [#40 2026-08-09] QING_exam_mint_banner_laureate - mint a FREE, UNPOSTED banner Translation
# Laureate (fanyi_jinshi) into the CHI court, so the amban PICKER has real candidates. The
# on_becoming_adult keju intake (QING_exam_sit_candidate) is the only OTHER source of a free court
# laureate, and it cannot reach the pre-existing adult generation at game start, so the amban bench
# starts (and stays) empty and a recalled amban is the sole eligible pick. This helper feeds it: the
# boot backfill (QING_amban_seed_spare_laureates) + the triennial cohort call it. Scope: COUNTRY (CHI).
# Mirrors QING_amban_seed_one's shape EXACTLY: create + add_trait, NO bind, NOT added to
# qing_scholar_pool (that is the CIVIL office pool with an age-55 retirement tick + a civil-office fill
# picker that must NOT draw a bannerman). NO qing_needs_bind (review MED: the deferred-bind tick only
# iterates qing_scholar_pool, so a non-pool marker never binds and just lingers; and the amban pickers
# score on charisma/finesse/degree only, never qing_char_affinity, so an unbound laureate cannot flood).
# #90-safe: no modifiers inside create_character; called only from the DEFERRED day-32 event + the
# triennial cohort, never at construction.
# ======================================================================
QING_exam_mint_banner_laureate = {
	create_character = {
		age = { 30 48 }
		culture = manchu
		religion = vajrayana
		# a translation-examination laureate is a court BROKER (amban stock): charisma leads, finesse
		# follows, martial low - matches qing_amban_rank_svalue (#61: charisma x2 + finesse).
		add_charisma = { 5 8 }
		add_finesse = { 4 7 }
		add_martial = { 1 3 }
		add_zeal = { 2 5 }
		add_trait = fanyi_jinshi
		save_scope_as = qing_new_banner_laureate
	}
	# Mirror QING_amban_seed_one EXACTLY: a char created in the CHI country scope already defaults to
	# employer = ROOT (what the picker's `employer = ROOT` gate checks — no set_home_country needed, and
	# seed_one does not use it), demoted to minor standing until posted. NO move_country (he stays a free
	# CHI court laureate, unlike seed_one which then wires him to a subject).
	scope:qing_new_banner_laureate = {
		# [#40 review HIGH-fix 2026-08-10] set_home_country = ROOT so the picker's `employer = ROOT` gate
		# actually matches — the sibling stay-at-court mint QING_exam_mint_scholar:174 does exactly this, and
		# every other stay-at-court mint in the repo sets home country too. QING_amban_seed_one does NOT (it
		# move_country's to a subject, which sets employer that way) — so it is a false precedent for a
		# character that STAYS at the CHI court. Without this, employer defaults are untested here and the
		# picker could silently find nobody (the exact empty-bench bug #40 fixes). Both call sites run in CHI
		# country scope, so ROOT = CHI is always valid (no ROOT-is-a-character vacate hazard here).
		set_home_country = ROOT
		set_as_minor_character = THIS   # a waiting laureate is a minor court figure until posted (as the amban seed/fallback do)
	}
	LOG_line = { sys = QING  msg = "exam: minted a spare banner Translation Laureate for the amban bench for" }
}

# ======================================================================
# [#114 2026-08-09] QING_keju_compute_convene — Scope: COUNTRY (CHI). Store the convene-time
# DISPLAY vars the "Examinations Convene" prompt (qing_keju.1) shows the player: the live academy
# HALL count, the "test-takers" gathered across that network, the two option COSTS, and the expected
# graduate count. Called from qing_keju.1's IMMEDIATE, which runs BEFORE the event window renders —
# so these vars exist for both the option-preview tooltips and the desc loc. Same-tick country
# set-then-read is proven safe (this file, QING_exam_compute_pass_rate reads back qing_exam_pass_drag
# the same tick). The gate + charge themselves read the svalues LIVE (not these vars), so a stale var
# can never desync the money; these vars are for the TEXT only.
#
# The gate/charge svalues (QING_keju_cost_full_svalue / _modest_svalue) and the cohort-size thresholds
# (>=16 / >=28) are calibrated in the #114 design doc. The expected-graduate ladder here MUST stay in
# lockstep with the cohort-mint ladder in QING_exam_graduate_cohort (the coupled pair).
# ======================================================================
QING_keju_compute_convene = {
	LOG_enter = { sys = QING  fn = "QING_keju_compute_convene" }
	QING_exam_init = yes
	set_variable = { name = qing_keju_hall_count   value = QING_academy_count }
	# "test-takers": the candidates gathering across the hall network — a narrative figure
	# (hall count × 500 ≈ the licentiates/graduates of the whole ladder), the readable face
	# of the same academy count that drives cost + graduates. ~9,500 at the 1763 baseline.
	set_variable = { name = qing_keju_candidates   value = QING_academy_count }
	change_variable = { name = qing_keju_candidates  multiply = 500 }
	set_variable = { name = qing_keju_cost_full     value = QING_keju_cost_full_svalue }
	set_variable = { name = qing_keju_cost_modest   value = QING_keju_cost_modest_svalue }
	# refresh the pass-rate here so the cohort at .2 (and any GUI reader) sees a current value at
	# convene time (cheap; idempotent). The .1 desc itself shows halls/candidates/cost/expected only.
	QING_exam_compute_pass_rate = yes
	# expected graduates = 1 lead + hall-band extras (mirrors QING_exam_graduate_cohort's >=16/>=28).
	set_variable = { name = qing_keju_expected_grads  value = 1 }
	if = { limit = { var:qing_keju_hall_count >= 16 }  change_variable = { name = qing_keju_expected_grads  add = 1 } }
	if = { limit = { var:qing_keju_hall_count >= 28 }  change_variable = { name = qing_keju_expected_grads  add = 1 } }
	LOG_line = { sys = QING  msg = "exam: convene figures computed (halls/candidates/cost/expected) for" }
	LOG_exit = { sys = QING  fn = "QING_keju_compute_convene"  result = OK }
}

# ======================================================================
# QING_exam_graduate_cohort — mint the cycle's graduate cohort. Scope: COUNTRY (CHI).
# Called from the palace examination (qing_keju.2). [#114 2026-08-09] The two axes are now
# separated cleanly: degree QUALITY tracks the PASS-RATE (ladder health net of corruption/捐納
# drag — a vigorous ladder leads with a jinshi, a clogged one with a juren), while graduate COUNT
# tracks the concrete ACADEMY HALL network (a wider system physically seats and passes more men).
# This is the #114 model — halls = quantity, integrity = quality — so corruption now debases the
# DEGREES awarded rather than shrinking the cohort. Bounded (at most 3 minted per cycle: 1 lead +
# 2 hall-band extras) so the pool cannot balloon past today's healthy-band ceiling — the age-55
# retirement tick + office draws keep it in a steady band. Self-computes the hall count, so a caller
# outside the convene path (e.g. a grace exam) still mints correctly.
# ======================================================================
QING_exam_graduate_cohort = {
	LOG_enter = { sys = QING  fn = "QING_exam_graduate_cohort" }
	QING_exam_init = yes
	# [#321 deferred] pass-rate metric (ladder health net of corruption/捐納 drag), recomputed by the
	# triennial scheduler before the cycle convenes — refreshed here too so a cohort minted outside
	# the triennial path (e.g. a grace exam) still reads a current value.
	QING_exam_compute_pass_rate = yes
	# [#114] self-contained hall count — the COUNT axis, independent of any convene-time var.
	set_variable = { name = qing_keju_hall_count  value = QING_academy_count }
	# LEAD graduate — QUALITY by the pass-rate band (jinshi if the ladder passes merit, else juren).
	if = {
		limit = { var:qing_exam_pass_rate >= 30 }
		QING_exam_mint_scholar = { degree = jinshi }
	}
	else = {
		QING_exam_mint_scholar = { degree = juren }
	}
	# EXTRA graduates — COUNT scales directly with the academy network (literal-RHS thresholds).
	# 19 halls → +1 (baseline total 2, matching the historic middling cohort); ≥28 → +2 (total 3,
	# the healthy-band ceiling); <16 → +0 (a thin/war-shrunk network yields the lone laureate).
	if = { limit = { var:qing_keju_hall_count >= 16 }  QING_exam_mint_scholar = { degree = juren } }
	if = { limit = { var:qing_keju_hall_count >= 28 }  QING_exam_mint_scholar = { degree = juren } }
	# [#40 2026-08-09] BANNER-TRACK laureate — the triennial 繙譯 (translation) examination also seats an
	# occasional banner Translation Laureate (fanyi_jinshi), refilling the amban bench that the boot backfill
	# (QING_amban_seed_spare_laureates) starts. One per cycle keeps the pipeline self-sustaining as posted
	# seats turn over, without flooding the court (the civil cohort above is the bulk of each cycle). Gated on
	# a live academy network (>=16 halls) — the same threshold the second civil extra uses — so a war-shrunk
	# ladder yields no spare banner laureate that cycle.
	if = { limit = { var:qing_keju_hall_count >= 16 }  QING_exam_mint_banner_laureate = yes }
	LOG_line = { sys = QING  msg = "exam: a graduate cohort was seated (lead by pass-rate, size by hall count) for" }
	LOG_exit = { sys = QING  fn = "QING_exam_graduate_cohort"  result = OK }
}

# ======================================================================
# [#111] QING_exam_pool_draw_one — draw ONE EXISTING office-less jinshi into the scholar
# hire-pool. Scope: COUNTRY (CHI). The tick half of the #111 caller split: the boot event
# legitimately MINTS fresh graduates (QING_exam_seed_hanlin_pool, unchanged, called only
# from qing_force_setup.12); the quarterly tick must never re-mint — it used to call
# QING_exam_seed_hanlin_pool too, which re-ran create_character (QING_exam_mint_scholar)
# every quarter to refill the drained bench, an unbounded aether-spawn of scholar
# characters. This effect draws from the LIVING population instead, mirroring
# QING_opium_commissioner_appoint's proven ordered_character idiom (eligibility limit +
# order_by finesse + max 1), plus the pool-scholar exclusion (an already-pooled man is not
# a fresh draw) and an age ceiling matching this pool's own retirement threshold
# (QING_exam_pool_tick age >= 55 retires a man; a 55+ jinshi is not a fresh recruit).
#
# STALE-SCOPE GUARD: this effect is meant to be called repeatedly from a `while` loop
# (QING_exam_pool_draw_refill, below). save_scope_as only writes on an actual
# ordered_character match — a round that matches nobody leaves scope:qing_new_pool_draw
# pointing at whatever the PREVIOUS successful round drew. Re-checking
# NOT = { has_variable = qing_is_pool_scholar } on that saved scope closes this: a genuine
# fresh draw never carries the marker (ordered_character's own limit excludes marked men),
# but a stale leftover from an earlier round in the same loop DOES (we stamp it during
# processing) — so the stale case falls through to else/no-op instead of double-processing
# the same man. Under-full is honest; a round with no eligible candidate does nothing.
# ======================================================================
QING_exam_pool_draw_one = {
	ordered_character = {
		limit = {
			employer = ROOT
			is_adult = yes
			is_alive = yes
			is_ruler = no
			age < 55
			has_trait = jinshi
			NOT = { has_variable = qing_is_pool_scholar }
			NOT = { QING_char_holds_court_position = yes }
			NOT = { AND = { exists = ROOT.primary_heir  this = ROOT.primary_heir } }
		}
		order_by = finesse
		max = 1
		save_scope_as = qing_new_pool_draw
	}
	if = {
		limit = {
			exists = scope:qing_new_pool_draw
			scope:qing_new_pool_draw = { NOT = { has_variable = qing_is_pool_scholar } }
		}
		scope:qing_new_pool_draw = {
			set_variable = { name = qing_is_pool_scholar  value = 1 }
			# a drawn man already existed before this tick, so the same-tick read-back
			# restriction that forces QING_exam_mint_scholar's deferred bind does NOT apply
			# here (se_QING_AFFINITY.txt QING_char_bind — dozens of call sites bind
			# pre-existing characters inline; only a create_character'd-THIS-tick character
			# cannot have its own variables read back the same tick).
			QING_char_bind = yes
		}
		add_to_variable_list = { name = qing_scholar_pool  target = scope:qing_new_pool_draw }
		change_variable = { name = qing_scholar_pool_count  add = 1 }
		LOG_line = { sys = QING  msg = "exam: drew an office-less jinshi into the hiring pool for" }
	}
	else = {
		LOG_fail = { sys = QING  fn = "QING_exam_pool_draw_one"  reason = "no eligible office-less jinshi found to draw" }
	}
}

# EXAMINATION-ADMISSION LAW's establishment (qing_law_hanlin_cap) by DRAWING existing
# office-less jinshi (QING_exam_pool_draw_one), never by minting. Scope: COUNTRY (CHI).
# The tick-side replacement for QING_exam_seed_hanlin_pool at the caller-split (#111) —
# same three branch targets (9 broad / 3 restricted / 6 customary default), same
# while+count+limit idiom, so the establishment law reads identically from either path.
# Called from QING_exam_pool_tick ONLY. Under-full is honest: a realm without enough
# office-less jinshi walking around simply keeps a thinner bench until one is produced
# by the exam cycle itself (QING_exam_graduate_cohort) or vacates an office.

QING_exam_pool_draw_refill = {

		limit = { has_variable = qing_law_hanlin_cap  var:qing_law_hanlin_cap = 1 }
		while = {
			limit = { var:qing_scholar_pool_count < 9 }
			count = 9
			QING_exam_pool_draw_one = yes
		}
	}
	else_if = {
		# RESTRICTED establishment (減額): a tight bench of 3.
		limit = { has_variable = qing_law_hanlin_cap  var:qing_law_hanlin_cap = 2 }
		while = {
			limit = { var:qing_scholar_pool_count < 3 }
			count = 3
			QING_exam_pool_draw_one = yes
		}
	}
	else = {
		# CUSTOMARY establishment (循例), the default: a bench of 6.
		while = {
			limit = { var:qing_scholar_pool_count < 6 }
			count = 6
			QING_exam_pool_draw_one = yes
		}

# ======================================================================

# EXAMINATION-ADMISSION LAW sets (qing_law_hanlin_cap: the establishment of degree-holders the
# throne keeps on the bench awaiting a post). Scope: COUNTRY (CHI). Populated the SAME way the
# Grand Council is (QING_council_autofill_office): mint purpose-built pool scholars via the proven
# QING_exam_mint_scholar (create_character age 28 / culture han / religion confucianism / add_trait
# — #90-safe, runs from the DEFERRED day-32 event qing_force_setup.12, NOT construction). Called
# once at game start and as a top-up in the quarterly pool tick, so the pool never sits empty (the
# boot-test "No laureate waits" — the old autofill events that filled it were removed on purpose;

QING_exam_seed_hanlin_pool = {
	QING_exam_init = yes
	# The EXAMINATION-ADMISSION LAW (qing_law_hanlin_cap: 0 customary / 1 broad / 2 restricted) sets
	# how many degree-holders the throne keeps on the bench. Resolve the target COUNT as a literal
	# (RHS-comparison rule: never a var: on a comparison RHS — so we branch on the selector and use a
	# literal `count`, NOT a var-driven cap comparison). Each branch tops the pool up to its target:
	# `count` mints that many, but QING_exam_mint_scholar is only reached while the pool is below the
	# target via the guarded while `limit` on the LITERAL threshold. Default (unset/0) = 6.

			count = 9

		}
	}
	else_if = {
		# RESTRICTED establishment (減額): a tight bench of 3.
		limit = { has_variable = qing_law_hanlin_cap  var:qing_law_hanlin_cap = 2 }
		while = {
			limit = { var:qing_scholar_pool_count < 3 }
			count = 3
			QING_exam_mint_scholar = { degree = jinshi }
		}
	}
	else = {
		# CUSTOMARY establishment (循例), the default: a bench of 6.
		while = {
			limit = { var:qing_scholar_pool_count < 6 }
			count = 6
			QING_exam_mint_scholar = { degree = jinshi }
		}
	}
	LOG_line = { sys = QING  msg = "exam: seeded the waiting Hanlin pool to the admission-law establishment for" }
}

# ======================================================================
# QING_exam_pool_drop_member — [#321 review-fix] call in a SEATED scholar's scope.
# THE single pool-exit path for a man who takes office. Hooked into the tail of
# QING_office_appoint so EVERY appointment route (the GC-panel autofill, the manual
# per-office appoint, the keju laureate first-vacant, and keju.6's pool fill) drops
# a pool scholar from qing_scholar_pool + decrements the count + strips his markers
# the instant he is actually seated — closing the leak where an autofilled/manually-
# appointed scholar stayed listed, marked, and counted (a phantom count that kept
# keju.6 firing and let the fill-picker re-select a man already in office). Self-
# guards on the qing_is_pool_scholar marker, so it is a safe no-op for the ~all
# appointees who were never pool scholars. this = the just-seated appointee.
# ======================================================================
QING_exam_pool_drop_member = {
	if = {
		limit = {
			has_variable = qing_is_pool_scholar
			exists = employer
		}
		save_scope_as = qing_seated_scholar
		remove_variable = qing_is_pool_scholar
		employer = {
			if = {
				limit = { has_variable_list = qing_scholar_pool }
				remove_list_variable = { name = qing_scholar_pool  target = scope:qing_seated_scholar }
				if = { limit = { var:qing_scholar_pool_count > 0 }  change_variable = { name = qing_scholar_pool_count  subtract = 1 } }
				LOG_line = { sys = QING  msg = "exam: a pool scholar took office, removed from the hiring pool for" }
			}
		}
	}
}

# ======================================================================
# QING_exam_fill_first_vacant_from_pool — Scope: COUNTRY (CHI). Draw the ablest
# waiting pool scholar and seat him in the FIRST vacant great office (the canonical
# order QING_office_appoint_first_vacant uses). Used by qing_keju.6 (the talent-
# draw event). [#321 review-fix] pool-EXIT is NOT done here: QING_office_appoint's
# tail (QING_exam_pool_drop_member) removes him only when he is ACTUALLY seated, so
# if QING_office_appoint_first_vacant no-ops (every office's holder-var still set —
# e.g. a stale var for a holder captured in war, so recompute reports a vacancy the
# appoint picker does not see) the scholar STAYS in the pool rather than being
# silently consumed. We also exclude sitting office-holders from the pick defensively.
# ======================================================================
QING_exam_fill_first_vacant_from_pool = {
	LOG_enter = { sys = QING  fn = "QING_exam_fill_first_vacant_from_pool" }
	QING_exam_init = yes
	if = {
		limit = {
			has_variable_list = qing_scholar_pool
			any_in_list = { variable = qing_scholar_pool  is_alive = yes  employer = ROOT  NOT = { has_variable = qing_office_held } }
		}
		ordered_in_list = {
			variable = qing_scholar_pool
			limit = { is_alive = yes  employer = ROOT  NOT = { has_variable = qing_office_held } }
			order_by = combined_stats_council_svalue
			check_range_bounds = no
			max = 1
			save_scope_as = qing_pool_pick
		}
		if = {
			limit = { exists = scope:qing_pool_pick }
			scope:qing_pool_pick = {
				# seat him in the first vacant office; QING_office_appoint's tail drops him
				# from the pool iff he is actually seated (no consumption on a no-op).
				QING_office_appoint_first_vacant = yes
			}
			LOG_line = { sys = QING  msg = "exam: raised a pool scholar to the first vacant office for" }
		}
	}
	LOG_exit = { sys = QING  fn = "QING_exam_fill_first_vacant_from_pool"  result = OK }
}

# ======================================================================
# QING_exam_pool_tick — quarterly maintenance. Scope: COUNTRY (CHI). Called from
# qing_mechanics_pulse_on_action. Retires scholars who have sat un-placed for ~15
# years (a graduate who never gains office drifts back to the gentry), keeps the
# count var honest against the live list, and logs. Self-guards on the list.
# ======================================================================
QING_exam_pool_tick = {
	if = {
		limit = { has_variable_list = qing_scholar_pool }
		LOG_enter = { sys = QING  fn = "QING_exam_pool_tick" }
		# retire the aged-out (age >= 55 — decades on the shelf, created at 28) or the
		# dead-but-listed. Removing from the list while iterating it is the PROVEN idiom
		# (Terra-Indomita COL_sguis.txt:37 every_in_list + remove_list_variable target=prev).
		every_in_list = {
			variable = qing_scholar_pool
			limit = {
				OR = {
					is_alive = no
					AND = { is_alive = yes  age >= 55 }
				}
			}
			ROOT = {
				remove_list_variable = { name = qing_scholar_pool  target = prev }
				if = { limit = { var:qing_scholar_pool_count > 0 }  change_variable = { name = qing_scholar_pool_count  subtract = 1 } }
			}
			# a living retiree keeps his degree but loses the pool marker.
			if = { limit = { is_alive = yes }  remove_variable = qing_is_pool_scholar }
		}
		# [playtest logfix] DEFERRED BIND: score + bind each pool scholar the tick AFTER he is
		# minted (see QING_exam_mint_scholar — the affinity read cannot run in the create tick,
		# so binding inline there flooded the log with qing_char_affinity/qing_aff_zeal_gap
		# "not being set"). Gate on the dedicated qing_needs_bind marker stamped at mint, and
		# clear it as we bind — a clean one-shot per scholar that is NOT confused by the bare
		# affinity SCORING that QING_office_appoint performs on a man seated before this tick.
		every_in_list = {
			variable = qing_scholar_pool
			limit = {
				is_alive = yes
				has_variable = qing_needs_bind
			}
			QING_char_bind = yes
			remove_variable = qing_needs_bind
		}
		# [Hanlin pool 2026-07-25] top the waiting bench back up to the admission-law establishment
		# after retirements/appointments drain it — the pool is LAW-governed, not filled by the old
		# (removed) autofill events. [#111 caller-split 2026-08-11] the tick DRAWS existing
		# office-less jinshi rather than minting — QING_exam_seed_hanlin_pool (create_character)
		# stays the BOOT-ONLY seed (qing_force_setup.12); re-running it here every quarter re-minted
		# fresh graduates to refill a bench drained by appointments, an unbounded character-count
		# leak (the exam-creation standing rule: create_character only at boot-seed + the exam itself,
		# never a maintenance tick). QING_exam_pool_draw_refill only draws while below the law's target.
		QING_exam_pool_draw_refill = yes
		LOG_line = { sys = QING  msg = "exam: pool tick (retired the aged/dead un-placed, topped up to the establishment) for" }
		LOG_exit = { sys = QING  fn = "QING_exam_pool_tick"  result = OK }
	}
}

# ======================================================================
# [#34] QING_exam_sit_candidate — THE COMING-OF-AGE INTAKE. Scope: CHARACTER (called
# from on_becoming_adult, so root = the maturing character, NOT the country). This is
# what makes the keju the SOURCE of the empire's civil characters: EVERY Qing court
# character sits the examination on reaching adulthood and — if he passes — carries a
# keju degree for life, so the candidate benches the council / ambanate / legations pick
# from are populated by graduates BY CONSTRUCTION (the approved population-source model:
# governors / researchers / engine-diplomats are not script-appointable, so the keju
# gates the POOL of eligible men, not the individual seat). The soft-preference weight
# in qing_degree_prestige_svalue then floats those graduates to the top of every picker.
#
# TWO TRACKS (per the approved banner-exam design):
#   • Han and the general court sit the CIVIL examination (文科) → civil degrees
#     (jiansheng→shengyuan→juren→gongshi→jinshi→hanlin); these staff the metropolitan
#     offices, governorships, and legations.
#   • Bannermen — Manchu (jurchen group) and Mongol (mongolic group) — sit the
#     TRANSLATION examination (翻譯科) → the fanyi_jinshi banner degree; these are the
#     stock the amban / resident-commissioner posts draw on (see se_QING_AMBAN.txt).
#
# PASS/FAIL is scaled by the realm's metropolitan pass-rate (employer.var:qing_exam_pass_rate,
# maintained by QING_exam_compute_pass_rate) AND the candidate's own finesse+charisma
# (qing_keju_pass_chance_svalue). A failed candidate takes NO degree (he joins the pool of
# degreeless commoners a picker falls back on only when no graduate is free — so the realm
# never soft-locks for want of a laureate), save a small chance of the purchased 監生 floor.
# The degree TIER on a pass follows the same pass-rate bands the cohort-mint uses.
#
# NOTE (scope): this runs at character root, so it does NOT call QING_char_init_identity /
# QING_char_bind (both read root.current_ruler, i.e. the COUNTRY's ruler — invalid when
# root is a character). Conferring the degree is the whole job here; a graduate's throne-
# affinity + Manchu/Han identity are set by the office/council machinery when he is drawn.
# Marks qing_sat_keju so a character sits exactly once. CHI court only; safe no-op elsewhere.
# ======================================================================
QING_exam_sit_candidate = {
	if = {
		limit = {
			# a Qing court character who has not yet sat, is a commoner subject (not the
			# reigning sovereign), free, and holds no degree yet. He need not be a pool
			# scholar — every court child sits the keju on coming of age.
			exists = employer
			employer = { tag = CHI }
			NOT = { has_variable = qing_sat_keju }
			is_ruler = no
			prisoner = no
			# not already degreed (defensive; a fresh adult never is).
			NOT = { has_trait = jiansheng }
			NOT = { has_trait = shengyuan }
			NOT = { has_trait = juren }
			NOT = { has_trait = gongshi }
			NOT = { has_trait = jinshi }
			NOT = { has_trait = hanlin }
			NOT = { has_trait = fanyi_jinshi }
			# [#48] nor already a military-exam degree-holder (parallel guard).
			NOT = { has_trait = wu_shengyuan }
			NOT = { has_trait = wu_juren }
			NOT = { has_trait = wu_jinshi }
			NOT = { has_trait = wu_zhuangyuan }
		}
		# sat once, whatever the outcome.
		set_variable = { name = qing_sat_keju  value = 1 }
		# refresh the pass-rate the roll reads (cheap; idempotent) so a child maturing between
		# triennia still sits against a current metric.
		employer = { QING_exam_compute_pass_rate = yes }
		if = {
			# ---------- THE BANNER TRACK (翻譯科) — Manchu + Mongol bannermen ----------
			limit = {
				OR = {
					culture.culture_group = culture_group:jurchen
					culture.culture_group = culture_group:mongolic
				}
			}
			random = {
				chance = qing_keju_pass_chance_svalue
				# a banner pass: the translation laureate (amban stock), or a lower licentiate.
				if = {
					limit = { exists = employer  employer = { has_variable = qing_exam_pass_rate  var:qing_exam_pass_rate >= 45 } }
					random_list = {
						70 = { add_trait = fanyi_jinshi }
						30 = { add_trait = shengyuan }
					}
				}
				else = {
					random_list = {
						40 = { add_trait = fanyi_jinshi }
						60 = { add_trait = shengyuan }
					}
				}
				LOG_line = { sys = QING  msg = "keju: a bannerman passed the translation examination (翻譯科) for" }
			}
		}
		else = {
			# ---------- HAN & THE GENERAL COURT — civil OR military track ----------
			# [#48] A martially-inclined man (martial the dominant skill) sat the 武舉 (military
			# exam) rather than the civil keju; everyone else sat the civil examination. Bare
			# skill-token comparisons on the RHS (martial > finesse) are the proven idiom
			# (se_POLITICS.txt:387, diplomatic_play_events.txt). The wuju was a HAN path into the
			# Green Standard officer corps; bannermen keep the translation-exam branch above.
			if = {
				limit = {
					martial > finesse
					martial > charisma
				}
				# ---------- THE MILITARY TRACK (武科) → the wuju degrees ----------
				random = {
					chance = qing_wuju_pass_chance_svalue
					set_variable = { name = qing_wuju_passed  value = 1 }
				}
				if = {
					limit = { has_variable = qing_wuju_passed }
					# martial degree tier by the realm's pass-rate band (parallels the civil ladder).
					if = {
						# vigorous ladder: a slim chance at the 武狀元 apex, else a strong martial degree.
						limit = { exists = employer  employer = { has_variable = qing_exam_pass_rate  var:qing_exam_pass_rate >= 60 } }
						random_list = {
							5  = { add_trait = wu_zhuangyuan }
							35 = { add_trait = wu_jinshi }
							60 = { add_trait = wu_juren }
						}
					}
					else_if = {
						# middling ladder.
						limit = { exists = employer  employer = { has_variable = qing_exam_pass_rate  var:qing_exam_pass_rate >= 30 } }
						random_list = {
							10 = { add_trait = wu_jinshi }
							40 = { add_trait = wu_juren }
							50 = { add_trait = wu_shengyuan }
						}
					}
					else = {
						# choked ladder: only the lower rungs.
						random_list = {
							30 = { add_trait = wu_juren }
							70 = { add_trait = wu_shengyuan }
						}
					}
					remove_variable = qing_wuju_passed
					LOG_line = { sys = QING  msg = "wuju: a candidate passed the military examination (武科) for" }
				}
				# a failed military candidate takes NO degree — there is no purchased-rank floor on the
				# martial ladder (捐監 was a CIVIL rank); he joins the degreeless the pickers fall back on.
			}
			else = {
				# ---------- THE CIVIL TRACK (文科) ----------
				# NOTE: `random` has NO `else` branch (else is a sibling of if/else_if only), so we
				# roll the pass into a one-shot marker, then branch pass/fail with a proper if/else.
				random = {
					chance = qing_keju_pass_chance_svalue
					set_variable = { name = qing_keju_passed  value = 1 }
				}
			if = {
				limit = { has_variable = qing_keju_passed }
				# degree tier by the realm's pass-rate band (mirrors QING_exam_graduate_cohort).
				if = {
					# vigorous ladder: a chance at the Hanlin apex, else a strong civil degree.
					limit = { exists = employer  employer = { has_variable = qing_exam_pass_rate  var:qing_exam_pass_rate >= 60 } }
					random_list = {
						10 = { add_trait = hanlin }
						30 = { add_trait = jinshi }
						30 = { add_trait = gongshi }
						30 = { add_trait = juren }
					}
				}
				else_if = {
					# middling ladder.
					limit = { exists = employer  employer = { has_variable = qing_exam_pass_rate  var:qing_exam_pass_rate >= 30 } }
					random_list = {
						5  = { add_trait = jinshi }
						25 = { add_trait = gongshi }
						40 = { add_trait = juren }
						30 = { add_trait = shengyuan }
					}
				}
				else = {
					# choked ladder: only the lower rungs.
					random_list = {
						30 = { add_trait = juren }
						45 = { add_trait = shengyuan }
						25 = { add_trait = jiansheng }
					}
				}
				remove_variable = qing_keju_passed
				LOG_line = { sys = QING  msg = "keju: a candidate passed the civil examination (文科) for" }
			}
			else = {
				# a failed candidate takes no merit degree — but a well-off family may still
				# purchase the 監生 floor (捐監), the historical sale-of-rank escape. Most fail
				# outright and join the degreeless commoners the pickers fall back on.
				random = {
					chance = 15
					add_trait = jiansheng
						LOG_line = { sys = QING  msg = "keju: a failed candidate purchased the 監生 rank (捐監) for" }
					}
				}
			}
		}
	}
}

QING_grant_martial_degree = {
