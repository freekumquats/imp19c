########################################################################
# se_ECON_LOG.txt — economic diagnostic tracing.
#
# A thin, domain-specific wrapper over the se_LOG module (sys = ECON). Its job
# is to make the economy pipeline OBSERVABLE: the big trade/currency/wealth
# effects run with no tracing, so a wrong quarterly number has nothing to grep.
# This module emits a compact, greppable per-country snapshot each quarter plus
# phase-boundary markers, so a -debug_mode session produces a trace we can read.
#
# IMPORTANT: every line here ultimately calls se_LOG's debug_log, which ONLY
# outputs under the -debug_mode launch option. In a normal session this whole
# module is a silent no-op — zero file I/O, zero string formatting. It cannot
# affect gameplay and cannot regress performance.
#
# Extract just the economic trace with:
#     grep "\[IMP19C\]\[ECON\]" debug.log > econ_trace.log
# Isolate a slice, e.g. the per-country snapshots:
#     grep "\[IMP19C\]\[ECON\] \[SNAP\]" debug.log
#
# SYNTAX NOTE: inside a debug_log effect the only verified-working promptable
# form for a scripted variable is the same one used across the Qing suite:
#     [ROOT.MakeScope.GetVariable('name').GetValue|0]
# (The Player.MakeScope / GuiScope forms seen in GUI tooltips are a GUI context
# and are NOT valid here.) So every value below is first STAGED into a ROOT
# temp variable via ordinary cross-scope reads, then logged in that form, then
# cleaned up. Temp vars are prefixed ECON_LOG_tmp_ and always removed.
#
#   ROOT / this = the country being traced
# Prefix = ECON_LOG · sys tag = ECON
########################################################################

# ======================================================================
# PHASE MARKERS — cheap ENTER/EXIT lines to bracket a pipeline stage, so the
# trace shows the order stages actually ran and where one aborted. Pass the
# stage name in $stage$. Called from the quarterly on_action chain.
# ======================================================================
ECON_LOG_phase_enter = {
	# Scope: any
	LOG_enter = { sys = ECON  fn = "$stage$" }
}

ECON_LOG_phase_exit = {
	# Scope: any
	LOG_exit = { sys = ECON  fn = "$stage$"  result = OK }
}

# ======================================================================
# COUNTRY SNAPSHOT — treasury + national private wealth + whether the country
# runs a monetary economy. Country scope. Reads only already-computed state.
# WEALTH_total_private_moveable_wealth is the national-wealth var cached by
# WEALTH_cache_national_wealth (se_ECON_wealth.txt:6), so this must be called
# AFTER that cache in the quarterly pulse (it is — see wire-up).
# ======================================================================
ECON_LOG_country_snapshot = {
	# Scope: country
	# [#98-fix task #98] this composite runs inside every_country in the quarterly
	# pulse (oa_wealth_changes.txt:185), so `this` = the iterated country but the
	# LOG_line reads via [ROOT.MakeScope...] / tags [ROOT.GetTag] — ROOT being the
	# on_action root, NOT the iterated country. The temp vars were staged on bare
	# `this` yet read off ROOT, so the line logged the wrong tag and read a
	# stale/zero temp. Stage a self-scope handle and stage/read the temps on ROOT
	# consistently (the ROOT.MakeScope form is the only promptable one here — see the
	# file's SYNTAX NOTE; the paired ECON_LOG_production_snapshot already does this).
	save_scope_as = econ_log_country
	# stage treasury (readable as a plain value — se_INCOME.txt:469) and wealth
	root = {
		set_variable = { name = ECON_LOG_tmp_treasury  value = scope:econ_log_country.treasury }
		set_variable = { name = ECON_LOG_tmp_wealth    value = 0 }
	}
	if = {
		limit = { has_variable = WEALTH_total_private_moveable_wealth }
		root = { set_variable = { name = ECON_LOG_tmp_wealth  value = scope:econ_log_country.var:WEALTH_total_private_moveable_wealth } }
	}
	LOG_line = {
		sys = ECON
		msg = " treasury= natl_wealth="
	}
	root = {
		remove_variable = ECON_LOG_tmp_treasury
		remove_variable = ECON_LOG_tmp_wealth
	}
}

# ======================================================================
# JOBS SNAPSHOT — employment used vs unemployed, aggregated across owned
# provinces. Country scope. Uses the inline "value = 0 ... every_owned_province
# = { add }" accumulation idiom (JOBS_svalues.txt:198). JOBS_num_used_slots and
# JOBS_unemployed_pops_province are province-scope script values
# (JOBS_svalues.txt:346/:351).
# ======================================================================
ECON_LOG_jobs_snapshot = {
	# Scope: country
	# [#98-fix task #98] same every_country scope fix as ECON_LOG_country_snapshot.
	# Keep the natural-scope every_owned_province accumulation (the JOBS_svalues idiom)
	# on `this`, then COPY the totals into ROOT temps via the proven cross-scope
	# var-read form so the LOG_line's [ROOT.MakeScope...] read matches what was staged,
	# and label the line with the iterated country instead of ROOT.
	save_scope_as = econ_log_country
	set_variable = {
		name = ECON_LOG_tmp_used
		value = 0
		every_owned_province = {
			add = JOBS_num_used_slots
		}
	}
	set_variable = {
		name = ECON_LOG_tmp_unemployed
		value = 0
		every_owned_province = {
			add = JOBS_unemployed_pops_province
		}
	}
	root = {
		set_variable = { name = ECON_LOG_tmp_used        value = scope:econ_log_country.var:ECON_LOG_tmp_used }
		set_variable = { name = ECON_LOG_tmp_unemployed  value = scope:econ_log_country.var:ECON_LOG_tmp_unemployed }
	}
	LOG_line = {
		sys = ECON
		msg = " used= unemployed="
	}
	# clean both the this-scope accumulators and the ROOT read-temps (harmless
	# double-remove if ROOT == this in a non-iterated call).
	remove_variable = ECON_LOG_tmp_used
	remove_variable = ECON_LOG_tmp_unemployed
	root = {
		remove_variable = ECON_LOG_tmp_used
		remove_variable = ECON_LOG_tmp_unemployed
	}
}

# ======================================================================
# CURRENCY SNAPSHOT — the country's official currency backing value and
# purchasing power, read off the currency province via the confirmed cross-
# scope form var:official_currency.var:X (cf. se_INCOME.txt:544). Only countries
# with a monetary economy have official_currency; others log a NONE line so the
# trace still records that the country was visited.
# ======================================================================
ECON_LOG_currency_snapshot = {
	# Scope: country
	# [#98-fix task #98] same every_country scope fix as the sibling snapshots: stage
	# the temps on ROOT (where the LOG_line reads them) and label the line with the
	# iterated country, not ROOT.
	save_scope_as = econ_log_country
	if = {
		limit = { has_variable = official_currency }
		root = {
			set_variable = { name = ECON_LOG_tmp_backing  value = 0 }
			set_variable = { name = ECON_LOG_tmp_pp        value = 0 }
		}
		if = {
			limit = { var:official_currency = { has_variable = backing_value } }
			root = { set_variable = { name = ECON_LOG_tmp_backing  value = scope:econ_log_country.var:official_currency.var:backing_value } }
		}
		if = {
			limit = { var:official_currency = { has_variable = purchasing_power } }
			root = { set_variable = { name = ECON_LOG_tmp_pp  value = scope:econ_log_country.var:official_currency.var:purchasing_power } }
		}
		LOG_line = {
			sys = ECON
			msg = " backing= purchasing_power="
		}
		root = {
			remove_variable = ECON_LOG_tmp_backing
			remove_variable = ECON_LOG_tmp_pp
		}
	}
	else = {
		LOG_line = { sys = ECON  msg = " no official currency (metal/barter economy)" }
	}
}

# ======================================================================
# QUARTER — the single composite call the pulse makes per country. Country
# scope. Guarded by ECON_LOG_enabled so the whole snapshot can be toggled from
# one place at zero cost even when running in -debug_mode (avoids trace spam
# when only phase markers are wanted). The guard variable is optional: if never
# set, snapshots run whenever -debug_mode is on (the debug_log gate still applies).
# ======================================================================
ECON_LOG_quarter = {
	# Scope: country
	ECON_LOG_country_snapshot = yes
	ECON_LOG_jobs_snapshot = yes
	ECON_LOG_currency_snapshot = yes
}

# ======================================================================
# PRODUCTION SNAPSHOT (#72) — the quarterly goods-production path
# (GOODS_governorship_produce_all in se_GOODS.txt) previously ran with ZERO
# tracing despite being bracketed by the "quarterly_production" phase markers,
# so a wrong or zeroed production number had nothing to grep. This logs, per
# governorship, the cached industrialisation multiplier (the value the divide-
# by-zero fix #70 guards and the cache #71 stores) plus a couple of
# representative stockpile totals, so the trace records what production actually
# saw. Governorship scope. Reads only already-computed state; silent no-op
# outside -debug_mode like the rest of this module.
# ======================================================================
ECON_LOG_production_snapshot = {
	# Scope: country. Aggregates the per-governorship cached industrialisation
	# multiplier (the value #70 guards against divide-by-zero and #71 caches) across
	# the country's governorships into ROOT temp vars, then logs via the verified
	# ROOT.MakeScope form (the only promptable form confirmed to work here — see the
	# SYNTAX NOTE at the top of this file; This/Player/GuiScope forms are NOT valid).
	# So production now has a greppable per-country line where it had none.
	#
	# [logfix #19 R2 — the REAL bug] This effect runs inside `every_country` (oa_wealth_changes.txt:166),
	# so `this` = the iterated country while `root` = the on_action origin. The temp vars were INITIALISED
	# on bare `this` (set_variable below) but INCREMENTED inside `root = {}` (change_variable) — two
	# DIFFERENT scopes. So change_variable hit a var that was never set on root -> "Variable not of the
	# 'value' scope type. Type: empty", ~1860 lines/session. The July-11 fix corrected only the READ idiom
	# (prev.var -> scope:x.var) and never noticed init-scope != modify-scope. Fix: stage ALL temp-var ops
	# (init, increment, cleanup) on ROOT consistently — exactly what the working ECON_LOG_country_snapshot
	# above does. The per-governorship value is still read via the proven scope:x.var: idiom (se_SELL.txt:1569).
	root = {
		set_variable = { name = ECON_LOG_tmp_gov_count  value = 0 }
		set_variable = { name = ECON_LOG_tmp_bonus_sum  value = 0 }
	}
	every_governorships = {
		limit = { has_variable = industrialisation_bonus_cached }
		save_scope_as = econ_log_snapshot_gov
		root = {
			change_variable = { name = ECON_LOG_tmp_gov_count  add = 1 }
			change_variable = { name = ECON_LOG_tmp_bonus_sum   add = scope:econ_log_snapshot_gov.var:industrialisation_bonus_cached }
		}
	}
	LOG_line = {
		sys = ECON
		msg = " governorships_producing= summed_industrialisation_bonus="
	}
	root = {
		remove_variable = ECON_LOG_tmp_gov_count
		remove_variable = ECON_LOG_tmp_bonus_sum
	}
}

# ======================================================================
# COTTAGE EMPTY-VAR PROBE (#MG-flood-diag) — read-only diagnostic for the long-
# standing cottage-production empty-var flood (change_variable "Type: empty" in
# COTTAGEIND_scale_production, ~30 goods, a FIXED ~8 scopes/boot, runtime-gated,
# predates all #133/#139/#144 MG work). Does NOT modify Sobisonator's cottage
# effects — it re-walks the SAME governorships the quarterly pass hits and, for
# each, records every operand in the propagation chain that produces the empty:
#
#   GOODS_governorship_<raw>_produced  (value=0, then multiply=owner.MODIFIER_*)
#        └─ if owner unresolved / MODIFIER_* empty -> raw svalue returns EMPTY
#   -> COTTAGEIND_raw_<input>   (cache_all_values: set = the raw svalue)      EMPTY
#   -> COTTAGEIND_produced_<good> (produce_X: set = var:COTTAGEIND_raw_input) EMPTY
#   -> COTTAGEIND_scale_production line-8 change_variable multiply -> "Type: empty"
#
# Scope: COUNTRY (called inside every_country in the quarterly pulse, same site
# as ECON_LOG_production_snapshot). For each of the country's governorships it
# logs a scope dump (identity + owner) ONLY when a probe operand is empty/absent,
# so the trace pinpoints exactly which scopes flood and why. Silent no-op outside
# -debug_mode (all output via LOG_*/debug_log). Uses clothing's inputs
# (textile_fibres + silk) as the representative recipe — it is one of the flooding
# goods and its raws are the most common, so a scope empty here is empty for all.
# ======================================================================
ECON_LOG_cottage_empty_probe = {
	# Scope: country
	save_scope_as = econ_log_probe_country
	every_governorships = {
		save_scope_as = econ_log_probe_gov
		# --- 1. OWNER RESOLUTION: the suspected root. GOODS_governorship_<raw>_produced
		#        does `multiply = owner.MODIFIER_agriculture_productivity` — if owner does
		#        not resolve, the whole svalue returns empty and propagates. Log any
		#        governorship whose owner link is unset (the prime hypothesis for the 8).
		if = {
			limit = { NOT = { exists = owner } }
			LOG_fail = {
				sys = ECON
				fn = "cottage_empty_probe"
				reason = "GOVERNORSHIP HAS NO OWNER — owner.MODIFIER_* returns empty -> raw svalue empty -> cottage flood (scope dump: identity of this ownerless governorship)"
			}
		}
		else = {
			# owner resolves — capture the modifier values that the raw svalue multiplies
			# by. If either is empty/absent, that is the empty source instead.
			owner = {
				save_scope_as = econ_log_probe_owner
				if = {
					limit = { NOT = { exists = var:tribute_income_modifier } }
					# MODIFIER_agriculture_productivity reads modifier:tribute_income_modifier;
					# a modifier (not a variable) — logged here only as context, not a var.
					LOG_line = { sys = ECON  msg = "cottage_empty_probe: owner present, agriculture-modifier context (see scope dump for tag)" }
				}
			}
		}
		# --- 2. RAW-INPUT CACHE STATE: were the raw vars actually seeded this pass?
		#        cache_all_values sets COTTAGEIND_raw_<input> = GOODS_governorship_<raw>_produced.
		#        If the svalue returned empty, the set produced an EMPTY var (has_variable can
		#        still be true). Stage the value onto the country ROOT and log it.
		if = {
			limit = { NOT = { has_variable = COTTAGEIND_raw_textile_fibres } }
			LOG_fail = {
				sys = ECON
				fn = "cottage_empty_probe"
				reason = "COTTAGEIND_raw_textile_fibres NOT SET in this governorship — cache_all_values never seeded it (scope dump)"
			}
		}
		if = {
			limit = { NOT = { has_variable = COTTAGEIND_raw_silk } }
			LOG_fail = {
				sys = ECON
				fn = "cottage_empty_probe"
				reason = "COTTAGEIND_raw_silk NOT SET in this governorship (scope dump)"
			}
		}
		# --- 3. POPS-OUTPUT CACHE STATE: the OTHER multiply operand in scale_production
		#        line-6 (multiply = var:COTTAGEIND_pops_output). Log if absent.
		if = {
			limit = { NOT = { has_variable = COTTAGEIND_pops_output } }
			LOG_fail = {
				sys = ECON
				fn = "cottage_empty_probe"
				reason = "COTTAGEIND_pops_output NOT SET in this governorship — scale_production line-6 multiply operand missing (scope dump)"
			}
		}
		# --- 4. PRODUCED-VAR STATE after a real produce pass would have run: log the
		#        clothing produced var's presence. Absent => the set_variable from an
		#        empty raw yielded nothing. (State capture; the scope dump names the gov.)
		if = {
			limit = { NOT = { has_variable = COTTAGEIND_produced_clothing } }
			LOG_line = { sys = ECON  msg = "cottage_empty_probe: COTTAGEIND_produced_clothing absent post-pass (see scope dump)" }
		}
		# --- 5. STRATA/POP STATE: an ownerless or pop-less governorship is the likely
		#        shared trait of the 8. Record whether the pop-count vars the pops_output
		#        svalue depends on are present, so we can correlate the 8 scopes.
		if = {
			limit = { NOT = { has_variable = governorship_middle_strata } }
			LOG_line = { sys = ECON  msg = "cottage_empty_probe: governorship_middle_strata absent (pop-state anomaly; see scope dump)" }
		}
	}
	root = { remove_variable = econ_log_probe_country }
}

# ======================================================================
# STOCKPILE-WRITE EMPTY-VAR PROBE (#MG-flood-diag) — the SECOND, distinct flood:
# GOODS_governorship_produce (se_GOODS.txt) does `change_variable name =
# $into_stockpile$` guarded ONLY by `has_variable = $do_if$` (the produces_X
# marker), NOT by whether the stockpile var itself was seeded. This is the raw/
# manufactured stockpile write erroring at GOODS_governorship_produce line 6.
# This probe walks the governorships and logs which <good>_stockpile vars are
# absent while their produces_ marker is present — the exact guard-gap mismatch.
# Representative goods chosen from the observed error offsets (raw + manufactured).
# Read-only; does not touch se_GOODS.txt. Scope: country.
# ======================================================================
ECON_LOG_stockpile_gap_probe = {
	# Scope: country
	every_governorships = {
		save_scope_as = econ_log_probe_gov
		# wool / chocolate / peat were among the named erroring stockpiles; clothing/
		# construction_materials among the manufactured. Log any produces_-marked good
		# whose _stockpile var is missing (guard passes, mutate target empty).
		if = {
			limit = { has_variable = produces_wool  NOT = { has_variable = wool_stockpile } }
			LOG_fail = { sys = ECON  fn = "stockpile_gap_probe"  reason = "produces_wool set but wool_stockpile NOT set — GOODS_governorship_produce mutate-empty (scope dump)" }
		}
		if = {
			limit = { has_variable = produces_chocolate  NOT = { has_variable = chocolate_stockpile } }
			LOG_fail = { sys = ECON  fn = "stockpile_gap_probe"  reason = "produces_chocolate set but chocolate_stockpile NOT set (scope dump)" }
		}
		if = {
			limit = { has_variable = produces_peat  NOT = { has_variable = peat_stockpile } }
			LOG_fail = { sys = ECON  fn = "stockpile_gap_probe"  reason = "produces_peat set but peat_stockpile NOT set (scope dump)" }
		}
	}
}

# ======================================================================
ECON_LOG_flood_probe = {
	# Scope: country
	ECON_LOG_cottage_empty_probe = yes
	ECON_LOG_stockpile_gap_probe = yes
}
