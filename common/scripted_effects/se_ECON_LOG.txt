########################################################################
# se_ECON_LOG.txt — economic diagnostic tracing.
#
# A thin, domain-specific wrapper over the se_LOG module (sys = ECON). Its job
# is to make the economy pipeline OBSERVABLE: the big trade/currency/wealth
# effects run with no tracing, so a wrong quarterly number has nothing to grep.
# This module emits a compact, greppable per-country snapshot each quarter plus
# phase-boundary markers, so a -debug_mode session produces a trace we can read.
#
# IMPORTANT: every line here ultimately calls se_LOG's debug_log, which ONLY
# outputs under the -debug_mode launch option. In a normal session this whole
# module is a silent no-op — zero file I/O, zero string formatting. It cannot
# affect gameplay and cannot regress performance.
#
# Extract just the economic trace with:
#     grep "\[IMP19C\]\[ECON\]" debug.log > econ_trace.log
# Isolate a slice, e.g. the per-country snapshots:
#     grep "\[IMP19C\]\[ECON\] \[SNAP\]" debug.log
#
# SYNTAX NOTE: inside a debug_log effect the only verified-working promptable
# form for a scripted variable is the same one used across the Qing suite:
#     [ROOT.MakeScope.GetVariable('name').GetValue|0]
# (The Player.MakeScope / GuiScope forms seen in GUI tooltips are a GUI context
# and are NOT valid here.) So every value below is first STAGED into a ROOT
# temp variable via ordinary cross-scope reads, then logged in that form, then
# cleaned up. Temp vars are prefixed ECON_LOG_tmp_ and always removed.
#
#   ROOT / this = the country being traced
# Prefix = ECON_LOG · sys tag = ECON
########################################################################

# ======================================================================
# CURRENCY SNAPSHOT — the country's official currency backing value and
# purchasing power, read off the currency province via the confirmed cross-
# scope form var:official_currency.var:X (cf. se_INCOME.txt:544). Only countries
# with a monetary economy have official_currency; others log a NONE line so the
# trace still records that the country was visited.
# ======================================================================
ECON_LOG_currency_snapshot = {
	# Scope: country
	# [#98-fix task #98] same every_country scope fix as the sibling snapshots: stage
	# the temps on ROOT (where the LOG_line reads them) and label the line with the
	# iterated country, not ROOT.
	# [#log-noise-fix 2026-08-03] The has_variable=official_currency branch RENDER was removed —
	# it staged backing/purchasing_power only to feed the unresolvable ROOT.MakeScope render (see
	# ECON_LOG_country_snapshot note; "Could not find promote" noise, no value).
	# [#28 2026-08-07] REFILLED the if-branch with BAND-BUCKETING (no numeric render — the render is
	# infeasible in a script debug_log, proven by the removal notes above; IGNORE the stale top-of-file
	# SYNTAX NOTE that still advertises ROOT.MakeScope). CHI-only (the #23 deflation report is CHI; avoids
	# flooding every monetary tag). Stage each script_value into a temp var (proven not to error), then
	# classify with sentinel-guarded band ladders (the ECON_LOG_fx_classify_good idiom, :322) — the band
	# ladder is NESTED inside the > -999999999 sentinel so an EMPTY/Div0 staged value is labelled EMPTY,
	# NOT mislabelled as the lowest band. Serves #23: shows whether private_cash_ratio is pinned at the
	# deflation floor and whether the denominator (private_cash_needed) is what moves.
	if = {
		limit = { has_variable = official_currency  tag = CHI }
		# --- private_cash_ratio: the core signal (0..~2.6; deflation floor at <0.1 ⇒ ~-10%) ---
		set_variable = { name = ECON_LOG_cur_ratio  value = CURRENCY_private_cash_ratio }
		if = { limit = { NOT = { has_variable = ECON_LOG_cur_ratio } }        debug_log = "IMP19C CURR ratio = UNSET" }
		else_if = {
			limit = { var:ECON_LOG_cur_ratio > -999999999 }
			if = { limit = { var:ECON_LOG_cur_ratio >= 1 }        debug_log = "IMP19C CURR ratio >= 1.0 (inflation side)" }
			else_if = { limit = { var:ECON_LOG_cur_ratio >= 0.5 } debug_log = "IMP19C CURR ratio 0.5-1.0 (mild deflation)" }
			else_if = { limit = { var:ECON_LOG_cur_ratio >= 0.1 } debug_log = "IMP19C CURR ratio 0.1-0.5 (serious deflation)" }
			else = { debug_log = "IMP19C CURR ratio < 0.1 (deflation FLOOR ~ -10%)" }
		}
		else = { debug_log = "IMP19C CURR ratio = EMPTY" }
		remove_variable = ECON_LOG_cur_ratio
		# --- private_cash_needed: the DENOMINATOR (does IT move → #23 hypothesis). Coarse magnitude bands. ---
		set_variable = { name = ECON_LOG_cur_need  value = CURRENCY_private_cash_needed }
		if = { limit = { NOT = { has_variable = ECON_LOG_cur_need } }        debug_log = "IMP19C CURR needed = UNSET" }
		else_if = {
			limit = { var:ECON_LOG_cur_need > -999999999 }
			if = { limit = { var:ECON_LOG_cur_need >= 100 }       debug_log = "IMP19C CURR needed >= 100" }
			else_if = { limit = { var:ECON_LOG_cur_need >= 50 }   debug_log = "IMP19C CURR needed 50-100" }
			else_if = { limit = { var:ECON_LOG_cur_need >= 25 }   debug_log = "IMP19C CURR needed 25-50" }
			else_if = { limit = { var:ECON_LOG_cur_need >= 10 }   debug_log = "IMP19C CURR needed 10-25" }
			else_if = { limit = { var:ECON_LOG_cur_need >= 0 }    debug_log = "IMP19C CURR needed 0-10" }
			else = { debug_log = "IMP19C CURR needed < 0 (negative denominator)" }
		}
		else = { debug_log = "IMP19C CURR needed = EMPTY" }
		remove_variable = ECON_LOG_cur_need
		# --- amt_circulated_scaled: the NUMERATOR (money in circulation). Coarse magnitude bands. ---
		set_variable = { name = ECON_LOG_cur_circ  value = CURRENCY_amt_circulated_scaled }
		if = { limit = { NOT = { has_variable = ECON_LOG_cur_circ } }        debug_log = "IMP19C CURR circ = UNSET" }
		else_if = {
			limit = { var:ECON_LOG_cur_circ > -999999999 }
			if = { limit = { var:ECON_LOG_cur_circ >= 100 }       debug_log = "IMP19C CURR circ >= 100" }
			else_if = { limit = { var:ECON_LOG_cur_circ >= 50 }   debug_log = "IMP19C CURR circ 50-100" }
			else_if = { limit = { var:ECON_LOG_cur_circ >= 25 }   debug_log = "IMP19C CURR circ 25-50" }
			else_if = { limit = { var:ECON_LOG_cur_circ >= 10 }   debug_log = "IMP19C CURR circ 10-25" }
			else = { debug_log = "IMP19C CURR circ < 10" }
		}
		else = { debug_log = "IMP19C CURR circ = EMPTY" }
		remove_variable = ECON_LOG_cur_circ
	}
	if = {
		limit = { NOT = { has_variable = official_currency } }
		LOG_line = { sys = ECON  msg = " no official currency (metal/barter economy)" }
	}
}

# ======================================================================
# [#23 2026-08-07] EXACT-VALUE CURRENCY FORENSIC — the whole causal chain as EXACT numbers.
#
# WHY: numeric render into debug_log is IMPOSSIBLE in scripted-effect context (proven: the
# [ROOT.MakeScope.GetVariable().GetValue] form emits "Could not find promote" and renders
# NOTHING — see the removal notes throughout this file; the 1.6M-line Aug-7 trace contains
# zero rendered svalue magnitudes). So #28's band snapshot could only bucket, not measure —
# it cannot tell "modest inflation → severe deflation" (the reported truth) from noise, and
# cannot show which upstream input changed. This probe recovers EXACT values render-free.
#
# METHOD (proven idioms only): for each chain value V, stage floor(V * SCALE) into a temp var,
# then emit that integer as a run of marker lines via a capped `while` (count+limit+change_variable
# is the proven idiom, se_CURRENCY.txt:1422). Read back EXACTLY:
#     (per quarter, between two "CURX QUARTER-MARK" lines)
#     grep -c "IMP19C CURX tick ratio"   debug.log   == floor(ratio   * 1000)
#     grep -c "IMP19C CURX tick defl"    debug.log   == floor(deflation% * 1000)   (0..100 => 0..~100k? no: %*1000)
# so ratio 0.087 -> 87 ticks; deflation 0.0993 -> 99 ticks; a MODEST inflation ratio 1.02 -> 1020 ticks.
# The CAP (max count) bounds the loop so a huge value can't hang the boot; a value hitting the cap
# logs a "CAPPED" line so we know to raise it. $param$ interpolation into debug_log is proven
# (ECON_LOG_fx_classify_good uses $tradegood$); value = $SV$ macro param is proven (se_ECON_wealth,
# se_QING_MINISTRY). Silent outside -debug_mode. CHI-only. Read-only (temp vars removed).
#
# CHAIN LOGGED (every operand from symptom back to primitives — so a boot can NEVER end in
# "I still don't know which input moved"):
#   OUTPUTS:   deflation%, inflation%              (the reported symptom, *1000)
#   RATIO:     private_cash_ratio                  (the core signal, *1000)
#   NUMERATOR: amt_circulated_scaled               (money supply, *100)
#   DENOM:     private_cash_needed                 (*100)
#     its inputs:  essentials_buying_power (*100), country_population (*1),
#                  trade_wealth_outgoing (*100), wealth_generated (*100)
#   RESERVE:   reserve_ratio_total (*1000), wealth_value_1_unit_scaled_by_reserve_ratio (*100)
#   FEEDBACK:  inflation_cash_selloff_amt (*100), deflation_money_demand_amt (*1), minting_rate (*1)
# ======================================================================

# --- exact-integer emitter via DIGIT DECOMPOSITION (magnitude-independent). ---------------
# Scope: country. Emits floor($val$ * $scale$) as one line PER DECIMAL DIGIT, most-significant
# first, tagged with its position — so ANY magnitude costs ~7 lines, not O(value), and no cap
# can truncate. Read back per quarter with, e.g.:
#     grep "IMP19C CURX digit ratio " <quarter-window>
#     -> lines like "...digit ratio p3 =7", "...p2 =0", "...p1 =2", "...p0 =0"  => 7020  (=ratio*1000)
# Method: peel the highest power of ten <= value, emit its leading digit, subtract, repeat.
# Bounded loop (fixed 9 positions => values up to 1e9-1). Sign emitted separately. Proven idioms:
# while+count+limit (se_CURRENCY.txt:1422), round=floor at svalue top-level (ECON_LOG_curx_floor_sv,
# ECON_svalues.txt — the ONLY legal spot for round in this codebase), $param$ value source
# (se_ECON_wealth), $param$ into debug_log (ECON_LOG_fx_classify_good). Digit equality (var = N) is
# safe because every digit is a clean integer 0..9 produced by the floor svalue.
ECON_LOG_curx_emit = {
	# Scope: country
	# floor(val*scale): stage the raw product, then floor it via a proven top-level svalue
	# (round=floor is legal ONLY at svalue top-level in this codebase — grep-confirmed it is
	# never used inside set/change_variable — so do it there, NOT in a value sub-block).
	set_variable = { name = ECON_LOG_curx_raw  value = $val$  multiply = $scale$ }
	set_variable = { name = ECON_LOG_curx_v    value = ECON_LOG_curx_floor_sv }
	# [review-fix CRITICAL] EMPTY guard — MUST mirror the band-snapshot sentinel (:118-164). A
	# set-but-empty svalue (Div0 / empty operand — the EXACT failure #23 is chasing) stores an
	# empty var whose digit ladder would silently emit nine "=9" lines = a bogus 999999999. Gate
	# the whole decomposition behind the > -999999999 sentinel: an empty errors→false→EMPTY line.
	if = {
		limit = { var:ECON_LOG_curx_v > -999999999 }
		if = {
			limit = { var:ECON_LOG_curx_v < 0 }
			debug_log = "IMP19C CURX sign $tag$ = NEGATIVE"
			change_variable = { name = ECON_LOG_curx_v  multiply = -1 }
		}
		# FIXED start at 10^8 and peel down to 10^0 — always 9 digit lines (pos 8..0), leading zeros
		# included (strip them when reading). This makes EVERY comparison RHS a LITERAL (pw >= 1),
		# avoiding the illegal var/block-on-RHS form (memory: RHS-comparison-operator-rule). Values
		# above 10^9-1 are out of range for all currency chain terms here (ratio*1000 <= ~3000,
		# circ*100/need*100/pop <= ~1e6), so 9 places is ample.
		set_variable = { name = ECON_LOG_curx_pw   value = 100000000 }
		set_variable = { name = ECON_LOG_curx_pos  value = 8 }
		# peel digits most-significant first: digit = floor(rem / pw); rem -= digit*pw; pw /= 10
		set_variable = { name = ECON_LOG_curx_rem  value = var:ECON_LOG_curx_v }
		while = {
			count = 9
			limit = { var:ECON_LOG_curx_pw >= 1 }
			# digit = floor(rem / pw): rem/pw is exact-integer-valued here (both are integers, pw a
			# power of ten) so the fixed-point quotient's fractional part is 0 for the leading digit
			# ONLY when rem<pw*10; to be safe, floor via the proven svalue form.
			set_variable = { name = ECON_LOG_curx_raw  value = var:ECON_LOG_curx_rem  divide = var:ECON_LOG_curx_pw }
			set_variable = { name = ECON_LOG_curx_d    value = ECON_LOG_curx_floor_sv }
			# [review-fix MEDIUM] clamp digit to 0..9. If PDX divide rounds-to-nearest (vs truncates),
			# a rem in the top ~5e-6 of pw at pw>=1e6 could floor to 10 and drive rem negative,
			# poisoning trailing digits. A hard 0..9 clamp makes the peel robust regardless of the
			# engine's divide-rounding mode (min=0 floor, max=9 ceiling on the digit).
			change_variable = { name = ECON_LOG_curx_d  min = 0  max = 9 }
			# emit this digit at its position. One line per digit; grep + read the digit off the tail.
			# (position lets us reassemble even if lines interleave with other tags.)
			ECON_LOG_curx_digitline = { tag = $tag$ }
			# rem -= d * pw
			change_variable = {
				name = ECON_LOG_curx_rem
				subtract = { value = var:ECON_LOG_curx_d  multiply = var:ECON_LOG_curx_pw }
			}
			# pw /= 10 (pw is an exact power of ten, so integer division is exact — no round needed); pos--
			change_variable = { name = ECON_LOG_curx_pw   divide = 10 }
			change_variable = { name = ECON_LOG_curx_pos  subtract = 1 }
		}
	}
	else = {
		# empty/Div0 operand — flag it, exactly like the band snapshot's "= EMPTY". This is a
		# FINDING for #23, not noise: it means this chain term went empty at the sampled moment.
		debug_log = "IMP19C CURX $tag$ = EMPTY (set-but-empty / Div0 — this operand went empty)"
	}
	remove_variable = ECON_LOG_curx_raw
	remove_variable = ECON_LOG_curx_v
	remove_variable = ECON_LOG_curx_pw
	remove_variable = ECON_LOG_curx_pos
	remove_variable = ECON_LOG_curx_rem
	remove_variable = ECON_LOG_curx_d
}

# --- emit ONE digit line. debug_log can't render a number, but the DIGIT (0..9) and POSITION
# (0..8) are tiny, so we emit them as their own greppable tick counts on a sub-marker line —
# actually simplest: emit `pos` p-ticks and `digit` d-ticks is O(100). Instead we exploit that
# 0..9 is only ten cases: a fixed if-ladder prints the literal digit with NO render. Same for pos.
ECON_LOG_curx_digitline = {
	# Scope: country. Reads ECON_LOG_curx_pos (0..8) and ECON_LOG_curx_d (0..9).
	# Build the position label.
	if = { limit = { var:ECON_LOG_curx_pos = 0 } ECON_LOG_curx_dv = { tag = $tag$  pos = p0 } }
	else_if = { limit = { var:ECON_LOG_curx_pos = 1 } ECON_LOG_curx_dv = { tag = $tag$  pos = p1 } }
	else_if = { limit = { var:ECON_LOG_curx_pos = 2 } ECON_LOG_curx_dv = { tag = $tag$  pos = p2 } }
	else_if = { limit = { var:ECON_LOG_curx_pos = 3 } ECON_LOG_curx_dv = { tag = $tag$  pos = p3 } }
	else_if = { limit = { var:ECON_LOG_curx_pos = 4 } ECON_LOG_curx_dv = { tag = $tag$  pos = p4 } }
	else_if = { limit = { var:ECON_LOG_curx_pos = 5 } ECON_LOG_curx_dv = { tag = $tag$  pos = p5 } }
	else_if = { limit = { var:ECON_LOG_curx_pos = 6 } ECON_LOG_curx_dv = { tag = $tag$  pos = p6 } }
	else_if = { limit = { var:ECON_LOG_curx_pos = 7 } ECON_LOG_curx_dv = { tag = $tag$  pos = p7 } }
	else = { ECON_LOG_curx_dv = { tag = $tag$  pos = p8 } }
}

# --- emit the digit value 0..9 as a literal on the line (fixed ladder, no render). ---------
ECON_LOG_curx_dv = {
	# Scope: country. Reads ECON_LOG_curx_d.
	if = { limit = { var:ECON_LOG_curx_d = 0 } debug_log = "IMP19C CURX digit $tag$ $pos$ =0" }
	else_if = { limit = { var:ECON_LOG_curx_d = 1 } debug_log = "IMP19C CURX digit $tag$ $pos$ =1" }
	else_if = { limit = { var:ECON_LOG_curx_d = 2 } debug_log = "IMP19C CURX digit $tag$ $pos$ =2" }
	else_if = { limit = { var:ECON_LOG_curx_d = 3 } debug_log = "IMP19C CURX digit $tag$ $pos$ =3" }
	else_if = { limit = { var:ECON_LOG_curx_d = 4 } debug_log = "IMP19C CURX digit $tag$ $pos$ =4" }
	else_if = { limit = { var:ECON_LOG_curx_d = 5 } debug_log = "IMP19C CURX digit $tag$ $pos$ =5" }

# --- the full-chain dump, one call per quarter (CHI only). ------------------------------
ECON_LOG_curx_dump = {
	# Scope: country
	if = {
		limit = { has_variable = official_currency  tag = CHI }
		# quarter boundary marker — segment the log into per-quarter windows for grep -c.
		# $phase$ distinguishes the two call sites: PRE = ECON_LOG_quarter (~oa_wealth_changes:209,
		# LAST quarter's state, before this quarter's reserve buy/sell + circulation update) vs
		# POST = the fresh site (~:356, THIS quarter's recomputed state). The PRE→POST delta across
		# one recompute is exactly what reveals which input moved when the ratio flips.
		# (No in-game date on log lines; the wall-clock timestamp is the segmentation key.)
		debug_log = "IMP19C CURX QUARTER-MARK $phase$"
		# OUTPUTS — the reported symptom, *1000 (so 0.0993 deflation -> 99, modest 0.02 infl -> 20)
		ECON_LOG_curx_emit = { tag = defl   val = CURRENCY_amt_circulated_deflation  scale = 1000 }
		ECON_LOG_curx_emit = { tag = infl   val = CURRENCY_amt_circulated_inflation   scale = 1000 }
		# RATIO — core signal, *1000 (deflation floor ratio 0.01 -> 10; modest-infl 1.02 -> 1020)
		ECON_LOG_curx_emit = { tag = ratio  val = CURRENCY_private_cash_ratio  scale = 1000 }
		# NUMERATOR — money supply, *100
		ECON_LOG_curx_emit = { tag = circ   val = CURRENCY_amt_circulated_scaled  scale = 100 }
		# DENOMINATOR — *100 — and every one of its inputs, so we can see WHICH moved
		ECON_LOG_curx_emit = { tag = need   val = CURRENCY_private_cash_needed  scale = 100 }
		ECON_LOG_curx_emit = { tag = ess    val = CURRENCY_essentials_buying_power  scale = 100 }
		ECON_LOG_curx_emit = { tag = pop    val = country_population  scale = 1 }
		ECON_LOG_curx_emit = { tag = trout  val = CURRENCY_trade_wealth_outgoing_currency_value  scale = 100 }
		ECON_LOG_curx_emit = { tag = wgen   val = CURRENCY_wealth_generated_country_as_currency_value  scale = 100 }
		# RESERVE channel — the denominator's other driver (reserve ratio scales cost-of-living)
		ECON_LOG_curx_emit = { tag = rratio val = CURRENCY_reserve_ratio_total  scale = 1000 }
		ECON_LOG_curx_emit = { tag = wvunit val = CURRENCY_wealth_value_1_unit_scaled_by_reserve_ratio  scale = 100 }
		# FEEDBACK amounts — the correction magnitudes (the damping-fix targets)
		ECON_LOG_curx_emit = { tag = selloff val = CURRENCY_inflation_cash_selloff_amt  scale = 100 }
		ECON_LOG_curx_emit = { tag = mdemand val = CURRENCY_deflation_money_demand_amt  scale = 1 }
		if = {
			limit = { has_variable = CURRENCY_minting_rate }
			ECON_LOG_curx_emit = { tag = mint  val = var:CURRENCY_minting_rate  scale = 1 }
		}
	}
}

# COTTAGE EMPTY-VAR PROBE (#MG-flood-diag) — read-only diagnostic for the long-
# standing cottage-production empty-var flood (change_variable "Type: empty" in
# COTTAGEIND_scale_production, ~30 goods, a FIXED ~8 scopes/boot, runtime-gated,
# predates all #133/#139/#144 MG work). Does NOT modify Sobisonator's cottage
# effects — it re-walks the SAME governorships the quarterly pass hits and, for
# each, records every operand in the propagation chain that produces the empty:
#
#   GOODS_governorship_<raw>_produced  (value=0, then multiply=owner.MODIFIER_*)
#        └─ if owner unresolved / MODIFIER_* empty -> raw svalue returns EMPTY
#   -> COTTAGEIND_raw_<input>   (cache_all_values: set = the raw svalue)      EMPTY
#   -> COTTAGEIND_produced_<good> (produce_X: set = var:COTTAGEIND_raw_input) EMPTY
#   -> COTTAGEIND_scale_production line-8 change_variable multiply -> "Type: empty"
#
# Scope: COUNTRY (called inside every_country in the quarterly pulse, same site
# as ECON_LOG_production_snapshot). For each of the country's governorships it
# logs a scope dump (identity + owner) ONLY when a probe operand is empty/absent,
# so the trace pinpoints exactly which scopes flood and why. Silent no-op outside
# -debug_mode (all output via LOG_*/debug_log). Uses clothing's inputs
# (textile_fibres + silk) as the representative recipe — it is one of the flooding
# goods and its raws are the most common, so a scope empty here is empty for all.
# ======================================================================
ECON_LOG_cottage_empty_probe = {
	# Scope: country
	save_scope_as = econ_log_probe_country
	every_governorships = {
		save_scope_as = econ_log_probe_gov
		# --- 1. OWNER RESOLUTION: the suspected root. GOODS_governorship_<raw>_produced
		#        does `multiply = owner.MODIFIER_agriculture_productivity` — if owner does
		#        not resolve, the whole svalue returns empty and propagates. Log any
		#        governorship whose owner link is unset (the prime hypothesis for the 8).
		if = {
			limit = { NOT = { exists = owner } }
			LOG_fail = {
				sys = ECON
				fn = "cottage_empty_probe"
				reason = "GOVERNORSHIP HAS NO OWNER — owner.MODIFIER_* returns empty -> raw svalue empty -> cottage flood (scope dump: identity of this ownerless governorship)"
			}
		}
		else = {
			# owner resolves — capture the modifier values that the raw svalue multiplies
			# by. If either is empty/absent, that is the empty source instead.
			owner = {
				save_scope_as = econ_log_probe_owner
				if = {
					limit = { NOT = { exists = var:tribute_income_modifier } }
					# MODIFIER_agriculture_productivity reads modifier:tribute_income_modifier;
					# a modifier (not a variable) — logged here only as context, not a var.
					LOG_line = { sys = ECON  msg = "cottage_empty_probe: owner present, agriculture-modifier context (see scope dump for tag)" }
				}
			}
		}
		# --- 2. RAW-INPUT CACHE STATE: were the raw vars actually seeded this pass?
		#        cache_all_values sets COTTAGEIND_raw_<input> = GOODS_governorship_<raw>_produced.
		#        If the svalue returned empty, the set produced an EMPTY var (has_variable can
		#        still be true). Stage the value onto the country ROOT and log it.
		if = {
			limit = { NOT = { has_variable = COTTAGEIND_raw_textile_fibres } }
			LOG_fail = {
				sys = ECON
				fn = "cottage_empty_probe"
				reason = "COTTAGEIND_raw_textile_fibres NOT SET in this governorship — cache_all_values never seeded it (scope dump)"
			}
		}
		if = {
			limit = { NOT = { has_variable = COTTAGEIND_raw_silk } }
			LOG_fail = {
				sys = ECON
				fn = "cottage_empty_probe"
				reason = "COTTAGEIND_raw_silk NOT SET in this governorship (scope dump)"
			}
		}
		# --- 3. POPS-OUTPUT CACHE STATE: the OTHER multiply operand in scale_production
		#        line-6 (multiply = var:COTTAGEIND_pops_output). Log if absent.
		if = {
			limit = { NOT = { has_variable = COTTAGEIND_pops_output } }
			LOG_fail = {
				sys = ECON
				fn = "cottage_empty_probe"
				reason = "COTTAGEIND_pops_output NOT SET in this governorship — scale_production line-6 multiply operand missing (scope dump)"
			}
		}
		# --- 4. PRODUCED-VAR STATE after a real produce pass would have run: log the
		#        clothing produced var's presence. Absent => the set_variable from an
		#        empty raw yielded nothing. (State capture; the scope dump names the gov.)
		if = {
			limit = { NOT = { has_variable = COTTAGEIND_produced_clothing } }
			LOG_line = { sys = ECON  msg = "cottage_empty_probe: COTTAGEIND_produced_clothing absent post-pass (see scope dump)" }
		}
		# --- 5. STRATA/POP STATE: an ownerless or pop-less governorship is the likely
		#        shared trait of the 8. Record whether the pop-count vars the pops_output
		#        svalue depends on are present, so we can correlate the 8 scopes.
		if = {
			limit = { NOT = { has_variable = governorship_middle_strata } }
			LOG_line = { sys = ECON  msg = "cottage_empty_probe: governorship_middle_strata absent (pop-state anomaly; see scope dump)" }
		}
	}
	root = { remove_variable = econ_log_probe_country }
}

# ======================================================================
# COMPREHENSIVE STOCKPILE FORENSIC (#MG-flood-diag v3) — supersedes the old
# has_variable-only stockpile_gap_probe. That probe (and the cottage_empty_probe
# above) could only ever see UNSET vs present; it was BLIND to the actual failure,
# which log analysis (Aug-3) proved is a var that is SET-but-EMPTY (has_variable
# TRUE, value empty). This forensic instead CLASSIFIES every relevant node into
# one of four buckets using CONTROL FLOW ONLY (never value-printing, which does
# not render in debug_log here — see the SYNTAX NOTE / the blank snapshot lines):
#
#   UNSET       = NOT has_variable            (never seeded)
#   REAL        = has_variable AND value >= a very-negative sentinel is TRUE
#   EMPTY       = has_variable AND that comparison ERRORS->FALSE (set-but-empty)
#   (a scriptvalue that Div/0s or reads an empty operand yields EMPTY when staged.)
#
# Method: for a var X, `has_variable` splits UNSET from present; then a guarded
# comparison `var:X > -999999999` is TRUE for any real number and engine-errors
# (=> FALSE, caught by else) for an empty. This is the ONLY way to see set-but-
# empty (memory: has_variable is true for set-to-empty). Precedent for comparing
# a staged scriptvalue behind a has_variable guard: se_GLOBALTRADE_split.txt:784.
#
# Emits ONE greppable line per (governorship, good, node) via DIRECT debug_log
# (bare literals — no $macro$ value interpolation, which is the thing that does
# not work) plus debug_log_scopes for governorship+owner identity. Runs for ALL
# tradegoods (every_tradegood_complex) so raw AND manufactured are covered in one
# pass, settling whether the 3 observed populations share a cause. Grep the trace:
#   grep "IMP19C FORENSIC" debug.log
# Silent no-op outside -debug_mode (all output via debug_log). Read-only: sets
# only ECON_LOG_fx_* temps and removes them. Remove the ECON_LOG_flood_probe call
# site (oa_wealth_changes.txt) + this block to revert.
# ======================================================================

# --- per-(governorship, good) node classifier. Scope: governorship. -----------
# Classifies FOUR nodes for $tradegood$: the stockpile TARGET, its seed-source
# svalue GOODS_governorship_$tradegood$_produced, the produces_ marker, and the
# cottage COTTAGEIND_produced_ var. Each line is prefixed so a single grep +
# sort|uniq -c yields the full (good x bucket) matrix.
ECON_LOG_fx_classify_good = {
	# Scope: governorship
	# NODE 1 — the stockpile TARGET var (what every erroring change_variable mutates).
	if = {
		limit = { NOT = { has_variable = $tradegood$_stockpile } }
		debug_log = "IMP19C FORENSIC stockpile $tradegood$ = UNSET"
		debug_log_scopes = yes
	}
	else_if = {
		limit = { var:$tradegood$_stockpile > -999999999 }
		debug_log = "IMP19C FORENSIC stockpile $tradegood$ = REAL"
	}
	else = {
		debug_log = "IMP19C FORENSIC stockpile $tradegood$ = EMPTY"
		debug_log_scopes = yes
	}
	# NODE 2 — the SEED-SOURCE scriptvalue the setup set_variable and the quarterly
	# FUNC re-seed both read. Stage it into a temp so an empty/Div0 svalue becomes a
	# detectable set-but-empty var (a raw svalue that hits an empty owner.MODIFIER_*,
	# or a manufactured one that Div/0s in the industrialisation/employment chain,
	# lands here). Staging a possibly-empty svalue: set_variable does NOT error, it
	# stores empty (setup proves this — 0 errors at seed time).
	set_variable = { name = ECON_LOG_fx_src  value = GOODS_governorship_$tradegood$_produced }
	if = {
		limit = { NOT = { has_variable = ECON_LOG_fx_src } }
		debug_log = "IMP19C FORENSIC src_produced $tradegood$ = UNSET"
	}
	else_if = {
		limit = { var:ECON_LOG_fx_src > -999999999 }
		debug_log = "IMP19C FORENSIC src_produced $tradegood$ = REAL"
	}
	else = {
		debug_log = "IMP19C FORENSIC src_produced $tradegood$ = EMPTY (seed-source svalue empty -> seeds stockpile empty)"
		debug_log_scopes = yes
	}
	remove_variable = ECON_LOG_fx_src
	# NODE 3 — the produces_ marker (the guard GOODS_governorship_produce trusts).
	if = {
		limit = { has_variable = produces_$tradegood$ }
		debug_log = "IMP19C FORENSIC marker produces_$tradegood$ = PRESENT"
	}
	# NODE 4 — the cottage produced var (manufactured goods only will have it).
	if = {
		limit = { has_variable = COTTAGEIND_produced_$tradegood$ }
		if = {
			limit = { var:COTTAGEIND_produced_$tradegood$ > -999999999 }
			debug_log = "IMP19C FORENSIC cottage_produced $tradegood$ = REAL"
		}
		else = {
			debug_log = "IMP19C FORENSIC cottage_produced $tradegood$ = EMPTY"
			debug_log_scopes = yes
		}
	}
}

# --- per-governorship SHARED-SCOPE classifier. Scope: governorship. -----------
# The operands EVERY good's seed-source svalue shares. If one of these is empty,
# it explains a whole-governorship flood (all goods at once). Covers owner
# resolution, the three owner.MODIFIER_* productivity svalues the RAW _produced
# svalues multiply by, and the manufactured-only chain (industrialisation bonus
# + its cache + employment ratio + its cache) — plus the pop/province inputs that
# feed the Div/0-guarded denominators.
ECON_LOG_fx_classify_scope = {
	# Scope: governorship
	if = {
		limit = { NOT = { exists = owner } }
		debug_log = "IMP19C FORENSIC scope owner = NONE"
		debug_log_scopes = yes
	}
	else = {
		debug_log = "IMP19C FORENSIC scope owner = PRESENT"
		# the three country-scope modifier svalues the raw _produced svalues multiply by
		owner = {
			set_variable = { name = ECON_LOG_fx_agri  value = MODIFIER_agriculture_productivity }
			set_variable = { name = ECON_LOG_fx_mine  value = MODIFIER_mining_productivity }
			set_variable = { name = ECON_LOG_fx_text  value = MODIFIER_textiles_productivity }
		}
		if = {
			limit = { owner = { var:ECON_LOG_fx_agri > -999999999 } }
			debug_log = "IMP19C FORENSIC scope MODIFIER_agriculture_productivity = REAL"
		}
		else = {
			debug_log = "IMP19C FORENSIC scope MODIFIER_agriculture_productivity = EMPTY (raw _produced multiply operand empty)"
			debug_log_scopes = yes
		}
		if = {
			limit = { owner = { var:ECON_LOG_fx_mine > -999999999 } }
			debug_log = "IMP19C FORENSIC scope MODIFIER_mining_productivity = REAL"
		}
		else = {
			debug_log = "IMP19C FORENSIC scope MODIFIER_mining_productivity = EMPTY"
			debug_log_scopes = yes
		}
		if = {
			limit = { owner = { var:ECON_LOG_fx_text > -999999999 } }
			debug_log = "IMP19C FORENSIC scope MODIFIER_textiles_productivity = REAL"
		}
		else = {
			debug_log = "IMP19C FORENSIC scope MODIFIER_textiles_productivity = EMPTY"
			debug_log_scopes = yes
		}
		owner = {
			remove_variable = ECON_LOG_fx_agri
			remove_variable = ECON_LOG_fx_mine
			remove_variable = ECON_LOG_fx_text
		}
	}
	# manufactured-only chain: industrialisation bonus (Div/0-guarded) + its cache
	set_variable = { name = ECON_LOG_fx_ind  value = GOODS_governorship_bonus_to_industrial_production_from_industrialisation }
	if = {
		limit = { var:ECON_LOG_fx_ind > -999999999 }
		debug_log = "IMP19C FORENSIC scope industrialisation_bonus = REAL"
	}
	else = {
		debug_log = "IMP19C FORENSIC scope industrialisation_bonus = EMPTY (Div/0 or empty operand in the mechanised chain)"
		debug_log_scopes = yes
	}
	remove_variable = ECON_LOG_fx_ind
	if = {
		limit = { has_variable = industrialisation_bonus_cached }
		if = {
			limit = { var:industrialisation_bonus_cached > -999999999 }
			debug_log = "IMP19C FORENSIC scope industrialisation_bonus_cached = REAL"
		}
		else = {
			debug_log = "IMP19C FORENSIC scope industrialisation_bonus_cached = EMPTY (cache holds an empty)"
			debug_log_scopes = yes
		}
	}
	else = {
		debug_log = "IMP19C FORENSIC scope industrialisation_bonus_cached = UNSET"
	}
	# employment ratio (Div/0-guarded) + its cache
	set_variable = { name = ECON_LOG_fx_emp  value = INDUSTRY_employment_ratio }
	if = {
		limit = { var:ECON_LOG_fx_emp > -999999999 }
		debug_log = "IMP19C FORENSIC scope employment_ratio = REAL"
	}
	else = {
		debug_log = "IMP19C FORENSIC scope employment_ratio = EMPTY"
		debug_log_scopes = yes
	}
	remove_variable = ECON_LOG_fx_emp
	if = {
		limit = { has_variable = industry_employment_ratio_cached }
		if = {
			limit = { var:industry_employment_ratio_cached > -999999999 }
			debug_log = "IMP19C FORENSIC scope employment_ratio_cached = REAL"
		}
		else = {
			debug_log = "IMP19C FORENSIC scope employment_ratio_cached = EMPTY"
			debug_log_scopes = yes
		}
	}
	# pop/province Div/0 denominators the industrialisation/employment computes divide by
	if = {
		limit = { governorship_total_provinces > 0 }
		debug_log = "IMP19C FORENSIC scope total_provinces = POSITIVE"
	}
	else = {
		debug_log = "IMP19C FORENSIC scope total_provinces = ZERO (Div/0 source -> empty industrialisation svalue)"
		debug_log_scopes = yes
	}
	if = {
		limit = { has_variable = COTTAGEIND_pops_output }
		if = {
			limit = { var:COTTAGEIND_pops_output > -999999999 }
			debug_log = "IMP19C FORENSIC scope pops_output = REAL"
		}
		else = {
			debug_log = "IMP19C FORENSIC scope pops_output = EMPTY"
			debug_log_scopes = yes
		}
	}
	else = {
		debug_log = "IMP19C FORENSIC scope pops_output = UNSET"
	}
	# --- VALUE DUMP (additive, separate line) -----------------------------------
	# [#log-noise-fix 2026-08-03] The numeric value-DUMP that used to sit here was REMOVED. The
	# UNSET/REAL/EMPTY buckets above ARE the diagnosis (control-flow is the only thing that tells an
	# empty from a real 0). The dump merely tried to add magnitudes for the REAL nodes via
	# [ROOT.MakeScope.GetVariable(...).GetValue], which the engine cannot resolve in scripted-effect
	# debug_log — it emitted "Could not find promote for 'MakeScope'" ×5 vars per erroring scope

	# needed, re-add them only with a render form proven to work in effect context.
}
