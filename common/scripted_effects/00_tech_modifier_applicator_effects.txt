apply_tech_modifier_country = {
	tech_modifier_applicator_$tech$ = yes	
}

tech_modifier_applicator_tech_bloomery = {
	# [fix 2026-08-22] this effect was called (00_civic_inventions.txt:608) but never written --
	# "Unknown effect" every boot, so Bloomery granted none of its 3 documented bonuses.
	# Implements bonus 1 (RGO output for Iron/Copper), mirroring tech_dynamite below. Bonuses 2
	# (+10% cottage industry output for Bronze/Naval supplies/Early artillery) and 3 (coal demand
	# per RGO) are implemented separately, per design/DESIGN_BLOOMERY_COTTAGE_COAL_FOLLOWUP.md --
	# bonus 2 in se_COTTAGEIND.txt (inside COTTAGEIND_produce_bronze/_naval_supplies/
	# _early_artillery), bonus 3 as DEMAND_bloomery_coal in DEMAND_svalues.txt.
	every_owned_province = {
		limit = {
			has_province_modifier = tech_bloomery_modifier
		}
		remove_province_modifier = tech_bloomery_modifier
	}
	every_owned_province = {
		limit = {
			OR = {
				trade_goods = iron
				trade_goods = copper
			}
		}
		FUNC_add_province_modifier_stack = {
			MOD = tech_bloomery_modifier
			NUM = num_of_IND_resource_gathering_operation
			DUR = -1
		}
	}
}
