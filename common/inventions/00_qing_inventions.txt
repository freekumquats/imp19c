########################################################################
# 00_qing_inventions.txt — QING-SPECIFIC TECHNOLOGY (清朝專屬技術).
#
# A civic_tech invention group holding the technologies at which High-Qing China (c.1763,
# mid-Qianlong) genuinely LED the world, plus an INTEGRATION-SCIENCE branch that turns the
# empire's many cultures + many subjects into a research lever (the user's brief: "Qing has
# plenty of cultures and plenty of subjects, so there should be lots of Qing-specific
# features"). Historical basis: overnight_technology.md Phase 1b digest.
#
# TWO layers:
#   1. DISTINCTIVE 1763 NODES — the things the Qing was best-in-world at: imperial porcelain,
#      Jiangnan silk, the tea/Canton trade node, Grand-Canal hydraulics, woodblock/四庫全書
#      compilation, Jesuit court mathematics, the Eight Banners, Columbian-crop agronomy, and
#      variolation. Gated `allow = { tag = CHI }` (per Invictus hundred_four_inv `allow={tag=CAR}`
#      idiom: country-specificity goes in ALLOW, never potential). Qing starts able to research these
#      immediately; nobody else can.
#   2. INTEGRATION-SCIENCE — one node per non-Han/non-Manchu culture family the Qing rules.
#      Each unlocks either for the country_culture_group directly OR (the many-subjects vector)
#      once you've INTEGRATED a culture of that family AND armed the matching X_science variable
#      via a culture decision (culture_decisions/qing_integration_science.txt, Part D). This is
#      the exact Invictus greek_advanced_mechanics_inv idiom (00_civic_inventions.txt:
#      OR { country_culture_group  custom_tooltip { text has_variable=X_science
#      any_integrated_culture { ... } } }).
#
# NO icon block (existing inventions leave it off; engine auto-derives — avoids blank icons).
# NO on_activate create_character (boot-crash rule). Modifier keys are ONLY those proven valid
# in the mod's own inventions/heritage/modifiers (validated 2026-07-25). Loc lives in
# localization/english/technology_l_english.yml (qing_tech_* + _desc).
#
# TAB indentation throughout (Paradox house style).
########################################################################

qing_inventions = {
	technology = civic_tech
	color = hsv { 0.13 0.85 0.85 }

	######################################################################
	# LAYER 1 — DISTINCTIVE 1763 QING NODES (allow = { tag = CHI })
	######################################################################

	# 景德鎮御窯廠 — the Imperial Kilns at Jingdezhen, the world's premier ceramic
	# industry: a proto-industrial complex of division-of-labour workshops feeding a
	# vast export trade. Reward: commerce + citizen (artisan) output.
	qing_tech_imperial_kilns = {
		icon_override = qing_tech_imperial_kilns	# [#118] -> GFX_qing_tech_imperial_kilns
		modifier = {
			global_commerce_modifier = 0.05
			global_citizen_output = 0.05
		}
		allow = { tag = CHI }
	}

	# 江南織造 — the Imperial Silk Manufactories of Suzhou/Hangzhou/Nanjing, the
	# state sericulture-and-weaving bureaux. Reward: export commerce + citizen output.
	qing_tech_imperial_silk = {
		icon_override = qing_tech_imperial_silk	# [#118] -> GFX_qing_tech_imperial_silk
		modifier = {
			global_export_commerce_modifier = 0.1
			global_citizen_output = 0.05
		}
		requires = { qing_tech_imperial_kilns }
		allow = { tag = CHI }
	}

	# 茶政與十三行 — the Tea Administration and the Canton System (廣州十三行): the
	# Qing was the sole global source of tea, funnelled through the Thirteen Hongs.
	# Reward: commerce + import commerce (the silver inflow).
	qing_tech_tea_canton = {
		icon_override = qing_tech_tea_canton	# [#118] -> GFX_qing_tech_tea_canton

			global_commerce_modifier = 0.1
			global_import_commerce_modifier = 0.05
		}
		requires = { qing_tech_imperial_kilns }
		allow = { tag = CHI }
	}

	# 河工與漕運 — River Conservancy and the Grand Canal grain transport: the most
	# sophisticated state hydraulic-engineering apparatus on earth (Yellow-River works,
	# the 大運河). Reward: supply limit + cheaper construction (canal/waterworks).
	qing_tech_grand_canal = {
		icon_override = qing_tech_grand_canal	# [#118] -> GFX_qing_tech_grand_canal
		modifier = {

			build_cost = -0.05
		}
		allow = { tag = CHI }
	}

	# 四庫全書與雕版 — the great state woodblock-printing and scholarly-compilation
	# apparatus that produced the 四庫全書 (Complete Library of the Four Treasuries).

	qing_tech_siku_compilation = {

			research_points_modifier = 0.05
			global_monthly_civilization = 0.02
		}
		allow = { tag = CHI }
	}

	# 欽天監 — the Imperial Astronomical Bureau, where Jesuit-transmitted mathematics
	# and astronomy were preserved at court (Verbiest's instruments, the 曆象考成).
	# Reward: research efficiency + points (rigorous but court-bound learning).
	qing_tech_court_mathematics = {
		icon_override = qing_tech_court_mathematics	# [#118] -> GFX_qing_tech_court_mathematics
		modifier = {
			max_research_efficiency = 0.25
			research_points_modifier = 0.05
		}
		requires = { qing_tech_siku_compilation }
		allow = { tag = CHI }
	}

	# 八旗 — the Eight Banners, the Manchu military-organizational system that
	# garrisoned the empire. Reward: manpower + a defensive garrison bonus.
	qing_tech_eight_banners = {
		icon_override = qing_tech_eight_banners	# [#118] -> GFX_qing_tech_eight_banners
		modifier = {
			global_manpower_modifier = 0.05
			global_defensive = 0.05
		}
		allow = { tag = CHI }
	}

	# 番薯與玉米 — the diffusion of Columbian crops (maize, sweet potato, peanut) onto
	# marginal upland soils, the agronomy behind the High-Qing population boom.
	# Reward: population capacity + food.
	qing_tech_columbian_crops = {
		icon_override = qing_tech_columbian_crops	# [#118] -> GFX_qing_tech_columbian_crops
		modifier = {
			global_population_capacity_modifier = 0.1
			global_monthly_food_modifier = 0.1
		}
		allow = { tag = CHI }
	}

	# 種痘 — variolation, the Chinese smallpox-inoculation practice (nasal insufflation)
	# long predating Jenner, promoted at the Qing court. Reward: population growth +
	# same-religion happiness (a benevolent, order-keeping public-health measure).
	qing_tech_variolation = {

			global_population_capacity_modifier = 0.05

	}

	######################################################################
	# LAYER 2 — INTEGRATION-SCIENCE (the many-cultures / many-subjects vector)
	# Idiom = Invictus greek_advanced_mechanics_inv: unlock for the culture_group
	# directly, OR once a culture of that family is INTEGRATED and the X_science
	# variable is armed (set by culture_decisions/qing_integration_science.txt).
	# CHI's own group is jurchen (manchu) — so the jurchen node is near-free, and
	# the others reward integrating Han/Mongol/Tibetan/Uyghur subjects.
	######################################################################

	# 滿洲騎射 — Manchu horsemanship-and-archery (騎射), the martial ethos of the
	# ruling banner people. Unlocks for the jurchen group (CHI's own).
	qing_tech_manchu_science = {
		icon_override = qing_tech_manchu_science	# [#118] -> GFX_qing_tech_manchu_science
		modifier = {
			conscripts_offensive = 0.1
			global_manpower_modifier = 0.05
		}
		requires = { qing_tech_eight_banners }
		allow = {
			OR = {
				country_culture_group = jurchen
				custom_tooltip = {
					text = qing_integrated_any_manchu_culture
					has_variable = manchu_science
					any_integrated_culture = {
						this.culture.culture_group = culture_group:jurchen
					}
				}
			}
		}
	}

	# 儒家治術 — Confucian statecraft and the examination bureaucracy, the Han
	# administrative science. Unlocks for the chinese_group OR an integrated Han culture.
	qing_tech_han_science = {
		icon_override = qing_tech_han_science	# [#118] -> GFX_qing_tech_han_science
		modifier = {
			global_pop_promotion_speed_modifier = 0.1
			research_points_modifier = 0.05
		}
		requires = { qing_tech_siku_compilation }
		allow = {
			OR = {
				country_culture_group = jurchen        # the Qing (Manchu) run the Confucian bureaucracy too

				custom_tooltip = {
					text = qing_integrated_any_han_culture
					has_variable = han_science

			}
		}

	# [#76] 官票 — official-note fiscal administration. A modern statecraft study of
	# note-issuing bureaus (the Ming 大明寶鈔 precedent + Western banking practice),

	# gain for the study itself. Its REAL effect is a gate: se_QING_DECLINE.txt reads
	# `invention = qing_tech_official_banking` as an ALTERNATE, PROACTIVE path to
	# unlock paper currency (qing_paper_money_unlocked) — the player who researches
	# this may adopt paper money on his own initiative, not only when forced by an
	# acute currency crisis (qing_currency_stress >= 70). The crisis path stays as
	# the reactive fallback for a player who never researched it. Both paths remain
	# gated on the existing current_date >= 1850.1.1 floor (no paper at the High-Qing
	# zenith — research/RESEARCH_QING_PAPER_MONEY_67), so researching this early does
	# NOT unlock paper money before 1850 either way.

		}
