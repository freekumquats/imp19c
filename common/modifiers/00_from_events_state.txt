# [#223] State-investment modifiers — backing the invest_in_state_button_* province buttons
# (common/scripted_guis/invest_in_state_buttons.txt) + the state_improvement.N event chain
# (events/state_improvement.txt). This subsystem is vanilla Imperator: Rome content that Terra
# Indomita wires live; imp19c copied the buttons + the state_improvement.N event calls but the
# modifier bodies live in the vanilla base game (which this total conversion does not ship), so
# they were undefined -> boot errors "Cannot find state_improvement_* in modifier database".
# state_improvement_military is the one body upstream preserves (Invictus 00_from_events_state_inv.txt);
# civic/oratory/religious are authored here from the vanilla loc intent, reskinned to the Qing setting.
# state_modification_in_progress is the 182-day "investment underway" guard the buttons check.

# "Fortify the Province" — garrison works, drill grounds, provincial procurators.
state_improvement_military = {
	local_fort_limit = 1
	local_monthly_state_loyalty = 0.01
}

# "Fund Yamen Works" — public infrastructure: granaries, courthouses, roads, waterworks.
state_improvement_civic = {
	local_tax_modifier = 0.05
	local_monthly_state_loyalty = 0.01
}

# "Charter Merchant Guilds" — encourage commerce and open new trade custom.
state_improvement_oratory = {
	local_state_trade_routes = 2
}

# "Endow Temples & Academies" — patronise shrines, examination halls, and local learning.
# [review #223] local_happiness_for_same_culture_modifier + local_population_capacity_modifier are
# proven state-legal via Invictus state_improvement_religious_KSH (applied by add_state_modifier);
# local_population_happiness has no state-modifier precedent upstream (province/mission only) and
# risked a "cannot be used in state modifier" boot error.
state_improvement_religious = {
	local_happiness_for_same_culture_modifier = 0.03
	local_population_capacity_modifier = 0.015
}

# Progress guard: applied for 182 days while an investment is underway, blocking a second one.
state_modification_in_progress = {
	show_in_outliner = yes
}
