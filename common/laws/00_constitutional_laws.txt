constitutional_monarchy_laws = {
	potential = {
		NOT = {
			OR = {
				government = viceroyalty
				government = communist_dictatorship
				# Below = governments where the monarch is head of government
				government = absolute_monarchy
				government = imperial_monarchy
			}
		}
	}
	no_monarchy = {
		allow = {
			NOT = {
				has_law = oligarchy_multiple_monarchy
			}
		}
		modifier = {
		}
	} 
	# [task law-adjacency-audit 2026-08-30] sequential prerequisite: constitutional_monarchy_laws
	# is an ordered spectrum of monarchical power (symbolic -> advisory -> administrative), so a
	# country must hold the immediately preceding tier to advance one step — added
	# has_law = <preceding tier> alongside the pre-existing NOT-has_law ratchet (which separately
	# makes abandoning the monarchy via no_monarchy irreversible through this law; no_monarchy is
	# a one-way EXIT from this ladder, not its baseline rung, so it is not itself gated and cannot
	# be required as a "preceding tier"). Same has_law-in-allow idiom proven on Women's Rights
	# (00_social_laws.txt). symbolic_monarchy is the baseline rung of the power-ladder and carries
	# no forward-adjacency requirement of its own.
	symbolic_monarchy = {
		allow = {
			NOT = {
				has_law = oligarchy_multiple_monarchy
			}
			NOT = {
				has_law = no_monarchy
			}
		}
		modifier = {
		}
	}
	advisory_monarchy = {
		allow = {
			NOT = {
				has_law = oligarchy_multiple_monarchy
			}
			NOT = {
				has_law = no_monarchy
			}
			has_law = symbolic_monarchy
		}
		modifier = {
			force_through_senate_cost_modifier = 0.1
		}
	}
	administrative_monarchy = {
		allow = {
			NOT = {
				has_law = no_monarchy
			}
			has_law = advisory_monarchy
		}
		modifier = {
			force_through_senate_cost_modifier = 0.5
		}
	}
}

	1_year_terms = {
		modifier = {
			election_term_duration = -3
			monthly_tyranny = -0.02

	# the default/reversion tier.

regional_government_law = {

	potential = {
		always = yes
	}
	
	vicarious_regional_governments = { # Regional governments are centrally managed and act as an extension of central government
		modifier = {
		monthly_governor_wage = 0.1
		monthly_political_influence_modifier = -0.1
		monthly_corruption = 0.05
		manpower_recovery_speed = 0.1
		}
	}
	
	# [task law-adjacency-audit 2026-08-30] sequential prerequisite: regional_government_law is an
	# ordered spectrum of central->local devolution (vicarious -> feudal -> federal -> devolved),
	# so a country must hold the immediately preceding tier to advance one step. Same
	# has_law-in-allow idiom proven on Women's Rights (00_social_laws.txt).
	# vicarious_regional_governments carries no allow, so it stays reachable as the default/
	# reversion tier.
	feudal_regional_governments = { # Regional governments are minor substates bound by a set of obligations to and guarantees from central government, but often with extensive local autonomy
		allow = {
			has_law = vicarious_regional_governments
		}
		modifier = {
		monthly_governor_wage = 0.25
		monthly_corruption = 0.1
		monthly_political_influence_modifier = -0.1
		force_through_senate_cost_modifier = 0.5
		army_maintenance_cost = -0.1
		global_upper_strata_happyness = 0.1
		}
	}

	federal_regional_governments = { # Regional governments are run by local legislatures who fall under a legal hierarchy within the greater state
		allow = {
			has_law = feudal_regional_governments
		}
		modifier = {
		monthly_governor_wage = 0.15
		force_through_senate_cost_modifier = 0.25
		diplomatic_reputation = 1
		}
	}

	devolved_regional_governments = { # Regional governments are empowered to diverge from central government on a large number of civil and economic matters, even forming different constitutional laws, but are bound by the powers granted by central government.
		allow = {
			has_law = federal_regional_governments
		}
		modifier = {
		monthly_governor_wage = 0.25
		force_through_senate_cost_modifier = 0.15
		diplomatic_reputation = 1
		global_capital_trade_routes = 1
		}
	}

}
