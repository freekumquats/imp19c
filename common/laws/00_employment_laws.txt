# [MO#11 2026-07-29] Employment laws — worldwide (potential = is_tribal = no). Integrate with the
# EXISTING building + manufactured-goods economy, not a standalone modifier layer:
#  - working_hours_law             -> trades strata output vs happiness/manpower/growth (production labour)
#  - child_and_bonded_labour_law   -> targets the indentured pop-type + build_cost (bonded-labour -> construction)
#  - workplace_safety_and_tenure_law -> biases the employment fill-ratio via EMPLOY_tenure_bias, hooked into
#                                       JOBS_buildings_cap (the capacity gate every buildable building's allow= reads)
# Tokens verified as real country modifiers (00_economic_laws.txt / 00_social_laws.txt / inventions).
# NOTE (review): use global_population_growth (NOT population_growth); default options of the bias-var
# group reset the var to 0 in on_enact, matching the proven Qing bias-law idiom (qing_military_upkeep_law).

working_hours_law = {
	potential = {
		is_tribal = no
	}
	# DEFAULT: no statutory limit on the working day. No modifier either way.
	unregulated_shifts = {
		modifier = { }
	}
	# Maximal extraction: long shifts push output up now, at the cost of worker
	# morale and rising unrest.
	sweatshop_hours = {
		modifier = {
			global_proletariat_output = 0.08
			global_lower_strata_output = 0.04
			global_proletariat_happyness = -0.15
			global_lower_strata_happyness = -0.08
			global_unrest = 0.3
		}
	}
	# A ten-hour act: a small output sacrifice for steadier, healthier labour.
	statutory_ten_hour_act = {
		allow = {
			invention = tech_manufactories
		}
		modifier = {
			global_proletariat_output = -0.02
			global_lower_strata_output = -0.01
			global_proletariat_happyness = 0.05
			manpower_recovery_speed = 0.03
		}
	}
	# The eight-hour movement: modern labour relations — happier, more productive-per-hour
	# workers and a small research/growth dividend, at a real output cost.
	eight_hour_movement = {
		allow = {
			invention = tech_electricity
		}
		modifier = {
			global_proletariat_output = -0.05
			global_lower_strata_output = -0.02
			global_proletariat_happyness = 0.12
			global_middle_strata_happyness = 0.03
			global_population_growth = 0.01
			research_points_modifier = 0.02
		}
	}
}

child_and_bonded_labour_law = {
	potential = {
		is_tribal = no
	}
	# DEFAULT: customary practice — whatever the local economy has always done.
	customary_practice = {
		modifier = { }
	}
	# Protective age limits: bar the youngest from the mills. A morale and human-capital
	# gain, a small output cost.
	protective_labour_age = {
		allow = {
			invention = tech_manufactories
		}
		modifier = {
			global_lower_strata_happyness = 0.08
			global_proletariat_happyness = 0.05
			global_lower_strata_output = -0.03
			global_population_growth = 0.01
			research_points_modifier = 0.02
		}
	}
	# Indentured apprenticeship: bond the young poor to a trade. Lifts indentured output
	# and cheapens construction, at the cost of their and the lower strata's contentment.
	# (Sized BELOW non_tribal_land_law/indentured_tenancy's +-0.1/+-0.25 to bound stacking.)
	indentured_apprenticeship = {
		modifier = {
			global_indentured_output = 0.15
			global_indentured_happyness = -0.05
			global_lower_strata_happyness = -0.03
			build_cost = -0.03
		}
	}
	# Workhouse system: the harshest extraction of bonded labour — strong indentured and
	# proletarian output and cheap building, but a heavy morale and unrest penalty.
	workhouse_system = {
		modifier = {
			global_indentured_output = 0.25
			global_proletariat_output = 0.05
			global_indentured_happyness = -0.15
			global_proletariat_happyness = -0.10
			global_unrest = 0.4
			build_cost = -0.05
		}
	}
}

workplace_safety_and_tenure_law = {
	potential = {
		is_tribal = no
	}
	# DEFAULT: customary hiring — no statutory tenure. Reset the fill-ratio bias to 0 so
	# reverting from any option below restores baseline (proven Qing bias-law idiom).
	customary_hiring = {
		on_enact = {
			set_variable = { name = EMPLOY_tenure_bias  value = 0 }
		}
		modifier = { }
	}
	# Flexible hire-and-fire: employers churn labour freely — more jobs can be filled
	# (positive fill-ratio bias) and building is quicker, but workers feel the insecurity.
	flexible_hire_and_fire = {
		on_enact = {
			set_variable = { name = EMPLOY_tenure_bias  value = 0.04 }
			custom_tooltip = flexible_hire_and_fire_bias_tt
		}
		modifier = {
			global_proletariat_happyness = -0.05
			build_time = -0.05
		}
	}
	# Regulated tenure: notice periods and cause requirements. Steadier, happier labour;
	# no fill-ratio bias, slightly slower building.
	regulated_tenure = {
		allow = {
			invention = tech_manufactories
		}
		on_enact = {
			set_variable = { name = EMPLOY_tenure_bias  value = 0 }
		}
		modifier = {
			global_proletariat_happyness = 0.05
			global_lower_strata_happyness = 0.03
			build_time = 0.03
		}
	}
	# Protected employment: strong job security and safety inspection. Contented, orderly
	# labour (unrest down), but a harder-to-flex workforce shrinks the effective job pool
	# (negative fill-ratio bias) and slows construction.
	protected_employment = {
		allow = {
			civic_tech >= 6
		}
		on_enact = {
			set_variable = { name = EMPLOY_tenure_bias  value = -0.04 }
			custom_tooltip = protected_employment_bias_tt
		}
		modifier = {
			global_proletariat_happyness = 0.12
			global_lower_strata_happyness = 0.06
			global_unrest = -0.2
			build_time = 0.08
			monthly_corruption = -0.02
		}
	}
}
