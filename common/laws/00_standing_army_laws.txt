standing_army_laws = {
	potential = {
		is_tribal = no
	}

	no_standing_army = {
			modifier = {
				levy_size_multiplier = 0.015
			}
			on_enact = {
				set_legion_recruitment = disabled
				custom_tooltip = legions_in_no_region
			}
	}

	# [task law-adjacency-audit 2026-08-30] sequential prerequisite: standing_army_laws is an
	# ordered spectrum of increasing legion-recruitment scope (none -> capital-only -> anywhere;
	# levy_size_multiplier falls monotonically 0.015 -> 0.005 -> 0), so a country must hold the
	# immediately preceding tier to advance one step. Same has_law-in-allow idiom proven on
	# Women's Rights (00_social_laws.txt). no_standing_army carries no allow, so it stays
	# reachable as the default/reversion tier.
	limited_army = {
			allow = {
				has_law = no_standing_army
			}
			modifier = {
				discipline = 0.05
				levy_size_multiplier = 0.005
			}
			on_enact = {
				custom_tooltip = legions_in_capital_region
				set_legion_recruitment = capital
			}
	}

	standing_army = {
			allow = {
				has_law = limited_army
			}
			modifier = {
				army_maintenance_cost = -0.15
			}
			on_enact = {
				custom_tooltip = legions_in_any_region
				set_legion_recruitment = enabled
			}
	}


}
