cultural_protections_law = {
	potential = {
		always = yes
	}
	dominant_culture_primacy = { # The dominant culture is protected above all others
		modifier = {
			happiness_for_same_culture_modifier = 0.30
		}
	}
	# [task law-adjacency-audit 2026-08-30] sequential prerequisite: cultural_protections_law is
	# an ordered spectrum of increasing minority-culture protection (dominant culture primacy ->
	# limited recognition -> federalism -> multiculturalism; happiness_for_same_culture_modifier
	# falls monotonically 0.30 -> 0.15 -> -0.10 -> -0.30 and happiness_for_wrong_culture_modifier
	# rises monotonically 0 -> 0.30 -> 0.50 -> 0.60), so a country must hold the immediately
	# preceding tier to advance one step. Same has_law-in-allow idiom proven on Women's Rights
	# (below). dominant_culture_primacy carries no allow, so it stays reachable as the default/
	# reversion tier.
	limited_cultural_recognition = { # Non-dominant cultures are recognised in law but enjoy very limited rights
		allow = {
			has_law = dominant_culture_primacy
		}
		modifier = {
			happiness_for_same_culture_modifier = 0.15
			global_unrest = -0.25
			happiness_for_wrong_culture_modifier = 0.30
			happiness_for_wrong_culture_group_modifier =0.20
			global_pop_assimilation_speed=-0.025
		}
	}
	cultural_federalism = { # Cultures are separate but equal under the law
		allow = {
			has_law = limited_cultural_recognition
		}
		modifier = {
			agressive_expansion_impact = -0.1
			happiness_for_same_culture_modifier = -0.10
			happiness_for_wrong_culture_modifier = 0.50
			happiness_for_wrong_culture_group_modifier =0.30
			global_pop_assimilation_speed=-0.05
		}
	}

religious_law = {
	potential = {
		always = yes
	}
	enforced_state_religion_law = {
		modifier = {
			happiness_for_same_religion_modifier = 0.30
			global_population_happiness=-0.10
		}
		allow = {
			NOT = { religion = secular }
		}
	}
	# [task law-adjacency-audit 2026-08-30] sequential prerequisite: religious_law is an ordered
	# spectrum of state religious ideology, from devout establishment through secular neutrality
	# to enforced atheism (enforced state religion -> nominal state religion -> secularism ->
	# atheism). Added has_law-in-allow (same idiom proven on Women's Rights, below) alongside the
	# pre-existing religion= gates. [review-note] this group is NOT a clean ladder like the others
	# this audit fixed: state_secularism/state_atheism's pre-existing `religion = secular` gate is
	# the binding constraint on the normal non-secular->secular path (a country holding
	# nominal_state_religion_law is by definition non-secular, so the new has_law line there is
	# usually redundant, not the deciding condition) -- it only bites on an edge path where religion
	# turns secular (e.g. via the ideology channel) while an earlier law is still enacted. Left in
	# place as a harmless belt-and-braces addition rather than removed, since it never LOOSENS any
	# existing restriction; not re-modeling this group's religion-state interaction, which predates
	# this audit and is out of its scope.
	nominal_state_religion_law = {
		modifier = {
			happiness_for_same_religion_modifier=-0.05
			global_population_happiness=0.10
			global_pop_conversion_speed_modifier=-0.50
		}
		allow = {
			NOT = { religion = secular }
			has_law = enforced_state_religion_law
		}
	}
	state_secularism = {
		modifier = {
			global_population_happiness=0.20
			global_pop_conversion_speed_modifier=-0.90
		}
		on_enact = {
			set_country_religion = secular
		}
		allow = {
			religion = secular
			# [#61 2026-07-23] Design 3 reconcile: never while the country holds an IDEOLOGY (a coexisting
			# religion entry) — enacting this would set_country_religion=secular and silently WIPE the
			# ideology. The ideology channel and the secularism laws both own the country religion slot; this
			# gate keeps them mutually exclusive. (Already gated religion=secular, so this is belt-and-braces.)
			NOT = { is_ideology_religion = yes }
			has_law = nominal_state_religion_law
		}
	}
	state_atheism = {
		modifier = {
			happiness_for_same_religion_modifier=0.30
			global_population_happiness=-0.10
		}
		on_enact = {
			set_country_religion = atheism
		}
		allow = {
			religion = secular
			# [#61 2026-07-23] Design 3 reconcile: never while holding an IDEOLOGY (would wipe it via
			# set_country_religion=atheism). Keeps the ideology channel + secularism laws mutually exclusive.
			NOT = { is_ideology_religion = yes }
			has_law = state_secularism
		}
	}
}

# [scan Tier-A] healthcare_law: was four empty modifier bodies. Filled with proven keys only —

# against the strata that bear the cost (the original comments named "increased administration
# demand" + "upper strata pay", modelled here as upper-strata discontent + manpower/growth gains,
# since admin-demand is an svalue system we deliberately do NOT touch on a no-break pass). Growth
# magnitudes sit within the 0.002–0.05 band already used across the mod's modifiers.

	potential = {
		always = yes

		}

	# spectrum of increasing socialisation of healthcare cost and administration (charitable ->

	# immediately preceding tier to advance one step. Same has_law-in-allow idiom proven on
