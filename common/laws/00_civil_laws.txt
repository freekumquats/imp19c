citizens_rights = {
	potential = {
		is_tribal = no
	}
	judicial_discretion_rights = {
		modifier = {
			global_unrest = 1
			senate_influence = 0.1
		}
	}

	# [task law-adjacency-audit 2026-08-30] sequential prerequisite: citizens_rights is an ordered
	# spectrum of increasing formalisation/universality of rights (ad-hoc judicial discretion ->
	# wealth-gated rights -> universal statutory rights -> entrenched constitutional rights), so a
	# country must hold the immediately preceding tier to advance one step. Same has_law-in-allow
	# idiom proven on Women's Rights (00_social_laws.txt). judicial_discretion_rights carries no
	# allow, so it stays reachable as the default/reversion tier.
	wealth_based_rights = {
		allow = {
			has_law = judicial_discretion_rights
		}
		modifier = {
			global_upper_strata_happyness = 0.25
			global_middle_strata_happyness = 0.05
		}
	}

	bill_of_rights = {
		allow = {
			has_law = wealth_based_rights
		}
		modifier = {
			global_upper_strata_happyness = 0.1
			global_middle_strata_happyness = 0.1
			global_lower_strata_happyness = 0.1
			global_proletariat_happyness = 0.1
		}
	}

	constitutional_rights = {
		allow = {
			has_law = bill_of_rights
		}
		modifier = {
			global_upper_strata_happyness = 0.1
			global_middle_strata_happyness = 0.1
			global_lower_strata_happyness = 0.1
			global_proletariat_happyness = 0.1
		}
	}
}
