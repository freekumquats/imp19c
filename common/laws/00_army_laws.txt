army_recruitment_law = {
	potential = {
		is_tribal = no
	}
	
	voluntary_service = { # The army is comprised of volunteers who go through training out of their own choice
		modifier = {
			discipline = 0.1
			land_morale_modifier = 0.33
			global_cohort_start_experience = 0.1

			global_manpower_modifier = -0.50
			manpower_recovery_speed = -0.15
		}
	}

	# [task law-adjacency-audit 2026-08-30] sequential prerequisite: army_recruitment_law is an
	# ordered spectrum of state coercion in recruitment (voluntary -> militia -> national service
	# -> conscription), so a country must hold the immediately preceding tier to advance one step.
	# Same has_law-in-allow idiom proven on Women's Rights (00_social_laws.txt). voluntary_service
	# carries no allow, so it stays reachable as the default/reversion tier.
	regional_militias = { # The army is drawn from regional militias organised on the local level
		allow = {
			has_law = voluntary_service
		}
		modifier = {
			army_maintenance_cost = -0.15
			manpower_recovery_speed = 0.05
			global_defensive = 0.1
			global_unrest = -1
			global_monthly_state_loyalty = 0.03

			discipline = -0.1
			monthly_governor_wage = 0.1
			cohort_reinforcement_speed = -0.15
			stability_monthly_change = -0.025
		}
	}

	national_service_law = { # Individuals are expected to train and serve in the military for a short period in their youth, saving exceptions
		allow = {
			has_law = regional_militias
		}
		modifier = {
			global_manpower_modifier = 0.05
			discipline = 0.05
			global_cohort_start_experience = 0.1
			army_maintenance_cost = 0.1
		}
	}

	conscription_law = { # Individuals may be called into the army at any time
		allow = {
			has_law = national_service_law
		}
		modifier = {
			global_manpower_modifier = 0.1
			manpower_recovery_speed = 0.1
			cohort_reinforcement_speed = 0.1
			global_cohort_start_experience = -0.1
		
			discipline = -0.1
			global_unrest = 1
		}
	}
}

army_supply_law = {
	potential = {
		always = yes
	}

	self_supplied_army = {
		modifier = {
			army_maintenance_cost = -0.75

			discipline = -0.33
			land_morale_modifier = -0.15
		}
	}

	# [task law-adjacency-audit 2026-08-30] sequential prerequisite: army_supply_law is an ordered
	# spectrum of state involvement in army supply (self-supplied -> irregular state -> regular
	# state), same has_law-in-allow idiom as above. self_supplied_army carries no allow, so it
	# stays reachable as the default/reversion tier.
	irregular_state_supplied_army = {
		# The state sponsors the purchasing of equipment for its soldiers
		allow = {
			has_law = self_supplied_army
		}
		modifier = {
			army_maintenance_cost = -0.5

			discipline = -0.1
			land_morale_modifier = -0.05
		}
	}

	regular_state_supplied_army = {
		# The state commissions and regulates all military equipment
		allow = {
			has_law = irregular_state_supplied_army
		}
		modifier = {
			manpower_recovery_speed = -0.1

			cohort_reinforcement_speed = 0.1
			global_cohort_start_experience = 0.1
		}
	}
}
