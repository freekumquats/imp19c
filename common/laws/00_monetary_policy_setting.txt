monetary_policy_setting = {
	# [#63] Reachable by republics (legislative monetary policy) OR the Qing (tag=CHI), so the Qing — a monarchy —
	# can set a minting stance. Scoping to CHI (not broadening to all monetary laws) keeps the group off every
	# other non-tribal monarchy AI (China-fine/ROW-abstraction). Republics unchanged.
	potential = {
		OR = {
			has_law = legislative_monetary_policy
			tag = CHI
		}
	}
	# [#63] NEUTRAL DEFAULT — the auto-default-held FIRST option MUST carry no modifier, else the Qing (now in the
	# potential) would silently gain the old first option's modifier at boot (not byte-identical). Empty modifier +
	# on_enact sets qing_monetary_bias 0 (the "return to neutral" stance). qing_monetary_bias is read by
	# QING_DECLINE (stress level + mint ceiling); on_enact fires only on active enactment, not default-hold, so a
	# realm that never legislates leaves the var unset = byte-identical.
	standard_minting = {
		modifier = {}
		on_enact = {
			if = {
				limit = { tag = CHI }
				set_variable = { name = qing_monetary_bias  value = 0 }
				if = { limit = { has_variable = qing_paper_regime_active }  remove_variable = qing_paper_regime_active }  # [#67] leave paper regime
			}
		}
	}

	# [law #49 fill 2026-07-24] Deflationary sound-money: pull debased coin, restore confidence — commerce
	# contracts but stability rises. [#63] bias -8: lowers monetary stress + tightens the Qing mint ceiling.
	currency_recall = {
		modifier = {
			global_commerce_modifier = -0.05
			stability_monthly_change = 0.03
		}
		on_enact = {
			if = {
				limit = { tag = CHI }
				set_variable = { name = qing_monetary_bias  value = -8 }
				if = { limit = { has_variable = qing_paper_regime_active }  remove_variable = qing_paper_regime_active }  # [#67] leave paper regime
			}
		}
	}

	# Tight minting — disciplined issuance keeps graft down and the propertied classes content.
	# [#63] bias -4: modest sound-money discipline.
	limited_minting = {
		modifier = {
			monthly_corruption = -0.03
			global_upper_strata_happyness = 0.03
		}
		on_enact = {
			if = {
				limit = { tag = CHI }
				set_variable = { name = qing_monetary_bias  value = -4 }
				if = { limit = { has_variable = qing_paper_regime_active }  remove_variable = qing_paper_regime_active }  # [#67] leave paper regime
			}
		}
	}

	# Loose minting — more coin greases trade and tax take, at a cost to price stability.
	# [#63] bias +4: raises monetary stress (over-minting debases). Does NOT raise mint volume (shared cap holds).
	more_minting = {
		modifier = {
			global_tax_modifier = 0.05
			global_commerce_modifier = 0.05
			stability_monthly_change = -0.02
		}
		on_enact = {
			if = {
				limit = { tag = CHI }
				set_variable = { name = qing_monetary_bias  value = 4 }
				if = { limit = { has_variable = qing_paper_regime_active }  remove_variable = qing_paper_regime_active }  # [#67] leave paper regime
			}
		}
	}

	# State bonds — deficit finance fuels commerce + capital trade, but opens the door to graft.
	# [#63] bias +8: strongest debasement pressure (deficit finance). Does NOT flip paper_money_allowed (R1).
	issue_bonds = {
		modifier = {
			global_commerce_modifier = 0.1
			global_capital_trade_routes = 1
			monthly_corruption = 0.03
		}
		on_enact = {
			if = {
				limit = { tag = CHI }
				set_variable = { name = qing_monetary_bias  value = 8 }
				# [#67] leaving the paper regime for bonds — clear the regime flag so the depreciation pulse stops.
				if = { limit = { has_variable = qing_paper_regime_active }  remove_variable = qing_paper_regime_active }
			}
		}
	}

	# [#67] PAPER CURRENCY (寶鈔 / 官票) — the Xianfeng-crisis fiat regime: print notes against the silver
	# reserve for deficit finance, at the risk of a depreciation doom-loop. UNAVAILABLE at 1763 (allow-gated on
	# the crisis unlock, per RESEARCH_QING_PAPER_MONEY_67: no govt paper 1661-1853). The mint LEVER is NOT this
	# bias — it is the CHI+regime-gated finite paper allowance added to the BOUNDED branch of
	# CURRENCY_minting_rate_cap (CURRENCY_svalues.txt), referenced to the silver reserve so it can't compound;
	# paper_money_allowed stays FALSE (never the 99999 uncap, so #23/#60 hold). bias +8 here only feeds the
	# EXISTING monetary-stress term (paper is as stressful as bonds, worsening as it depreciates — see the
	# ≤+15 depreciation term in se_QING_DECLINE). qing_paper_regime_active drives the allowance + depreciation.
	paper_currency = {
		# [#67] per-option allow-gate (Imperator law OPTIONS support `allow`, NOT per-option `potential` —
		# 00_qing_statutes_laws.txt precedent). Greys the option until qing_paper_crisis sets the unlock var.
		allow = {
			has_variable = qing_paper_money_unlocked
		}
		modifier = {
			global_commerce_modifier = 0.1
			global_capital_trade_routes = 1
			monthly_corruption = 0.04
		}
		on_enact = {
			if = {
				limit = { tag = CHI }
				set_variable = { name = qing_monetary_bias  value = 8 }
				set_variable = { name = qing_paper_regime_active  value = 1 }
			}
		}
	}
}
