# [MO#12 2026-07-29] Industrialization laws — companion to the employment laws. Integrate with the
# EXISTING building + manufactured-goods economy:
#  - industrial_capacity_policy_law   -> moves global_import_commerce_modifier, which feeds
#      MODIFIER_industry_productivity (MODIFIER_svalues.txt) -> INDUSTRY_province_industry_capacity, the
#      ceiling every IND_/qing_industry_ building's allow= checks; plus build_cost/build_time (all buildings)
#  - enterprise_ownership_model_law   -> worldwide state-vs-private axis; MUTUALLY EXCLUSIVE with the Qing
#      qing_industrial_encouragement_law via potential NOT = { tag = CHI } (same axis, would double-apply)
#  - cottage_to_factory_transition_law -> biases TECH_cottage_industry_overall_bonus via INDUS_cottage_bias
#      (the cottage-industry throughput multiplier in se_COTTAGEIND.txt)
# Review fixes applied: bias-var default options reset the var to 0 in on_enact (proven idiom).

industrial_capacity_policy_law = {
	potential = {
		is_tribal = no
	}
	# DEFAULT: laissez-faire — the state neither drives nor restrains the pace of industry.
	laissez_faire_pace = {
		modifier = { }
	}
	# State-subsidised industrialisation: the treasury underwrites factory building (cheaper,
	# higher capacity ceiling) at the cost of tax yield and some graft.
	state_subsidized_industrialisation = {
		allow = {
			invention = tech_manufactories
		}
		modifier = {
			build_cost = -0.15
			global_tax_modifier = -0.05
			monthly_corruption = 0.03
			global_import_commerce_modifier = 0.05
		}
	}
	# Forced-pace industrialisation: break every bottleneck — fastest building and the
	# highest capacity ceiling — but the social cost is steep (morale, unrest).
	forced_pace_industrialisation = {
		allow = {
			invention = tech_mechanical_tools
		}
		modifier = {
			build_time = -0.20
			global_import_commerce_modifier = 0.10
			global_proletariat_happyness = -0.10
			global_middle_strata_happyness = -0.05
			global_unrest = 0.2
		}
	}
	# Cautious gradualism: deliberately restrained growth — stability and elite comfort,
	# but pricier building and a slightly lower capacity ceiling.
	cautious_gradualism = {
		allow = {
			invention = tech_manufactories
		}
		modifier = {
			build_cost = 0.05
			global_upper_strata_happyness = 0.05
			stability_monthly_change = 0.01
			global_import_commerce_modifier = -0.02
		}
	}
}

enterprise_ownership_model_law = {
	potential = {
		is_tribal = no
		NOT = { tag = CHI }   # CHI runs qing_industrial_encouragement_law on the same axis
	}
	# DEFAULT: a mixed economy of state and private ventures.
	mixed_enterprise = {
		modifier = { }
	}
	# State-directed industry: the ministries run the works — good for coordinated research
	# and revenue direction, weaker on commercial dynamism.
	state_directed_industry = {
		allow = {
			invention = tech_manufactories
		}
		modifier = {
			research_points_modifier = 0.05
			global_commerce_modifier = -0.03
			global_tax_modifier = -0.02
		}
	}
	# Private capital industry: entrepreneurs own the factories — strong commerce and
	# middle/upper output, but the state's own research direction suffers.
	private_capital_industry = {
		allow = {
			invention = tech_manufactories
		}
		modifier = {
			global_commerce_modifier = 0.05
			global_middle_strata_output = 0.05
			global_upper_strata_output = 0.05
			research_points_modifier = -0.02
		}
	}
	# Free incorporation: joint-stock capitalism unleashed — the strongest commerce and
	# elite output and an extra trade route, at a cost to labour's share and mood.
	free_incorporation = {
		allow = {
			civic_tech >= 6
		}
		modifier = {
			global_commerce_modifier = 0.10
			global_upper_strata_output = 0.10
			global_capital_trade_routes = 1
			global_proletariat_happyness = -0.10
		}
	}
}

cottage_to_factory_transition_law = {
	potential = {
		is_tribal = no
	}
	# DEFAULT: an unguided transition. Reset the cottage bias to 0 so reverting from any
	# option below restores baseline (proven Qing bias-law idiom).
	unguided_transition = {
		on_enact = {
			set_variable = { name = INDUS_cottage_bias  value = 0 }
		}
		modifier = { }
	}
	# Protect cottage industry: shelter the household workshops — a strong cottage-output
	# bias and happier artisans/lower strata, but the wider industrial ceiling is held back.
	protect_cottage_industry = {
		on_enact = {
			set_variable = { name = INDUS_cottage_bias  value = 0.10 }
			custom_tooltip = protect_cottage_industry_bias_tt
		}
		modifier = {
			global_import_commerce_modifier = -0.03
			global_middle_strata_happyness = 0.05
			global_lower_strata_happyness = 0.05
		}
	}
	# Promote mechanisation: tilt away from the cottage toward the factory — a small negative
	# cottage bias, more proletarian output and a higher capacity ceiling, at the artisans' cost.
	promote_mechanisation = {
		allow = {
			invention = tech_manufactories
		}
		on_enact = {
			set_variable = { name = INDUS_cottage_bias  value = -0.05 }
			custom_tooltip = promote_mechanisation_bias_tt
		}
		modifier = {
			global_import_commerce_modifier = 0.05
			global_proletariat_output = 0.03
			global_middle_strata_happyness = -0.03
		}
	}
	# Full mechanisation drive: gut the cottage sector for the factory — the strongest
	# negative cottage bias and capacity boost and cheap building, but a real social cost.
	full_mechanisation_drive = {
		allow = {
			invention = tech_mechanical_tools
		}
		on_enact = {
			set_variable = { name = INDUS_cottage_bias  value = -0.10 }
			custom_tooltip = full_mechanisation_drive_bias_tt
		}
		modifier = {
			global_import_commerce_modifier = 0.10
			global_proletariat_output = 0.06
			global_middle_strata_happyness = -0.08
			global_lower_strata_happyness = -0.05
			build_cost = -0.05
		}
	}
}
