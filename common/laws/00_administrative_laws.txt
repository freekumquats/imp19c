monetary_policy_law = {
	potential = {
		is_tribal = no
	}
	# [law #48 fill 2026-07-24] Executive control of minting — the crown/treasury sets monetary policy directly.
	# Firm hand (stability) but the mint is a patronage trough (mild graft). Default-ish for a monarchy.
	executive_monetary_policy = {
		modifier = {
			stability_monthly_change = 0.02
			monthly_corruption = 0.02
		}
	}

	# Delegated to a competent ministry/board — professional coinage management lifts commerce + revenue.
	# [task law-adjacency-audit 2026-08-30] sequential prerequisite: monetary_policy_law is an
	# ordered spectrum of who controls monetary policy (executive -> delegated -> legislative),
	# so a country must hold the immediately preceding tier to advance one step — no jumping
	# straight from executive to legislative control. Same has_law-in-allow idiom already proven
	# on Women's Rights (00_social_laws.txt) and succession_law. executive_monetary_policy carries
	# no allow, so it stays reachable as the default/reversion tier; only forward progression is
	# gated.
	delegated_monetary_policy = {
		allow = {
			has_law = executive_monetary_policy
		}
		modifier = {
			global_commerce_modifier = 0.03
			global_tax_modifier = 0.02
		}
	}

	# Legislative (republics only) — an accountable but slower mint: strong commerce + a little research,
	# at a small standing stability cost (deliberation delays decisive action). Gates monetary_policy_setting.
	legislative_monetary_policy = {
		allow = {
			is_republic = yes
			has_law = delegated_monetary_policy
		}
		modifier = {
			global_commerce_modifier = 0.05
			research_points_modifier = 0.02
			stability_monthly_change = -0.02
		}
	}
}
