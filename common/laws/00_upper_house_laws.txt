upper_house_composition_law = {

    potential = {
        is_tribal = no
        has_law = bicameral_legislature
    }

    # [law #51 fill 2026-07-24] Hereditary + spiritual peers — the old order: upper strata + ritual
    # legitimacy, at the cost of the commons feeling shut out. (monthly_legitimacy replaces the WRONG
    # omen_power, which is MINING output in this TC per imp19c_modifier_equivalencies.txt.)

        modifier = {
			global_upper_strata_happyness = 0.05
			monthly_legitimacy = 0.05
			global_middle_strata_happyness = -0.02
		}
    }

    # Crown-appointed chamber — a lever of patronage: political influence + a content aristocracy.
    # [task law-adjacency-audit 2026-08-30] sequential prerequisite: upper_house_composition_law
    # is an ordered spectrum of democratisation (hereditary/spiritual appointment -> plain
    # appointment -> election -> state-representative election), so a country must hold the
    # immediately preceding tier to advance one step. Same has_law-in-allow idiom proven on
    # Women's Rights (00_social_laws.txt). appointed_hereditary_spiritual carries no allow, so it
    # stays reachable as the default/reversion tier.
    appointed = {
        allow = {
            has_law = appointed_hereditary_spiritual
        }
        modifier = {
			monthly_political_influence = 0.05
			global_upper_strata_happyness = 0.02
		}
    }

    # Elected upper house — the middle + lower strata gain a voice, at a standing stability cost.
    elected = {
        allow = {
            has_law = appointed
        }
        modifier = {
			global_middle_strata_happyness = 0.05
			global_lower_strata_happyness = 0.03
			stability_monthly_change = -0.02
		}
    }

    # A federal chamber of state representatives — binds a diverse realm together: assimilation + standing.
    state_representatives = {
        allow = {
            has_law = elected
        }
        modifier = {
			global_pop_assimilation_speed_modifier = 0.05
			diplomatic_reputation = 0.5
		}
    }

}
