# QING_kok_conquered_trigger — TRUE once Kokand is neutralised BY FORCE: subjugated (a CHI
# subject, wherever it sits) OR its heartland seat held (CHI or a CHI subject owns Kokand
# city p:110 — covers annexation and a march carving off Fergana). A LIVE derive: flips back
# false if a subjugated KOK later breaks vassalage AND CHI does not hold p:110.
QING_kok_conquered_trigger = {
	OR = {
		AND = {
			exists = c:KOK
			c:KOK = { is_subject_of = ROOT }
		}
		owns_or_subject_owns = 110
	}
}

# QING_kok_yielded_flag — TRUE once the caravan.3 settlement was concluded (dictate terms OR
# absorb the route). PERMANENT (the end-state flags are never cleared) — so a Kokand settled
# via caravan.3 stays settled even if it later breaks free, keeping both the caravan crisis
# chains and the khoja scare suppressed forever (design §NEW-MEDIUM-2).
QING_kok_yielded_flag = {
	OR = {
		has_variable = qing_caravan_kok_terms_dictated
		has_variable = qing_caravan_kok_absorbed
	}
}
