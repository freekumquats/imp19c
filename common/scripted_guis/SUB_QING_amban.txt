# SUB_QING_amban.txt — manual AMBAN management (#163), per-subject.
#
# These are the gated entry points the DIPLOMATIC-VIEW buttons call to manage
# the amban (駐紮大臣, resident commissioner) posted to an Inner Asian dependency
# BY HAND, overriding the automatic lifecycle in se_QING_AMBAN.txt. Each is
# scoped to the player (ROOT = overlord CHI) and reads scope:target as the
# subject currently shown in the diplomacy view, passed from the GUI via:
#     .AddScope('target', DiplomaticView.GetTargetCountry.MakeScope )
# — the SAME proven pattern the qing_subject_promote/demote/integrate buttons
# use (SUB_QING_subject_interactions.txt).
#
# COEXISTENCE WITH THE AUTO-LAYER (locked design, Option A):
#   • Posting/recalling/replacing an amban by hand sets qing_amban_manual on the
#     SUBJECT. While that flag is present the quarterly auto-sweep
#     (QING_amban_post_sweep) skips the dependency and the auto turnover/gone-
#     native/recall event DISPATCH in QING_amban_evaluate is frozen for it — but
#     the friction/affinity SCORING and opinion effects still run, so a poorly
#     chosen resident keeps its quarterly consequences (see the guards in
#     se_QING_AMBAN.txt tagged [#163]).
#   • "Return to automatic" clears the flag and hands the post back to the pulse.
#   • Free recall / repost, no cooldown (this panel IS the manual override).
#
# AI is disabled on every interaction (ai_is_valid = { always = no }) — CHI-only,
# player-only, exactly like the subject-interaction set.
#
# Prefix = qing_amban_manage_
########################################################################

# ======================================================================
# ELIGIBILITY (shared, read as an is_shown block below): scope:target is a
# subject of ROOT (CHI) whose ruler is of an Inner Asian culture group
# (mongolic / bodish) — the Lhasa / Urga / Kashgar march the Lifan Yuan
# governs. Mirrors the auto-sweep's random_subject limit so the manual and
# automatic layers agree on which dependencies take a resident.
# ======================================================================

# ---- POST: dispatch a resident to this dependency by hand -------------
# Shown only when the dependency is eligible and currently has NO amban. Gated
# on the Lifan Yuan office being filled (a vacant 理藩院 has no man to send — the
# same precondition the auto-sweep enforces) plus a modest influence cost.
qing_amban_manage_post_button = {
	scope = country
	saved_scopes = { target }

	is_shown = {
		tag = CHI
		exists = scope:target
		scope:target = {
			is_subject_of = ROOT
			# [#34 2026-08-07] use the SHARED warrant trigger (the SAME predicate that builds the
			# Lifan Yuan vacant-dependencies list + the amban post-sweep, #392) instead of an inline
			# culture-only test. The old inline OR{mongolic,bodish} MISSED the Manchu-ruled
			# autonomous-governorship posts ILI (伊犁將軍) / ULS (烏里雅蘇台將軍) that #392 widened the
			# warrant/sweep to cover — so those dependencies listed in the vacant roster with a
			# button whose is_shown returned false (a dead, button-less row). Aligning to the shared
			# trigger makes button ↔ list ↔ sweep one predicate.
			QING_amban_warrants_resident_trigger = yes
			# no LIVE resident currently posted here. A dead/departed amban may leave the var SET but
			# pointing at a non-employed char — treat that as vacant too, matching the vacant-list's
			# else-branch. Guard the var-read so it only fires when the var exists (avoid reading an
			# unset var: the first branch = truly-unset; the second = set-but-not-a-live-employee).
			OR = {
				NOT = { has_variable = qing_amban_here }
				AND = {
					has_variable = qing_amban_here
					NOT = { var:qing_amban_here = { is_alive = yes  employer = scope:target } }
				}
			}
		}
	}

	ai_is_valid = { always = no }

	is_valid = {
		# the Lifan Yuan supervisor must be in place to dispatch a resident
		custom_tooltip = {
			text = qing_amban_needs_lifanyuan_tt
			has_variable = qing_office_lifanyuan_holder
			var:qing_office_lifanyuan_holder = { is_alive = yes  employer = ROOT }
		}
		custom_tooltip = {
			text = qing_amban_needs_influence_25_tt
			political_influence >= 25
		}
	}

	effect = {
		# [#26 2026-08-07] The post button now OPENS A CHARACTER PICKER (like the Censorate's Commission an
		# Inspector) instead of auto-drawing the ablest graduate. This effect only PREPARES the picker:
		# it records the destination subject + builds the candidate list. The 25 PI charge + qing_amban_manual
		# + the actual post are MOVED to the row-click handler (qing_amban_appoint_selected) — so closing the
		# picker without choosing costs nothing (the is_valid PI>=25 gate on the button still blocks opening
		# when unaffordable). The .gui button also fires gui.createwidget to open qing_amban_picker_window.
		LOG_line = { sys = QING  msg = "amban: MANUAL post picker opened by" }
		# remember which subject the picker will post to (a country-ref var on CHI the row-handler reads).
		# [#26 review-fix] the row handler re-derives qing_amban_manual_target itself from this var, so no
		# save_scope here — the button's execution ends when the picker opens, and a saved scope would not
		# survive to the (later) row click anyway.
		set_variable = { name = qing_amban_picker_subject  value = scope:target }
		# build the eligible-candidate list the picker window walks.
		QING_amban_refresh_candidates = yes
	}
}

# ---- ROW-CLICK: appoint the PICKED candidate as amban to the stored subject ----
# [#26 2026-08-07] Character-scope handler for a picker row. this = the chosen candidate; scope:player = CHI.
# Reads qing_amban_picker_subject (set by the post button). Pays the 25 PI, claims the post
# (qing_amban_manual), and dispatches the PICKED char via the qing_amban.6 trampoline (NOT inlined — the
# #34 rule; and the wire runs inside the event). Mirrors qing_gov_office_appoint_selected.
qing_amban_appoint_selected = {
	scope = character
	saved_scopes = { player }
	ai_is_valid = { always = no }
	is_shown = {
		scope:player = { tag = CHI  has_variable = qing_amban_picker_subject }
		employer = scope:player
	}
	is_valid = {
		is_adult = yes
		is_alive = yes
		# not already holding a court position (one man, one post) + still CHI-employed.
		NOT = { QING_char_holds_court_position = yes }
		custom_tooltip = {
			text = qing_amban_needs_influence_25_tt
			scope:player = { political_influence >= 25 }
		}
		# [#26 review-fix] the Lifan Yuan supervisor must still be in place — mirror the button's gate + the
		# auto-sweep precondition, so a 理藩院 holder dying while the picker is open can't post an unsupervised
		# amban (the race the button's is_valid alone didn't cover).
		custom_tooltip = {
			text = qing_amban_needs_lifanyuan_tt
			scope:player = { has_variable = qing_office_lifanyuan_holder  var:qing_office_lifanyuan_holder = { is_alive = yes  employer = scope:player } }
		}
		# the stored destination must still be a valid, amban-less subject of CHI.
		scope:player = {
			var:qing_amban_picker_subject = {
				is_subject_of = scope:player
				NOT = { has_variable = qing_amban_here }
			}
		}
	}
	effect = {
		LOG_line = { sys = QING  msg = "amban: player PICKED a resident to post" }
		# save the CHOSEN character (this) for the trampoline.
		save_scope_as = qing_amban_picked
		scope:player = {
			add_political_influence = -25
			# claim the post (freeze the auto-sweep for this dependency).
			var:qing_amban_picker_subject = { set_variable = { name = qing_amban_manual  value = 1 } }
			# hand the subject to the trampoline via a saved scope (the picked char is saved above).
			var:qing_amban_picker_subject = { save_scope_as = qing_amban_manual_target }
			trigger_event = { id = qing_amban.6  days = 0 }
			# clear the picker-subject marker so stray rows can't re-fire against a stale destination.
			remove_variable = qing_amban_picker_subject
		}
	}
}

# ---- RECALL: withdraw the resident and leave the post vacant ----------
# Shown when a resident is posted here. Recalling claims the post (sets
# qing_amban_manual) so the auto-sweep does not simply re-staff it next quarter —
# the player has chosen to leave the march unattended until they act again (or
# press "return to automatic"). Free of influence cost (recalling your own man
# is the throne's prerogative).
qing_amban_manage_recall_button = {
	scope = country
	saved_scopes = { target }

	is_shown = {
		tag = CHI
		exists = scope:target
		scope:target = {
			is_subject_of = ROOT
			has_variable = qing_amban_here
		}
	}

	ai_is_valid = { always = no }

	is_valid = {
		# the stored resident must still be alive and in this subject's employ
		custom_tooltip = {
			text = qing_amban_recall_valid_tt
			scope:target = {
				var:qing_amban_here = { is_alive = yes  employer = scope:target }
			}
		}
		# [BT-19] recall now costs 10 PI (see effect) — gate the button on affordability
		custom_tooltip = {
			text = qing_amban_recall_needs_influence_10_tt
			political_influence >= 10
		}
	}

	effect = {
		LOG_line = { sys = QING  msg = "amban: MANUAL recall from by" }
		# [BT-19] Recalling a resident is a direct intervention in the frontier bureaucracy:
		# it costs 10 political influence (was 0) and a little tyranny (add_tyranny = 1). Proven
		# idioms: add_political_influence = -N (anoint_heir_button.txt:98), add_tyranny = N country
		# scope (se_QING_CENSORATE.txt:155). PI gate added to is_valid above matches the cost.
		add_political_influence = -10
		add_tyranny = 1
		# claim the post so the auto-sweep leaves the vacancy alone
		scope:target = { set_variable = { name = qing_amban_manual  value = 1 } }
		QING_amban_recall = { subject = scope:target  reason = "manual-recall" }
	}
}

# ---- REPLACE: recall the sitting resident and dispatch a fresh one ----
# One-click recall + post (posting over a live resident would strand the old man,
# so the recall MUST run first). Shown when a resident is posted here; gated like
# the post button (Lifan Yuan filled + influence). Claims the post.
qing_amban_manage_replace_button = {  # [#34] is_shown culture test aligned to QING_amban_warrants_resident_trigger below
	scope = country
	saved_scopes = { target }

	is_shown = {
		tag = CHI
		exists = scope:target
		scope:target = {
			is_subject_of = ROOT
			has_variable = qing_amban_here
			# [#34 2026-08-07] shared warrant trigger instead of the inline culture-only OR (same
			# alignment as the post button above) so the replace action is offered on the Manchu-ruled
			# autonomous-governorship posts ILI/ULS too, not just mongolic/bodish-ruled dependencies.
			QING_amban_warrants_resident_trigger = yes
		}
	}

	ai_is_valid = { always = no }

	is_valid = {
		custom_tooltip = {
			text = qing_amban_needs_lifanyuan_tt
			has_variable = qing_office_lifanyuan_holder
			var:qing_office_lifanyuan_holder = { is_alive = yes  employer = ROOT }
		}
		custom_tooltip = {
			text = qing_amban_needs_influence_25_tt
			political_influence >= 25
		}
	}

	effect = {
		LOG_line = { sys = QING  msg = "amban: MANUAL replace at by" }
		add_political_influence = -25
		# [BT-19] replacing a sitting resident is a heavier intervention than recall — a little
		# more tyranny (add_tyranny = 2 vs recall's 1). Country scope (se_QING_CENSORATE.txt:155).
		add_tyranny = 2
		scope:target = { set_variable = { name = qing_amban_manual  value = 1 } }
		# recall the old man FIRST (clears qing_amban_here) — recall carries no sorting iterator,
		# so it is safe to inline. Then dispatch the new resident via the qing_amban.5 trampoline
		# ([#34 CRASHFIX] — QING_amban_post now inlines an ordered_character sorting iterator that
		# access-violates a scripted_gui at boot; see the post-button note above).
		QING_amban_recall = { subject = scope:target  reason = "manual-replace" }
		scope:target = { save_scope_as = qing_amban_manual_target }
		trigger_event = { id = qing_amban.5  days = 0 }
	}
}

# ---- RETURN TO AUTOMATIC: hand the post back to the quarterly pulse ----
# Shown when the player has claimed this dependency (qing_amban_manual set).
# Clearing the flag re-enrols it in the auto-sweep and auto-turnover: if the post
# is vacant the pulse will staff it; if occupied, the lifecycle events resume.
qing_amban_manage_return_auto_button = {
	scope = country
	saved_scopes = { target }

	is_shown = {
		tag = CHI
		exists = scope:target
		scope:target = {
			is_subject_of = ROOT
			has_variable = qing_amban_manual
		}
	}

	ai_is_valid = { always = no }

	is_valid = { always = yes }

	effect = {
		LOG_line = { sys = QING  msg = "amban: returned to AUTOMATIC management by" }
		scope:target = { remove_variable = qing_amban_manual }
	}
}
