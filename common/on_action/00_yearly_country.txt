yearly_country_pulse =
{
	on_actions = {
		prominent_actors_setup
		delay = { days = 150 }

		# [#94] Japanese bakumatsu -> Boshin pulse. AI-AUTONOMOUS: self-gates on
		# tag = TKG (NOT is_ai), so it drives Japan's road to the Meiji Restoration
		# for an AI Japan (the player is the Qing). Registered in
		# japan_bakumatsu_on_actions.txt. This is the fix for the #81 AI gap.
		japan_bakumatsu_pulse_on_action

		# [#276] Dynastic-marriage-diplomacy pulse. AI-AUTONOMOUS: self-gates on
		# MARRIAGE_eligible_realm_trigger (a Western dynastic monarchy, NOT is_ai),
		# so every eligible European house drives its own royal-marriage diplomacy
		# (pacts -> unions -> inheritance) while the player runs the Qing. Registered
		# in marriage_on_actions.txt; MARRIAGE_pulse self-throttles ~5 years.
		marriage_diplomacy_pulse_on_action

		# [#27] Frontier-march autonomous pulses. AI-AUTONOMOUS: self-gate on
		# is_subject_type = frontier_protectorate (NOT is_ai — the marches are AI-run
		# subjects that expand + consolidate on their own). Registered in
		# qing_march_on_actions.txt; se_QING_MARCH_PULSE.txt. Low yearly chances.
		qing_march_expand_pulse_on_action
		qing_march_integrate_pulse_on_action
	}
	
	effect = {
		if = {
			limit = {
				tag = CHI
				has_variable = new_era_variable
			}
			set_variable = {
				name = regnal_year_var
				value = 1
			}
			remove_variable = new_era_variable
			remove_variable = is_jiaqing_era
			remove_variable = is_daoguang_era
			remove_variable = is_xianfeng_era
			remove_variable = is_qixiang_era
			remove_variable = is_tongzhi_era
			remove_variable = is_guangxu_era
			remove_variable = is_xuantong_era
			switch = {
				trigger = has_variable
				will_be_daoguang_era = {
					set_variable = is_daoguang_era
					remove_variable = will_be_daoguang_era
				}
				will_be_xianfeng_era = {
					set_variable = is_xianfeng_era
					remove_variable = will_be_xianfeng_era
				}
				will_be_qixiang_era = {
					set_variable = is_qixiang_era
					remove_variable = will_be_qixiang_era
				}
				will_be_tongzhi_era = {
					set_variable = is_tongzhi_era
					remove_variable = will_be_tongzhi_era
				}
				will_be_guangxu_era = {
					set_variable = is_guangxu_era
					remove_variable = will_be_guangxu_era
				}
				will_be_xuantong_era = {
					set_variable = is_xuantong_era
					remove_variable = will_be_xuantong_era
				}
			}
		}
		else_if = {
			limit = {
				tag = CHI
				NOT = { has_variable = new_era_variable }
			}
			change_variable = {
				name = regnal_year_var
				add = 1
			}
		}
	}

	# [ADMIN-refresh 2026-08-04] Re-apply the administration-capacity -> state-loyalty impact
	# (ADMIN_state_loyalty_from_province_drain / _gain) on a BIANNUAL cadence. It was applied ONCE
	# at economy setup and then frozen — the monthly refresh (monthly_administration_pulse) was
	# disabled for "unacceptable monthly slowdowns" (oa_wealth_changes.txt:133), so the impact never
	# tracked population growth or admin-district construction after game start. This restores a LIVE
	# refresh at a cadence cheap enough to afford: the effect walks every_governorships ->
	# every_governorship_state, so a self-throttle var (730 days) keeps each country to one refresh
	# per ~2 years rather than the every-month walk that was too slow. yearly_country_pulse.effect
	# fires per country, so the work is naturally spread across countries and years. Country scope.
	if = {
		limit = { NOT = { has_variable = admin_loyalty_refresh_cd } }
		set_variable = { name = admin_loyalty_refresh_cd  days = 730 }
		ADMIN_set_loyalty_impact_all_states = yes
	}

	# §11 Layer 3 (Qing): the heavy annual demographic scan that re-derives
	# the ethnic-tension TARGET from the realm's ethnic geography. Runs once a
	# year, CHI + human only (the quarterly pulse cheaply drifts the live
	# counter toward this target). Gated separately from the regnal block above.
	if = {
		limit = {
			tag = CHI
			is_ai = no
		}
		QING_DECLINE_scan_ethnic_target = yes
		# §11 L4 ext: the annual civic-nation pass — extend citizenship (integrate a
		# restive culture) and, at high civic identity, a small popular-sinicization
		# trickle. Runs right after the scan, same annual CHI+human cadence.
		QING_DECLINE_civic_assimilation = yes
	}
	
	random_events = {

	}
	
	events = {
			  
	}
}
