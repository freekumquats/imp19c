	effect = {
		if = {
			limit = {

				}
			}
			CURRENCY_mint_currency = yes
		}
		# [#28 2026-08-07] monthly minting trace (CHI-only, -debug_mode-gated inside the effect) — logs

		# month's clamp. Self-gated (tag=CHI + debug_log's -debug_mode gate); a no-op otherwise.
		ECON_LOG_minting_snapshot = yes

		# ratio stress meter, apply the band, offer the crisis event. Self-gated to
		# non-CHI monetary economies and throttled to a quarterly drift by its own
		# cooldown, so for most countries most months this is one has_variable check.

		# backing_type (se_MONSTD.txt). Uniform for every country; its own quarterly cooldown
		# keeps the common-case cost to one has_variable check.

	}

quarterly_trade_pulse = {
	effect = {

		if = {
			limit = {
				NOT = { has_global_variable = done_quarterly_trade }
			}
			remove_global_variable = trade_ranks_sorted # Used by event piechart_update.2
			ECON_LOG_phase_enter = { stage = quarterly_production }
			# Commence Production
			every_country = {
				every_governorships = {
					limit = {
						NOT = {
							has_variable = internal_trade_scope
						}
					}
					# [logfix 2026-08-04] REMOVED a bare `CURRENCY_governorship_send_to_reserves = yes` here:
					# the macro requires a $tradegood$ arg (used ~19x), so the bare call compile-failed at load
					# ("missing arguments: tradegood") and did nothing. It was also MISPLACED — this is the
					# first-time governorship trade-scope SETUP block; reserve-sending is a quarterly FLOW done
					# correctly (gold + silver) by CURRENCY_all_governorships_send_to_reserves at line ~335 of
					# this same pulse. Removing the dead, misplaced call: no behaviour change, kills the error.
					TRADE_set_internal_trade_scope = yes
					TRADE_update_governorship_TZs = yes # In case new governorships are formed. TODO: slot this into the territory ownership changed effect once that is made
					
				}
				every_governorships = {
					TRADE_governorship_get_pops_this_quarter = yes
					GOODS_governorship_produce_all = yes
					COTTAGEIND_produce_all = yes
					# [fix, task #103-follow-up 2026-08-16] direct, undiscounted per-building military-
					# goods bonus -- see design/DESIGN_103_FOLLOWUP_COTTAGE_MILITARY_BOOST.md Part 2.
					COTTAGEIND_military_goods_building_bonus = yes
				}
				# #72: trace the production path (no-op outside -debug_mode). Reads the
				# per-governorship industrialisation bonus cached inside produce_all above.
				ECON_LOG_production_snapshot = yes
				# [#MG-flood-diag] read-only probe: after produce_all has run this pass,
				# walk the governorships and log (with scope dumps) exactly which scopes
				# hit the cottage empty-var flood + the stockpile-write guard-gap, and why

				# no-op outside -debug_mode. Does not touch Sobisonator's cottage effects.

			WEALTH_pay_wages_all_countries = yes
			# Cache currency power and currency power trade bonuses before trading
			every_province = {
				limit = {
					has_variable = currency_name
				}
				CURRENCY_cache_power = yes
			}
			every_province = {
				limit = {
					has_variable = currency_name
				}
				CURRENCY_cache_power_trade_bonus = yes
			}
			every_country = {
				INCOME_collect_all_taxes_country = yes
				INCOME_update_treasury_country = yes
				WEALTH_cache_national_wealth = yes
				MILITARY_update_supplies = yes
				INCOME_calculate_and_distribute_military_procurement_wealth_owed = yes
				MILITARY_supplies_log_diagnostic = yes
				# Diagnostic snapshot (se_ECON_LOG.txt): treasury / national wealth /
				# jobs / currency, read AFTER WEALTH_cache_national_wealth so the
				# numbers are the freshly-cached ones. No-op outside -debug_mode.
				ECON_LOG_quarter = yes
			}
			DIPLOMACY_update_global_power_status = yes
			# [task #21 fix 2026-08-26] AI_update_all_diplomatic_plays was renamed to
			# DIPLOMACY_update_all_diplomatic_plays (se_DIPLOMACY.txt:501); the old name doesn't

			# plays never got their per-province status/event update from this pulse.

			ECON_LOG_phase_exit = { stage = quarterly_production }

			set_global_variable = {
				name = done_quarterly_trade
				days = 1
			}

quarterly_reset_trade_transaction_totals = {
	effect = {
		if = {
			limit = {
				NOT = { has_global_variable = trade_transaction_totals_reset }
			}
			every_trade_center = {
		    	set_variable = {
		    		name = TZ_shipping
		    		value = 0
		    	}
		    	set_variable = {
		    		name = TZ_this_quarter_transport_pool
		    		value = 0
		    	}
		    }
		    ECON_LOG_phase_enter = { stage = trade_reset_totals }
		    GT_split_setup_global_shipping_costs_pool = yes
			every_country = {
				every_governorships = {
					WEALTH_cache_shipping_trade_values = yes
					DEMAND_cache_and_update_elasticity_impact = yes
					GT_reset_trade_transaction_totals = yes
					# [#139 Finding 1 2026-07-30] Cache governorship infrastructure capacity ONCE per
					# quarter here, alongside the sibling once-per-quarter caches. It was previously
					# refreshed inside GT_split_do_global_trade_split (se_GLOBALTRADE_split.txt:12), which
					# runs once per governorship PER CATEGORY PASS -> 7x/quarter. The svalue
					# (TRADE_governorship_infrastructure_capacity_svalue) is 7 addends each walking every
					# state province (canal/commerce/railway/depot/port/river-port + shipping-power), so it
					# was the most expensive per-pass recompute in the hot loop, yet it changes only when a
					# building finishes construction — never within a quarter's 7 passes. All readers use
					# var:governorship_infrastructure_capacity (se_GLOBALTRADE_split.txt:782/797/2132/2147),
					# so this is transparent to them. TEMPORAL CAVEAT (accepted, same class as the sibling
					# WEALTH_cache_shipping_trade_values / DEMAND_cache_and_update_elasticity_impact caches
					# right here): a building completed mid-quarter now takes effect NEXT quarter's trade
					# rather than the next category pass within the same quarter. See manufactured_goods.md 13.
					GT_split_cache_governorship_infrastructure_capacity = yes
				}
			}
			SHIPPING_update_TZ_overview_piecharts = yes
			GT_split_update_governorship_vars = yes
			GT_split_cache_TZ_penetration_values = yes
			# [#145 Part A 2026-07-30] Quarterly refresh of global_mean_price_<good>. This global
			# was computed ONCE at day-0 setup (PRICE_update_TZ_prices -> PRICE_update_all_global_mean_prices,
			# oa_economy_setup.txt:2360) and never refreshed, so I7's input-cost factoring
			# (PRICE_factor_raw_input_costs_X, se_PRICE.txt) read a frozen day-0 mean forever even though
			# local_price_<good> IS recomputed every quarter by GT_set_tradegood_price (se_GLOBALTRADE_split.txt)
			# on the trade-center provinces. Call ONLY the averaging pass here — NOT the full
			# PRICE_update_TZ_prices, which would also re-run PRICE_update_price_TZ/PRICE_normalise_price and
			# double-recompute local_price via the defunct SELL formula. Runs at quarter-start (this block is
			# globally gated to fire once per quarter), so it reads the PRIOR quarter's final local prices —
			# the same "stable prior snapshot" ordering I7 documented, now refreshed each quarter. It touches
			# only global vars (writes global_mean_price_*, reads var:local_price_* on trade centers), so it
			# is scope-safe here. See manufactured_goods.md section 11.
			PRICE_update_all_global_mean_prices = yes
		}
		set_global_variable = {
			name = trade_transaction_totals_reset
			days = 1
		}
	}
}

quarterly_apply_trade_changes_and_consume = {
	effect = {
		ECON_LOG_phase_enter = { stage = trade_apply_consume }
		every_governorships = {
			# Actually apply the changes to stockpiles and get shortages
			CONSUME_all_stockpiles = yes
			GT_do_all_wealth_distribution = yes
			GT_save_final_quarterly_wealth_values = yes
		}
		# [fix, task #103-follow-up 2026-08-16] cottage/Military Supplies probe -- country scope
		# (this on_action's own scope), placed right after the governorship block above closes so
		# shortage_<good>/COTTAGEIND_produced_<good> reflect THIS quarter's real consumption. See
		# design/DESIGN_103_FOLLOWUP_COTTAGE_MILITARY_BOOST.md Part 3.
		# [review-fix] CHI-only guard -- this on_action runs once PER COUNTRY/quarter; without this
		# the probe emitted for all ~200 countries, corrupting the CHI read-back and flooding the log.
		if = {
			limit = { tag = CHI }
			ECON_LOG_curx_cottage_military = yes
		}
		# [fix, task #9 2026-08-16] real wage payment for every Qing character-appointment post --
		# monthly_wage_for_character never actually paid anyone (see se_QING_WAGES.txt header).
		# CHI-only guard, same reasoning as the cottage/Military Supplies probe above.
		if = {
			limit = { tag = CHI }
			QING_pay_officeholder_wages = yes
		}
		CURRENCY_all_governorships_send_to_reserves = yes # Collect all the precious metals demanded for the gold and silver reserves
		if = {
			limit = {
				has_variable = official_currency
			}
			CURRENCY_private_purchase_or_sell_reserves = yes
			CURRENCY_apply_inflation_wealth_malus_country = yes
			CURRENCY_update_amt_circulated = yes
			# [CPI 2026-08-18] DISPLAY-ONLY Consumer Price Index update -- once per country per
			# quarter, right after the circulation recompute. Writes only CPI_* display vars; see
			# design/DESIGN_CPI_INFLATION_READOUT.md. NOT wired at the global quarterly_trade_pulse
			# site (a double call would break the alpha/annualise math).
			CPI_update = yes
			# [#28 2026-08-07 review-fix] currency band trace at the FRESH-value site — right after the
			# quarterly recompute (ratio/needed/circ reflect THIS quarter's reserve buy/sell + circulation
			# update), unlike the ECON_LOG_quarter call (~:209) which runs earlier = last quarter's state.
			# Idempotent band debug_logs; self-gated tag=CHI + -debug_mode inside the effect.
			ECON_LOG_currency_snapshot = yes
			# [#23 EXACT-VALUE forensic] exact-integer dump of the whole currency chain at the FRESH
			# post-recompute site (POST). Paired with the PRE dump in ECON_LOG_quarter (~:209), the
			# PRE→POST delta across this quarter's recompute shows exactly which input moved when the
			# ratio flips modest-inflation → deflation-floor. CHI-only, -debug_mode-gated inside.
			ECON_LOG_curx_dump_post = yes
		}
		# CACHE OFT-REUSED VALUES
		set_variable = {
			name = TRADE_national_expenditure
			value = TRADE_national_expenditure
		}
		set_variable = {
			name = TRADE_national_income
			value = TRADE_national_income
		}

		# [logistics P1/P3] military-logistics consequence layer. Runs here in
		# country scope AFTER every_governorships above refreshed each governorship's
		# shortage_* variables (se_CONSUME.txt). Reads the worst military-goods / coal
		# shortage and stamps tiered attrition-morale-movement and naval-attrition
		# country modifiers (se_LOGISTICS.txt). Additive to the engine food/attrition
		# supply loop; self-clears when supply recovers.
		LOGISTICS_quarter = yes

	}
}
