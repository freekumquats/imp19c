# qing_march_on_actions.txt — [#27] autonomous yearly pulses for the frontier
# marches (都護府, frontier_protectorate). se_QING_MARCH_PULSE.txt.
#
# AI-AUTONOMOUS, like the Japan/marriage pulses: each self-gates on
# `is_subject_type = frontier_protectorate` (NEVER is_ai — the marches are AI-run
# subjects that must expand + consolidate on their own). Registered in
# 00_yearly_country.txt's yearly_country_pulse.on_actions LIST (the merge form —
# only ONE file's inline effect runs, but on_actions lists DO merge, per #254).
# yearly_country_pulse.effect already fires for every country; these on_actions are
# chained from that list, so the subject-type limit is the only gate needed and the
# cost for every other country is a single rejected trigger check.
########################################################################

# The march's yearly EXPANSION roll: a low-chance war of conquest (gated on peace) +
# a colonisation roll on unowned bordering land. se_QING_MARCH_PULSE.txt (§3.4).
qing_march_expand_pulse_on_action = {
	trigger = {
		is_subject_type = frontier_protectorate
	}
	effect = {
		QING_march_expand_pulse = yes
	}
}

# The march's yearly INTEGRATION roll: slowly digest one princely state toward direct
# rule (改土歸流) via the march-specific progress var + SUBJ_QING_absorb_subject at
# threshold. se_QING_MARCH_PULSE.txt (§3.5).
qing_march_integrate_pulse_on_action = {
	trigger = {
		is_subject_type = frontier_protectorate
		any_subject = { count >= 1 }
	}
	effect = {
		QING_march_integrate_pulse = yes
	}
}
